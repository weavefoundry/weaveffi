1//! The value-type attributes: `#[weaveffi::record]`,
2//! `#[weaveffi::enumeration]`, `#[weaveffi::error]`,
3//! `#[weaveffi::interface]`, and `#[weaveffi::custom]`.
4//!
5//! Each implements the runtime's value traits for the type under the crate's
6//! tag (a record's or rich enum's `Codec`, a C-style enum's `CEnum`, an
7//! error domain's `ErrorType`, an interface's object traits) and exports its
8//! metadata frame and contract function.
9

24fn no_args(attr: &TokenStream, what: &str) -> syn::Result<()> {
25    if attr.is_empty() {
26        Ok(())
27    } else {
28        Err(syn::Error::new(
29            attr.span(),
30            format!("weaveffi: `#[weaveffi::{what}]` takes no arguments"),
31        ))
32    }
33}

35fn no_generics(generics: &syn::Generics) -> syn::Result<()> {
36    if generics.params.is_empty() {
37        Ok(())
38    } else {
39        Err(syn::Error::new(
40            generics.span(),
41            "weaveffi: an exported type can't be generic",
42        ))
43    }
44}

46/// The local each of `n` fields binds to in a generated `match` arm:
47/// `Self::V { key: __wv_f0 }`, never the shorthand `Self::V { key }`, which
48/// a constant named `key` in scope would turn into a constant pattern.
49fn field_locals(n: usize) -> Vec<syn::Ident> {
50    (0..n).map(|i| format_ident!("__wv_f{i}")).collect()
51}

53/// The `Name` expression of a field's type.
54fn field_name(ty: &syn::Type) -> TokenStream {
55    let tag = tag();
56    quote!(<#ty as ::weaveffi::__rt::FfiType<#tag>>::NAME)
57}

59/// A named field: its identifier, type, and docs.
60struct NamedField<'a> {
61    ident: &'a syn::Ident,
62    ty: &'a syn::Type,
63    attrs: &'a [syn::Attribute],
64}
65
66fn named_fields<'a>(fields: &'a syn::Fields, what: &str) -> syn::Result<Vec<NamedField<'a>>> {
67    match fields {
68        syn::Fields::Named(named) => named
69            .named
70            .iter()
71            .map(|f| {
72                no_member_cfg(&f.attrs, &format!("{what} field"))?;
73                Ok(NamedField {
74                    ident: f.ident.as_ref().expect("a named field"),
75                    ty: &f.ty,
76                    attrs: &f.attrs,
77                })
78            })
79            .collect(),
80        syn::Fields::Unit => Ok(Vec::new()),
81        syn::Fields::Unnamed(u) => Err(syn::Error::new(
82            u.span(),
83            format!("weaveffi: {what} fields must be named"),
84        )),
85    }
86}

88fn field_ir(fb: &mut FrameBuilder, fields: &[NamedField<'_>]) -> Vec<Field> {
89    fields
90        .iter()
91        .map(|f| Field {
92            name: f.ident.to_string().trim_start_matches("r#").to_string(),
93            ty: fb.ty(field_name(f.ty)),
94            doc: docs(f.attrs),
95        })
96        .collect()
97}

99/// `#[weaveffi::record]`.
100pub(crate) fn record(attr: TokenStream, item: syn::ItemStruct) -> syn::Result<TokenStream> {
101    no_args(&attr, "record")?;
102    no_generics(&item.generics)?;
103    let prefix = prefix()?;
104    let tag = tag();
105    let name = item.ident.to_string();
106    let ty = &item.ident;
107    let fields = named_fields(&item.fields, "a record's")?;
108    if fields.is_empty() {
109        return Err(syn::Error::new(
110            item.ident.span(),
111            "weaveffi: a record needs at least one field",
112        ));
113    }
114    let writes = fields.iter().map(|f| {
115        let (id, fty) = (f.ident, f.ty);
116        quote!(<#fty as ::weaveffi::__rt::FfiType<#tag>>::write(&self.#id, w);)
117    });
118    let reads = fields.iter().map(|f| {
119        let (id, fty) = (f.ident, f.ty);
120        quote!(#id: unsafe { <#fty as ::weaveffi::__rt::FfiType<#tag>>::read(r) }?)
121    });
122    let lit = name_lit(&name);
123    let codec = quote! {
124        #[allow(unsafe_code, unused_unsafe)]
125        impl ::weaveffi::__rt::Codec<#tag> for #ty {
126            const NAME: ::weaveffi::__rt::Name = #lit;
127            fn encode(&self, w: &mut ::weaveffi::__rt::BufferWriter) {
128                #(#writes)*
129            }
130            unsafe fn decode(
131                r: &mut ::weaveffi::__rt::BufferReader<'_>,
132            ) -> ::core::result::Result<Self, ::weaveffi::__rt::BufferDecodeError> {
133                ::core::result::Result::Ok(Self { #(#reads),* })
134            }
135        }
136        ::weaveffi::__weaveffi_buffered!(#tag, #ty);
137    };
138    let mut fb = FrameBuilder::new();
139    let def = RecordDef {
140        name: name.clone(),
141        doc: docs(&item.attrs),
142        deprecated: deprecated(&item.attrs),
143        fields: field_ir(&mut fb, &fields),
144    };
145    let frame = fb.finish(&prefix, location(item.ident.span()), Item::Record(def));
146    let sig = Sig::record(&name, fields.iter().map(|f| field_name(f.ty)).collect());
147    let contract = contract(&prefix, &name, sig);
148    let cfg = cfgs(&item.attrs);
149    let generated = scoped(&cfg, quote!(#codec #frame #contract));
150    Ok(quote!(#item #generated))
151}

153/// The discriminant of each variant: its written `= value`, or one more
154/// than the previous variant's (`0` for the first).
155fn discriminants(item: &syn::ItemEnum) -> syn::Result<Vec<i32>> {
156    let mut next = 0i32;
157    item.variants
158        .iter()
159        .map(|v| {
160            let value = match &v.discriminant {
161                Some((_, expr)) => int_literal(expr)?,
162                None => next,
163            };
164            next = value.wrapping_add(1);
165            Ok(value)
166        })
167        .collect()
168}
169
170fn int_literal(expr: &syn::Expr) -> syn::Result<i32> {
171    let err = || {
172        syn::Error::new(
173            expr.span(),
174            "weaveffi: a discriminant must be an integer literal that fits in an i32",
175        )
176    };
177    match expr {
178        syn::Expr::Lit(syn::ExprLit {
179            lit: syn::Lit::Int(i),
180            ..
181        }) => i.base10_parse::<i32>().map_err(|_| err()),
182        syn::Expr::Unary(syn::ExprUnary {
183            op: syn::UnOp::Neg(_),
184            expr,
185            ..
186        }) => int_literal(expr).map(|v| -v),
187        syn::Expr::Group(g) => int_literal(&g.expr),
188        _ => Err(err()),
189    }
190}

192/// `#[weaveffi::enumeration]`: a C-style enum, or a rich enum whose
193/// variants carry named fields.
194pub(crate) fn enumeration(attr: TokenStream, item: syn::ItemEnum) -> syn::Result<TokenStream> {
195    no_args(&attr, "enumeration")?;
196    no_generics(&item.generics)?;
197    let prefix = prefix()?;
198    let tag = tag();
199    let name = item.ident.to_string();
200    let ty = &item.ident;
201    let values = discriminants(&item)?;
202    let variants: Vec<(&syn::Variant, Vec<NamedField<'_>>)> = item
203        .variants
204        .iter()
205        .map(|v| {
206            no_member_cfg(&v.attrs, "an enum variant")?;
207            Ok((v, named_fields(&v.fields, "an enum variant's")?))
208        })
209        .collect::<syn::Result<_>>()?;
210    if variants.is_empty() {
211        return Err(syn::Error::new(
212            item.ident.span(),
213            "weaveffi: an enum needs at least one variant",
214        ));
215    }
216    let rich = variants.iter().any(|(_, f)| !f.is_empty());
217    let lit = name_lit(&name);
218    let impls = if rich {
219        let writes = variants.iter().zip(&values).map(|((v, fields), value)| {
220            let vid = &v.ident;
221            let ids: Vec<_> = fields.iter().map(|f| f.ident).collect();
222            let locals = field_locals(ids.len());
223            let tys = fields.iter().map(|f| f.ty);
224            let pat = if matches!(v.fields, syn::Fields::Unit) {
225                quote!(Self::#vid)
226            } else {
227                quote!(Self::#vid { #(#ids: #locals),* })
228            };
229            quote! {
230                #pat => {
231                    w.write_i32(#value);
232                    #(<#tys as ::weaveffi::__rt::FfiType<#tag>>::write(#locals, w);)*
233                }
234            }
235        });
236        let reads = variants.iter().zip(&values).map(|((v, fields), value)| {
237            let vid = &v.ident;
238            if matches!(v.fields, syn::Fields::Unit) {
239                return quote!(#value => Self::#vid,);
240            }
241            let reads = fields.iter().map(|f| {
242                let (id, fty) = (f.ident, f.ty);
243                quote!(#id: unsafe { <#fty as ::weaveffi::__rt::FfiType<#tag>>::read(r) }?)
244            });
245            quote!(#value => Self::#vid { #(#reads),* },)
246        });
247        quote! {
248            #[allow(unsafe_code, unused_unsafe)]
249            impl ::weaveffi::__rt::Codec<#tag> for #ty {
250                const NAME: ::weaveffi::__rt::Name = #lit;
251                fn encode(&self, w: &mut ::weaveffi::__rt::BufferWriter) {
252                    match self {
253                        #(#writes)*
254                    }
255                }
256                unsafe fn decode(
257                    r: &mut ::weaveffi::__rt::BufferReader<'_>,
258                ) -> ::core::result::Result<Self, ::weaveffi::__rt::BufferDecodeError> {
259                    ::core::result::Result::Ok(match r.read_i32()? {
260                        #(#reads)*
261                        _ => {
262                            return ::core::result::Result::Err(::weaveffi::__rt::BufferDecodeError {
263                                context: "unknown enum tag",
264                            })
265                        }
266                    })
267                }
268            }
269            ::weaveffi::__weaveffi_buffered!(#tag, #ty);
270        }
271    } else {
272        let to = variants.iter().zip(&values).map(|((v, _), value)| {
273            let vid = &v.ident;
274            quote!(Self::#vid => #value,)
275        });
276        let from = variants.iter().zip(&values).map(|((v, _), value)| {
277            let vid = &v.ident;
278            quote!(#value => Self::#vid,)
279        });
280        quote! {
281            impl ::weaveffi::__rt::CEnum<#tag> for #ty {
282                const NAME: ::weaveffi::__rt::Name = #lit;
283                fn to_i32(&self) -> i32 {
284                    match self {
285                        #(#to)*
286                    }
287                }
288                fn from_i32(value: i32) -> ::core::option::Option<Self> {
289                    ::core::option::Option::Some(match value {
290                        #(#from)*
291                        _ => return ::core::option::Option::None,
292                    })
293                }
294            }
295            ::weaveffi::__weaveffi_c_enum!(#tag, #ty);
296        }
297    };
298    let mut fb = FrameBuilder::new();
299    let def = EnumDef {
300        name: name.clone(),
301        doc: docs(&item.attrs),
302        deprecated: deprecated(&item.attrs),
303        variants: variants
304            .iter()
305            .zip(&values)
306            .map(|((v, fields), value)| EnumVariant {
307                name: v.ident.to_string(),
308                value: Some(*value),
309                doc: docs(&v.attrs),
310                fields: field_ir(&mut fb, fields),
311            })
312            .collect(),
313    };
314    let frame = fb.finish(&prefix, location(item.ident.span()), Item::Enum(def));
315    let sig = Sig::enumeration(
316        &name,
317        variants
318            .iter()
319            .zip(&values)
320            .map(|((v, fields), value)| {
321                (
322                    v.ident.to_string(),
323                    *value,
324                    fields.iter().map(|f| field_name(f.ty)).collect(),
325                )
326            })
327            .collect(),
328    );
329    let contract = contract(&prefix, &name, sig);
330    let cfg = cfgs(&item.attrs);
331    let generated = scoped(&cfg, quote!(#impls #frame #contract));
332    Ok(quote!(#item #generated))
333}

335/// The `#[weaveffi(message = "...")]` template of an error variant.
336fn message_template(attrs: &[syn::Attribute]) -> syn::Result<Option<syn::LitStr>> {
337    let mut found = None;
338    for attr in attrs.iter().filter(|a| is_helper(a)) {
339        attr.parse_nested_meta(|meta| {
340            if meta.path.is_ident("message") {
341                found = Some(meta.value()?.parse::<syn::LitStr>()?);
342                Ok(())
343            } else {
344                Err(meta.error("weaveffi: an error variant takes `#[weaveffi(message = \"...\")]`"))
345            }
346        })?;
347    }
348    Ok(found)
349}

351/// The names a format template's `{name}` and `{name:spec}` placeholders
352/// refer to (`{{` escapes skipped).
353fn placeholders(template: &str) -> Vec<String> {
354    let mut out = Vec::new();
355    let mut chars = template.chars().peekable();
356    while let Some(c) = chars.next() {
357        if c != '{' {
358            continue;
359        }
360        if chars.peek() == Some(&'{') {
361            chars.next();
362            continue;
363        }
364        let name: String =
365            std::iter::from_fn(|| chars.next_if(|c| *c == '_' || c.is_alphanumeric())).collect();
366        if !name.is_empty() && !out.contains(&name) {
367            out.push(name);
368        }
369    }
370    out
371}

373/// `#[weaveffi::error]` (or `#[weaveffi::error(no_display)]`): an error
374/// domain.
375pub(crate) fn error(attr: TokenStream, mut item: syn::ItemEnum) -> syn::Result<TokenStream> {
376    let no_display = match attr.to_string().as_str() {
377        "" => false,
378        "no_display" => true,
379        _ => {
380            return Err(syn::Error::new(
381                attr.span(),
382                "weaveffi: `#[weaveffi::error]` takes nothing or `no_display`",
383            ))
384        }
385    };
386    no_generics(&item.generics)?;
387    let prefix = prefix()?;
388    let tag = tag();
389    let name = item.ident.to_string();
390    let ty = item.ident.clone();
391    let mut codes = Vec::new();
392    for v in &item.variants {
393        no_member_cfg(&v.attrs, "an error code")?;
394        let Some((_, expr)) = &v.discriminant else {
395            return Err(syn::Error::new(
396                v.ident.span(),
397                "weaveffi: every error code needs an explicit positive value (`NotFound = 1`)",
398            ));
399        };
400        let value = int_literal(expr)?;
401        if value <= 0 {
402            return Err(syn::Error::new(
403                expr.span(),
404                "weaveffi: an error code must be positive; 0 means success and negative codes \
405                 belong to the runtime",
406            ));
407        }
408        let fields: Vec<(syn::Ident, syn::Type, Vec<syn::Attribute>)> =
409            named_fields(&v.fields, "an error code's")?
410                .into_iter()
411                .map(|f| (f.ident.clone(), f.ty.clone(), f.attrs.to_vec()))
412                .collect();
413        let template = message_template(&v.attrs)?;
414        let default_message = docs(&v.attrs)
415            .and_then(|d| d.lines().next().map(str::to_string))
416            .unwrap_or_else(|| v.ident.to_string());
417        codes.push((v.clone(), value, fields, template, default_message));
418    }
419    // The helper attributes are this macro's; the enum is re-emitted
420    // without them.
421    for v in &mut item.variants {
422        v.attrs.retain(|a| !is_helper(a));
423    }
424
425    let code_arms = codes.iter().map(|(v, value, fields, ..)| {
426        let vid = &v.ident;
427        if fields.is_empty() && matches!(v.fields, syn::Fields::Unit) {
428            quote!(Self::#vid => #value,)
429        } else {
430            quote!(Self::#vid { .. } => #value,)
431        }
432    });
433    let payload_arms = codes.iter().filter(|(_, _, f, ..)| !f.is_empty()).map(|(v, _, fields, ..)| {
434        let vid = &v.ident;
435        let ids: Vec<_> = fields.iter().map(|f| &f.0).collect();
436        let locals = field_locals(ids.len());
437        let tys = fields.iter().map(|f| &f.1);
438        quote! {
439            Self::#vid { #(#ids: #locals),* } => __wv_err.set_payload(&::weaveffi::__rt::encode(|w| {
440                #(<#tys as ::weaveffi::__rt::FfiType<#tag>>::write(#locals, w);)*
441            })),
442        }
443    });
444    let read_arms = codes.iter().map(|(v, value, fields, ..)| {
445        let vid = &v.ident;
446        if matches!(v.fields, syn::Fields::Unit) {
447            return quote!(#value => Self::#vid,);
448        }
449        let reads = fields.iter().map(|(id, fty, _)| {
450            quote!(#id: unsafe { <#fty as ::weaveffi::__rt::FfiType<#tag>>::read(&mut __wv_r) }.ok()?)
451        });
452        quote!(#value => Self::#vid { #(#reads),* },)
453    });
454    let lit = name_lit(&name);
455    // The message is the domain's `Display`: spanned at the type, a missing
456    // one (with `no_display`) is reported there.
457    let to_string = quote_spanned!(ty.span()=> <#ty as ::std::string::ToString>::to_string);
458    let domain_impl = quote_spanned! {ty.span()=>
459        #[allow(unsafe_code, unused_unsafe, unreachable_patterns)]
460        impl ::weaveffi::__rt::ErrorType<#tag> for #ty
461    };
462    let mut impls = quote! {
463        #domain_impl {
464            const DOMAIN: ::weaveffi::__rt::Name = #lit;
465            fn into_ffi(self) -> ::weaveffi::__rt::FfiError {
466                let __wv_code = match &self {
467                    #(#code_arms)*
468                };
469                let mut __wv_err = ::weaveffi::__rt::FfiError::new(
470                    __wv_code,
471                    &#to_string(&self),
472                );
473                match &self {
474                    #(#payload_arms)*
475                    _ => {}
476                }
477                __wv_err
478            }
479            fn from_code(__wv_code: i32, __wv_payload: &[u8]) -> ::core::option::Option<Self> {
480                let mut __wv_r = ::weaveffi::__rt::payload_reader(__wv_payload);
481                let __wv_value = match __wv_code {
482                    #(#read_arms)*
483                    _ => return ::core::option::Option::None,
484                };
485                __wv_r.expect_end().ok()?;
486                ::core::option::Option::Some(__wv_value)
487            }
488        }
489    };
490    if !no_display {
491        let arms = codes.iter().map(|(v, _, fields, template, default)| {
492            let vid = &v.ident;
493            let ids: Vec<_> = fields.iter().map(|f| &f.0).collect();
494            let locals = field_locals(ids.len());
495            let pat = if matches!(v.fields, syn::Fields::Unit) {
496                quote!(Self::#vid)
497            } else {
498                quote!(Self::#vid { #(#ids: #locals),* })
499            };
500            let body = match template {
501                Some(t) => {
502                    let named = placeholders(&t.value());
503                    let args = ids
504                        .iter()
505                        .zip(&locals)
506                        .filter(|(id, _)| named.contains(&id.to_string()))
507                        .map(|(id, local)| quote!(, #id = #local));
508                    quote!(::std::write!(__wv_f, #t #(#args)*))
509                }
510                None => quote!(__wv_f.write_str(#default)),
511            };
512            quote! {
513                #[allow(unused_variables)]
514                #pat => #body,
515            }
516        });
517        let error_impl = quote_spanned! {ty.span()=>
518            impl ::std::error::Error for #ty {}
519        };
520        impls.extend(quote! {
521            impl ::std::fmt::Display for #ty {
522                fn fmt(&self, __wv_f: &mut ::std::fmt::Formatter<'_>) -> ::std::fmt::Result {
523                    match self {
524                        #(#arms)*
525                    }
526                }
527            }
528            #error_impl
529        });
530    }
531
532    let mut fb = FrameBuilder::new();
533    let def = ErrorDomain {
534        name: name.clone(),
535        doc: docs(&item.attrs),
536        codes: codes
537            .iter()
538            .map(|(v, value, fields, template, default)| ErrorCode {
539                name: v.ident.to_string(),
540                code: *value,
541                message: template
542                    .as_ref()
543                    .map_or_else(|| default.clone(), syn::LitStr::value),
544                doc: docs(&v.attrs),
545                fields: fields
546                    .iter()
547                    .map(|(id, fty, attrs)| Field {
548                        name: id.to_string(),
549                        ty: fb.ty(field_name(fty)),
550                        doc: docs(attrs),
551                    })
552                    .collect(),
553            })
554            .collect(),
555    };
556    let frame = fb.finish(&prefix, location(item.ident.span()), Item::Error(def));
557    let mut contracts = contract(&prefix, &name, Sig::text(format!("errors {name}")));
558    for (v, value, fields, ..) in &codes {
559        let code = v.ident.to_string();
560        let sig = Sig::code(
561            &code,
562            *value,
563            fields.iter().map(|f| field_name(&f.1)).collect(),
564        );
565        let one = contract(&prefix, &format!("{name}.{code}"), sig);
566        // Each contract function lives in its own block.
567        contracts.extend(scoped(&[], one));
568    }
569    let cfg = cfgs(&item.attrs);
570    let generated = scoped(&cfg, quote!(#impls #frame #contracts));
571    Ok(quote!(#item #generated))
572}

574/// `#[weaveffi::interface]`: a reference-counted object type.
575pub(crate) fn interface(attr: TokenStream, item: syn::ItemStruct) -> syn::Result<TokenStream> {
576    no_args(&attr, "interface")?;
577    no_generics(&item.generics)?;
578    let prefix = prefix()?;
579    let tag = tag();
580    let name = item.ident.to_string();
581    let ty = &item.ident;
582    let clone = format!("{prefix}_{name}_clone");
583    let destroy = format!("{prefix}_{name}_destroy");
584    let assert = quote_spanned! {ty.span()=>
585        const _: fn() = || {
586            fn shared<T: ::core::marker::Send + ::core::marker::Sync + 'static>() {}
587            shared::<#ty>();
588        };
589    };
590    let impls = quote! {
591        #assert
592        ::weaveffi::__weaveffi_interface!(#tag, #ty, #name);
593
594        #[doc(hidden)]
595        #[allow(unsafe_code, clippy::missing_safety_doc)]
596        #[unsafe(export_name = #clone)]
597        pub unsafe extern "C" fn __weaveffi_clone(ptr: *const #ty) -> *const #ty {
598            unsafe { ::weaveffi::__rt::object_clone(ptr) }.cast_const()
599        }
600
601        #[doc(hidden)]
602        #[allow(unsafe_code, clippy::missing_safety_doc)]
603        #[unsafe(export_name = #destroy)]
604        pub unsafe extern "C" fn __weaveffi_destroy(ptr: *mut #ty) {
605            unsafe { ::weaveffi::__rt::object_destroy(ptr) }
606        }
607    };
608    let fb = FrameBuilder::new();
609    let def = InterfaceDef {
610        name: name.clone(),
611        doc: docs(&item.attrs),
612        deprecated: deprecated(&item.attrs),
613        constructors: Vec::new(),
614        methods: Vec::new(),
615        statics: Vec::new(),
616    };
617    let frame = fb.finish(&prefix, location(item.ident.span()), Item::Interface(def));
618    let contract = contract(&prefix, &name, Sig::text(format!("interface {name}")));
619    let cfg = cfgs(&item.attrs);
620    let generated = scoped(&cfg, quote!(#impls #frame #contract));
621    Ok(quote!(#item #generated))
622}

624/// The arguments of `#[weaveffi::custom(repr = R, lift = f, lower = g)]`.
625struct CustomArgs {
626    repr: syn::Type,
627    lift: syn::Expr,
628    lower: syn::Expr,
629}
630
631impl syn::parse::Parse for CustomArgs {
632    fn parse(input: syn::parse::ParseStream<'_>) -> syn::Result<Self> {
633        let (mut repr, mut lift, mut lower) = (None, None, None);
634        let args =
635            syn::punctuated::Punctuated::<syn::MetaNameValue, syn::Token![,]>::parse_terminated(
636                input,
637            )?;
638        for arg in args {
639            let key = arg
640                .path
641                .get_ident()
642                .map(ToString::to_string)
643                .unwrap_or_default();
644            match key.as_str() {
645                "repr" => {
646                    let tokens = quote!(#arg).to_string();
647                    let value = arg.value;
648                    repr = Some(syn::parse2(quote!(#value)).map_err(|_| {
649                        syn::Error::new(
650                            arg.path.span(),
651                            format!("weaveffi: `{tokens}` is not a type"),
652                        )
653                    })?);
654                }
655                "lift" => lift = Some(arg.value),
656                "lower" => lower = Some(arg.value),
657                _ => {
658                    return Err(syn::Error::new(
659                        arg.path.span(),
660                        "weaveffi: `#[weaveffi::custom]` takes `repr`, `lift`, and `lower`",
661                    ))
662                }
663            }
664        }
665        let missing = |what| {
666            syn::Error::new(
667                input.span(),
668                format!(
669                    "weaveffi: `#[weaveffi::custom]` needs `{what}`: \
670                     `#[weaveffi::custom(repr = R, lift = f, lower = g)]`"
671                ),
672            )
673        };
674        Ok(Self {
675            repr: repr.ok_or_else(|| missing("repr"))?,
676            lift: lift.ok_or_else(|| missing("lift"))?,
677            lower: lower.ok_or_else(|| missing("lower"))?,
678        })
679    }
680}

682/// Reject a custom alias of a type that already crosses the ABI (a
683/// scalar, `String`, a container, a borrow, ...): its trait impls would
684/// conflict with the built-in ones.
685fn builtin_alias(t: &syn::ItemType) -> syn::Result<()> {
686    const BUILTIN: &[&str] = &[
687        "i8", "i16", "i32", "i64", "u8", "u16", "u32", "u64", "f32", "f64", "bool", "char",
688        "usize", "isize", "String", "Vec", "Option", "HashMap", "BTreeMap", "Arc", "Box",
689    ];
690    let target = crate::shape::ungroup(&t.ty);
691    let builtin = match target {
692        syn::Type::Path(p) if p.qself.is_none() => p
693            .path
694            .segments
695            .last()
696            .is_some_and(|s| BUILTIN.contains(&s.ident.to_string().as_str())),
697        syn::Type::Reference(_)
698        | syn::Type::Slice(_)
699        | syn::Type::Array(_)
700        | syn::Type::Tuple(_)
701        | syn::Type::Ptr(_) => true,
702        _ => false,
703    };
704    if !builtin {
705        return Ok(());
706    }
707    let (name, target) = (&t.ident, quote!(#target).to_string());
708    Err(syn::Error::new(
709        t.ty.span(),
710        format!(
711            "weaveffi: `{target}` already crosses the ABI, so an alias of it can't be a custom \
712             type; wrap it in a newtype (`pub struct {name}(pub {target});`) and put \
713             `#[weaveffi::custom]` on that"
714        ),
715    ))
716}
717
718/// `#[weaveffi::custom(repr = R, lift = f, lower = g)]` on a type alias or
719/// a struct: the type crosses the ABI as `R`.
720pub(crate) fn custom(attr: TokenStream, item: syn::Item) -> syn::Result<TokenStream> {
721    let args: CustomArgs = syn::parse2(attr)?;
722    let (ty, attrs) = match &item {
723        syn::Item::Type(t) => {
724            builtin_alias(t)?;
725            (&t.ident, &t.attrs)
726        }
727        syn::Item::Struct(s) => (&s.ident, &s.attrs),
728        other => {
729            return Err(syn::Error::new(
730                other.span(),
731                "weaveffi: `#[weaveffi::custom]` goes on a type alias or a struct",
732            ))
733        }
734    };
735    let tag = tag();
736    let (repr, lift, lower) = (&args.repr, &args.lift, &args.lower);
737    let cfg = cfgs(attrs);
738    let generated = scoped(
739        &cfg,
740        quote!(::weaveffi::__weaveffi_custom!(#tag, #ty, #repr, #lift, #lower);),
741    );
742    Ok(quote!(#item #generated))
743}
