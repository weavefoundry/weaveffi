1//! IDL validation. This module owns the [`ValidationError`] catalog and the
2//! [`validate`] entry point, which checks an [`Api`] against the library's
3//! [`Identity`] and builds the one [`Model`] every generator consumes. The
4//! work is split across submodules: `rules` (name, type, and shape checks),
5//! `diagnostic` (miette span attachment), and `warnings` (advisory lints).
6//!
7//! Validation collects *every* rule violation before failing, so a document
8//! with several problems reports them all in one run rather than one per
9//! invocation.
10

27/// Every way an [`Api`] can fail validation.
28///
29/// [`validate`] collects every variant it encounters. Each variant carries
30/// the names needed to render an actionable diagnostic, and the
31/// `#[error]`/`#[diagnostic]` attributes supply the message and help text
32/// shown to the user. The variant serializes as an object whose `code` is the
33/// variant name, with the variant's fields alongside (the shape of
34/// `weaveffi validate --format json` failures).
35#[derive(Debug, thiserror::Error, serde::Serialize)]
36#[cfg_attr(feature = "idl", derive(Diagnostic))]

533/// Every validation failure found in one pass, each wrapped as a
534/// [`ValidationDiagnostic`] carrying an optional source span.
535///
536/// `Display` renders the first failure; miette renderers reach the others
537/// through [`Diagnostic::related`], and [`diagnostics`](Self::diagnostics)
538/// holds them all.
539#[derive(Debug)]
540pub struct ValidationDiagnostics {
541    /// The individual failures, in the order they were found. Never empty.
542    pub diagnostics: Vec<ValidationDiagnostic>,
543}
544
545impl ValidationDiagnostics {
546    /// The first failure, which every report is guaranteed to contain.
547    pub fn first(&self) -> &ValidationDiagnostic {
548        &self.diagnostics[0]
549    }
550}
551
552impl std::fmt::Display for ValidationDiagnostics {
553    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
554        std::fmt::Display::fmt(self.first(), f)
555    }
556}
557
558impl std::error::Error for ValidationDiagnostics {}
559
560#[cfg(feature = "idl")]
561impl Diagnostic for ValidationDiagnostics {
562    fn code<'a>(&'a self) -> Option<Box<dyn std::fmt::Display + 'a>> {
563        self.first().code()
564    }
565
566    fn help<'a>(&'a self) -> Option<Box<dyn std::fmt::Display + 'a>> {
567        self.first().help()
568    }
569
570    fn source_code(&self) -> Option<&dyn miette::SourceCode> {
571        self.first().source_code()
572    }
573
574    fn labels(&self) -> Option<Box<dyn Iterator<Item = miette::LabeledSpan> + '_>> {
575        self.first().labels()
576    }
577
578    fn related<'a>(&'a self) -> Option<Box<dyn Iterator<Item = &'a dyn Diagnostic> + 'a>> {
579        if self.diagnostics.len() <= 1 {
580            return None;
581        }
582        Some(Box::new(
583            self.diagnostics[1..].iter().map(|d| d as &dyn Diagnostic),
584        ))
585    }
586}

588/// The IDL source files a document came from, used to locate each
589/// diagnostic in the file that declares the offending item.
590#[derive(Debug, Clone, Default, PartialEq, Eq)]
591pub struct Sources {
592    /// Every file, as `(name, contents)`, the root document first.
593    pub files: Vec<(String, String)>,
594    /// The file (an index into [`files`](Self::files)) declaring each
595    /// top-level declaration, by name. A name missing here is in the first
596    /// file.
597    pub origin: BTreeMap<String, usize>,
598}
599
600impl Sources {
601    /// One file, holding every declaration.
602    #[must_use]
603    pub fn single(name: &str, contents: &str) -> Self {
604        Self {
605            files: vec![(name.to_string(), contents.to_string())],
606            origin: BTreeMap::new(),
607        }
608    }
609
610    /// The file declaring the top-level declaration `name`.
611    #[must_use]
612    pub fn file_of(&self, name: Option<&str>) -> Option<(&str, &str)> {
613        let index = name.and_then(|n| self.origin.get(n)).copied().unwrap_or(0);
614        self.files.get(index).map(|(n, c)| (n.as_str(), c.as_str()))
615    }
616}

618/// Validate an [`Api`] against the library's `identity`, reporting **every**
619/// rule violation found, and build its [`Model`].
620///
621/// This is the one checked way to obtain the [`Model`] every generator
622/// consumes, and it builds the model exactly once. `sources`, when the API
623/// came from IDL files, locates each diagnostic in its file.
624///
625/// # Errors
626///
627/// Returns [`ValidationDiagnostics`] carrying one [`ValidationDiagnostic`]
628/// per violation in the catalog above.
629pub fn validate(
630    api: &Api,
631    identity: &Identity,
632    sources: Option<&Sources>,
633) -> Result<Model, ValidationDiagnostics> {
634    validate_scoped(api, identity).map_err(|errors| ValidationDiagnostics {
635        diagnostics: errors
636            .into_iter()
637            .map(|(error, scope)| {
638                let source = sources.and_then(|s| s.file_of(scope.first().map(String::as_str)));
639                ValidationDiagnostic::new(error, &scope, source)
640            })
641            .collect(),
642    })
643}

645/// A rule violation paired with the declaration path that encloses it (the
646/// declaration, then member and parameter or field names), which locates
647/// the offending text.
648pub type Found = (ValidationError, Vec<String>);
649
650/// Validate `api` against `identity` exactly like [`validate`], returning
651/// each violation with the declaration path that encloses it instead of a
652/// located diagnostic.
653///
654/// # Errors
655///
656/// Returns every rule violation found, each paired with its declaration
657/// path.
658pub fn validate_scoped(api: &Api, identity: &Identity) -> Result<Model, Vec<Found>> {
659    if api.version != CURRENT_SCHEMA_VERSION {
660        // A wrong-schema document is checked no further: the rules below
661        // assume the current schema's shape.
662        return Err(vec![(
663            ValidationError::UnsupportedSchemaVersion {
664                version: api.version.clone(),
665                supported: CURRENT_SCHEMA_VERSION.to_string(),
666            },
667            vec![],
668        )]);
669    }
670    let types = crate::model::index(api);
671    let mut found = Vec::new();
672    rules::check(api, &types, &mut found);
673    if !found.is_empty() {
674        // The model's lowering assumes every rule holds, so the symbol
675        // table is only built for an otherwise valid document.
676        return Err(found);
677    }
678    let model = match crate::model::build(api, identity.clone(), types) {
679        Ok(model) => model,
680        Err(error) => return Err(vec![(error, vec![])]),
681    };
682    check_symbol_collisions(&model, &mut found);
683    if found.is_empty() {
684        Ok(model)
685    } else {
686        Err(found)
687    }
688}

690/// Reject any C identifier the model claims twice, or that falls in a
691/// family the runtime and the C headers reserve.
692fn check_symbol_collisions(model: &Model, found: &mut Vec<Found>) {
693    let prefix = model.prefix();
694    let mut seen: BTreeMap<String, SymbolOwner> = BTreeMap::new();
695    for symbol in model.c_symbols() {
696        if symbol.owner != SymbolOwner::Runtime {
697            let family = symbol
698                .name
699                .strip_prefix(&format!("{prefix}_"))
700                .and_then(|rest| {
701                    RESERVED_SYMBOL_FAMILIES
702                        .iter()
703                        .find(|f| rest.starts_with(*f))
704                });
705            if let Some(family) = family {
706                found.push((
707                    ValidationError::SymbolCollision {
708                        symbol: symbol.name.clone(),
709                        first: format!("the reserved names '{prefix}_{family}*'"),
710                        second: symbol.owner.to_string(),
711                    },
712                    vec![],
713                ));
714            }
715        }
716        match seen.get(&symbol.name) {
717            Some(first) => found.push((
718                ValidationError::SymbolCollision {
719                    first: first.to_string(),
720                    second: symbol.owner.to_string(),
721                    symbol: symbol.name,
722                },
723                vec![],
724            )),
725            None => {
726                seen.insert(symbol.name, symbol.owner);
727            }
728        }
729    }
