1//! Non-fatal, lint-style checks over a validated [`Api`].
2//!
3//! These are distinct from the hard validation errors in the parent
4//! [`crate::validate`] module: errors *reject* an API, whereas warnings flag
5//! ergonomic concerns (deep nesting, huge enums, undocumented declarations)
6//! that `weaveffi validate --warn` surfaces and the user may ignore.
7
8use crate::ir::{Api, Function, TypeRef};
9
10/// Nesting deeper than this (`[[[[i32]]]]`) is flagged.
11const MAX_NESTING: usize = 3;
12
13/// Enums with more variants than this are flagged.
14const MAX_ENUM_VARIANTS: usize = 100;
15
16/// A non-fatal advisory emitted by [`collect_warnings`].
17#[derive(Debug, Clone, PartialEq, Eq, serde::Serialize)]
18#[serde(tag = "code")]
19pub enum ValidationWarning {
20    /// An enum has more than 100 variants.
21    LargeEnumVariantCount {
22        /// The enum.
23        enum_name: String,
24        /// How many variants it declares.
25        count: usize,
26    },
27    /// A type is nested more than 3 levels deep.
28    DeepNesting {
29        /// Where the type appears (`Store.get.key`, `Entry.tags`).
30        location: String,
31        /// The nesting depth.
32        depth: usize,
33    },
34    /// A top-level declaration has no documentation, so its bindings have
35    /// no doc comment.
36    Undocumented {
37        /// The declaration.
38        name: String,
39    },
40}
41
42impl std::fmt::Display for ValidationWarning {
43    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
44        match self {
45            Self::LargeEnumVariantCount { enum_name, count } => write!(
46                f,
47                "enum '{enum_name}' has {count} variants (more than {MAX_ENUM_VARIANTS})"
48            ),
49            Self::DeepNesting { location, depth } => write!(
50                f,
51                "deep type nesting at {location} (depth {depth}, at most {MAX_NESTING} \
52                 recommended)"
53            ),
54            Self::Undocumented { name } => write!(f, "'{name}' has no documentation"),
55        }
56    }
57}
58
59impl ValidationWarning {
60    /// The warning's code, the variant name (`--format json` reports it).
61    #[must_use]
62    pub fn code(&self) -> &'static str {
63        match self {
64            Self::LargeEnumVariantCount { .. } => "LargeEnumVariantCount",
65            Self::DeepNesting { .. } => "DeepNesting",
66            Self::Undocumented { .. } => "Undocumented",
67        }
68    }
69}
70
71/// Collect every advisory warning for `api`, which has passed validation.
72#[must_use]
73pub fn collect_warnings(api: &Api) -> Vec<ValidationWarning> {
74    let mut warnings = Vec::new();
75    for e in &api.enums {
76        if e.variants.len() > MAX_ENUM_VARIANTS {
77            warnings.push(ValidationWarning::LargeEnumVariantCount {
78                enum_name: e.name.clone(),
79                count: e.variants.len(),
80            });
81        }
82    }
83    let mut deep = |ty: &TypeRef, location: String| {
84        let depth = nesting_depth(ty);
85        if depth > MAX_NESTING {
86            warnings.push(ValidationWarning::DeepNesting { location, depth });
87        }
88    };
89    for r in &api.records {
90        for field in &r.fields {
91            deep(&field.ty, format!("{}.{}", r.name, field.name));
92        }
93    }
94    let mut callable = |owner: Option<&str>, f: &Function| {
95        let path = owner.map_or_else(|| f.name.clone(), |o| format!("{o}.{}", f.name));
96        for p in &f.params {
97            deep(&p.ty, format!("{path}.{}", p.name));
98        }
99        if let Some(ret) = &f.returns {
100            deep(ret, format!("{path} (return)"));
101        }
102    };
103    for f in &api.functions {
104        callable(None, f);
105    }
106    for i in &api.interfaces {
107        for f in i.constructors.iter().chain(&i.methods).chain(&i.statics) {
108            callable(Some(&i.name), f);
109        }
110    }
111    let docs = api
112        .records
113        .iter()
114        .map(|d| (&d.name, &d.doc))
115        .chain(api.enums.iter().map(|d| (&d.name, &d.doc)))
116        .chain(api.errors.iter().map(|d| (&d.name, &d.doc)))
117        .chain(api.interfaces.iter().map(|d| (&d.name, &d.doc)))
118        .chain(api.callbacks.iter().map(|d| (&d.name, &d.doc)))
119        .chain(api.functions.iter().map(|d| (&d.name, &d.doc)));
120    for (name, doc) in docs {
121        if doc.is_none() {
122            warnings.push(ValidationWarning::Undocumented { name: name.clone() });
123        }
124    }
125    warnings
126}
127
128fn nesting_depth(ty: &TypeRef) -> usize {
129    match ty {
130        TypeRef::Optional(inner) | TypeRef::List(inner) | TypeRef::Iterator(inner) => {
131            1 + nesting_depth(inner)
132        }
133        TypeRef::Map(k, v) => 1 + nesting_depth(k).max(nesting_depth(v)),
134        _ => 0,
135    }
136}

138#[cfg(test)]
139mod tests {
140    use super::*;
141
142    #[test]
143    fn deep_nesting_large_enums_and_missing_docs_warn() {
144        let api: Api = serde_json::from_str(
145            r#"{"version":"0.13.0",
146                "records":[{"name":"R","doc":"r","fields":[{"name":"x","type":"[[[[i32]]]]"}]}],
147                "functions":[{"name":"f"}]}"#,
148        )
149        .unwrap();
150        let warnings = collect_warnings(&api);
151        assert_eq!(
152            warnings,
153            [
154                ValidationWarning::DeepNesting {
155                    location: "R.x".into(),
156                    depth: 4
157                },
158                ValidationWarning::Undocumented { name: "f".into() },
159            ]
160        );
161        assert_eq!(warnings[1].code(), "Undocumented");
162    }
