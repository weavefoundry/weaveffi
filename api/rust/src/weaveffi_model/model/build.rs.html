1//! Building the [`Model`]: index every type declaration, resolve each written
2//! [`TypeRef`] to its positional type, and lower every declaration to its C
3//! symbols and ABI signatures in one walk.
4//!
5//! The build runs only on a document every validation rule accepted, so each
6//! step that can fail (an undeclared name, a callback interface or an
7//! iterator in a value position) is one a rule already reported. The steps
8//! still return the matching [`ValidationError`] instead of panicking, so a
9//! missing rule surfaces as a diagnostic, never as a crash.
10

28/// Index every type and error-domain declaration in `api`. The first
29/// declaration of a name wins; validation reports any later one.
30pub(crate) fn index(api: &Api) -> TypeIndex {
31    let mut types = TypeIndex::default();
32    for (i, r) in api.records.iter().enumerate() {
33        types.declare(&r.name, TypeKind::Record, i);
34    }
35    for (i, e) in api.enums.iter().enumerate() {
36        let kind = if e.is_rich() {
37            TypeKind::RichEnum
38        } else {
39            TypeKind::Enum
40        };
41        types.declare(&e.name, kind, i);
42    }
43    for (i, d) in api.errors.iter().enumerate() {
44        types.declare(&d.name, TypeKind::ErrorDomain, i);
45    }
46    for (i, iface) in api.interfaces.iter().enumerate() {
47        types.declare(&iface.name, TypeKind::Interface, i);
48    }
49    for (i, c) in api.callbacks.iter().enumerate() {
50        types.declare(&c.name, TypeKind::CallbackInterface, i);
51    }
52    types
53}

55/// Build the model of `api` under `identity` from an index built by
56/// [`index`].
57///
58/// # Errors
59///
60/// Returns the violation a validation rule should already have reported:
61/// an undeclared name, or a callback interface, error domain, or iterator
62/// where a value type belongs.
63pub(crate) fn build(
64    api: &Api,
65    identity: Identity,
66    types: TypeIndex,
67) -> Result<Model, ValidationError> {
68    let b = Builder {
69        types: &types,
70        prefix: &identity.prefix,
71    };
72    let errors = api
73        .errors
74        .iter()
75        .map(|d| b.error_domain(d))
76        .collect::<Result<_, _>>()?;
77    let enums = api
78        .enums
79        .iter()
80        .map(|e| b.enum_def(e))
81        .collect::<Result<_, _>>()?;
82    let records = api
83        .records
84        .iter()
85        .map(|r| {
86            Ok(RecordBinding {
87                name: r.name.clone(),
88                doc: r.doc.clone(),
89                deprecated: r.deprecated.clone(),
90                c_tag: b.tag(&r.name),
91                fields: b.fields(&r.fields)?,
92            })
93        })
94        .collect::<Result<_, ValidationError>>()?;
95    let interfaces = api
96        .interfaces
97        .iter()
98        .map(|i| b.interface(i))
99        .collect::<Result<_, _>>()?;
100    let callbacks = api
101        .callbacks
102        .iter()
103        .map(|c| b.callback(c))
104        .collect::<Result<_, _>>()?;
105    let functions = api
106        .functions
107        .iter()
108        .map(|f| b.callable(f, CallableKind::Function, None))
109        .collect::<Result<_, _>>()?;
110    Ok(Model {
111        version: api.version.clone(),
112        doc: api.doc.clone(),
113        errors,
114        enums,
115        records,
116        interfaces,
117        callbacks,
118        functions,
119        identity,
120        types,
121    })
122}

124/// The error strategy a written `throws` declares.
125fn error_strategy(throws: Option<&Throws>) -> ErrorStrategy {
126    match throws {
127        None => ErrorStrategy::Trap,
128        Some(Throws::Domain(name)) => ErrorStrategy::Domain(name.clone()),
129        Some(Throws::Any) => ErrorStrategy::Untyped,
130    }
131}

133/// The per-build context: the type index and the symbol prefix.
134struct Builder<'a> {
135    types: &'a TypeIndex,
136    prefix: &'a str,
137}
138
139impl Builder<'_> {
140    /// `{prefix}_{name}`.
141    fn tag(&self, name: &str) -> String {
142        format!("{}_{name}", self.prefix)
143    }
144
145    /// Resolve a written type in a value position.
146    fn value(&self, ty: &TypeRef) -> Result<Ty, ValidationError> {
147        Ok(match ty {
148            TypeRef::Prim(p) => Ty::Prim(*p),
149            TypeRef::Named(name) => {
150                let name = name.clone();
151                match self.types.kind(&name) {
152                    Some(TypeKind::Record) => Ty::Record(name),
153                    Some(TypeKind::Enum) => Ty::Enum(name),
154                    Some(TypeKind::RichEnum) => Ty::RichEnum(name),
155                    Some(TypeKind::Interface) => Ty::Interface(name),
156                    Some(TypeKind::CallbackInterface) => {
157                        return Err(ValidationError::CallbackInterfaceInInvalidPosition {
158                            name,
159                            location: "a value position".to_string(),
160                        })
161                    }
162                    Some(TypeKind::ErrorDomain) => {
163                        return Err(ValidationError::ErrorDomainAsType { name })
164                    }
165                    None => return Err(ValidationError::UnknownTypeRef { name }),
166                }
167            }
168            TypeRef::Optional(inner) => Ty::Optional(Box::new(self.value(inner)?)),
169            TypeRef::List(inner) => Ty::List(Box::new(self.value(inner)?)),
170            TypeRef::Map(k, v) => Ty::Map(Box::new(self.value(k)?), Box::new(self.value(v)?)),
171            TypeRef::Iterator(_) => {
172                return Err(ValidationError::IteratorInInvalidPosition {
173                    location: "a value position".to_string(),
174                })
175            }
176        })
177    }
178
179    /// Resolve a written parameter type: a bare or optional callback
180    /// interface, or a value.
181    fn param_ty(&self, ty: &TypeRef) -> Result<ParamTy, ValidationError> {
182        let (top, nullable) = match ty {
183            TypeRef::Optional(inner) => (inner.as_ref(), true),
184            other => (other, false),
185        };
186        if let TypeRef::Named(name) = top {
187            if self.types.kind(name) == Some(TypeKind::CallbackInterface) {
188                return Ok(ParamTy::Callback {
189                    name: name.clone(),
190                    nullable,
191                });
192            }
193        }
194        Ok(ParamTy::Value(self.value(ty)?))
195    }
196
197    /// Resolve a written return type: an iterator, or a value.
198    fn ret_ty(&self, ty: &TypeRef) -> Result<RetTy, ValidationError> {
199        Ok(match ty {
200            TypeRef::Iterator(elem) => RetTy::Iterator(self.value(elem)?),
201            other => RetTy::Value(self.value(other)?),
202        })
203    }
204
205    fn error_domain(&self, domain: &ErrorDomain) -> Result<ErrorBinding, ValidationError> {
206        let tag = self.tag(&domain.name);
207        let codes = domain
208            .codes
209            .iter()
210            .map(|c| {
211                Ok(ErrorCodeBinding {
212                    name: c.name.clone(),
213                    value: c.code,
214                    message: c.message.clone(),
215                    doc: c.doc.clone(),
216                    c_const: format!("{tag}_{}", c.name),
217                    fields: self.fields(&c.fields)?,
218                })
219            })
220            .collect::<Result<_, ValidationError>>()?;
221        Ok(ErrorBinding {
222            name: domain.name.clone(),
223            type_name: crate::errors::type_name(&domain.name, "Error"),
224            doc: domain.doc.clone(),
225            c_tag: tag,
226            codes,
227        })
228    }
229
230    fn fields(&self, fields: &[Field]) -> Result<Vec<FieldBinding>, ValidationError> {
231        fields
232            .iter()
233            .map(|f| {
234                Ok(FieldBinding {
235                    name: f.name.clone(),
236                    doc: f.doc.clone(),
237                    ty: self.value(&f.ty)?,
238                })
239            })
240            .collect()
241    }
242
243    fn enum_def(&self, e: &EnumDef) -> Result<EnumBinding, ValidationError> {
244        let tag = self.tag(&e.name);
245        let variants = e
246            .variants
247            .iter()
248            .zip(e.values())
249            .map(|(v, value)| {
250                Ok(EnumVariantBinding {
251                    name: v.name.clone(),
252                    value,
253                    doc: v.doc.clone(),
254                    c_const: format!("{tag}_{}", v.name),
255                    fields: self.fields(&v.fields)?,
256                })
257            })
258            .collect::<Result<_, ValidationError>>()?;
259        Ok(EnumBinding {
260            name: e.name.clone(),
261            doc: e.doc.clone(),
262            deprecated: e.deprecated.clone(),
263            c_tag: tag,
264            variants,
265        })
266    }
267
268    fn callback(&self, c: &CallbackDef) -> Result<CallbackBinding, ValidationError> {
269        let tag = self.tag(&c.name);
270        let methods = c
271            .methods
272            .iter()
273            .enumerate()
274            .map(|(slot, m)| self.callback_method(slot, m))
275            .collect::<Result<_, _>>()?;
276        Ok(CallbackBinding {
277            name: c.name.clone(),
278            doc: c.doc.clone(),
279            deprecated: c.deprecated.clone(),
280            vtable_tag: format!("{tag}_vtable"),
281            c_tag: tag,
282            methods,
283        })
284    }
285
286    /// Lower one callback method to its vtable entry: `ctx`, the
287    /// parameters, then `out_err`, returning the value.
288    fn callback_method(
289        &self,
290        slot: usize,
291        m: &Function,
292    ) -> Result<CallbackMethodBinding, ValidationError> {
293        let params = m
294            .params
295            .iter()
296            .map(|p| {
297                Ok(FieldBinding {
298                    name: p.name.clone(),
299                    doc: p.doc.clone(),
300                    ty: self.value(&p.ty)?,
301                })
302            })
303            .collect::<Result<Vec<_>, ValidationError>>()?;
304        let ret = m.returns.as_ref().map(|r| self.value(r)).transpose()?;
305        let mut slots = vec![ctx_param()];
306        slots.extend(
307            params
308                .iter()
309                .map(|p| AbiParam::new(&p.name, CType::of(&p.ty, Position::Transfer))),
310        );
311        slots.push(error_out_param());
312        Ok(CallbackMethodBinding {
313            name: m.name.clone(),
314            doc: m.doc.clone(),
315            deprecated: m.deprecated.clone(),
316            slot,
317            abi: AbiFn {
318                symbol: m.name.clone(),
319                params: slots,
320                ret: ret
321                    .as_ref()
322                    .map_or(CType::Void, |t| CType::of(t, Position::Transfer)),
323            },
324            params,
325            ret,
326            error: error_strategy(m.throws.as_ref()),
327        })
328    }
329
330    /// Lower an interface: constructors return the interface, methods gain
331    /// the `self` slot, and every member symbol hangs off its C tag.
332    fn interface(&self, iface: &InterfaceDef) -> Result<InterfaceBinding, ValidationError> {
333        let tag = self.tag(&iface.name);
334        let owner = Some(iface.name.as_str());
335        let members = |list: &[Function], kind| {
336            list.iter()
337                .map(|f| self.callable(f, kind, owner))
338                .collect::<Result<Vec<_>, _>>()
339        };
340        Ok(InterfaceBinding {
341            name: iface.name.clone(),
342            doc: iface.doc.clone(),
343            deprecated: iface.deprecated.clone(),
344            constructors: members(&iface.constructors, CallableKind::Constructor)?,
345            methods: members(&iface.methods, CallableKind::Method)?,
346            statics: members(&iface.statics, CallableKind::Static)?,
347            clone_symbol: format!("{tag}_clone"),
348            destroy_symbol: format!("{tag}_destroy"),
349            c_tag: tag,
350        })
351    }

353    /// Lower one callable (free function or interface member).
354    fn callable(
355        &self,
356        f: &Function,
357        kind: CallableKind,
358        owner: Option<&str>,
359    ) -> Result<FnBinding, ValidationError> {
360        let prefix = self.prefix;
361        // The symbol without its prefix: `name`, or `Owner_name`.
362        let core = match owner {
363            Some(o) => format!("{o}_{}", f.name),
364            None => f.name.clone(),
365        };
366        let path = match owner {
367            Some(o) => format!("{o}.{}", f.name),
368            None => f.name.clone(),
369        };
370        let symbol = format!("{prefix}_{core}");
371        let params = f
372            .params
373            .iter()
374            .map(|p| {
375                Ok(ParamBinding {
376                    name: p.name.clone(),
377                    ty: self.param_ty(&p.ty)?,
378                    doc: p.doc.clone(),
379                })
380            })
381            .collect::<Result<Vec<_>, ValidationError>>()?;
382        let ret = match (kind, owner) {
383            (CallableKind::Constructor, Some(o)) => {
384                Some(RetTy::Value(Ty::Interface(o.to_string())))
385            }
386            _ => f.returns.as_ref().map(|r| self.ret_ty(r)).transpose()?,
387        };
388        let mut slots = Vec::new();
389        if kind == CallableKind::Method {
390            let o = owner.expect("a method has an owner");
391            slots.push(AbiParam::new(
392                "self",
393                CType::const_ptr(CType::Object(o.to_string())),
394            ));
395        }
396        for p in &params {
397            slots.push(AbiParam::new(
398                &p.name,
399                match &p.ty {
400                    ParamTy::Value(ty) => CType::of(ty, Position::Param),
401                    ParamTy::Callback { name, .. } => CType::Callback(name.clone()),
402                },
403            ));
404        }
405
406        let (abi, shape) = if f.r#async {
407            let value = match &ret {
408                None => None,
409                Some(RetTy::Value(ty)) => Some(ty),
410                Some(RetTy::Iterator(_)) => {
411                    return Err(ValidationError::AsyncIteratorReturn { function: path })
412                }
413            };
414            if f.cancellable {
415                slots.push(cancel_token_param());
416            }
417            let callback_core = format!("{core}_callback");
418            slots.push(AbiParam::new(
419                "callback",
420                CType::Named(callback_core.clone()),
421            ));
422            slots.push(context_param());
423            let mut callback_params = vec![
424                context_param(),
425                AbiParam::new("err", CType::ptr(CType::Error)),
426            ];
427            if let Some(ty) = value {
428                callback_params.push(AbiParam::new("result", CType::of(ty, Position::Transfer)));
429            }
430            let callback_type = format!("{prefix}_{callback_core}");
431            let shape = Shape::Async(AsyncBinding {
432                callback: AbiFn {
433                    symbol: callback_type.clone(),
434                    params: callback_params,
435                    ret: CType::Void,
436                },
437                callback_type,
438                cancellable: f.cancellable,
439            });
440            let launch = AbiFn {
441                symbol,
442                params: slots,
443                ret: CType::Void,
444            };
445            (launch, shape)
446        } else if let Some(RetTy::Iterator(elem)) = &ret {
447            let iter_core = match owner {
448                Some(o) => format!("{o}_{}Iterator", f.name.to_upper_camel_case()),
449                None => format!("{}Iterator", f.name.to_upper_camel_case()),
450            };
451            let iter_tag = format!("{prefix}_{iter_core}");
452            let handle = CType::ptr(CType::Named(iter_core));
453            slots.push(error_out_param());
454            let launch = AbiFn {
455                symbol,
456                params: slots,
457                ret: handle.clone(),
458            };
459            let iterator = IteratorBinding {
460                elem: elem.clone(),
461                next: AbiFn {
462                    symbol: format!("{iter_tag}_next"),
463                    params: vec![
464                        AbiParam::new("iter", handle),
465                        AbiParam::new("out_item", CType::ptr(CType::of(elem, Position::Transfer))),
466                        error_out_param(),
467                    ],
468                    ret: CType::Scalar(crate::ty::Prim::I32),
469                },
470                destroy_symbol: format!("{iter_tag}_destroy"),
471                iter_tag,
472            };
473            (launch, Shape::Iterator(iterator))
474        } else {
475            slots.push(error_out_param());
476            let c_ret = ret
477                .as_ref()
478                .and_then(RetTy::value)
479                .map_or(CType::Void, |t| CType::of(t, Position::Transfer));
480            let call = AbiFn {
481                symbol,
482                params: slots,
483                ret: c_ret,
484            };
485            (call, Shape::Sync)
486        };
487
488        Ok(FnBinding {
489            name: f.name.clone(),
490            doc: f.doc.clone(),
491            deprecated: f.deprecated.clone(),
492            kind,
493            owner: owner.map(str::to_string),
494            path,
495            params,
496            ret,
497            error: error_strategy(f.throws.as_ref()),
498            abi,
499            shape,
500        })
501    }
