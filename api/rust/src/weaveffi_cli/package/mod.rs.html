1//! The packaging layer: the artifacts a target assembles from a set of
2//! per-platform builds, and the driver that writes them to the dist
3//! directory.
4//!
5//! `weaveffi generate` emits binding *source* that a consumer must compile or
6//! point at a native library themselves. `weaveffi package` produces the
7//! next artifact up: an installable package for each ecosystem (a wheel, an
8//! npm tarball, a gem, a SwiftPM `XCFramework` archive, …) with the prebuilt
9//! native library for each platform inside, so
10//! `pip install`, `npm install`, and `gem install` work with no local
11//! toolchain.
12//!
13//! A target opts in by overriding [`Target::package`](crate::targets::Target::package),
14//! returning [`Artifact`]s: a directory tree, an archive, or a single file,
15//! each a list of [`PackagedFile`]s. Packaging returns values and leaves the
16//! writing to `weaveffi package`, so package layouts are testable exactly
17//! like generated source.
18

31/// The contents of one [`PackagedFile`].
32#[derive(Debug, Clone, PartialEq, Eq)]
33pub enum FileContent {
34    /// Rendered text (a manifest, loader, README, or binding source file).
35    Text(String),
36    /// Raw bytes rendered in memory.
37    Bytes(Vec<u8>),
38    /// A file copied byte-for-byte from this path, such as a prebuilt native
39    /// library. Keeping binaries out of memory until the artifact is written
40    /// means rendering never holds a multi-megabyte library as a value.
41    Copy(Utf8PathBuf),
42}
43
44impl FileContent {
45    /// The file's bytes, reading a [`Copy`](Self::Copy) source from disk.
46    ///
47    /// # Errors
48    ///
49    /// Returns an error when a copied file can't be read.
50    pub fn bytes(&self) -> Result<Cow<'_, [u8]>> {
51        Ok(match self {
52            Self::Text(text) => Cow::Borrowed(text.as_bytes()),
53            Self::Bytes(bytes) => Cow::Borrowed(bytes),
54            Self::Copy(source) => Cow::Owned(read_file(source)?),
55        })
56    }
57}

59/// One file in an [`Artifact`]: its path inside the artifact and its
60/// contents.
61#[derive(Debug, Clone, PartialEq, Eq)]
62pub struct PackagedFile {
63    /// Path relative to the artifact's root, with `/` separators.
64    pub path: Utf8PathBuf,
65    /// What the file contains.
66    pub content: FileContent,
67}
68
69impl PackagedFile {
70    /// A file whose contents are rendered text.
71    pub fn text(path: impl Into<Utf8PathBuf>, contents: impl Into<String>) -> Self {
72        Self {
73            path: normalize_separators(path.into()),
74            content: FileContent::Text(contents.into()),
75        }
76    }
77
78    /// A file whose contents are bytes rendered in memory.
79    pub fn bytes(path: impl Into<Utf8PathBuf>, contents: Vec<u8>) -> Self {
80        Self {
81            path: normalize_separators(path.into()),
82            content: FileContent::Bytes(contents),
83        }
84    }
85
86    /// A file copied from `source` on disk (a prebuilt native library).
87    pub fn copy(path: impl Into<Utf8PathBuf>, source: impl Into<Utf8PathBuf>) -> Self {
88        Self {
89            path: normalize_separators(path.into()),
90            content: FileContent::Copy(source.into()),
91        }
92    }
93
94    /// True when this entry copies in a file from disk rather than holding
95    /// rendered contents.
96    pub fn is_binary(&self) -> bool {
97        matches!(self.content, FileContent::Copy(_))
98    }
99}

101/// Normalize a package path to use `/` separators on every host, so package
102/// layouts are identical (and testable) across platforms.
103fn normalize_separators(path: Utf8PathBuf) -> Utf8PathBuf {
104    if path.as_str().contains('\\') {
105        Utf8PathBuf::from(path.as_str().replace('\\', "/"))
106    } else {
107        path
108    }
109}

111/// How an [`Artifact`]'s files are materialized.
112#[derive(Debug, Clone, PartialEq, Eq)]
113pub enum ArtifactKind {
114    /// A directory tree (a Gradle project, a Go module, a pub package).
115    Directory,
116    /// A gzip-compressed tarball whose entries all sit under `prefix/`
117    /// (`package/` for npm).
118    TarGz {
119        /// The directory every entry is placed under.
120        prefix: String,
121    },
122    /// A zip archive.
123    Zip,
124    /// A Python wheel: a zip whose `.dist-info` directory the writer adds.
125    Wheel(WheelMeta),
126    /// A RubyGems package: a tar of the gzipped specification and data.
127    Gem(GemSpec),
128    /// A single file: the contents of the artifact's one entry.
129    File,
130}
131
132/// One installable artifact a target produces.
133#[derive(Debug, Clone, PartialEq, Eq)]
134pub struct Artifact {
135    /// Where the artifact goes, relative to the dist directory: the
136    /// directory for [`ArtifactKind::Directory`], else the archive's file.
137    pub path: Utf8PathBuf,
138    /// How the files are written.
139    pub kind: ArtifactKind,
140    /// The artifact's files, with paths relative to its root.
141    pub files: Vec<PackagedFile>,
142}
143
144impl Artifact {
145    /// A directory artifact at `path`.
146    pub fn directory(path: impl Into<Utf8PathBuf>, files: Vec<PackagedFile>) -> Self {
147        Self {
148            path: normalize_separators(path.into()),
149            kind: ArtifactKind::Directory,
150            files,
151        }
152    }
153
154    /// A `.tar.gz` artifact at `path` with every entry under `prefix/`.
155    pub fn tar_gz(
156        path: impl Into<Utf8PathBuf>,
157        prefix: impl Into<String>,
158        files: Vec<PackagedFile>,
159    ) -> Self {
160        Self {
161            path: normalize_separators(path.into()),
162            kind: ArtifactKind::TarGz {
163                prefix: prefix.into(),
164            },
165            files,
166        }
167    }
168
169    /// A single file at `path` holding `contents`.
170    pub fn single_file(path: impl Into<Utf8PathBuf>, contents: Vec<u8>) -> Self {
171        let path = normalize_separators(path.into());
172        let name = path.file_name().unwrap_or_default().to_string();
173        Self {
174            path,
175            kind: ArtifactKind::File,
176            files: vec![PackagedFile::bytes(name, contents)],
177        }
178    }
179
180    /// The file at `path` inside this artifact, if any.
181    pub fn file(&self, path: &str) -> Option<&PackagedFile> {
182        self.files.iter().find(|f| f.path == path)
183    }
184}

186/// An artifact skipped because an external tool it needs is missing.
187#[derive(Debug, Clone, PartialEq, Eq)]
188pub(crate) struct Skipped {
189    /// The artifact (`the NuGet package`, `the linux-x64 library`).
190    pub(crate) what: String,
191    /// Why it was skipped.
192    pub(crate) reason: String,
193}
194
195/// The artifacts a `build` or `package` run skipped, collected for the
196/// summary at the end of the run.
197#[derive(Debug, Default)]
198pub(crate) struct Skips(Mutex<Vec<Skipped>>);
199
200impl Skips {
201    /// Record that `what` was skipped for `reason`, warning about it now.
202    pub(crate) fn skip(&self, what: impl Into<String>, reason: impl std::fmt::Display) {
203        let skipped = Skipped {
204            what: what.into(),
205            reason: reason.to_string(),
206        };
207        eprintln!("warning: skipping {}: {}", skipped.what, skipped.reason);
208        self.0
209            .lock()
210            .unwrap_or_else(std::sync::PoisonError::into_inner)
211            .push(skipped);
212    }
213
214    /// Everything skipped so far, in order.
215    pub(crate) fn list(&self) -> Vec<Skipped> {
216        self.0
217            .lock()
218            .unwrap_or_else(std::sync::PoisonError::into_inner)
219            .clone()
220    }
221}

223/// Everything a target's [`package`](crate::targets::Target::package) hook
224/// needs beyond the [`Model`](weaveffi_model::model::Model) and its own
225/// configuration.
226#[derive(Debug, Clone, Copy)]
227pub struct PackageContext<'a> {
228    /// The per-platform builds to bundle.
229    pub(crate) binaries: &'a BinarySet,
230    /// The macOS deployment target the libraries were built for (`"11.0"`),
231    /// which wheel tags and Swift platforms carry.
232    pub(crate) macos_deployment_target: &'a str,
233    /// Where skipped artifacts are recorded.
234    pub(crate) skips: Option<&'a Skips>,
235}
236
237impl<'a> PackageContext<'a> {
238    /// A context for `binaries` built for the default macOS deployment
239    /// target (11.0).
240    #[cfg(test)]
241    pub(crate) fn new(binaries: &'a BinarySet) -> Self {
242        Self {
243            binaries,
244            macos_deployment_target: "11.0",
245            skips: None,
246        }
247    }
248
249    /// Record that the artifact `what` was skipped because an external tool
250    /// it needs (named in `reason`) is missing. `weaveffi package` warns,
251    /// lists every skipped artifact at the end, and fails only with
252    /// `--strict`.
253    pub fn skip(&self, what: impl Into<String>, reason: impl std::fmt::Display) {
254        match self.skips {
255            Some(skips) => skips.skip(what, reason),
256            None => eprintln!("warning: skipping {}: {reason}", what.into()),
257        }
258    }
259}

261/// The producer library of every build `keep` accepts, laid out as
262/// `{dir}/<platform-id>/<library file>`, plus the import library MSVC links
263/// a Windows DLL through.
264pub(crate) fn per_platform_libraries(
265    binaries: &BinarySet,
266    dir: &str,
267    keep: impl Fn(Platform) -> bool,
268) -> Vec<PackagedFile> {
269    let mut files = Vec::new();
270    for nb in binaries.binaries.iter().filter(|nb| keep(nb.platform)) {
271        let platform_dir = format!("{dir}/{}", nb.platform.id());
272        files.push(PackagedFile::copy(
273            format!("{platform_dir}/{}", binaries.bundled_filename(nb.platform)),
274            nb.library.clone(),
275        ));
276        if let Some(import) = &nb.import_library {
277            files.push(PackagedFile::copy(
278                format!("{platform_dir}/{}.dll.lib", binaries.lib_name),
279                import.clone(),
280            ));
281        }
282    }
283    files
284}

286/// Write `artifact` under `dist`, returning the path written.
287///
288/// A directory artifact replaces any previous directory at its path, so
289/// files from an earlier run never linger; an archive is built in memory and
290/// written in one go.
291///
292/// # Errors
293///
294/// Returns an error when a directory can't be created or replaced, a file
295/// can't be read or written, or an archive can't be assembled.
296pub(crate) fn write_artifact(dist: &Utf8Path, artifact: &Artifact) -> Result<Utf8PathBuf> {
297    let path = dist.join(&artifact.path);
298    if let ArtifactKind::Directory = artifact.kind {
299        if path.exists() {
300            std::fs::remove_dir_all(path.as_std_path())
301                .into_diagnostic()
302                .wrap_err_with(|| format!("failed to replace {path}"))?;
303        }
304        for file in &artifact.files {
305            let dest = path.join(&file.path);
306            create_parent(&dest)?;
307            match &file.content {
308                FileContent::Copy(source) => {
309                    std::fs::copy(source.as_std_path(), dest.as_std_path())
310                        .into_diagnostic()
311                        .wrap_err_with(|| format!("failed to copy {source} to {dest}"))?;
312                }
313                content => std::fs::write(dest.as_std_path(), content.bytes()?)
314                    .into_diagnostic()
315                    .wrap_err_with(|| format!("failed to write {dest}"))?,
316            }
317        }
318        return Ok(path);
319    }
320    let bytes = match &artifact.kind {
321        ArtifactKind::Directory => unreachable!("handled above"),
322        ArtifactKind::TarGz { prefix } => {
323            let entries = archive::entries(&artifact.files, Some(prefix))?;
324            archive::gzip(&archive::tar(&entries)?)?
325        }
326        ArtifactKind::Zip => archive::zip(&archive::entries(&artifact.files, None)?)?,
327        ArtifactKind::Wheel(meta) => archive::wheel(meta, &artifact.files)?,
328        ArtifactKind::Gem(spec) => archive::gem(spec, &artifact.files)?,
329        ArtifactKind::File => match artifact.files.as_slice() {
330            [file] => file.content.bytes()?.into_owned(),
331            _ => {
332                return Err(miette::miette!(
333                    "the file artifact {path} must hold one entry"
334                ))
335            }
336        },
337    };
338    create_parent(&path)?;
339    std::fs::write(path.as_std_path(), bytes)
340        .into_diagnostic()
341        .wrap_err_with(|| format!("failed to write {path}"))?;
342    Ok(path)
343}
344
345fn create_parent(path: &Utf8Path) -> Result<()> {
346    if let Some(parent) = path.parent() {
347        std::fs::create_dir_all(parent.as_std_path())
348            .into_diagnostic()
349            .wrap_err_with(|| format!("failed to create {parent}"))?;
350    }
351    Ok(())
352}

358    #[test]
359    fn directories_are_replaced_and_binaries_copied() {
360        let dir = tempfile::tempdir().unwrap();
361        let root = Utf8Path::from_path(dir.path()).unwrap();
362        let src = root.join("src-lib.bin");
363        std::fs::write(&src, b"\x00native\x01").unwrap();
364        std::fs::create_dir_all(root.join("dist/pkg")).unwrap();
365        std::fs::write(root.join("dist/pkg/stale.txt"), "old").unwrap();
366
367        let artifact = Artifact::directory(
368            "pkg",
369            vec![
370                PackagedFile::text("manifest.json", "{\"name\":\"x\"}"),
371                PackagedFile::copy("native/lib.bin", src.clone()),
372            ],
373        );
374        let written = write_artifact(&root.join("dist"), &artifact).unwrap();
375        assert_eq!(written, root.join("dist/pkg"));
376        assert_eq!(
377            std::fs::read_to_string(written.join("manifest.json")).unwrap(),
378            "{\"name\":\"x\"}"
379        );
380        assert_eq!(
381            std::fs::read(written.join("native/lib.bin")).unwrap(),
382            b"\x00native\x01"
383        );
384        assert!(!written.join("stale.txt").exists());
385    }

387    #[test]
388    fn missing_copy_sources_are_errors() {
389        let dir = tempfile::tempdir().unwrap();
390        let root = Utf8Path::from_path(dir.path()).unwrap();
391        let artifact = Artifact::tar_gz(
392            "x.tar.gz",
393            "x",
394            vec![PackagedFile::copy("lib.bin", root.join("missing.bin"))],
395        );
396        let err = write_artifact(root, &artifact).unwrap_err();
397        assert!(format!("{err:#}").contains("missing.bin"), "{err:#}");
398    }

401    fn paths_are_forward_slashed() {
402        let text = PackagedFile::text(Utf8Path::new("dotnet").join("x.cs"), "x");
403        assert_eq!(text.path.as_str(), "dotnet/x.cs");
404        let artifact = Artifact::directory("a\\b", vec![]);
405        assert_eq!(artifact.path.as_str(), "a/b");
406    }
