1//! Locating the Android NDK and its per-API-level compilers.
2
3use camino::{Utf8Path, Utf8PathBuf};
4use miette::{bail, IntoDiagnostic, Result, WrapErr};
5
6use crate::platform::Platform;
7
8/// An Android NDK installation.
9#[derive(Debug, Clone, PartialEq, Eq)]
10pub struct Ndk {
11    /// The NDK's root directory (holding `toolchains/` and
12    /// `source.properties`).
13    pub root: Utf8PathBuf,
14}
15
16impl Ndk {
17    /// Find the NDK: `ANDROID_NDK_HOME`, then `ANDROID_NDK_ROOT`, then the
18    /// newest `ndk/<version>` (or `ndk-bundle`) of the Android SDK named by
19    /// `ANDROID_HOME` or `ANDROID_SDK_ROOT`, or installed at the Android
20    /// Studio default location.
21    ///
22    /// # Errors
23    ///
24    /// Returns an error explaining how to install or point at an NDK when
25    /// none is found.
26    pub fn locate() -> Result<Self> {
27        let var = |name: &str| std::env::var(name).ok().filter(|v| !v.is_empty());
28        for name in ["ANDROID_NDK_HOME", "ANDROID_NDK_ROOT"] {
29            if let Some(dir) = var(name) {
30                let root = Utf8PathBuf::from(dir);
31                if !root.join("toolchains").is_dir() {
32                    bail!("{name} is set to {root}, which isn't an Android NDK (no toolchains/)");
33                }
34                return Ok(Self { root });
35            }
36        }
37        let home = var("HOME")
38            .or_else(|| var("USERPROFILE"))
39            .map(Utf8PathBuf::from);
40        let mut sdks: Vec<Utf8PathBuf> = ["ANDROID_HOME", "ANDROID_SDK_ROOT"]
41            .into_iter()
42            .filter_map(|n| var(n).map(Utf8PathBuf::from))
43            .collect();
44        if let Some(home) = &home {
45            sdks.push(home.join("Library/Android/sdk"));
46            sdks.push(home.join("Android/Sdk"));
47        }
48        if let Some(local) = var("LOCALAPPDATA") {
49            sdks.push(Utf8PathBuf::from(local).join("Android/Sdk"));
50        }
51        for sdk in &sdks {
52            if let Some(root) = newest_version(&sdk.join("ndk")) {
53                return Ok(Self { root });
54            }
55            let bundle = sdk.join("ndk-bundle");
56            if bundle.join("toolchains").is_dir() {
57                return Ok(Self { root: bundle });
58            }
59        }
60        Err(miette::miette!(
61            "the Android NDK wasn't found: set ANDROID_NDK_HOME to an NDK directory, or install \
62             one with the Android SDK manager (`sdkmanager --install \"ndk;27.2.12479018\"`) so \
63             it lands in $ANDROID_HOME/ndk/<version>"
64        ))
65    }
66
67    /// The directory of the NDK's LLVM toolchain binaries for this host.
68    ///
69    /// # Errors
70    ///
71    /// Returns an error when the NDK has no prebuilt LLVM toolchain.
72    pub fn bin_dir(&self) -> Result<Utf8PathBuf> {
73        let prebuilt = self.root.join("toolchains/llvm/prebuilt");
74        let host = prebuilt
75            .read_dir_utf8()
76            .into_diagnostic()
77            .wrap_err_with(|| format!("the NDK at {} has no {prebuilt}", self.root))?
78            .filter_map(Result::ok)
79            .map(|e| e.path().to_path_buf())
80            .find(|p| p.join("bin").is_dir());
81        match host {
82            Some(host) => Ok(host.join("bin")),
83            None => Err(miette::miette!(
84                "the NDK at {} has no LLVM toolchain under {prebuilt}",
85                self.root
86            )),
87        }
88    }
89
90    /// The C compiler driver targeting `platform` at Android `api_level`
91    /// (`aarch64-linux-android21-clang`).
92    ///
93    /// # Errors
94    ///
95    /// Returns an error when the NDK has no compiler for that platform and
96    /// API level.
97    pub fn clang(&self, platform: Platform, api_level: u32) -> Result<Utf8PathBuf> {
98        let suffix = if cfg!(windows) { ".cmd" } else { "" };
99        let clang = self.bin_dir()?.join(format!(
100            "{}{api_level}-clang{suffix}",
101            platform.rust_target()
102        ));
103        if !clang.is_file() {
104            bail!(
105                "the NDK at {} has no compiler for {} at API level {api_level} ({clang}); \
106                 check `[build] android_api`",
107                self.root,
108                platform.id()
109            );
110        }
111        Ok(clang)
112    }
113
114    /// The NDK's `llvm-ar`.
115    ///
116    /// # Errors
117    ///
118    /// Returns an error when the NDK has no prebuilt LLVM toolchain.
119    pub fn ar(&self) -> Result<Utf8PathBuf> {
120        let suffix = if cfg!(windows) { ".exe" } else { "" };
121        Ok(self.bin_dir()?.join(format!("llvm-ar{suffix}")))
122    }
123}
124
125/// The newest NDK under an SDK's `ndk/` directory, comparing the numeric
126/// components of each version directory's name.
127fn newest_version(dir: &Utf8Path) -> Option<Utf8PathBuf> {
128    let key = |name: &str| -> Vec<u64> {
129        name.split('.')
130            .map(|part| part.parse().unwrap_or(0))
131            .collect()
132    };
133    dir.read_dir_utf8()
134        .ok()?
135        .filter_map(Result::ok)
136        .map(|e| e.path().to_path_buf())
137        .filter(|p| p.join("toolchains").is_dir())
138        .max_by_key(|p| key(p.file_name().unwrap_or("")))
139}

141#[cfg(test)]
142mod tests {
143    use super::*;
144
145    #[test]
146    fn the_newest_ndk_version_wins() {
147        let dir = tempfile::tempdir().unwrap();
148        let ndk = Utf8Path::from_path(dir.path()).unwrap().join("ndk");
149        for v in ["25.2.9519653", "27.1.12297006", "9.0.1", "notes"] {
150            let toolchains = ndk.join(v).join("toolchains");
151            if v != "notes" {
152                std::fs::create_dir_all(toolchains).unwrap();
153            } else {
154                std::fs::create_dir_all(ndk.join(v)).unwrap();
155            }
156        }
157        assert_eq!(newest_version(&ndk), Some(ndk.join("27.1.12297006")));
158        assert_eq!(newest_version(&ndk.join("missing")), None);
159    }
160
161    #[test]
162    fn compilers_are_named_by_triple_and_api_level() {
163        let dir = tempfile::tempdir().unwrap();
164        let root = Utf8Path::from_path(dir.path()).unwrap().to_path_buf();
165        let bin = root.join("toolchains/llvm/prebuilt/test-host/bin");
166        std::fs::create_dir_all(&bin).unwrap();
167        let suffix = if cfg!(windows) { ".cmd" } else { "" };
168        std::fs::write(
169            bin.join(format!("aarch64-linux-android24-clang{suffix}")),
170            "",
171        )
172        .unwrap();
173        let ndk = Ndk { root };
174        assert_eq!(
175            ndk.clang(Platform::AndroidArm64, 24).unwrap(),
176            bin.join(format!("aarch64-linux-android24-clang{suffix}"))
177        );
178        let err = ndk.clang(Platform::AndroidArm64, 19).unwrap_err();
179        assert!(err.to_string().contains("android_api"), "{err}");
180    }
