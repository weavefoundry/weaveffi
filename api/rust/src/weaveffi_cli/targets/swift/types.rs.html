13/// The Swift spelling of a user-chosen identifier in a lowerCamel position
14/// (parameters, fields, enum cases, wrapper names): camel-cased, then
15/// keyword-escaped through the shared rule, so a parameter named `in`
16/// becomes `in_` rather than emitting broken Swift.
17pub(crate) fn swift_ident(name: &str) -> String {
18    lang::escape_ident(&name.to_lower_camel_case(), lang::SWIFT_KEYWORDS)
19}
20
21/// Members every interface wrapper class declares: the stored `ptr`, the
22/// internal `clonePtr()`, and the `WvCodable` conformance's `wvRead` and
23/// `wvWrite`. (`init` and `deinit` are keywords, so [`swift_ident`] already
24/// escapes them.)
25const OBJECT_MEMBERS: &[&str] = &["clonePtr", "ptr", "wvRead", "wvWrite"];
26
27/// The Swift spelling of an interface member (a factory constructor,
28/// method, or static): [`swift_ident`], with a trailing `_` when it would
29/// redeclare a member the wrapper class declares (`clone_ptr` is
30/// `clonePtr_`).
31pub(crate) fn swift_member(name: &str) -> String {
32    lang::escape_member(&swift_ident(name), OBJECT_MEMBERS)
33}

35/// Escape a string for embedding inside a Swift double-quoted literal.
36pub(crate) fn swift_str(s: &str) -> String {
37    let mut out = String::with_capacity(s.len());
38    for c in s.chars() {
39        match c {
40            '\\' => out.push_str("\\\\"),
41            '"' => out.push_str("\\\""),
42            '\n' => out.push_str("\\n"),
43            '\r' => out.push_str("\\r"),
44            '\t' => out.push_str("\\t"),
45            c if c.is_control() => out.push_str(&format!("\\u{{{:x}}}", c as u32)),
46            c => out.push(c),
47        }
48    }
49    out
50}

52/// The Swift spelling of a scalar, string, or bytes type.
53pub(crate) fn prim_swift_type(p: Prim) -> &'static str {
54    match p {
55        Prim::I8 => "Int8",
56        Prim::I16 => "Int16",
57        Prim::I32 => "Int32",
58        Prim::I64 => "Int64",
59        Prim::U8 => "UInt8",
60        Prim::U16 => "UInt16",
61        Prim::U32 => "UInt32",
62        Prim::U64 => "UInt64",
63        Prim::F32 => "Float",
64        Prim::F64 => "Double",
65        Prim::Bool => "Bool",
66        Prim::String => "String",
67        Prim::Bytes => "Data",
68    }
69}

71/// Facts about the whole API that spellings consult: the generated
72/// package's names and every declared identifier (for doc text).
73pub(crate) struct SwiftCtx<'a> {
74    /// The model being rendered.
75    pub(crate) model: &'a Model,
76    /// The C symbol prefix, which names the run and optional-scalar
77    /// structs the C module declares.
78    pub(crate) prefix: &'a str,
79    /// The error type for failures outside any declared domain
80    /// (`KvstoreRuntimeError`).
81    pub(crate) runtime_error: String,
82    /// The namespace type of the library itself (`KvstoreLibrary`), which
83    /// carries `check()`.
84    pub(crate) library_type: String,
85    /// Every declared identifier, for rewriting doc text.
86    names: ApiNames,
87}
88
89impl<'a> SwiftCtx<'a> {
90    /// Collect the API-wide facts from the model.
91    pub(crate) fn new(model: &'a Model, swift_module: &str) -> SwiftCtx<'a> {
92        SwiftCtx {
93            model,
94            prefix: model.prefix(),
95            runtime_error: runtime_error_name(swift_module),
96            library_type: library_type_name(swift_module),
97            names: ApiNames::new(model),
98        }
99    }
100
101    /// The Swift type of a value type.
102    pub(crate) fn swift_type(&self, t: &Ty) -> String {
103        match t {
104            Ty::Prim(p) => prim_swift_type(*p).to_string(),
105            Ty::Enum(name) | Ty::Record(name) | Ty::RichEnum(name) | Ty::Interface(name) => {
106                name.clone()
107            }
108            Ty::Optional(inner) => format!("{}?", self.swift_type(inner)),
109            Ty::List(inner) => format!("[{}]", self.swift_type(inner)),
110            Ty::Map(k, v) => format!("[{}: {}]", self.swift_type(k), self.swift_type(v)),
111        }
112    }
113
114    /// The Swift type of a parameter. A callback interface is an existential
115    /// (`any Listener`, or `(any Listener)?`).
116    pub(crate) fn param_type(&self, t: &ParamTy) -> String {
117        match t {
118            ParamTy::Value(t) => self.swift_type(t),
119            ParamTy::Callback { name, nullable } => {
120                if *nullable {
121                    format!("(any {name})?")
122                } else {
123                    format!("any {name}")
124                }
125            }
126        }
127    }
128
129    /// The Swift type of a return: an iterator is the runtime's generic
130    /// `NativeSequence`.
131    pub(crate) fn ret_type(&self, t: &RetTy) -> String {
132        match t {
133            RetTy::Value(t) => self.swift_type(t),
134            RetTy::Iterator(elem) => format!("NativeSequence<{}>", self.swift_type(elem)),
135        }
136    }
137
138    /// The Swift spelling of the one C value a value crosses as, as the C
139    /// module imports it: the scalar (`Int32` for a C-style enum, whose C
140    /// type is an `int32_t` typedef), the `{p}_opt_{prim}` or `{p}_run`
141    /// struct, or a nullable `OpaquePointer` for an object.
142    pub(crate) fn c_value_type(&self, crossing: &Crossing) -> String {
143        match crossing {
144            Crossing::Scalar(p) => prim_swift_type(*p).to_string(),
145            Crossing::Enum(_) => "Int32".to_string(),
146            Crossing::OptScalar(p) => format!("{}_opt_{}", self.prefix, p.snake()),
147            Crossing::OptEnum(_) => format!("{}_opt_i32", self.prefix),
148            Crossing::Utf8 | Crossing::Bytes | Crossing::Slice(_) | Crossing::Buffer(_) => {
149                "WvRun".to_string()
150            }
151            Crossing::Object { .. } => "OpaquePointer?".to_string(),
152        }
153    }
154
155    /// The zero value of [`c_value_type`](Self::c_value_type): what a
156    /// callback method returns when it fails, and what an iterator's item
157    /// slot starts as.
158    pub(crate) fn c_zero(&self, crossing: &Crossing) -> String {
159        match crossing {
160            Crossing::Scalar(Prim::Bool) => "false".to_string(),
161            Crossing::Scalar(_) | Crossing::Enum(_) => "0".to_string(),
162            Crossing::Object { .. } => "nil".to_string(),
163            other => format!("{}()", self.c_value_type(other)),
164        }
165    }
166
167    /// The Swift spelling of an API identifier named in doc text, or `None`
168    /// to keep it as written.
169    fn spell(&self, ident: &str) -> Option<String> {
170        if ident.contains('.') {
171            return None;
172        }
173        match self.names.kind(ident)? {
174            IdentKind::Function
175            | IdentKind::Member
176            | IdentKind::CallbackMethod
177            | IdentKind::Param
178            | IdentKind::Field
179            | IdentKind::Variant
180            | IdentKind::ErrorCode => Some(swift_ident(ident)),
181            IdentKind::ErrorDomain => Some(type_name(ident, "Error")),
182            IdentKind::Type => None,
183        }
184    }
185
186    /// A doc text with every backticked API identifier in Swift spelling.
187    pub(crate) fn doc(&self, doc: &Option<String>) -> Option<String> {
188        Doc::new(doc, &None).text(|i| self.spell(i))
189    }
190
191    /// The `@available(*, deprecated, ...)` attribute of a deprecated
192    /// declaration, its message in Swift spelling.
193    pub(crate) fn deprecated_attr(&self, deprecated: &Option<String>) -> Option<String> {
194        Doc::new(&None, deprecated)
195            .deprecation(|i| self.spell(i))
196            .map(|msg| {
197                format!(
198                    "@available(*, deprecated, message: \"{}\")",
199                    swift_str(&msg)
200                )
201            })
202    }
203
204    /// The Swift type of the error domain named `domain`.
205    pub(crate) fn domain_type(&self, domain: &str) -> String {
206        self.model.error_domain(domain).type_name.clone()
207    }
208}
209
210/// The error type for failures outside any declared domain:
211/// `{Module}RuntimeError`.
212pub(crate) fn runtime_error_name(swift_module: &str) -> String {
213    format!("{swift_module}RuntimeError")
214}
215
216/// The namespace type of the library: `{Module}Library`.
217pub(crate) fn library_type_name(swift_module: &str) -> String {
218    format!("{swift_module}Library")
219}
220
221/// The internal namespace holding the process-wide vtable for the callback
222/// interface named `name`: `Wv{Name}Vtable`.
223pub(crate) fn callback_vtable_name(name: &str) -> String {
224    format!("Wv{name}Vtable")
225}

231    #[test]
232    fn interface_members_avoid_the_wrapper_members() {
233        assert_eq!(swift_member("clone_ptr"), "clonePtr_");
234        assert_eq!(swift_member("ptr"), "ptr_");
235        assert_eq!(swift_member("wv_write"), "wvWrite_");
236        assert_eq!(swift_member("init"), "init_");
237        assert_eq!(swift_member("close"), "close");
238    }
239
240    #[test]
241    fn string_literals_escape_quotes_and_controls() {
242        assert_eq!(swift_str(r#"a "b" \c"#), r#"a \"b\" \\c"#);
243        assert_eq!(swift_str("x\ny\u{1}"), "x\\ny\\u{1}");
244    }
