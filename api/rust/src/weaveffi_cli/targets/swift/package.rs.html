1//! SwiftPM packaging surfaces: `Package.swift` (generated and packaged), the
2//! system-library module map, and the packaged README, framed with the
3//! generated-file prelude and trailer.
4
5use crate::utils::{render_prelude, render_trailer, CommentStyle};

10/// The SwiftPM tools version the manifest declares (Swift 5 language mode).
11const TOOLS_VERSION: &str = "5.9";
12
13/// Where a manifest's C module comes from when no `XCFramework` sits next to
14/// it.
15pub(crate) enum CSource<'a> {
16    /// The generated `Sources/C{Module}` system-library target, linking the
17    /// library from the linker search path.
18    SystemLibrary,
19    /// A published `XCFramework` archive.
20    Remote {
21        /// The archive's URL.
22        url: &'a str,
23        /// The archive's SHA-256 checksum.
24        checksum: &'a str,
25    },
26}
27
28/// The manifest's `platforms:` list from the configured minimum versions.
29pub(crate) fn platforms(config: &SwiftConfig) -> String {
30    format!(
31        ".macOS(\"{}\"), .iOS(\"{}\")",
32        config.min_macos, config.min_ios
33    )
34}
35
36/// Render `Package.swift`. The manifest picks the C module's source at
37/// resolution time: a prebuilt `C{Module}.xcframework` next to it when one
38/// exists, otherwise `source`.
39pub(crate) fn render_package_swift(
40    names: &Names,
41    config: &SwiftConfig,
42    source: &CSource,
43) -> String {
44    let c = names.c_module;
45    let (comment, fallback) = match source {
46        CSource::SystemLibrary => (
47            format!(
48                "// The C module `{c}` comes from one of two places. Next to this\n\
49                 // manifest, a prebuilt `{c}.xcframework` (from `weaveffi package`) wins.\n\
50                 // Otherwise `Sources/{c}` declares the generated header as a system\n\
51                 // library that links `{}` from the linker search path, e.g.\n\
52                 // `swift build -Xlinker -L/path/to/lib`.",
53                names.library
54            ),
55            format!(".systemLibrary(name: \"{c}\")"),
56        ),
57        CSource::Remote { url, checksum } => (
58            format!(
59                "// The C module `{c}` is the prebuilt `{c}.xcframework`, published as\n\
60                 // `{c}.xcframework.zip`. A copy unzipped next to this manifest wins,\n\
61                 // for local development."
62            ),
63            format!(
64                ".binaryTarget(\n        name: \"{c}\",\n        url: \"{url}\",\n        \
65                 checksum: \"{checksum}\"\n    )"
66            ),
67        ),
68    };
69    // `swift-tools-version` must be the very first line of the manifest, so
70    // the generated-file prelude follows it.
71    format!(
72        "// swift-tools-version:{TOOLS_VERSION}\n{}{}\n{}",
73        render_prelude(CommentStyle::DoubleSlash),
74        render_package(names, &comment, &fallback, &platforms(config)),
75        render_trailer(CommentStyle::DoubleSlash, "Package.swift"),
76    )
77}
78
79/// Render the module map of the `C{Module}` system-library target, which
80/// sits next to its copy of the C header.
81pub(crate) fn render_modulemap(names: &Names) -> String {
82    format!(
83        "{}{}\n{}",
84        render_prelude(CommentStyle::DoubleSlash),
85        render_module_map(names),
86        render_trailer(CommentStyle::DoubleSlash, "module.modulemap"),
87    )
88}

90/// README for the packaged Swift package: the `XCFramework` slices and how
91/// to publish the archive its binary target points at.
92pub(crate) fn render_packaged_readme(
93    names: &Names,
94    slices: &str,
95    url: &str,
96    checksum: &str,
97) -> String {
98    format!(
99        "{}{}\n{}",
100        render_prelude(CommentStyle::Xml),
101        render_readme(names, slices, url, checksum),
102        render_trailer(CommentStyle::Xml, "README.md"),
103    )
