2//! (dispatched on [`Crossing`]), the JNA type each C value crosses as, and
3//! the public Kotlin wrapper of every callable, with the borrow scopes that
4//! keep every object (receiver and arguments) alive and unreleased for the

18/// `expr` in receiver position: as is when it's a postfix chain (a name,
19/// member, call, call with a trailing lambda, or index, with no other space
20/// outside its brackets), else parenthesized.
21pub(crate) fn receiver(expr: &str) -> String {
22    let mut depth = 0usize;
23    for (i, c) in expr.char_indices() {
24        match c {
25            '(' | '[' | '{' => depth += 1,
26            ')' | ']' | '}' => depth = depth.saturating_sub(1),
27            c if c.is_whitespace() && depth == 0 && !expr[i..].trim_start().starts_with('{') => {
28                return format!("({expr})");
29            }
30            _ => {}
31        }
32    }
33    expr.to_string()
34}

36/// Where a C value comes from or goes to, which decides who owns a run.
37#[derive(Clone, Copy, PartialEq, Eq)]
38pub(crate) enum Flow {
39    /// A callable's parameter (lowered; the library borrows runs) or its
40    /// return, async result, or iterator item (lifted; the bindings release
41    /// runs).
42    Call,
43    /// A callback method's argument (lifted; runs are borrowed) or return
44    /// (lowered; the library adopts runs, allocated with `{p}_alloc`).
45    Callback,
46}

48/// The Kotlin type a C value crosses JNA as. In a callback (`callback`),
49/// the error slot is a raw `Pointer?` the dispatch reports through.
50pub(crate) fn jna_type(ct: &CType, callback: bool) -> String {
51    match ct {
52        CType::Scalar(p) => jna_prim(*p).to_string(),
53        CType::Size => "SizeT".to_string(),
54        CType::Void => "Unit".to_string(),
55        CType::Enum(_) => "Int".to_string(),
56        CType::Opt(p) => format!("{}.ByValue", opt_class(*p)),
57        CType::Run => "Run.ByValue".to_string(),
58        CType::Callback(_) => "CallbackArg".to_string(),
59        CType::Ptr { pointee, .. } => match pointee.as_ref() {
60            CType::Error if !callback => "NativeError?".to_string(),
61            // An iterator's out item.
62            CType::Run => "Run".to_string(),
63            CType::Opt(p) => opt_class(*p),
64            CType::Scalar(p) => format!("{}ByReference", jna_prim(*p)),
65            CType::Enum(_) => "IntByReference".to_string(),
66            CType::Ptr { .. } => "PointerByReference".to_string(),
67            _ => "Pointer?".to_string(),
68        },
69        CType::Named(_)
70        | CType::Object(_)
71        | CType::Vtable(_)
72        | CType::Error
73        | CType::CancelToken => {
74            unreachable!("{ct:?} never crosses by value")
75        }
76    }
77}

79/// The completion callback type stem for an async result's C type (`Run`,
80/// `OptI64`, `Object`, `Void`).
81pub(crate) fn completion_stem(result: Option<&CType>) -> String {
82    match result {
83        None => "Void".to_string(),
84        Some(CType::Scalar(p)) => p.pascal().to_string(),
85        Some(CType::Enum(_)) => "I32".to_string(),
86        Some(CType::Opt(p)) => opt_class(*p),
87        Some(CType::Run) => "Run".to_string(),
88        Some(_) => "Object".to_string(),
89    }
90}

92/// The `Completions` member holding the completion of an async result of
93/// C type `result` (`runDone`).
94pub(crate) fn completion_value(result: Option<&CType>) -> String {
95    let stem = completion_stem(result);
96    let mut chars = stem.chars();
97    let first = chars
98        .next()
99        .map(|c| c.to_ascii_lowercase())
100        .unwrap_or_default();
101    format!("{first}{}Done", chars.as_str())
102}

104/// The result slot of an async callable's completion.
105pub(crate) fn completion_result(f: &FnBinding) -> Option<&CType> {
106    let a = f.async_binding()?;
107    a.callback.params.get(2).map(|p| &p.ty)
108}

110/// The primitive-array stem of a typed-array element (`Int` for `IntArray`).
111fn array_stem(p: Prim) -> &'static str {
112    match p {
113        Prim::I8 | Prim::U8 => "Byte",
114        Prim::I16 | Prim::U16 => "Short",
115        Prim::I32 | Prim::U32 => "Int",
116        Prim::I64 | Prim::U64 => "Long",
117        Prim::F32 => "Float",
118        Prim::F64 => "Double",
119        _ => unreachable!("{p} is not a typed-array element"),
120    }
121}

136/// The conversion of a scalar's public value to its JNA carrier.
137fn lower_scalar(p: Prim, expr: &str) -> String {
138    let r = receiver(expr);
139    match p {
140        Prim::Bool => format!("{r}.asByte()"),
141        Prim::U8 => format!("{r}.toByte()"),
142        Prim::U16 => format!("{r}.toShort()"),
143        Prim::U32 => format!("{r}.toInt()"),
144        Prim::U64 => format!("{r}.toLong()"),
145        _ => expr.to_string(),
146    }
147}

149/// The conversion of a scalar's JNA carrier to its public value.
150fn lift_scalar(p: Prim, expr: &str) -> String {
151    let r = receiver(expr);
152    match p {
153        Prim::Bool => format!("{r}.asBoolean()"),
154        Prim::U8 => format!("{r}.toUByte()"),
155        Prim::U16 => format!("{r}.toUShort()"),
156        Prim::U32 => format!("{r}.toUInt()"),
157        Prim::U64 => format!("{r}.toULong()"),
158        _ => expr.to_string(),
159    }
160}

162/// The Kotlin expression lowering the public value `expr` of `t` to its C
163/// value. A run is staged for the library to borrow ([`Flow::Call`]) or
164/// allocated with `{p}_alloc` for it to adopt ([`Flow::Callback`]), and an
165/// object returned from a callback is a fresh strong reference.
166///
167/// # Panics
168///
169/// Panics for an object parameter of a call, which borrows the wrapper's
170/// reference instead (see [`emit_callable`]).
171pub(crate) fn lower(n: &Names, t: &Ty, flow: Flow, expr: &str) -> String {
172    let run = match flow {
173        Flow::Call => "runOf",
174        Flow::Callback => "allocRun",
175    };
176    let r = receiver(expr);
177    match crossing(t) {
178        Crossing::Scalar(p) => lower_scalar(p, expr),
179        Crossing::Enum(_) => format!("{r}.value"),
180        Crossing::OptScalar(p) => format!("{}({expr})", opt_fn(p)),
181        Crossing::OptEnum(_) => format!("{}({r}?.value)", opt_fn(Prim::I32)),
182        Crossing::Utf8 => format!("{run}(encodeUtf8({expr}))"),
183        Crossing::Bytes => format!("{run}({expr})"),
184        Crossing::Slice(p) => {
185            let bits = matches!(p, Prim::U16 | Prim::U32 | Prim::U64);
186            let suffix = if bits { "ArrayBits" } else { "Array" };
187            format!("{run}({r}.to{}{suffix}())", array_stem(p))
188        }
189        Crossing::Buffer(ty) => format!("{run}({})", encode_expr(n, ty, expr)),
190        Crossing::Object { nullable, .. } => {
191            assert!(
192                flow == Flow::Callback,
193                "a call borrows its object arguments"
194            );
195            if nullable {
196                format!("{r}?.cloneHandle()")
197            } else {
198                format!("{r}.cloneHandle()")
199            }
200        }
201    }
202}

204/// The Kotlin expression lifting `expr`, the C value of `t`, into its public
205/// value. A run the call returned ([`Flow::Call`]) is read and released; a
206/// callback argument's ([`Flow::Callback`]) is borrowed. An object is
207/// adopted: the wrapper owes the reference's release.
208pub(crate) fn lift(n: &Names, t: &Ty, flow: Flow, expr: &str) -> String {
209    let read = |body: &dyn Fn(&str) -> String| match flow {
210        Flow::Call => format!("take({expr}) {{ _run -> {} }}", body("_run")),
211        Flow::Callback => body(&receiver(expr)),
212    };
213    match crossing(t) {
214        Crossing::Scalar(p) => lift_scalar(p, expr),
215        Crossing::Enum(name) => format!("{}.fromValue({expr})", n.ty(name)),
216        Crossing::OptScalar(_) => format!("{}.get()", receiver(expr)),
217        Crossing::OptEnum(name) => format!(
218            "{}.get()?.let {{ _v -> {}.fromValue(_v) }}",
219            receiver(expr),
220            n.ty(name)
221        ),
222        Crossing::Utf8 => read(&|r| format!("{r}.string()")),
223        Crossing::Bytes => read(&|r| format!("{r}.bytes()")),
224        Crossing::Slice(p) => {
225            let list = match p {
226                Prim::U16 => "toUShortList()",
227                Prim::U32 => "toUIntList()",
228                Prim::U64 => "toULongList()",
229                _ => "asList()",
230            };
231            read(&|r| format!("{r}.{}().{list}", run_reader(p)))
232        }
233        Crossing::Buffer(ty) => read(&|r| decode_expr(n, ty, r)),
234        Crossing::Object {
235            interface,
236            nullable,
237        } => {
238            let class = n.ty(interface);
239            if nullable {
240                format!("{class}.fromHandleOrNull({expr})")
241            } else {
242                format!("{class}.fromHandle({expr})")
243            }
244        }
245    }
246}

248/// The JNA carrier class an `Any?` holding the C value of `t` (an async
249/// result) is cast to.
250fn carrier(t: &Ty) -> String {
251    match crossing(t) {
252        Crossing::Scalar(p) => jna_prim(p).to_string(),
253        Crossing::Enum(_) => "Int".to_string(),
254        Crossing::OptScalar(p) => opt_class(p),
255        Crossing::OptEnum(_) => opt_class(Prim::I32),
256        Crossing::Object { .. } => "Pointer?".to_string(),
257        _ => "Run".to_string(),
258    }
259}

261/// The out-item holder an iterator's `_next` fills for an element of `t`,
262/// and the expression reading the element from it.
263fn item_holder(t: &Ty) -> (String, &'static str) {
264    match crossing(t) {
265        Crossing::Scalar(p) => (format!("{}ByReference()", jna_prim(p)), "_item.value"),
266        Crossing::Enum(_) => ("IntByReference()".to_string(), "_item.value"),
267        Crossing::OptScalar(p) => (format!("{}()", opt_class(p)), "_item"),
268        Crossing::OptEnum(_) => (format!("{}()", opt_class(Prim::I32)), "_item"),
269        Crossing::Object { .. } => ("PointerByReference()".to_string(), "_item.value"),
270        _ => ("Run()".to_string(), "_item"),
271    }
272}

279/// Escape `s` for the inside of a Kotlin string literal.
280pub(crate) fn kt_string(s: &str) -> String {
281    s.replace('\\', "\\\\")
282        .replace('"', "\\\"")
283        .replace('$', "\\$")
284        .replace('\n', "\\n")
285        .replace('\r', "\\r")

288/// How a public wrapper is declared.
289pub(crate) struct Decl<'a> {
290    /// The Kotlin name (`count`, `invoke`).
291    pub name: &'a str,
292    /// Whether it's the companion's `operator fun invoke` (a `new`

300/// The pieces of one call: the locals staged before it, the borrow scopes
301/// around it, and its C arguments (without the error slot or, for an async
302/// call, the token, completion, and context).
303struct Args {
304    locals: Vec<String>,
305    opens: Vec<String>,
306    args: Vec<String>,
307}
308
309/// Lower every argument of `f`. Runs and optional scalars are staged into
310/// locals first (`key_` for `key`: generated temporaries start with `_`, so
311/// neither collides with a parameter), the receiver and every object argument open one `borrow`
312/// scope each, and a callback interface registers its implementation inline
313/// in the call, after everything that can fail, so a failure before the call
314/// never strands a registered implementation.
315fn lower_args(n: &Names, f: &FnBinding) -> Args {
316    let mut out = Args {
317        locals: Vec::new(),
318        opens: Vec::new(),
319        args: Vec::new(),
320    };
321    if f.has_self() {
322        out.opens.push("handle.borrow { _self ->".to_string());
323        out.args.push("_self".to_string());
324    }
325    for (i, p) in f.params.iter().enumerate() {
326        let name = kt_param(&p.name);
327        let t = match &p.ty {
328            ParamTy::Callback { name: cb, nullable } => {
329                let vtable = n.vtable_named(cb);
330                out.args.push(if *nullable {
331                    format!("{vtable}.argOrNone({name})")
332                } else {
333                    format!("{vtable}.arg({name})")
334                });
335                continue;
336            }
337            ParamTy::Value(t) => t,
338        };
339        match crossing(t) {
340            Crossing::Object { nullable, .. } => {
341                out.opens.push(if nullable {
342                    format!("{name}?.handle.borrowOrNull {{ _h{i} ->")
343                } else {
344                    format!("{name}.handle.borrow {{ _h{i} ->")
345                });
346                out.args.push(format!("_h{i}"));
347            }
348            Crossing::Scalar(_) | Crossing::Enum(_) => {
349                out.args.push(lower(n, t, Flow::Call, &name));
350            }
351            _ => {
352                out.locals
353                    .push(format!("val {name}_ = {}", lower(n, t, Flow::Call, &name)));
354                out.args.push(format!("{name}_"));
355            }
356        }
357    }
358    out
359}

361/// Write `body` (lines relative to depth 0) inside the borrow scopes
362/// `opens`, one level deeper per scope.
363fn nested(w: &mut CodeWriter, opens: &[String], body: &[String], first_prefix: &str) {
364    for (d, open) in opens.iter().enumerate() {
365        let open = if d == 0 {
366            format!("{first_prefix}{open}")
367        } else {
368            open.clone()
369        };
370        w.line(open);
371        w.indent();
372    }
373    for (i, line) in body.iter().enumerate() {
374        if opens.is_empty() && i == 0 {
375            w.line(format!("{first_prefix}{line}"));
376        } else {
377            w.line(line);
378        }
379    }
380    for _ in opens {
381        w.dedent();
382        w.line("}");
383    }
384}

386/// Emit the public wrapper for callable `f`, with its KDoc, annotations,
387/// and body.
388pub(crate) fn emit_callable(
389    w: &mut CodeWriter,
390    n: &Names,
391    sp: &Speller,
392    f: &FnBinding,
393    decl: Decl,
394) {
395    sp.fn_doc(
396        w,
397        sp.text(&f.doc),
398        f.params.iter().map(|p| (p.name.as_str(), &p.doc)),
399    );
400    if let Some(msg) = sp.deprecation(&f.deprecated) {
401        w.line(deprecated_line(&msg));
402    }
403    if let Some(exc) = n.thrown(&f.error) {
404        w.line(format!("@Throws({exc}::class)"));
405    }
406    if decl.jvm_static {
407        w.line("@JvmStatic");
408    }
409    let params: Vec<String> = f
410        .params
411        .iter()
412        .map(|p| format!("{}: {}", kt_param(&p.name), n.kt_param_type(&p.ty)))
413        .collect();
414    let ret = f
415        .ret
416        .as_ref()
417        .map(|t| format!(": {}", n.kt_ret_type(t)))
418        .unwrap_or_default();
419    let suspend = if f.is_async() { "suspend " } else { "" };
420    let operator = if decl.operator { "operator " } else { "" };
421    let sig = format!(
422        "{suspend}{operator}fun {}({}){ret}",
423        decl.name,
424        params.join(", ")
425    );
426    let domain = n.domain_index(&f.error);
427    let symbol = &f.abi.symbol;
428    let Args {
429        locals,
430        opens,
431        mut args,
432    } = lower_args(n, f);
433    let returns = f.ret.is_some();
434    let body: Vec<String> = match &f.shape {
435        Shape::Sync => {
436            args.push("_err".to_string());
437            let call = format!(
438                "callNative({domain}) {{ _err -> Lib.c.{symbol}({}) }}",
439                args.join(", ")
440            );
441            vec![match f.ret_value() {
442                Some(t) => lift(n, t, Flow::Call, &call),
443                None => call,
444            }]
445        }
446        Shape::Iterator(it) => {
447            args.push("_err".to_string());
448            let elem = n.kt_type(&it.elem);
449            let (holder, item) = item_holder(&it.elem);
450            vec![
451                format!("NativeSequence<{elem}>("),
452                format!(
453                    "    nonNull(callNative({domain}) {{ _err -> Lib.c.{symbol}({}) }}),",
454                    args.join(", ")
455                ),
456                format!("    Lib.c::{},", it.destroy_symbol),
457                ") { _iter ->".to_string(),
458                format!("    val _item = {holder}"),
459                format!(
460                    "    if (callNative({domain}) {{ _err -> Lib.c.{}(_iter, _item, _err) }} == 0) {{",
461                    it.next.symbol
462                ),
463                "        NativeSequence.DONE".to_string(),
464                "    } else {".to_string(),
465                format!("        {}", lift(n, &it.elem, Flow::Call, item)),
466                "    }".to_string(),
467                "}".to_string(),
468            ]
469        }
470        Shape::Async(a) => {
471            if a.cancellable {
472                args.push("_token".to_string());
473            }
474            let result = completion_result(f);
475            args.push(format!("Completions.{}", completion_value(result)));
476            args.push("_context".to_string());
477            let lift_fn = match f.ret_value() {
478                Some(t) => format!(
479                    "{{ _raw -> {} }}",
480                    lift(n, t, Flow::Call, &format!("_raw as {}", carrier(t)))
481                ),
482                None => "{ }".to_string(),
483            };
484            let token = if a.cancellable { "_token" } else { "_" };
485            let mut inner = CodeWriter::four_space();
486            nested(
487                &mut inner,
488                &opens,
489                &[format!("Lib.c.{symbol}({})", args.join(", "))],
490                "",
491            );
492            let mut lines = vec![format!(
493                "awaitNative({}, {domain}, {lift_fn}) {{ {token}, _context ->",
494                a.cancellable
495            )];
496            lines.extend(inner.finish().lines().map(|l| format!("    {l}")));
497            lines.push("}".to_string());
498            // The borrows live inside the launch, so none spans a suspension.
499            return emit_body(w, &sig, returns, &locals, &[], &lines);
500        }
501    };
502    emit_body(w, &sig, returns, &locals, &opens, &body);
503}

505/// Emit a function `sig` whose body stages `locals`, then evaluates `body`
506/// inside the borrow scopes `opens`: an expression body when it has neither
507/// locals nor scopes and returns a value, a block body otherwise.
508fn emit_body(
509    w: &mut CodeWriter,
510    sig: &str,
511    returns: bool,
512    locals: &[String],
513    opens: &[String],
514    body: &[String],
515) {
516    if returns && locals.is_empty() && opens.is_empty() {
517        w.line(format!("{sig} = {}", body[0]));
518        for line in &body[1..] {
519            w.line(line);
520        }
521        return;
522    }
523    w.block(format!("{sig} {{"), "}", |w| {
524        for local in locals {
525            w.line(local);
526        }
527        nested(w, opens, body, if returns { "return " } else { "" });
528    });
529}
