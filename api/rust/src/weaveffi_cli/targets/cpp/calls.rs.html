1//! Callable rendering: free functions and interface members in the sync,
2//! iterator, and async call shapes, each marshalled per value by its
3//! [`Crossing`] and failing per its [`ErrorStrategy`].
4//!
5//! Interface members are rendered twice: a declaration inside the class body
6//! (so the class is complete before any record that holds it by value) and an
7//! out-of-line `inline` definition after every value type and codec exists.
8//! Free functions are defined inline at namespace scope, rendered last.
9

27/// The C++ value of a `ty` the producer handed over as the C value `v` (a
28/// return, an async result, or an iterator item), adopting what it owns: a
29/// run is copied or decoded and then released, an object is adopted into
30/// its wrapper, an optional scalar is unpacked, and a C-style enum converts
31/// from its discriminant.
32pub(crate) fn lift(ty: &Ty, v: &str) -> String {
33    match crossing(ty) {
34        Crossing::Scalar(_) => v.to_string(),
35        Crossing::Enum(name) => format!("static_cast<{name}>({v})"),
36        Crossing::OptScalar(p) => format!("detail::from_opt<{}>({v})", cpp_prim(p)),
37        Crossing::OptEnum(name) => format!("detail::from_opt<{name}>({v})"),
38        Crossing::Utf8 => format!("detail::take_string({v})"),
39        Crossing::Bytes => format!("detail::take_bytes({v})"),
40        Crossing::Slice(p) => format!("detail::take_slice<{}>({v})", cpp_prim(p)),
41        Crossing::Buffer(ty) => format!("detail::take<{}>({v})", cpp_type(ty)),
42        Crossing::Object {
43            interface,
44            nullable: true,
45        } => format!("detail::adopt_optional<{interface}>({v})"),
46        Crossing::Object {
47            interface,
48            nullable: false,
49        } => format!("{interface}(adopt, {v})"),
50    }
51}

53/// The C type of a `ty` the producer hands over, which a local receiving
54/// it (an iterator's `out_item`) is declared with.
55fn transfer_ctype(ctx: &Ctx<'_>, ty: &Ty) -> String {
56    CType::of(ty, Position::Transfer).render_c(ctx.prefix)
57}

61/// The names a wrapper body declares, kept apart from its parameters: a
62/// local gets a trailing underscore while its preferred name is taken.
63struct Locals(HashSet<String>);
64
65impl Locals {
66    fn new(f: &FnBinding) -> Self {
67        Self(f.params.iter().map(|p| cpp_param_name(&p.name)).collect())
68    }
69
70    fn fresh(&mut self, base: &str) -> String {
71        let mut name = base.to_string();
72        while self.0.contains(&name) {
73            name.push('_');
74        }
75        self.0.insert(name.clone());
76        name
77    }
78}

82/// Append the setup statements for one parameter and return the C argument
83/// expression it passes, per its [`Crossing`]:
84///
85/// * A string, bytes, or typed array passes a run over its own storage (a
86///   string's UTF-8 without a terminator, so interior NULs survive).
87/// * An optional scalar or enum passes its `{p}_opt_*` struct.
88/// * A buffered value is encoded into a local `detail::BufferWriter` that
89///   outlives the call, and passes a run over it.
90/// * An object is borrowed: the wrapper's pointer is passed and the wrapper
91///   keeps its reference. `Interface?` passes null for none.
92/// * A callback interface moves the caller's `std::shared_ptr` into a heap
93///   box that becomes the `{p}_{Cb}` struct's `ctx`; the producer deletes it
94///   through the vtable's `free`. A `std::unique_ptr` owns the box until the
95///   argument list releases it, so a throw while marshalling a later
96///   parameter frees it. An optional callback interface passes a null
97///   vtable for none.
98fn emit_param_setup(
99    w: &mut CodeWriter,
100    ctx: &Ctx<'_>,
101    locals: &mut Locals,
102    ty: &ParamTy,
103    name: &str,
104) -> String {
105    let prefix = ctx.prefix;
106    let ty = match ty {
107        ParamTy::Callback {
108            name: iface,
109            nullable,
110        } => {
111            let tag = &ctx.model.callback(iface).c_tag;
112            let box_ = locals.fresh(&format!("{name}_ctx"));
113            let vtable = format!("&detail::Callbacks<{iface}>::vtable()");
114            if *nullable {
115                let vt = locals.fresh(&format!("{name}_vtable"));
116                w.line(format!("auto {box_} = detail::lend(std::move({name}));"));
117                w.line(format!("const auto* {vt} = {box_} ? {vtable} : nullptr;"));
118                return format!("{tag}{{{box_}.release(), {vt}}}");
119            }
120            w.line(format!(
121                "auto {box_} = detail::lend_required(std::move({name}), \"{name}\");"
122            ));
123            return format!("{tag}{{{box_}.release(), {vtable}}}");
124        }
125        ParamTy::Value(ty) => ty,
126    };
127    match crossing(ty) {
128        Crossing::Scalar(_) => name.to_string(),
129        Crossing::Enum(_) => format!("static_cast<int32_t>({name})"),
130        Crossing::OptScalar(p) => format!("detail::to_opt<{prefix}_opt_{}>({name})", p.snake()),
131        Crossing::OptEnum(_) => format!("detail::to_opt<{prefix}_opt_i32>({name})"),
132        Crossing::Utf8 => format!("detail::string_run({name})"),
133        Crossing::Bytes => format!("detail::bytes_run({name})"),
134        Crossing::Slice(_) => format!("detail::slice_run({name})"),
135        Crossing::Buffer(_) => {
136            let buf = locals.fresh(&format!("{name}_buf"));
137            w.line(format!("const auto {buf} = detail::encode({name});"));
138            format!("{buf}.view()")
139        }
140        Crossing::Object { nullable: true, .. } => format!("detail::handle_of({name})"),
141        Crossing::Object {
142            nullable: false, ..
143        } => format!("{name}.handle()"),
144    }
145}

149/// How a rendered callable is declared in the C++ surface.
150#[derive(Clone, Copy)]
151pub(crate) enum FnKind<'a> {
152    /// A namespace-scope free function.
153    Free,
154    /// An instance method: passes the wrapped pointer as the leading C
155    /// argument and is `const` (the ABI receiver is a const pointer).
156    Method {
157        /// The wrapper class name.
158        class: &'a str,
159    },
160    /// A static member: interface statics and the factory form of
161    /// constructors not named `new`.
162    Static {
163        /// The wrapper class name.
164        class: &'a str,
165    },
166    /// The interface constructor named `new`, rendered as a C++ constructor.
167    Ctor {
168        /// The wrapper class name.
169        class: &'a str,
170    },
171}
172
173/// The C++ return type of a callable: the mapped type (or `void`) for sync,
174/// `Range<T>` for an iterator, and `std::future<T>` for async. `None` for a
175/// constructor.
176fn return_type(f: &FnBinding, kind: FnKind<'_>) -> Option<String> {
177    if matches!(kind, FnKind::Ctor { .. }) {
178        return None;
179    }
180    let value = f.ret.as_ref().map_or("void".to_string(), cpp_ret_type);
181    Some(if f.is_async() {
182        format!("std::future<{value}>")
183    } else {
184        value
185    })
186}

188/// The C++ parameter list of a callable. A cancellable async call takes a
189/// trailing `const CancelToken&`; `with_defaults` adds its
190/// `= CancelToken::none()` default, which belongs on the first declaration
191/// only.
192fn param_decls(f: &FnBinding, with_defaults: bool) -> String {
193    let mut decls: Vec<String> = f
194        .params
195        .iter()
196        .map(|p| cpp_param_decl(&p.ty, &cpp_param_name(&p.name)))
197        .collect();
198    if f.cancellable() {
199        let default = if with_defaults {
200            " = CancelToken::none()"
201        } else {
202            ""
203        };
204        decls.push(format!("const CancelToken& cancel_token{default}"));
205    }
206    decls.join(", ")
207}

209/// The `@throws` line of a callable that declares errors.
210fn throws_tag(error: &ErrorStrategy) -> Option<String> {
211    match error {
212        ErrorStrategy::Trap => None,
213        ErrorStrategy::Untyped => Some(
214            "@throws Error with code -1 and the producer's message when the call fails.".into(),
215        ),
216        ErrorStrategy::Domain(name) => Some(format!(
217            "@throws {} (or one of its codes' classes) when the call fails with one of the \
218             domain's codes.",
219            cpp_error_class(name)
220        )),
221    }
222}

224/// Append the doc comment and any `[[deprecated]]` attribute of a callable:
225/// its IDL doc, `@param` for each documented parameter, what an iterator or
226/// async call returns, and the exceptions it throws.
227fn emit_callable_attrs(w: &mut CodeWriter, ctx: &Ctx<'_>, f: &FnBinding) {
228    let spell = |s: &str| ctx.spell(s);
229    let doc = Doc::new(&f.doc, &f.deprecated);
230    let mut sections: Vec<String> = doc.text(spell).into_iter().collect();
231    let mut tags: Vec<String> = f
232        .params
233        .iter()
234        .filter_map(|p| {
235            Doc::new(&p.doc, &None)
236                .text(spell)
237                .map(|d| format!("@param {} {d}", cpp_param_name(&p.name)))
238        })
239        .collect();
240    if f.cancellable() {
241        tags.push("@param cancel_token Cancels the call; the future then throws Cancelled.".into());
242    }
243    if f.iterator().is_some() {
244        tags.push("@return A lazy, single-pass range; each step makes one producer call.".into());
245    } else if f.is_async() {
246        tags.push(
247            "@return A future settled from a producer thread; get() rethrows the call's error."
248                .into(),
249        );
250    }
251    tags.extend(throws_tag(&f.error));
252    if !tags.is_empty() {
253        sections.push(tags.join("\n"));
254    }
255    let text = (!sections.is_empty()).then(|| sections.join("\n\n"));
256    w.doc(&text, DocCommentStyle::Javadoc);
257    if let Some(msg) = doc.deprecation(spell) {
258        let escaped = msg.replace('\\', "\\\\").replace('"', "\\\"");
259        w.line(format!("[[deprecated(\"{escaped}\")]]"));
260    }
261}

263/// Append the in-class declaration of an interface member, with its doc
264/// comment and deprecation marker.
265pub(crate) fn render_member_decl(
266    w: &mut CodeWriter,
267    ctx: &Ctx<'_>,
268    f: &FnBinding,
269    cpp_name: &str,
270    kind: FnKind<'_>,
271) {
272    emit_callable_attrs(w, ctx, f);
273    let params = param_decls(f, true);
274    match (kind, return_type(f, kind)) {
275        // Explicit, so no argument list converts into a new producer object
276        // behind the caller's back.
277        (FnKind::Ctor { class }, _) if f.params.is_empty() => w.line(format!("{class}();")),
278        (FnKind::Ctor { class }, _) => w.line(format!("explicit {class}({params});")),
279        (FnKind::Method { .. }, Some(ret)) => w.line(format!("{ret} {cpp_name}({params}) const;")),
280        (FnKind::Static { .. }, Some(ret)) => w.line(format!("static {ret} {cpp_name}({params});")),
281        _ => unreachable!("free functions are defined inline, never declared"),
282    };
283    w.blank();
284}

286/// Append the definition of a callable: an `inline` free function (with its
287/// doc comment), or the out-of-line `inline` definition of an interface
288/// member declared by [`render_member_decl`].
289///
290/// Wrappers are never `noexcept`: a call that declares no errors still
291/// surfaces a producer bug as `InternalError`.
292///
293/// Every free function, constructor, and static member begins with
294/// `check_library()`, which verifies the linked library once. Instance
295/// methods skip it: their receiver could only come from a checked call.
296pub(crate) fn render_definition(
297    w: &mut CodeWriter,
298    ctx: &Ctx<'_>,
299    f: &FnBinding,
300    cpp_name: &str,
301    kind: FnKind<'_>,
302) {
303    if matches!(kind, FnKind::Free) {
304        emit_callable_attrs(w, ctx, f);
305    }
306    let params = param_decls(f, matches!(kind, FnKind::Free));
307    let header = match (kind, return_type(f, kind)) {
308        (FnKind::Ctor { class }, _) => format!("inline {class}::{class}({params}) {{"),
309        (FnKind::Method { class }, Some(ret)) => {
310            format!("inline {ret} {class}::{cpp_name}({params}) const {{")
311        }
312        (FnKind::Static { class }, Some(ret)) => {
313            format!("inline {ret} {class}::{cpp_name}({params}) {{")
314        }
315        (FnKind::Free, Some(ret)) => format!("inline {ret} {cpp_name}({params}) {{"),
316        _ => unreachable!("every non-constructor callable has a return type"),
317    };
318    w.block(header, "}", |w| {
319        if !matches!(kind, FnKind::Method { .. }) {
320            w.line("check_library();");
321        }
322        let mut locals = Locals::new(f);
323        let mut c_args = Vec::new();
324        if matches!(kind, FnKind::Method { .. }) {
325            c_args.push("raw_.get()".to_string());
326        }
327        for p in &f.params {
328            c_args.push(emit_param_setup(
329                w,
330                ctx,
331                &mut locals,
332                &p.ty,
333                &cpp_param_name(&p.name),
334            ));
335        }
336        if let Some(a) = f.async_binding() {
337            emit_async_body(w, ctx, f, a, &mut locals, c_args);
338        } else if let Some(it) = f.iterator() {
339            emit_iterator_body(w, ctx, f, it, &mut locals, c_args);
340        } else {
341            emit_sync_body(w, ctx, f, kind, &mut locals, c_args);
342        }
343    });
344    w.blank();
345}

347/// Append a synchronous callable's call, error check, and return. A
348/// constructor adopts the returned reference. A failed call returns a zero
349/// value, which owns nothing, so checking before lifting leaks nothing.
350fn emit_sync_body(
351    w: &mut CodeWriter,
352    ctx: &Ctx<'_>,
353    f: &FnBinding,
354    kind: FnKind<'_>,
355    locals: &mut Locals,
356    mut c_args: Vec<String>,
357) {
358    let err = locals.fresh("err");
359    w.line(format!("{}_error {err}{{}};", ctx.prefix));
360    c_args.push(format!("&{err}"));
361    let call = format!("{}({})", f.abi.symbol, c_args.join(", "));
362    let check = format!("detail::check<{}>({err});", error_class(&f.error));
363    match (kind, f.ret_value()) {
364        (_, None) => {
365            w.line(format!("{call};"));
366            w.line(check);
367        }
368        (FnKind::Ctor { .. }, Some(_)) => {
369            let result = locals.fresh("result");
370            w.line(format!("auto* {result} = {call};"));
371            w.line(check);
372            w.line(format!("raw_.reset({result});"));
373        }
374        (_, Some(ty)) => {
375            let result = locals.fresh("result");
376            w.line(format!("auto {result} = {call};"));
377            w.line(check);
378            w.line(format!("return {};", lift(ty, &result)));
379        }
380    }
381}

385/// Append an iterator-returning callable's body: call the launcher, check
386/// its error, and wrap the producer iterator in a `Range<T>` with two
387/// captureless lambdas: one pulling an element through `_next` (receiving
388/// it like a return of its type and failing per the callable's
389/// [`ErrorStrategy`]), and one releasing the iterator with `_destroy`.
390fn emit_iterator_body(
391    w: &mut CodeWriter,
392    ctx: &Ctx<'_>,
393    f: &FnBinding,
394    it: &IteratorBinding,
395    locals: &mut Locals,
396    mut c_args: Vec<String>,
397) {
398    let prefix = ctx.prefix;
399    let elem = cpp_type(&it.elem);
400    let tag = &it.iter_tag;
401    let errors = error_class(&f.error);
402    let err = locals.fresh("err");
403    let iter = locals.fresh("iter");
404    w.line(format!("{prefix}_error {err}{{}};"));
405    c_args.push(format!("&{err}"));
406    w.line(format!(
407        "{tag}* {iter} = {}({});",
408        f.abi.symbol,
409        c_args.join(", ")
410    ));
411    w.line(format!("detail::check<{errors}>({err});"));
412    // The lambdas are captureless, but their names stay clear of the
413    // wrapper's parameters and locals so nothing shadows.
414    let raw = locals.fresh("raw");
415    let item = locals.fresh("item");
416    let out_item = locals.fresh("out_item");
417    let next_err = locals.fresh("next_err");
418    w.line(format!(
419        "return Range<{elem}>(adopt, {iter}, [](void* {raw}, std::optional<{elem}>& {item}) {{"
420    ));
421    w.scope(|w| {
422        w.line(format!("{prefix}_error {next_err}{{}};"));
423        w.line(format!("{} {out_item}{{}};", transfer_ctype(ctx, &it.elem)));
424        w.block(
425            format!(
426                "if ({}(static_cast<{tag}*>({raw}), &{out_item}, &{next_err}) == 0) {{",
427                it.next.symbol
428            ),
429            "}",
430            |w| {
431                w.line(format!("detail::check<{errors}>({next_err});"));
432                w.line("return false;");
433            },
434        );
435        w.line(format!("{item}.emplace({});", lift(&it.elem, &out_item)));
436        w.line("return true;");
437    });
438    w.line(format!(
439        "}}, [](void* {raw}) {{ {}(static_cast<{tag}*>({raw})); }});",
440        it.destroy_symbol
441    ));
442}

446/// Append an asynchronous callable's body: a `std::future` wrapper.
447///
448/// The parameters are marshalled first, then a heap-allocated promise
449/// travels through the C `context`, and the completion settles it exactly
450/// once through `detail::settle`, which adopts the promise and the boxed
451/// error, raises the call's exception for a failure, and otherwise receives
452/// the result like a return of its type. A cancellable call passes its
453/// token's native handle (null for `CancelToken::none()`); the producer
454/// takes its own reference.
455fn emit_async_body(
456    w: &mut CodeWriter,
457    ctx: &Ctx<'_>,
458    f: &FnBinding,
459    a: &AsyncBinding,
460    locals: &mut Locals,
461    mut c_args: Vec<String>,
462) {
463    let value = f.ret_value().map_or("void".to_string(), cpp_type);
464    let promise = locals.fresh("promise");
465    let future = locals.fresh("future");
466    let params: Vec<String> = a
467        .callback
468        .params
469        .iter()
470        .map(|p| slot_decl(p, ctx.prefix))
471        .collect();
472    let (context, err) = (
473        slot_name(&a.callback.params[0]),
474        slot_name(&a.callback.params[1]),
475    );
476    if a.cancellable {
477        c_args.push("cancel_token.handle()".to_string());
478    }
479    w.line(format!(
480        "auto {promise} = std::make_unique<std::promise<{value}>>();"
481    ));
482    w.line(format!("auto {future} = {promise}->get_future();"));
483    c_args.push(format!("[]({}) {{", params.join(", ")));
484    w.line(format!("{}({}", f.abi.symbol, c_args.join(", ")));
485    w.scope(|w| {
486        let settle = format!("detail::settle<{}, {value}>", error_class(&f.error));
487        match (f.ret_value(), a.callback.params.get(2)) {
488            (Some(ty), Some(result)) => {
489                w.line(format!(
490                    "{settle}({context}, {err}, [&] {{ return {}; }});",
491                    lift(ty, &slot_name(result))
492                ));
493            }
494            _ => {
495                w.line(format!("{settle}({context}, {err}, [] {{}});"));
496            }
497        }
498    });
499    w.line(format!("}}, {promise}.release());"));
500    w.line(format!("return {future};"));
