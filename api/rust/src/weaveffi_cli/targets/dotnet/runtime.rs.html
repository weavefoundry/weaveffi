1//! The fixed runtime (`runtime/Runtime.cs`, spliced with the library's
2//! names) and the names of its public classes.
3
4use crate::utils::{render_prelude, render_trailer, CommentStyle};
5use weaveffi_model::errors;
6use weaveffi_model::model::{Model, ABI_VERSION};
7
8/// The runtime source shared by every generated library.
9const RUNTIME_CS: &str = include_str!("runtime/Runtime.cs");
10
11/// The public classes of the runtime, named so they never collide with a
12/// type the API declares.
13pub(crate) struct RuntimeNames {
14    /// The root of every error a throwing call reports (`NativeException`).
15    pub exception: String,
16    /// The trap a failed non-throwing call raises (`NativeBugException`).
17    pub bug_exception: String,
18    /// The load-time failure (`NativeLoadException`).
19    pub load_exception: String,
20    /// The static class with `Check()`, `AbiVersion`, and the free
21    /// functions (`{Namespace}Library`).
22    pub library_class: String,
23}
24
25impl RuntimeNames {
26    /// The names for `model` in `namespace`. A name the API already declares
27    /// is qualified by the namespace's last segment (`KvstoreNativeException`).
28    pub(crate) fn new(model: &Model, namespace: &str) -> Self {
29        let taken = |name: &str| {
30            model.records.iter().any(|s| s.name == name)
31                || model.enums.iter().any(|e| e.name == name)
32                || model.interfaces.iter().any(|i| i.name == name)
33                || model
34                    .callbacks
35                    .iter()
36                    .any(|c| c.name == name || format!("I{}", c.name) == name)
37                || model
38                    .errors
39                    .iter()
40                    .any(|e| errors::exception_type_name(&e.name) == name)
41        };
42        let last = namespace.rsplit('.').next().unwrap_or(namespace);
43        let pick = |name: &str| {
44            if taken(name) {
45                format!("{last}{name}")
46            } else {
47                name.to_string()
48            }
49        };
50        let library = format!("{last}Library");
51        Self {
52            exception: pick("NativeException"),
53            bug_exception: pick("NativeBugException"),
54            load_exception: pick("NativeLoadException"),
55            library_class: if taken(&library) {
56                format!("{last}NativeLibrary")
57            } else {
58                library
59            },
60        }
61    }
62}
63
64/// Render `Runtime.cs`, loading `library`: the fixed runtime with every
65/// placeholder replaced.
66pub(crate) fn render_runtime(
67    model: &Model,
68    namespace: &str,
69    names: &RuntimeNames,
70    library: &str,
71    filename: &str,
72) -> String {
73    let identity = &model.identity;
74    let body = RUNTIME_CS
75        .replace("{{NAMESPACE}}", namespace)
76        .replace("{{BUG_EXCEPTION}}", &names.bug_exception)
77        .replace("{{LOAD_EXCEPTION}}", &names.load_exception)
78        .replace("{{EXCEPTION}}", &names.exception)
79        .replace("{{LIBRARY_CLASS}}", &names.library_class)
80        .replace("{{PREFIX}}", model.prefix())
81        .replace("{{LIBRARY_ENV}}", &identity.library_env_var())
82        .replace("{{LIBRARY}}", library)
83        .replace("{{ABI_VERSION}}", &ABI_VERSION.to_string());
84    debug_assert!(!body.contains("{{"), "unfilled placeholder in Runtime.cs");
85    format!(
86        "{}{body}\n{}",
87        render_prelude(CommentStyle::DoubleSlash),
88        render_trailer(CommentStyle::DoubleSlash, filename)
89    )
90}
