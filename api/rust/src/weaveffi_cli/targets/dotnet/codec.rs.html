1//! Value-buffer codec and value equality expressions.
2//!
3//! Nothing here is monomorphized per composite type: optionals, lists, and
4//! maps go through the runtime's generic `FfiBufferWriter.WriteList` /
5//! `FfiBufferReader.ReadList` (and friends), handed a `static` lambda for
6//! the element, so `[{string:[i32]}]` is three nested calls rather than a
7//! generated method per shape. Records and rich enums carry their own
8//! `WriteTo`/`ReadFrom` pair. An object token carries one strong reference:
9//! writing one clones the wrapper's handle, and reading one adopts the
10//! pointer into a new wrapper.
11
12use weaveffi_model::ty::{Prim, Ty};
13
14use crate::targets::dotnet::types::{cs_type, is_value_type, Cx};
15
16/// The C# expression (a `void` call) writing `value`, the surface value of
17/// `ty`, into the writer `writer`. `depth` numbers the lambda parameters of
18/// nested composites so they never shadow each other.
19pub(crate) fn write_call(ty: &Ty, writer: &str, value: &str, depth: usize) -> String {
20    match ty {
21        Ty::Prim(p) => format!("{writer}.Write{}({value})", p.pascal()),
22        Ty::Enum(_) => format!("{writer}.WriteI32((int){value})"),
23        Ty::Interface(_) => format!("{writer}.WriteObject({value}.CloneHandle())"),
24        Ty::Record(_) | Ty::RichEnum(_) => format!("{value}.WriteTo({writer})"),
25        Ty::Optional(inner) => {
26            let method = if is_value_type(inner) {
27                "WriteOptionalValue"
28            } else {
29                "WriteOptional"
30            };
31            format!("{writer}.{method}({value}, {})", write_lambda(inner, depth))
32        }
33        Ty::List(inner) => format!(
34            "{writer}.WriteList({value}, {})",
35            write_lambda(inner, depth)
36        ),
37        Ty::Map(k, v) => format!(
38            "{writer}.WriteMap({value}, {}, {})",
39            write_lambda(k, depth),
40            write_lambda(v, depth)
41        ),
42    }
43}
44
45/// A `static` lambda writing one value of `ty`: the element writer handed to
46/// the runtime's generic composite methods.
47pub(crate) fn write_lambda(ty: &Ty, depth: usize) -> String {
48    let (w, v) = (format!("w{depth}"), format!("v{depth}"));
49    format!("static ({w}, {v}) => {}", write_call(ty, &w, &v, depth + 1))
50}
51
52/// The C# expression reading a value of `ty` from the reader `reader`; the
53/// inverse of [`write_call`]. An object token is adopted into a new
54/// wrapper, which owes the reference's release.
55pub(crate) fn read_expr(cx: Cx<'_>, ty: &Ty, reader: &str, depth: usize) -> String {
56    match ty {
57        Ty::Prim(p) => format!("{reader}.Read{}()", p.pascal()),
58        Ty::Enum(name) => format!("({}){reader}.ReadI32()", cx.ty(name)),
59        Ty::Interface(name) => format!("{}.Adopt({reader}.ReadObject())", cx.ty(name)),
60        Ty::Record(name) | Ty::RichEnum(name) => format!("{}.ReadFrom({reader})", cx.ty(name)),
61        Ty::Optional(inner) => {
62            let method = if is_value_type(inner) {
63                "ReadOptionalValue"
64            } else {
65                "ReadOptional"
66            };
67            format!("{reader}.{method}({})", read_lambda(cx, inner, depth))
68        }
69        Ty::List(inner) => format!("{reader}.ReadList({})", read_lambda(cx, inner, depth)),
70        // A dictionary isn't covariant in its value type, so the reader
71        // builds it with the surface types spelled out.
72        Ty::Map(k, v) => format!(
73            "{reader}.ReadMap<{}, {}>({}, {})",
74            cs_type(k),
75            cs_type(v),
76            read_lambda(cx, k, depth),
77            read_lambda(cx, v, depth)
78        ),
79    }
80}

82/// A `static` lambda reading one value of `ty`.
83pub(crate) fn read_lambda(cx: Cx<'_>, ty: &Ty, depth: usize) -> String {
84    let r = format!("r{depth}");
85    format!("static {r} => {}", read_expr(cx, ty, &r, depth + 1))
86}

88/// True when the surface type of `ty` compares by reference in C# (a byte
89/// array, a list, or a map, possibly optional), so a record's value
90/// equality has to compare its contents explicitly.
91pub(crate) fn needs_deep_equality(ty: &Ty) -> bool {
92    match ty {
93        Ty::Prim(Prim::Bytes) | Ty::List(_) | Ty::Map(..) => true,
94        Ty::Optional(inner) => needs_deep_equality(inner),
95        _ => false,
96    }
97}
98
99/// The C# expression comparing two values of `ty` by value: contents for
100/// byte arrays, lists, and maps, default equality for everything else.
101pub(crate) fn equal_expr(ty: &Ty, a: &str, b: &str, depth: usize) -> String {
102    match ty {
103        Ty::Prim(Prim::Bytes) => format!("FfiEquality.Bytes({a}, {b})"),
104        Ty::Optional(inner) if needs_deep_equality(inner) => equal_expr(inner, a, b, depth),
105        Ty::List(inner) => match equal_lambda(inner, depth) {
106            Some(eq) => format!("FfiEquality.Lists({a}, {b}, {eq})"),
107            None => format!("FfiEquality.Lists({a}, {b})"),
108        },
109        Ty::Map(_, v) => match equal_lambda(v, depth) {
110            Some(eq) => format!("FfiEquality.Maps({a}, {b}, {eq})"),
111            None => format!("FfiEquality.Maps({a}, {b})"),
112        },
113        _ => format!("FfiEquality.Equal({a}, {b})"),
114    }
115}

117/// The element comparer of a collection whose elements need content
118/// equality, or `None` when default equality does.
119fn equal_lambda(ty: &Ty, depth: usize) -> Option<String> {
120    needs_deep_equality(ty).then(|| {
121        let (x, y) = (format!("x{depth}"), format!("y{depth}"));
122        format!("static ({x}, {y}) => {}", equal_expr(ty, &x, &y, depth + 1))
123    })
124}

126/// The C# expression a record's `GetHashCode` adds for one field: the value
127/// itself, or a hash consistent with [`equal_expr`] for collections.
128pub(crate) fn hash_expr(ty: &Ty, value: &str) -> String {
129    match ty {
130        Ty::Prim(Prim::Bytes) => format!("FfiEquality.BytesHash({value})"),
131        Ty::List(_) => format!("FfiEquality.ListHash({value})"),
132        Ty::Map(..) => format!("FfiEquality.MapHash({value})"),
133        Ty::Optional(inner) if needs_deep_equality(inner) => hash_expr(inner, value),
134        _ => value.to_string(),
135    }
136}

138#[cfg(test)]
139mod tests {
140    use super::*;
141
142    fn list(t: Ty) -> Ty {
143        Ty::List(Box::new(t))
144    }
145
146    #[test]
147    fn nested_composites_are_nested_generic_calls() {
148        let cx = Cx {
149            ns: "Kv",
150            base: "NativeException",
151            bug: "NativeBugException",
152        };
153        let ty = Ty::Map(
154            Box::new(Ty::Prim(Prim::String)),
155            Box::new(list(Ty::Optional(Box::new(Ty::Prim(Prim::I32))))),
156        );
157        assert_eq!(
158            write_call(&ty, "writer", "Value", 0),
159            "writer.WriteMap(Value, static (w0, v0) => w0.WriteString(v0), \
160             static (w0, v0) => w0.WriteList(v0, static (w1, v1) => \
161             w1.WriteOptionalValue(v1, static (w2, v2) => w2.WriteI32(v2))))"
162        );
163        assert_eq!(
164            read_expr(cx, &ty, "reader", 0),
165            "reader.ReadMap<string, IReadOnlyList<int?>>(static r0 => r0.ReadString(), \
166             static r0 => r0.ReadList(static r1 => r1.ReadOptionalValue(static r2 => r2.ReadI32())))"
167        );
168    }
169
170    #[test]
171    fn equality_compares_collection_contents() {
172        let bytes = Ty::Prim(Prim::Bytes);
173        assert_eq!(
174            equal_expr(&list(bytes.clone()), "A", "o.A", 0),
175            "FfiEquality.Lists(A, o.A, static (x0, y0) => FfiEquality.Bytes(x0, y0))"
176        );
177        assert_eq!(
178            equal_expr(&list(Ty::Prim(Prim::I32)), "A", "o.A", 0),
179            "FfiEquality.Lists(A, o.A)"
180        );
181        assert_eq!(
182            equal_expr(&Ty::Prim(Prim::F64), "A", "o.A", 0),
183            "FfiEquality.Equal(A, o.A)"
184        );
185        assert!(!needs_deep_equality(&Ty::Optional(Box::new(Ty::Prim(
186            Prim::I64
187        )))));
188        assert_eq!(
189            hash_expr(&Ty::Optional(Box::new(bytes)), "B"),
190            "FfiEquality.BytesHash(B)"
191        );
192    }
