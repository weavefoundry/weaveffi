1//! Callback-interface renderers: the consumer-facing C# `interface` plus the
2//! process-wide vtable and its `[UnmanagedCallersOnly]` trampolines.
3//!
4//! * A parameter of a callback interface crosses as an `FfiCallback`, the
5//!   `{p}_{Cb}` struct of a `ctx` and a vtable pointer, by value.
6//! * One static vtable per callback interface, allocated once in native
7//!   memory and never freed, so its address is stable for the process. It
8//!   starts with the `{size, flags, free}` header, `size` being the struct's
9//!   real size as C# lays it out and `flags` zero (methods may run on any
10//!   thread).
11//! * `ctx` is a `GCHandle` to the implementation, released by the vtable's
12//!   `free` entry, so the implementation lives exactly as long as the
13//!   producer holds it.
14//! * Arguments are borrowed for the call: strings and buffers are decoded,
15//!   typed arrays and bytes are spans over the producer's memory, and
16//!   objects are adopted into wrappers the implementation owns.
17//! * A return crosses as one C value: a scalar or optional scalar by
18//!   value, an object as a fresh strong reference, and everything else as a
19//!   `{p}_alloc` run the producer adopts.
20//! * A thrown exception never unwinds into native code: an exception of the
21//!   method's own domain reports its code and fields; any other is reported
22//!   with code -1 (or -4 from a method that can't fail).
23

35/// Render one callback interface: the `public interface I{Name}` the
36/// consumer implements, then the internal class hosting its vtable.
37pub(crate) fn render_callback_interface(
38    w: &mut CodeWriter,
39    model: &Model,
40    docs: &Docs,
41    cb: &CallbackBinding,
42    cx: Cx<'_>,
43) {
44    render_consumer_interface(w, model, docs, cb, cx);
45    render_vtable_class(w, model, cb, cx);
46}
47
48/// The consumer-facing interface: one method per callback method.
49fn render_consumer_interface(
50    w: &mut CodeWriter,
51    model: &Model,
52    docs: &Docs,
53    cb: &CallbackBinding,
54    cx: Cx<'_>,
55) {
56    let iface = callback_interface_cs(&cb.name);
57    docs.summary(w, &cb.doc);
58    w.line("/// <remarks>The native library may call any method from any thread until");
59    w.line("/// it releases its last reference to the implementation. Span arguments are");
60    w.line("/// valid only during the call; object arguments belong to the implementation.");
61    w.line("/// An exception thrown by a method fails the native call in progress: an");
62    w.line("/// exception of the domain a method documents is reported with its code and");
63    w.line("/// fields, and any other reaches the library as a failure carrying its message.</remarks>");
64    docs.obsolete(w, &cb.deprecated);
65    w.line(format!("public interface {iface}"));
66    w.block("{", "}", |w| {
67        for (i, m) in cb.methods.iter().enumerate() {
68            if i > 0 {
69                w.blank();
70            }
71            docs.summary(w, &m.doc);
72            for p in &m.params {
73                docs.param(w, &camel(&p.name), &p.doc);
74            }
75            if let ErrorStrategy::Domain(_) = &m.error {
76                let err = ErrCtx::new(model, &m.error, cx);
77                if let Some(exc) = &err.domain {
78                    w.line(format!(
79                        "/// <exception cref=\"{exc}\">Reported to the library with its code and fields.</exception>"
80                    ));
81                }
82            }
83            docs.obsolete(w, &m.deprecated);
84            let ret = m.ret.as_ref().map_or_else(|| "void".into(), owned_cs);
85            let sig: Vec<String> = m
86                .params
87                .iter()
88                .map(|p| format!("{} {}", lent_cs(&p.ty), camel(&p.name)))
89                .collect();
90            w.line(format!("{ret} {}({});", method_cs(&m.name), sig.join(", ")));
91        }
92    });
93    w.blank();
94}

96/// The internal class owning the vtable (the header, then function pointers
97/// in declaration order) and the trampolines it points at.
98fn render_vtable_class(w: &mut CodeWriter, model: &Model, cb: &CallbackBinding, cx: Cx<'_>) {
99    let iface = callback_interface_cs(&cb.name);
100    let class = vtable_class_cs(&cb.name);
101    w.line(format!(
102        "/// <summary>The process-wide <c>{}</c> and the trampolines behind it,",
103        cb.vtable_tag
104    ));
105    w.line(format!(
106        "/// adapting an <see cref=\"{iface}\"/> to the C ABI.</summary>"
107    ));
108    w.line(format!("internal static unsafe class {class}"));
109    w.block("{", "}", |w| {
110        w.line("[StructLayout(LayoutKind.Sequential)]");
111        w.line("private struct Layout");
112        w.block("{", "}", |w| {
113            w.line("public uint Size;");
114            w.line("public uint Flags;");
115            w.line("public delegate* unmanaged[Cdecl]<IntPtr, void> Free;");
116            for m in &cb.methods {
117                w.line(format!(
118                    "public {} {};",
119                    fn_pointer_type(cx.ns, &m.abi.params, &m.abi.ret),
120                    safe_cs_name(&m.name)
121                ));
122            }
123        });
124        w.blank();
125        w.line("/// <summary>The vtable's address, valid for the process lifetime.</summary>");
126        w.line("internal static readonly IntPtr Pointer = Allocate();");
127        w.blank();
128        w.line("private static IntPtr Allocate()");
129        w.block("{", "}", |w| {
130            w.line("var vtable = (Layout*)NativeMemory.AllocZeroed((nuint)sizeof(Layout));");
131            w.line("vtable->Size = (uint)sizeof(Layout);");
132            w.line("// Not thread-affine: every method may be called from any thread.");
133            w.line("vtable->Flags = 0;");
134            w.line("vtable->Free = &FfiFree;");
135            for m in &cb.methods {
136                w.line(format!(
137                    "vtable->{} = &{}Trampoline;",
138                    safe_cs_name(&m.name),
139                    method_cs(&m.name)
140                ));
141            }
142            w.line("return (IntPtr)vtable;");
143        });
144        w.blank();
145        w.line(UNMANAGED_CALLERS_ONLY);
146        w.line("private static void FfiFree(IntPtr ctx)");
147        w.block("{", "}", |w| {
148            w.line("Ffi.Unregister(ctx);");
149        });
150        for m in &cb.methods {
151            w.blank();
152            render_trampoline(w, model, cx, &iface, m);
153        }
154    });
155    w.blank();
156}

158/// One trampoline: receive the arguments, call the implementation, and
159/// hand its result back; any exception is reported through `out_err`.
160fn render_trampoline(
161    w: &mut CodeWriter,
162    model: &Model,
163    cx: Cx<'_>,
164    iface: &str,
165    m: &CallbackMethodBinding,
166) {
167    // The slots are `ctx`, one per parameter (named after it), then
168    // `out_err`; the C# names are the escaped C names.
169    let slot_name = |name: &str| format!("ffi_{name}");
170    let sig: Vec<String> = m
171        .abi
172        .params
173        .iter()
174        .map(|s| format!("{} {}", cs_ctype(cx.ns, &s.ty), slot_name(&s.name)))
175        .collect();
176    let ret = cs_ctype(cx.ns, &m.abi.ret);
177    let returns_value = m.ret.is_some();
178    let ctx = slot_name("ctx");
179    let out_err = slot_name("out_err");
180    let err = ErrCtx::new(model, &m.error, cx);
181    let fallback = match m.error {
182        ErrorStrategy::Trap => "ForeignErrorCode",
183        _ => "GenericErrorCode",
184    };
185    let method = method_cs(&m.name);
186    w.line(UNMANAGED_CALLERS_ONLY);
187    w.line(format!(
188        "private static {ret} {method}Trampoline({})",
189        sig.join(", ")
190    ));
191    w.block("{", "}", |w| {
192        w.line("try");
193        w.block("{", "}", |w| {
194            let args: Vec<String> = m
195                .params
196                .iter()
197                .map(|p| borrow(cx, &p.ty, &safe_cs_name(&slot_name(&p.name))))
198                .collect();
199            let call = format!("Ffi.Target<{iface}>({ctx}).{method}({})", args.join(", "));
200            match &m.ret {
201                None => w.line(format!("{call};")),
202                Some(ty) => w.line(format!("return {};", give(ty, &call))),
203            };
204        });
205        if let Some(exc) = &err.domain {
206            w.line(format!("catch ({} e)", cx.ty(exc)));
207            w.block("{", "}", |w| {
208                w.line(format!("Ffi.ReportDomain({out_err}, e);"));
209                if returns_value {
210                    w.line("return default;");
211                }
212            });
213        }
214        w.line("catch (Exception e)");
215        w.block("{", "}", |w| {
216            w.line(format!("Ffi.Report({out_err}, e, {}.{fallback});", cx.base));
217            if returns_value {
218                w.line("return default;");
219            }
220        });
221    });
