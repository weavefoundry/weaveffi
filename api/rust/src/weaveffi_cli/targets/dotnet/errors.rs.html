16/// How a call maps a failure to an exception, rendering its
17/// [`ErrorStrategy`]: a domain's `FromError` (typed codes, open to codes
18/// added later), the root exception's for `throws: any`, or the bug
19/// exception's for a call that can't fail.
20#[derive(Clone)]
21pub(crate) struct ErrCtx {
22    /// The `delegate*` expression of the mapping function.
23    pub map: String,
24    /// The exception a throwing call documents, with its description.
25    doc: Option<(String, String)>,
26    /// The domain exception a callback method reports with its code and
27    /// fields.
28    pub domain: Option<String>,
29}
30
31impl ErrCtx {
32    /// The error context of a callable (or callback method) with `error`.
33    pub(crate) fn new(model: &Model, error: &ErrorStrategy, cx: Cx<'_>) -> Self {
34        match error {
35            ErrorStrategy::Domain(name) => {
36                let exc = domain_exception(model, name);
37                ErrCtx {
38                    map: format!("&{}.FromError", cx.ty(&exc)),
39                    doc: Some((
40                        exc.clone(),
41                        format!("The call failed with a <c>{name}</c> code."),
42                    )),
43                    domain: Some(exc),
44                }
45            }
46            ErrorStrategy::Untyped => ErrCtx {
47                map: format!("&{}.FromError", cx.ty(cx.base)),
48                doc: Some((cx.base.to_string(), "The call failed.".into())),
49                domain: None,
50            },
51            ErrorStrategy::Trap => ErrCtx {
52                map: format!("&{}.FromError", cx.ty(cx.bug)),
53                doc: None,
54                domain: None,
55            },
56        }
57    }
58
59    /// The statement throwing when the local `FfiError` named `var` holds a
60    /// failure.
61    pub(crate) fn check(&self, var: &str) -> String {
62        format!(
63            "if ({var}.Code != 0) throw Ffi.TakeError(&{var}, {});",
64            self.map
65        )
66    }
67
68    /// Emit the `<exception>` doc line of a throwing wrapper.
69    pub(crate) fn write_doc(&self, w: &mut CodeWriter) {
70        if let Some((exc, text)) = &self.doc {
71            w.line(format!("/// <exception cref=\"{exc}\">{text}</exception>"));
72        }
73    }
74}

76/// The exception class of the error domain `name`: its stem with one
77/// `Exception` suffix (`KvError` is `KvException`).
78pub(crate) fn domain_exception(model: &Model, name: &str) -> String {
79    weaveffi_model::errors::exception_type_name(&model.error_domain(name).name)
80}

82/// Render every error domain's exception hierarchy.
83pub(crate) fn render_domains(w: &mut CodeWriter, model: &Model, docs: &Docs, cx: Cx<'_>) {
84    for table in tables(model, "Exception") {
85        render_domain(w, &table, docs, cx);
86    }
87}

89/// One domain: a class deriving from the root exception, with one nested
90/// sealed class per code. Each code class exposes its fields as typed
91/// properties and can be constructed by a callback-interface
92/// implementation, which reports it (code, message, and fields) to the
93/// native caller. The domain class itself stands for a code these bindings
94/// don't know: domains are open, so a library may add codes.
95fn render_domain(w: &mut CodeWriter, table: &ErrorTable<'_>, docs: &Docs, cx: Cx<'_>) {
96    let exc = &table.type_name;
97    let base = cx.base;
98    w.line(format!(
99        "/// <summary>The errors of the <c>{}</c> domain.",
100        table.domain.name
101    ));
102    w.line("/// Catch a nested class for one code, or this class for any of them. A code");
103    w.line("/// these bindings don't know (one the library added later) surfaces as this");
104    w.line(format!(
105        "/// class itself, with its <see cref=\"{base}.Code\"/> and message.</summary>"
106    ));
107    w.line(format!("public class {exc} : {base}"));
108    w.block("{", "}", |w| {
109        w.line(format!(
110            "internal {exc}(int code, string message) : base(code, message)"
111        ));
112        w.line("{");
113        w.line("}");
114        w.blank();
115        for row in &table.codes {
116            let class = pascal(&row.code.name);
117            if row.code.doc.is_some() {
118                docs.summary(w, &row.code.doc);
119            } else {
120                w.line(format!("/// <summary>{}</summary>", xml_text(&row.code.message)));
121            }
122            render_code_class(w, exc, &class, row.code, docs);
123        }
124        w.line("/// <summary>The exception for a raw code and its payload: a typed code");
125        w.line("/// class, this class for a code it doesn't know (or a payload that doesn't");
126        w.line("/// decode), or the root mapping for a runtime code.</summary>");
127        w.line(
128            "internal static new Exception FromError(int code, string message, FfiBufferReader payload)",
129        );
130        w.block("{", "}", |w| {
131            w.line("if (code <= 0)");
132            w.block("{", "}", |w| {
133                w.line(format!(
134                    "return {}.FromError(code, message, payload);",
135                    cx.ty(base)
136                ));
137            });
138            w.line("var given = message.Length == 0 ? null : message;");
139            w.line("try");
140            w.block("{", "}", |w| {
141                w.line("switch (code)");
142                w.block("{", "}", |w| {
143                    for row in &table.codes {
144                        let class = pascal(&row.code.name);
145                        w.line(format!("case {class}.ErrorCode:"));
146                        w.indent();
147                        if row.code.fields.is_empty() {
148                            w.line(format!("return new {class}(given);"));
149                        } else {
150                            let mut args: Vec<String> = row
151                                .code
152                                .fields
153                                .iter()
154                                .map(|f| read_expr(cx, &f.ty, "payload", 0))
155                                .collect();
156                            args.push("given".into());
157                            w.line(format!(
158                                "return payload.End(new {class}({}));",
159                                args.join(", ")
160                            ));
161                        }
162                        w.dedent();
163                    }
164                });
165            });
166            w.line(format!("catch ({} e) when (e.Code == {base}.MarshalErrorCode)", cx.bug));
167            w.block("{", "}", |w| {
168                w.line("// A payload that doesn't decode keeps the code and message.");
169            });
170            w.line(format!("return new {exc}(code, message);"));
171        });
172    });
173    w.blank();
174}

176/// One code's nested exception class.
177fn render_code_class(
178    w: &mut CodeWriter,
179    exc: &str,
180    class: &str,
181    code: &ErrorCodeBinding,
182    docs: &Docs,
183) {
184    let (value, message, fields) = (code.value, &code.message, &code.fields);
185    w.line(format!("public sealed class {class} : {exc}"));
186    w.block("{", "}", |w| {
187        w.line(format!(
188            "/// <summary>The domain code, <c>{value}</c>.</summary>"
189        ));
190        w.line(format!("public const int ErrorCode = {value};"));
191        w.blank();
192        for f in fields {
193            docs.summary(w, &f.doc);
194            w.line(format!(
195                "public {} {} {{ get; }}",
196                cs_type(&f.ty),
197                exception_field_cs(&f.name, class)
198            ));
199            w.blank();
200        }
201        let mut params: Vec<String> = fields
202            .iter()
203            .map(|f| format!("{} {}", cs_type(&f.ty), param_name(f)))
204            .collect();
205        params.push("string? message = null".into());
206        w.line("/// <summary>Creates the error, as a callback-interface implementation");
207        w.line("/// throws it to report this code.</summary>");
208        for f in fields {
209            docs.param(w, &param_name(f), &f.doc);
210        }
211        w.line(format!(
212            "/// <param name=\"message\">The message, or null for the default \"{}\".</param>",
213            xml_text(message)
214        ));
215        w.line(format!(
216            "public {class}({}) : base(ErrorCode, message ?? {})",
217            params.join(", "),
218            default_message(message, fields)
219        ));
220        w.block("{", "}", |w| {
221            for f in fields {
222                w.line(format!(
223                    "{} = {};",
224                    exception_field_cs(&f.name, class),
225                    param_name(f)
226                ));
227            }
228        });
229        if !fields.is_empty() {
230            w.blank();
231            w.line("internal override void WritePayload(FfiBufferWriter writer)");
232            w.block("{", "}", |w| {
233                for f in fields {
234                    let value = exception_field_cs(&f.name, class);
235                    w.line(format!("{};", write_call(&f.ty, "writer", &value, 0)));
236                }
237            });
238        }
239    });
240    w.blank();
241}

243/// The C# expression of a code's default message: the IDL message as a
244/// string literal, or, when it names the code's fields (`key not found:
245/// {key}`), an interpolated string filling them from the constructor's
246/// parameters.
247fn default_message(message: &str, fields: &[FieldBinding]) -> String {
248    let mut out = String::new();
249    let mut interpolated = false;
250    let mut rest = message;
251    while let Some(open) = rest.find('{') {
252        let (before, after) = rest.split_at(open);
253        out.push_str(&cs_str(before).replace('}', "}}"));
254        let field = after[1..]
255            .find('}')
256            .map(|close| &after[1..=close])
257            .and_then(|name| fields.iter().find(|f| f.name == name));
258        match field {
259            Some(f) => {
260                out.push_str(&format!("{{{}}}", param_name(f)));
261                rest = &after[f.name.len() + 2..];
262                interpolated = true;
263            }
264            None => {
265                out.push_str("{{");
266                rest = &after[1..];
267            }
268        }
269    }
270    out.push_str(&cs_str(rest).replace('}', "}}"));
271    if interpolated {
272        format!("$\"{out}\"")
273    } else {
274        format!("\"{}\"", cs_str(message))
275    }
276}
277
278/// A code field's constructor parameter.
279fn param_name(f: &FieldBinding) -> String {
280    let name = f.name.to_lower_camel_case();
281    if name == "message" {
282        "message_".into()
283    } else {
284        safe_cs_name(&name)
285    }
286}

288#[cfg(test)]
289mod tests {
290    use super::*;
291    use weaveffi_model::ty::{Prim, Ty};
292
293    #[test]
294    fn default_messages_fill_in_their_fields() {
295        let field = |name: &str| FieldBinding {
296            name: name.into(),
297            doc: None,
298            ty: Ty::Prim(Prim::String),
299        };
300        let fields = [field("key"), field("class")];
301        assert_eq!(
302            default_message("plain \"x\"", &fields),
303            "\"plain \\\"x\\\"\""
304        );
305        assert_eq!(
306            default_message("key not found: {key}", &fields),
307            "$\"key not found: {key}\""
308        );
309        assert_eq!(
310            default_message("{class} in {other} }", &fields),
311            "$\"{@class} in {{other}} }}\""
312        );
313        assert_eq!(default_message("{other}", &fields), "\"{other}\"");
314    }
