1//! .NET binding generator.
2//!
3//! Emits a standalone C# project over the C ABI, revision 6: a `.csproj`,
4//! the generated API (`{Namespace}.cs`), the fixed runtime (`Runtime.cs`,
5//! from `runtime/Runtime.cs`), and a README. The whole library is one C#
6//! namespace.
7//!
8//! * Native calls use source-generated `[LibraryImport]` declarations over
9//!   blittable slots with runtime marshalling disabled, so the bindings are
10//!   trim- and AOT-friendly. Every value is one C value: a string, bytes, a
11//!   typed array, or a value buffer is an `FfiRun` (`{p}_run`) by value over
12//!   pinned memory, an optional scalar an `FfiOpt*` struct (`{p}_opt_*`),
13//!   a C `bool` one `byte`, an object its `SafeHandle`, and a callback
14//!   interface an `FfiCallback` (`{p}_{Cb}`).
15//! * Every import resolves through a `DllImportResolver` that loads the
16//!   library (honoring `{PREFIX}_LIBRARY`) and, once, checks its ABI
17//!   revision and calls every declaration's contract function; a failure
18//!   is a `NativeLoadException` every call throws, and
19//!   `{Namespace}Library.Check()` runs the check up front.
20//! * Free functions are static methods of `{Namespace}Library`, the static
21//!   class that also holds `Check()` and `AbiVersion`.
22//! * Records are positional `sealed record`s whose list and map fields are
23//!   `IReadOnlyList<T>`/`IReadOnlyDictionary<K, V>`, compared by content;
24//!   rich enums are closed record hierarchies. Values are encoded by
25//!   per-type `WriteTo`/`ReadFrom` pairs and the runtime's generic composite
26//!   codec, with no per-shape code.
27//! * Interfaces are sealed `IDisposable` wrappers over a `SafeHandle`
28//!   subclass whose `ReleaseHandle` calls `_destroy`. Every import takes the
29//!   handle itself, so the interop stub keeps the object alive for the call.
30//! * A call's [`ErrorStrategy`](weaveffi_model::model::ErrorStrategy) picks
31//!   its exceptions: a domain's hierarchy (one nested class per code; an
32//!   unknown code is the domain class itself), the root `NativeException`
33//!   for `throws: any` and runtime codes, or `NativeBugException` for a call
34//!   that can't fail. Cancellation raises `OperationCanceledException`.
35//! * Callback interfaces are C# `interface`s; a passed implementation is
36//!   pinned in a `GCHandle` and paired with one static vtable of
37//!   `[UnmanagedCallersOnly]` trampolines.
38//! * Async functions return `Task`/`Task<T>` completed by an
39//!   `[UnmanagedCallersOnly]` completion and take a `CancellationToken`
40//!   (linked to the native cancel token when the function is cancellable).
41//!   `iter<T>` returns re-enumerable `IEnumerable<T>`s, one native iterator
42//!   per enumeration.

78/// Per-target configuration for [`DotnetGenerator`].
79#[derive(Debug, Clone, Default, Serialize, Deserialize)]
80#[serde(default, deny_unknown_fields)]
81pub struct DotnetConfig {
82    /// C# namespace, assembly name, and NuGet id. Defaults to the package
83    /// name in PascalCase (`my-kv` becomes `MyKv`).
84    pub name: Option<String>,
85}
86
87impl DotnetConfig {
88    /// The C# namespace: the configured one, else `PascalCase(name)`.
89    pub fn namespace(&self, model: &Model) -> String {
90        self.name
91            .clone()
92            .unwrap_or_else(|| model.identity.pascal_name())
93    }
94}

107/// The text files of a generated project, at `dir`, loading `library`.
108fn project_files(
109    model: &Model,
110    dir: &Utf8Path,
111    config: &DotnetConfig,
112    library: &str,
113    ctx: Option<&PackageContext>,
114) -> Vec<(Utf8PathBuf, String)> {
115    let namespace = config.namespace(model);
116    let project = Project {
117        identity: &model.identity,
118        namespace: &namespace,
119        library,
120    };
121    let names = RuntimeNames::new(model, &namespace);
122    let cs = format!("{namespace}.cs");
123    let csproj = format!("{namespace}.csproj");
124    let assets = if ctx.is_some() { NATIVE_ASSETS } else { "" };
125    vec![
126        (dir.join(&cs), render_csharp(model, &namespace, &names, &cs)),
127        (
128            dir.join("Runtime.cs"),
129            render_runtime(model, &namespace, &names, library, "Runtime.cs"),
130        ),
131        (dir.join(&csproj), render_csproj(&project, &csproj, assets)),
132        (
133            dir.join("README.md"),
134            render_readme(&project, &names.library_class, ctx),
135        ),
136    ]
137}

139impl Target for DotnetGenerator {
140    fn name(&self) -> &'static str {
141        "dotnet"
142    }
143
144    fn fixed_files(&self) -> &'static [&'static str] {
145        &["*.csproj", "README.md", "Runtime.cs"]
146    }
147
148    /// Run `dotnet pack` over the written project, producing the NuGet
149    /// package under `dotnet/`.
150    fn finish_package(
151        &self,
152        dist: &Utf8Path,
153        artifacts: &[Artifact],
154        ctx: &PackageContext<'_>,
155    ) -> Result<Vec<Utf8PathBuf>> {
156        pack::dotnet_pack(dist, artifacts, ctx)
157    }
158
159    fn render(&self, model: &Model) -> Vec<OutputFile> {
160        let library = &model.identity.library;
161        project_files(model, &Utf8PathBuf::new(), &self.config, library, None)
162            .into_iter()
163            .map(|(path, text)| OutputFile::new(path, text))
164            .collect()
165    }
166
167    /// The NuGet-ready project at `dotnet/{Namespace}/`, with every desktop
168    /// library under the `runtimes/<rid>/native/` layout NuGet resolves at
169    /// restore time; `weaveffi package` then runs `dotnet pack` on it.
170    /// Platforms without a RID (Android, iOS, `wasm32`) have no slot.
171    fn package(&self, model: &Model, ctx: &PackageContext<'_>) -> Result<Vec<Artifact>> {
172        let config = &self.config;
173        let library = &ctx.binaries.lib_name;
174        let mut natives = Vec::new();
175        for nb in &ctx.binaries.binaries {
176            let Some(rid) = nb.platform.nuget_rid() else {
177                continue;
178            };
179            natives.push(PackagedFile::copy(
180                format!(
181                    "runtimes/{rid}/native/{}",
182                    ctx.binaries.bundled_filename(nb.platform)
183                ),
184                nb.library.clone(),
185            ));
186        }
187        if natives.is_empty() {
188            return Ok(Vec::new());
189        }
190        let mut files: Vec<PackagedFile> =
191            project_files(model, Utf8Path::new(""), config, library, Some(ctx))
192                .into_iter()
193                .map(|(path, text)| PackagedFile::text(path, text))
194                .collect();
195        files.extend(natives);
196        Ok(vec![Artifact::directory(
197            format!("dotnet/{}", config.namespace(model)),
198            files,
199        )])
200    }
201}

203/// Render the generated API file: typed exceptions, the entities, the free
204/// functions (in the library class), and the generated half of
205/// `NativeMethods`.
206pub(crate) fn render_csharp(
207    model: &Model,
208    namespace: &str,
209    names: &RuntimeNames,
210    filename: &str,
211) -> String {
212    let cx = Cx {
213        ns: namespace,
214        base: &names.exception,
215        bug: &names.bug_exception,
216    };
217    let docs = Docs::new(model);
218    let mut w = CodeWriter::four_space();
219    w.line("#nullable enable");
220    w.blank();
221    for u in [
222        "System",
223        "System.Collections.Generic",
224        "System.Runtime.CompilerServices",
225        "System.Runtime.InteropServices",
226        "System.Threading",
227        "System.Threading.Tasks",
228    ] {
229        w.line(format!("using {u};"));
230    }
231    w.blank();
232    w.line(format!("namespace {namespace};"));
233    w.blank();
234
235    render_domains(&mut w, model, &docs, cx);
236    for e in &model.enums {
237        if e.is_rich() {
238            render_rich_enum(&mut w, cx, &docs, e);
239        } else {
240            render_enum(&mut w, &docs, e);
241        }
242    }
243    for s in &model.records {
244        render_record(&mut w, cx, &docs, s);
245    }
246    for cb in &model.callbacks {
247        render_callback_interface(&mut w, model, &docs, cb, cx);
248    }
249    for i in &model.interfaces {
250        render_interface(&mut w, model, &docs, i, cx);
251    }
252    render_functions(&mut w, model, &docs, &names.library_class, cx);
253    render_native_methods(&mut w, model, namespace);
254
255    let mut out = render_prelude(CommentStyle::DoubleSlash);
256    out.push_str(&tidy(&w.finish()));
257    out.push_str(&render_trailer(CommentStyle::DoubleSlash, filename));
258    out
259}

261/// Drop blank lines directly before a closing brace.
262fn tidy(text: &str) -> String {
263    let lines: Vec<&str> = text.lines().collect();
264    let mut out = String::with_capacity(text.len());
265    for (i, line) in lines.iter().enumerate() {
266        let next_closes = lines
267            .get(i + 1)
268            .is_some_and(|n| n.trim_start().starts_with('}'));
269        if line.is_empty() && next_closes {
270            continue;
271        }
272        out.push_str(line);
273        out.push('\n');
274    }
275    out
276}
