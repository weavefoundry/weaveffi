1//! Go type mapping: how value types, parameter and return types, zero
2//! values, and C ABI types are spelled in the generated package.
3
4use weaveffi_model::abi::CType;
5use weaveffi_model::ty::{ParamTy, Prim, RetTy, Ty};
6
7use crate::targets::go::names::pascal;
8
9/// The Go type spelling of a value type.
10///
11/// Records are value structs, rich enums sealed interfaces, and interfaces
12/// wrapper pointers (one Go value owns each strong reference). `T?` is `*T`
13/// unless `T` is already nil-able (see [`optional_derefs`]); that holds for
14/// optional scalars too, whichever way they cross.
15pub(crate) fn go_type(ty: &Ty) -> String {
16    match ty {
17        Ty::Prim(p) => prim_type(*p).into(),
18        Ty::Record(n) | Ty::RichEnum(n) | Ty::Enum(n) => pascal(n),
19        Ty::Interface(n) => format!("*{}", pascal(n)),
20        Ty::Optional(inner) if optional_derefs(inner) => format!("*{}", go_type(inner)),
21        Ty::Optional(inner) => go_type(inner),
22        Ty::List(inner) => format!("[]{}", go_type(inner)),
23        Ty::Map(k, v) => format!("map[{}]{}", go_type(k), go_type(v)),
24    }
25}
26
27/// The Go type of a parameter: a value type, or the Go interface a
28/// callback implementation satisfies (nil for an absent optional one).
29pub(crate) fn param_type(ty: &ParamTy) -> String {
30    match ty {
31        ParamTy::Value(t) => go_type(t),
32        ParamTy::Callback { name, .. } => pascal(name),
33    }
34}
35
36/// The Go type a wrapper returns for `ret` (before any `error`): a value
37/// type, or a lazy sequence of the element type.
38pub(crate) fn ret_type(ret: &RetTy, throws: bool) -> String {
39    match ret {
40        RetTy::Value(t) => go_type(t),
41        RetTy::Iterator(t) if throws => format!("iter.Seq2[{}, error]", go_type(t)),
42        RetTy::Iterator(t) => format!("iter.Seq[{}]", go_type(t)),
43    }
44}

46/// The Go type of a primitive.
47pub(crate) fn prim_type(p: Prim) -> &'static str {
48    match p {
49        Prim::I8 => "int8",
50        Prim::I16 => "int16",
51        Prim::I32 => "int32",
52        Prim::I64 => "int64",
53        Prim::U8 => "uint8",
54        Prim::U16 => "uint16",
55        Prim::U32 => "uint32",
56        Prim::U64 => "uint64",
57        Prim::F32 => "float32",
58        Prim::F64 => "float64",
59        Prim::Bool => "bool",
60        Prim::String => "string",
61        Prim::Bytes => "[]byte",
62    }
63}

65/// `true` when `T?` surfaces as `*T` in Go. Types that are already
66/// nil-able (rich enums, slices, maps, byte slices, and wrapper pointers)
67/// use nil as the none marker instead.
68pub(crate) fn optional_derefs(inner: &Ty) -> bool {
69    !matches!(
70        inner,
71        Ty::RichEnum(_) | Ty::List(_) | Ty::Map(_, _) | Ty::Prim(Prim::Bytes) | Ty::Interface(_)
72    )
73}
74
75/// The Go zero-value expression of a type, returned on an error path.
76pub(crate) fn go_zero(ty: &Ty) -> String {
77    match ty {
78        Ty::Prim(Prim::Bool) => "false".into(),
79        Ty::Prim(Prim::String) => "\"\"".into(),
80        Ty::Prim(Prim::Bytes) => "nil".into(),
81        Ty::Prim(_) | Ty::Enum(_) => "0".into(),
82        Ty::Record(n) => format!("{}{{}}", pascal(n)),
83        _ => "nil".into(),
84    }
85}

87/// The cgo spelling of a C scalar (`C.int32_t`, `C.double`, `C._Bool`).
88pub(crate) fn cgo_scalar(p: Prim) -> String {
89    match p {
90        Prim::Bool => "C._Bool".into(),
91        other => format!("C.{}", CType::scalar_c(other)),
92    }
93}
94
95/// The cgo spelling of one C ABI type: `C.int32_t`, `C.kv_run`,
96/// `*C.kv_Store` (cgo drops `const`), and `unsafe.Pointer` for `void*`.
97pub(crate) fn cgo_type(ct: &CType, prefix: &str) -> String {
98    match ct {
99        CType::Scalar(p) => cgo_scalar(*p),
100        CType::Size => "C.size_t".into(),
101        CType::Ptr { pointee, .. } if **pointee == CType::Void => "unsafe.Pointer".into(),
102        CType::Ptr { pointee, .. } => format!("*{}", cgo_type(pointee, prefix)),
103        CType::Void => unreachable!("a void slot is never spelled in Go"),
104        named => format!("C.{}", named.render_c(prefix)),
105    }
106}

108/// Quote `s` as a Go string literal.
109pub(crate) fn go_str(s: &str) -> String {
110    let mut out = String::with_capacity(s.len() + 2);
111    out.push('"');
112    for ch in s.chars() {
113        match ch {
114            '\\' => out.push_str("\\\\"),
115            '"' => out.push_str("\\\""),
116            '\n' => out.push_str("\\n"),
117            '\t' => out.push_str("\\t"),
118            _ => out.push(ch),
119        }
120    }
121    out.push('"');
122    out
123}
