18/// One `name rest` line inside a struct or const block, with its doc
19/// comment, for [`emit_aligned`].
20struct AlignedEntry {
21    doc: Option<String>,
22    name: String,
23    rest: String,
24}
25
26/// Emit `entries` as the body of a struct or const block the way gofmt lays
27/// it out: gofmt aligns the second column of consecutive lines and starts a
28/// fresh column after a comment line, so each run of entries following a
29/// documented one (or the start) is padded to the widest name in that run.
30fn emit_aligned(w: &mut CodeWriter, entries: Vec<AlignedEntry>) {
31    let mut i = 0;
32    while i < entries.len() {
33        let mut j = i + 1;
34        while j < entries.len() && entries[j].doc.is_none() {
35            j += 1;
36        }
37        let width = entries[i..j]
38            .iter()
39            .map(|e| e.name.len())
40            .max()
41            .unwrap_or(0);
42        for e in &entries[i..j] {
43            GoDoc::plain(e.doc.clone()).emit(w);
44            w.line(format!("{:<width$} {}", e.name, e.rest));
45        }
46        i = j;
47    }
48}

50/// The struct-body entries of `fields`, named by `name`.
51fn field_entries(
52    ctx: &Ctx,
53    fields: &[FieldBinding],
54    name: fn(&str) -> String,
55) -> Vec<AlignedEntry> {
56    fields
57        .iter()
58        .map(|f| AlignedEntry {
59            doc: ctx.docs.of(&f.doc, &None).0,
60            name: name(&f.name),
61            rest: go_type(&f.ty),
62        })
63        .collect()
64}

66/// Render a C-style enum: an `int32` type with one constant per variant,
67/// plus its codec pair when it appears inside a buffer.
68pub(crate) fn render_enum(w: &mut CodeWriter, ctx: &Ctx, e: &EnumBinding) {
69    let name = pascal(&e.name);
70    let (doc, deprecated) = ctx.docs.of(&e.doc, &e.deprecated);
71    GoDoc::decl(&name, doc).deprecated(deprecated).emit(w);
72    w.line(format!("type {name} int32"));
73    w.blank();
74    w.block("const (", ")", |w| {
75        let entries = e
76            .variants
77            .iter()
78            .map(|v| AlignedEntry {
79                doc: ctx.docs.of(&v.doc, &None).0,
80                name: format!("{name}{}", pascal(&v.name)),
81                rest: format!("{name} = {}", v.value),
82            })
83            .collect();
84        emit_aligned(w, entries);
85    });
86    w.blank();
87    if ctx.codecs.has_enum(&e.name) {
88        func(
89            w,
90            &format!("func wvWrite{name}(w *wvWriter, v {name})"),
91            "w.writeI32(int32(v))",
92        );
93        func(
94            w,
95            &format!("func wvRead{name}(r *wvReader) {name}"),
96            &format!("return {name}(r.readI32())"),
97        );
98    }
99}

101/// Render a rich enum as a sealed interface (`type Shape interface {
102/// isShape() }`) with one struct per variant (`ShapeCircle`), plus its codec
103/// pair: the `i32` tag, then the variant's fields in order.
104pub(crate) fn render_rich_enum(w: &mut CodeWriter, ctx: &Ctx, e: &EnumBinding) {
105    let package = ctx.package;
106    let name = pascal(&e.name);
107    let (doc, deprecated) = ctx.docs.of(&e.doc, &e.deprecated);
108    GoDoc::decl(&name, doc)
109        .para(&format!(
110            "{name} is a sealed sum type: exactly one of its variant structs is the \
111             value at a time."
112        ))
113        .deprecated(deprecated)
114        .emit(w);
115    w.block(format!("type {name} interface {{"), "}", |w| {
116        w.line(format!("is{name}()"));
117    });
118    w.blank();
119
120    for v in &e.variants {
121        let vn = format!("{name}{}", pascal(&v.name));
122        let text = ctx.docs.of(&v.doc, &None).0;
123        let doc = match text {
124            Some(_) => GoDoc::decl(&vn, text),
125            None => GoDoc::plain(None).para(&format!("{vn} is the {} variant of {name}.", v.name)),
126        };
127        doc.emit(w);
128        if v.fields.is_empty() {
129            w.line(format!("type {vn} struct{{}}"));
130        } else {
131            w.block(format!("type {vn} struct {{"), "}", |w| {
132                emit_aligned(w, field_entries(ctx, &v.fields, names::field));
133            });
134        }
135        w.blank();
136        w.line(format!("func ({vn}) is{name}() {{}}"));
137        w.blank();
138    }
139
140    let any_fields = e.variants.iter().any(|v| !v.fields.is_empty());
141    let switch = if any_fields {
142        "switch x := v.(type) {"
143    } else {
144        "switch v.(type) {"
145    };
146    w.block(
147        format!("func wvWrite{name}(w *wvWriter, v {name}) {{"),
148        "}",
149        |w| {
150            w.line(switch);
151            for v in &e.variants {
152                w.line(format!("case {name}{}:", pascal(&v.name)));
153                w.scope(|w| {
154                    w.line(format!("w.writeI32({})", v.value));
155                    for f in &v.fields {
156                        let expr = format!("x.{}", names::field(&f.name));
157                        w.line(write_stmt("w", &expr, &f.ty));
158                    }
159                });
160            }
161            w.line("default:");
162            w.scope(|w| {
163                w.line(format!(
164                    "panic(\"{package}: {name} value is not one of its variants\")"
165                ));
166            });
167            w.line("}");
168        },
169    );
170    w.blank();
171    w.block(
172        format!("func wvRead{name}(r *wvReader) {name} {{"),
173        "}",
174        |w| {
175            w.line("switch r.readI32() {");
176            for v in &e.variants {
177                let vn = format!("{name}{}", pascal(&v.name));
178                w.line(format!("case {}:", v.value));
179                w.scope(|w| {
180                    if v.fields.is_empty() {
181                        w.line(format!("return {vn}{{}}"));
182                    } else {
183                        w.line(format!("var x {vn}"));
184                        for f in &v.fields {
185                            w.line(format!(
186                                "x.{} = {}",
187                                names::field(&f.name),
188                                read_expr("r", &f.ty)
189                            ));
190                        }
191                        w.line("return x");
192                    }
193                });
194            }
195            w.line("}");
196            w.line(format!("wvMalformed(\"{name} tag out of range\")"));
197            w.line("return nil");
198        },
199    );
200    w.blank();
201}

203/// Render a record as a value struct with exported fields, plus its codec
204/// pair: the fields in declaration order.
205pub(crate) fn render_record(w: &mut CodeWriter, ctx: &Ctx, s: &RecordBinding) {
206    let name = pascal(&s.name);
207    let (doc, deprecated) = ctx.docs.of(&s.doc, &s.deprecated);
208    GoDoc::decl(&name, doc).deprecated(deprecated).emit(w);
209    w.block(format!("type {name} struct {{"), "}", |w| {
210        emit_aligned(w, field_entries(ctx, &s.fields, names::field));
211    });
212    w.blank();
213    w.block(
214        format!("func wvWrite{name}(w *wvWriter, v {name}) {{"),
215        "}",
216        |w| {
217            for f in &s.fields {
218                let expr = format!("v.{}", names::field(&f.name));
219                w.line(write_stmt("w", &expr, &f.ty));
220            }
221        },
222    );
223    w.blank();
224    w.block(
225        format!("func wvRead{name}(r *wvReader) {name} {{"),
226        "}",
227        |w| {
228            w.line(format!("var v {name}"));
229            for f in &s.fields {
230                w.line(format!(
231                    "v.{} = {}",
232                    names::field(&f.name),
233                    read_expr("r", &f.ty)
234                ));
235            }
236            w.line("return v");
237        },
238    );
239    w.blank();
240}

242/// Render an error domain: the sealed interface every code type implements
243/// (`KitchenError`), one `*{Code}Error` struct per code (its payload
244/// fields, a `Message`, `Error`, and `Code`), the `*Unknown{Domain}` type of
245/// a code these bindings don't declare, and the helper mapping a failure
246/// onto the domain (an `*Error` for a runtime code).
247pub(crate) fn render_error(w: &mut CodeWriter, ctx: &Ctx, e: &ErrorBinding) {
248    let gn = ctx.names;
249    let d = gn.domain_of(e);
250    let domain = &d.iface;
251    let marker = format!("is{domain}");
252    let codes: Vec<String> = e.codes.iter().map(|c| format!("*{}", gn.code(c))).collect();
253    GoDoc::plain(None)
254        .para(&format!(
255            "{domain} is an error of the {} domain: {}, or *{} for a code these \
256             bindings don't declare.",
257            e.name,
258            codes.join(", "),
259            d.unknown
260        ))
261        .para(&format!(
262            "Match one code with errors.As and its type, or any code of the domain \
263             with a {domain}. A callback method that declares the domain returns \
264             one to report that code."
265        ))
266        .emit(w);
267    w.block(format!("type {domain} interface {{"), "}", |w| {
268        w.line("error");
269        w.line("// Code returns the error's numeric code.");
270        w.line("Code() int32");
271        w.line(format!("{marker}()"));
272    });
273    w.blank();
274
275    for c in &e.codes {
276        let ty = gn.code(c);
277        let text = ctx
278            .docs
279            .of(&c.doc, &None)
280            .0
281            .filter(|d| d.trim() != c.message.trim());
282        let doc = match text {
283            Some(_) => GoDoc::decl(ty, text),
284            None => GoDoc::plain(None).para(&format!(
285                "{ty} is the {domain} code {} ({}): {}.",
286                c.name,
287                c.value,
288                c.message.trim_end_matches('.')
289            )),
290        };
291        doc.emit(w);
292        w.block(format!("type {ty} struct {{"), "}", |w| {
293            emit_aligned(w, field_entries(ctx, &c.fields, names::error_field));
294            w.line("// Message describes the failure; when it's empty, Error returns the");
295            w.line(format!(
296                "// code's default message, {}.",
297                go_str(&c.message)
298            ));
299            w.line("Message string");
300        });
301        w.blank();
302        w.line("// Error returns the error's message.");
303        w.block(format!("func (e *{ty}) Error() string {{"), "}", |w| {
304            w.block("if e.Message == \"\" {", "}", |w| {
305                w.line(format!("return {}", go_str(&c.message)));
306            });
307            w.line("return e.Message");
308        });
309        w.blank();
310        w.line(format!("// Code returns {}.", c.value));
311        func(
312            w,
313            &format!("func (*{ty}) Code() int32"),
314            &format!("return {}", c.value),
315        );
316        w.line(format!("func (*{ty}) {marker}() {{}}"));
317        w.blank();
318        if c.fields.is_empty() {
319            func(w, &format!("func (*{ty}) wvPayload() []byte"), "return nil");
320        } else {
321            w.block(format!("func (e *{ty}) wvPayload() []byte {{"), "}", |w| {
322                w.line("w := &wvWriter{}");
323                for f in &c.fields {
324                    let expr = format!("e.{}", names::error_field(&f.name));
325                    w.line(write_stmt("w", &expr, &f.ty));
326                }
327                w.line("return w.buf");
328            });
329            w.blank();
330        }
331    }
332
333    let unknown = &d.unknown;
334    GoDoc::plain(None)
335        .para(&format!(
336            "{unknown} is a {domain} code these bindings don't declare, which a \
337             newer library may report."
338        ))
339        .emit(w);
340    w.block(format!("type {unknown} struct {{"), "}", |w| {
341        w.line("code int32");
342        w.line("// Message is the library's message.");
343        w.line("Message string");
344    });
345    w.blank();
346    w.line("// Error returns the error's message.");
347    w.block(format!("func (e *{unknown}) Error() string {{"), "}", |w| {
348        w.block("if e.Message == \"\" {", "}", |w| {
349            w.line(format!("return wvCodeMessage({}, e.code)", go_str(&e.name)));
350        });
351        w.line("return e.Message");
352    });
353    w.blank();
354    w.line("// Code returns the code the library reported.");
355    func(
356        w,
357        &format!("func (e *{unknown}) Code() int32"),
358        "return e.code",
359    );
360    w.line(format!("func (*{unknown}) {marker}() {{}}"));
361    w.blank();
362    func(
363        w,
364        &format!("func (*{unknown}) wvPayload() []byte"),
365        "return nil",
366    );
367
368    w.block(
369        format!("func {}(f wvFailure) error {{", d.mapper),
370        "}",
371        |w| {
372            w.line("switch f.code {");
373            for c in &e.codes {
374                let ty = gn.code(c);
375                w.line(format!("case {}:", c.value));
376                w.scope(|w| {
377                    if c.fields.is_empty() {
378                        w.line(format!("return &{ty}{{Message: f.message}}"));
379                        return;
380                    }
381                    w.line(format!("e := &{ty}{{Message: f.message}}"));
382                    w.block("if f.payload != nil {", "}", |w| {
383                        w.line("r := &wvReader{buf: f.payload}");
384                        for f in &c.fields {
385                            w.line(format!(
386                                "e.{} = {}",
387                                names::error_field(&f.name),
388                                read_expr("r", &f.ty)
389                            ));
390                        }
391                        w.line("r.expectEnd()");
392                    });
393                    w.line("return e");
394                });
395            }
396            w.line("}");
397            w.block("if f.code > 0 {", "}", |w| {
398                w.line(format!(
399                    "return &{unknown}{{code: f.code, Message: f.message}}"
400                ));
401            });
402            w.line("return f.err()");
403        },
404    );
405    w.blank();
406}

408/// Render an interface as a reference-counted object wrapper embedding a
409/// `wvObject`, which owns one strong reference released by `Close`
410/// (idempotent, and safe to race with in-flight calls) or by a cleanup once
411/// the wrapper is unreachable; its codec pair when it appears inside a
412/// buffer (an object token carrying a fresh reference); and its members:
413/// constructors as factories (`OpenStore`), methods on the wrapper, and
414/// statics prefixed with the type (`StoreDefaultCapacity`).
415pub(crate) fn render_interface(w: &mut CodeWriter, ctx: &Ctx, iface: &InterfaceBinding) {
416    let package = ctx.package;
417    let name = pascal(&iface.name);
418    let c_tag = &iface.c_tag;
419    let (doc, deprecated) = ctx.docs.of(&iface.doc, &iface.deprecated);
420    let doc = match doc {
421        Some(_) => GoDoc::decl(&name, doc),
422        None => GoDoc::plain(None).para(&format!(
423            "{name} is a reference-counted object owned by the native library."
424        )),
425    };
426    doc.para(
427        "Each wrapper holds one strong reference to the object. Close releases \
428         it; a wrapper that's never closed releases it some time after it becomes \
429         unreachable.",
430    )
431    .deprecated(deprecated)
432    .emit(w);
433    w.block(format!("type {name} struct {{"), "}", |w| {
434        w.line("wvObject");
435    });
436    w.blank();
437
438    func(
439        w,
440        &format!("func wvDestroy{name}(ptr unsafe.Pointer)"),
441        &format!("C.{}((*C.{c_tag})(ptr))", iface.destroy_symbol),
442    );
443
444    w.line(format!(
445        "// wvAdopt{name} wraps one owned strong reference. A null pointer adopts to nil."
446    ));
447    w.block(
448        format!("func wvAdopt{name}(ptr *C.{c_tag}) *{name} {{"),
449        "}",
450        |w| {
451            w.block("if ptr == nil {", "}", |w| {
452                w.line("return nil");
453            });
454            w.line(format!("s := &{name}{{}}"));
455            w.line(format!("s.adopt(unsafe.Pointer(ptr), wvDestroy{name})"));
456            w.line("return s");
457        },
458    );
459    w.blank();
460
461    w.line("// native borrows the wrapper's pointer for one call; pair it with a");
462    w.line("// deferred s.release().");
463    w.block(
464        format!("func (s *{name}) native() *C.{c_tag} {{"),
465        "}",
466        |w| {
467            w.block("if s == nil {", "}", |w| {
468                w.line(format!("panic(\"{package}: nil *{name}\")"));
469            });
470            w.line(format!("return (*C.{c_tag})(s.acquire(\"{name}\"))"));
471        },
472    );
473    w.blank();
474
475    let in_buffers = ctx.codecs.has_interface(&iface.name);
476    let returned = ctx.model.callbacks.iter().any(|cb| {
477        cb.methods.iter().any(|m| {
478            m.ret.as_ref().is_some_and(|ty| {
479                matches!(crossing(ty), Crossing::Object { interface, .. } if interface == iface.name)
480            })
481        })
482    });
483    if in_buffers || returned {
484        w.line("// share returns a new strong reference to the wrapper's object for the");
485        w.line("// native library to adopt (the wrapper keeps its own), or null for nil.");
486        w.block(
487            format!("func (s *{name}) share() *C.{c_tag} {{"),
488            "}",
489            |w| {
490                w.block("if s == nil {", "}", |w| {
491                    w.line("return nil");
492                });
493                w.line("ptr := s.native()");
494                w.line("defer s.release()");
495                w.line(format!("return C.{}(ptr)", iface.clone_symbol));
496            },
497        );
498        w.blank();
499    }
500
501    if in_buffers {
502        func(
503            w,
504            &format!("func wvWrite{name}(w *wvWriter, v *{name})"),
505            "w.writeU64(uint64(uintptr(unsafe.Pointer(v.share()))))",
506        );
507        w.block(
508            format!("func wvRead{name}(r *wvReader) *{name} {{"),
509            "}",
510            |w| {
511                w.line("token := r.readU64()");
512                w.block("if token == 0 {", "}", |w| {
513                    w.line("wvMalformed(\"null object token\")");
514                });
515                w.line(format!(
516                    "return wvAdopt{name}((*C.{c_tag})(C.wvHandlePtr(C.uintptr_t(token))))"
517                ));
518            },
519        );
520        w.blank();
521    }
522
523    for c in &iface.constructors {
524        render_function(w, ctx, c, &constructor(&name, c), None);
525    }
526    for f in &iface.methods {
527        let go_name = names::method(&f.name);
528        render_function(w, ctx, f, &go_name, Some(Receiver { ty: &name }));
529    }
530    for f in &iface.statics {
531        render_function(w, ctx, f, ctx.names.function(f), None);
532    }
