1//! Rendering of the `{library}.h` header.
2//!
3//! The declarations come from [`crate::cabi`]; this module adds the framing
4//! (include guard, includes, visibility macros, and the calling-convention
5//! comment).
6
7use std::fmt::Write;
8
9use crate::cabi::{self, Decls};
10use crate::utils::{render_prelude, render_trailer, CommentStyle};
11use weaveffi_model::model::Model;
12
13/// Render the complete header for `model`; `filename` is the header's own
14/// name, which its trailer and the convention comment mention.
15pub(crate) fn render_c_header_from_model(model: &Model, filename: &str) -> String {
16    let prefix = model.prefix();
17    let guard = format!("{}_H", prefix.to_uppercase());
18    let decls = Decls::new(model);
19    let mut out = String::with_capacity(4096 + model.callables().count() * 512);
20    out.push_str(&render_prelude(CommentStyle::DoubleSlash));
21    let _ = write!(out, "#ifndef {guard}\n#define {guard}\n\n");
22    out.push_str("#include <stdint.h>\n");
23    out.push_str("#include <stddef.h>\n");
24    out.push_str("#include <stdbool.h>\n\n");
25    cabi::render_visibility_macros(&mut out, prefix);
26    out.push_str("#ifdef __cplusplus\nextern \"C\" {\n#endif\n\n");
27    decls.runtime(&mut out);
28    let helper = format!(
29        "{}_buffer.h",
30        filename.strip_suffix(".h").unwrap_or(filename)
31    );
32    let _ = write!(
33        out,
34        "/*\n \
35         * How values cross. Every value is one C value in every position\n \
36         * (parameter, return, async result, iterator item, callback argument):\n \
37         *   integer, float, bool, C-style enum   the value (an enum is int32_t)\n \
38         *   optional of one of those (T?)       {prefix}_opt_T: {{ bool has; T value; }}\n \
39         *   string, bytes, [i8] .. [f64]        {prefix}_run: {{ ptr, len }}, len in bytes\n \
40         *   (typed arrays), and every record,   (typed arrays are aligned for their\n \
41         *   rich enum, list, map, optional      element); the rest is a value buffer\n \
42         *   interface                           const Tag* as a parameter (borrowed),\n \
43         *                                       Tag* elsewhere (one reference moves)\n \
44         *   callback interface                  {prefix}_Cb: {{ ctx, vtable }}\n \
45         * Strings are UTF-8 without a NUL terminator; NULL with length 0 is the\n \
46         * empty run. When generated, {helper} declares a C struct and codec for\n \
47         * every value-buffer type. A run the producer returns is released with\n \
48         * {prefix}_run_free, and every one it allocates is 8-aligned. Every sync\n \
49         * call ends with `{prefix}_error* out_err`, which it overwrites.\n \
50         */\n\n"
51    );
52    decls.declarations(&mut out);
53    out.push_str("\n#ifdef __cplusplus\n}\n#endif\n\n");
54    let _ = write!(out, "#endif // {guard}\n\n");
55    out.push_str(&render_trailer(CommentStyle::DoubleSlash, filename));
56    out
57}
