1//! The C ABI runtime (revision 6) the producer macros expand against: the
2//! error struct, runs and optional scalars, the value layer, objects,
3//! callback vtables, iterators, cancel tokens, the async executor hook, and
4//! the compile-time metadata builder.
5//!
6//! Producers never name these items: the macros emit
7//! `::weaveffi::__rt::...`, so every `unsafe` pointer operation a producer
8//! performs has one audited home here. The module is hidden and isn't part
9//! of the crate's semver surface. The normative description of the contract
10//! it implements is `docs/src/reference/abi.md` in the WeaveFFI repository.
11#![allow(unsafe_code)]
12

26/// A raw pointer a test moves to another thread. The handles the tests
27/// share are thread-safe; the wrapper only lets the pointer cross, keeping
28/// its provenance (unlike a round trip through `usize`).
29#[cfg(test)]
30pub(crate) struct ThreadPtr<P>(pub(crate) P);
31
32#[cfg(test)]
33// SAFETY: only thread-safe handles are wrapped.
34unsafe impl<P> Send for ThreadPtr<P> {}
35
36#[cfg(test)]
37impl<P: Copy> ThreadPtr<P> {
38    /// The pointer. A method, so a closure captures the whole wrapper.
39    pub(crate) fn get(&self) -> P {
40        self.0
41    }
42}

74/// The revision of the WeaveFFI C ABI this runtime implements.
75///
76/// Every producer exports it as `{prefix}_abi_version()`, and every
77/// generated consumer compares it with the revision it was generated
78/// against before any other call. It changes only when the runtime surface
79/// or a calling convention changes incompatibly, independently of the crate
80/// and IDL schema versions.
81///
82/// Revision 6 passes every value as exactly one C value in every position
83/// (a scalar, a `{prefix}_opt_*` struct, a `{prefix}_run`, or an object
84/// pointer), passes a callback-interface argument as one struct of its
85/// context and vtable, gives every declaration its own contract function,
86/// and accepts vtables shorter than the producer's.
87pub const ABI_VERSION: u32 = 6;
