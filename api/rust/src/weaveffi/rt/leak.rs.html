1//! Debug leak counters behind the `leak-check` cargo feature.
2//!
3//! With the feature on, the runtime counts every resource it hands to a
4//! consumer and every release, so a test harness can assert that a consumer
5//! released everything it was given (after forcing its garbage collector or
6//! finalizers to run). `export_runtime!` exports the counters as
7//! `uint64_t {prefix}_debug_live(int32_t kind)`.
8//!
9//! The symbol is exported whether or not the feature is on, so a consumer
10//! never has to know how the producer was built: without the feature every
11//! counter reads `0`. Ask for kind [`COUNTING`] (`-1`) to tell "nothing is
12//! live" from "nothing is counted".
13
14/// Whether this build counts live resources (the `leak-check` feature).
15pub const ENABLED: bool = cfg!(feature = "leak-check");
16
17/// The pseudo-kind `{prefix}_debug_live` answers with `1` when the producer
18/// counts live resources ([`ENABLED`]) and `0` when it doesn't.
19pub const COUNTING: i32 = -1;
20
21/// Interface object references held by the consumer: one per pointer a
22/// thunk returns, per `_clone`, and per object token written into a value
23/// buffer; released by `_destroy` and by adopting a token.
24pub const OBJECTS: i32 = 0;
25
26/// Consumer callback-interface implementations the producer still holds
27/// (each is released through the vtable's `free` entry).
28pub const CALLBACKS: i32 = 1;
29
30/// Iterator handles a launcher returned that haven't been destroyed.
31pub const ITERATORS: i32 = 2;
32
33/// Cancel-token references, both the consumer's (`create` until `destroy`)
34/// and the producer's (one per in-flight cancellable call).
35pub const TOKENS: i32 = 3;
36
37/// Returned string, bytes, and value-buffer allocations (including error
38/// payloads) that haven't been passed to `{prefix}_run_free`, and runs a
39/// consumer allocated with `{prefix}_alloc` that the producer hasn't adopted.
40pub const ALLOCATIONS: i32 = 4;
41
42#[cfg(feature = "leak-check")]
43mod counters {
44    use std::sync::atomic::{AtomicI64, Ordering};
45
46    static LIVE: [AtomicI64; 5] = [
47        AtomicI64::new(0),
48        AtomicI64::new(0),
49        AtomicI64::new(0),
50        AtomicI64::new(0),
51        AtomicI64::new(0),
52    ];
53
54    pub(super) fn track(kind: i32, delta: i64) {
55        if let Some(c) = usize::try_from(kind).ok().and_then(|k| LIVE.get(k)) {
56            c.fetch_add(delta, Ordering::Relaxed);
57        }
58    }
59
60    pub(super) fn live(kind: i32) -> u64 {
61        usize::try_from(kind)
62            .ok()
63            .and_then(|k| LIVE.get(k))
64            .map_or(0, |c| u64::try_from(c.load(Ordering::Relaxed)).unwrap_or(0))
65    }
66}
67
68/// Adjust the live count of `kind` by `delta`. Compiles to nothing without
69/// the `leak-check` feature.
70#[inline]
71pub(crate) fn track(kind: i32, delta: i64) {
72    #[cfg(feature = "leak-check")]
73    counters::track(kind, delta);
74    #[cfg(not(feature = "leak-check"))]
75    let _ = (kind, delta);
76}
77
78/// The body of `{prefix}_debug_live`: how many resources of `kind` (one of
79/// [`OBJECTS`], [`CALLBACKS`], [`ITERATORS`], [`TOKENS`], or
80/// [`ALLOCATIONS`]) are live right now, or, for [`COUNTING`], `1` when this
81/// build counts at all.
82///
83/// Returns `0` for an unknown kind, and for every counter when the
84/// `leak-check` feature is off.
85#[must_use]
86pub fn debug_live(kind: i32) -> u64 {
87    if kind == COUNTING {
88        return u64::from(ENABLED);
89    }
90    #[cfg(feature = "leak-check")]
91    {
92        counters::live(kind)
93    }
94    #[cfg(not(feature = "leak-check"))]
95    {
96        let _ = kind;
97        0
98    }
99}
