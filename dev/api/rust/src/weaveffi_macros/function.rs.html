1//! `#[weaveffi::export]`: a free function, or the `pub fn`s of an
2//! interface's `impl` block.
3//!
4//! Each callable expands to its C entry point (a synchronous thunk, an async
5//! launcher, or an iterator launcher with its `_next` and `_destroy`), its
6//! metadata frame, and its contract function, inside an anonymous `const`
7//! block next to the item. Every value is lifted and lowered through its
8//! type's traits, so the thunk's signature is written in terms of the types
9//! the producer wrote and rustc resolves each one wherever it's declared.
10

25/// What kind of callable is being exported.
26#[derive(Clone, Copy, PartialEq, Eq)]
27enum Kind {
28    Function,
29    Constructor,
30    Method { arc: bool },
31    Static,
32}

34/// The interface an `impl` block's members belong to.
35struct Owner<'a> {
36    /// The written self type (`Store`).
37    ty: &'a syn::Type,
38    /// The interface's name.
39    name: String,
40}

42/// Expand `#[weaveffi::export]` on `item`.
43pub(crate) fn expand(attr: TokenStream, item: syn::Item) -> syn::Result<TokenStream> {
44    if !attr.is_empty() {
45        return Err(syn::Error::new(
46            attr.span(),
47            "weaveffi: `#[weaveffi::export]` takes no arguments (an async function that takes a \
48             `weaveffi::CancelToken` is cancellable)",
49        ));
50    }
51    let prefix = prefix()?;
52    match item {
53        syn::Item::Fn(f) => {
54            let generated = callable(&prefix, &f.attrs, &f.sig, Kind::Function, None)?;
55            Ok(quote!(#f #generated))
56        }
57        syn::Item::Impl(mut imp) => {
58            if imp.trait_.is_some() {
59                return Err(syn::Error::new(
60                    imp.span(),
61                    "weaveffi: export an interface's inherent `impl` block, not a trait impl",
62                ));
63            }
64            if !imp.generics.params.is_empty() {
65                return Err(syn::Error::new(
66                    imp.generics.span(),
67                    "weaveffi: an exported `impl` block can't be generic",
68                ));
69            }
70            let self_ty = (*imp.self_ty).clone();
71            let name = type_name(&self_ty)?;
72            let owner = Owner { ty: &self_ty, name };
73            let impl_cfgs: Vec<syn::Attribute> = cfgs(&imp.attrs).into_iter().cloned().collect();
74            let mut generated = TokenStream::new();
75            for member in &mut imp.items {
76                let syn::ImplItem::Fn(f) = member else {
77                    continue;
78                };
79                let skip = has_helper_word(&f.attrs, "skip")?;
80                f.attrs.retain(|a| !is_helper(a));
81                if skip || !matches!(f.vis, syn::Visibility::Public(_)) {
82                    continue;
83                }
84                let kind = member_kind(&f.sig, &owner)?;
85                let mut attrs = impl_cfgs.clone();
86                attrs.extend(f.attrs.iter().cloned());
87                generated.extend(callable(&prefix, &attrs, &f.sig, kind, Some(&owner))?);
88            }
89            Ok(quote!(#imp #generated))
90        }
91        other => Err(syn::Error::new(
92            other.span(),
93            "weaveffi: `#[weaveffi::export]` goes on a function or on an interface's `impl` block",
94        )),
95    }
96}

98/// The last identifier of a type path (`Store` for `crate::kv::Store`).
99fn type_name(ty: &syn::Type) -> syn::Result<String> {
100    match ungroup(ty) {
101        syn::Type::Path(p) if p.qself.is_none() => p
102            .path
103            .segments
104            .last()
105            .map(|s| s.ident.to_string())
106            .ok_or_else(|| syn::Error::new(ty.span(), "weaveffi: expected a type name")),
107        _ => Err(syn::Error::new(
108            ty.span(),
109            "weaveffi: export the `impl` block of a named #[weaveffi::interface] struct",
110        )),
111    }
112}

114/// Classify an `impl` member: a method (it has a receiver), a constructor
115/// (it's synchronous and returns the interface, possibly in an `Arc` or a
116/// `Result`), or a static (including an async factory, since constructors
117/// are synchronous).
118fn member_kind(sig: &syn::Signature, owner: &Owner<'_>) -> syn::Result<Kind> {
119    if let Some(receiver) = sig.receiver() {
120        let arc = match ungroup(&receiver.ty) {
121            syn::Type::Reference(r) if r.mutability.is_none() => false,
122            syn::Type::Path(p) if p.path.segments.last().is_some_and(|s| s.ident == "Arc") => true,
123            _ => {
124                return Err(syn::Error::new(
125                    receiver.span(),
126                    "weaveffi: an interface method takes `&self` or `self: Arc<Self>`; objects \
127                     are shared, so they can't be moved or mutably borrowed",
128                ))
129            }
130        };
131        return Ok(Kind::Method { arc });
132    }
133    let syn::ReturnType::Type(_, ty) = &sig.output else {
134        return Ok(Kind::Static);
135    };
136    if sig.asyncness.is_some() {
137        return Ok(Kind::Static);
138    }
139    let is_owner = |t: &syn::Type| {
140        let t = peel(t, "Arc").unwrap_or(t);
141        match ungroup(t) {
142            syn::Type::Path(p) => p
143                .path
144                .segments
145                .last()
146                .is_some_and(|s| s.ident == "Self" || s.ident == owner.name),
147            _ => false,
148        }
149    };
150    let ok = peel(ty, "Result").unwrap_or(ty);
151    Ok(if is_owner(ok) {
152        Kind::Constructor
153    } else {
154        Kind::Static
155    })
156}

158/// The first generic argument of `Name<..>`.
159fn peel<'a>(ty: &'a syn::Type, name: &str) -> Option<&'a syn::Type> {
160    let syn::Type::Path(p) = ungroup(ty) else {
161        return None;
162    };
163    let seg = p.path.segments.last()?;
164    if seg.ident != name {
165        return None;
166    }
167    match &seg.arguments {
168        syn::PathArguments::AngleBracketed(a) => a.args.iter().find_map(|g| match g {
169            syn::GenericArgument::Type(t) => Some(t),
170            _ => None,
171        }),
172        _ => None,
173    }
174}

176/// One parameter: its name, its written type, and its shape.
177struct Arg {
178    name: String,
179    param: Param,
180    span: Span,
181}

183/// Expand one callable.
184fn callable(
185    prefix: &str,
186    attrs: &[syn::Attribute],
187    sig: &syn::Signature,
188    kind: Kind,
189    owner: Option<&Owner<'_>>,
190) -> syn::Result<TokenStream> {
191    if !sig.generics.params.is_empty() {
192        return Err(syn::Error::new(
193            sig.generics.span(),
194            "weaveffi: an exported function can't be generic",
195        ));
196    }
197    let fix = |ty: &syn::Type| match owner {
198        Some(o) => replace_self(ty, o.ty),
199        None => ty.clone(),
200    };
201    let name = sig.ident.to_string();
202    let is_async = sig.asyncness.is_some();
203    let mut args = Vec::new();
204    for input in &sig.inputs {
205        let syn::FnArg::Typed(pat) = input else {
206            continue;
207        };
208        let syn::Pat::Ident(id) = &*pat.pat else {
209            return Err(syn::Error::new(
210                pat.pat.span(),
211                "weaveffi: an exported function's parameters are plain names",
212            ));
213        };
214        let param = Param::of(&fix(&pat.ty))?;
215        args.push(Arg {
216            name: id.ident.to_string().trim_start_matches("r#").to_string(),
217            param,
218            span: pat.ty.span(),
219        });
220    }
221    let token_at = args.iter().position(|a| matches!(a.param, Param::Token));
222    if let Some(at) = token_at {
223        if !is_async || at + 1 != args.len() {
224            return Err(syn::Error::new(
225                args[at].span,
226                "weaveffi: only an `async fn` takes a `CancelToken`, as its last parameter",
227            ));
228        }
229    }
230    let cancellable = token_at.is_some();
231    let output = match &sig.output {
232        syn::ReturnType::Type(arrow, ty) => syn::ReturnType::Type(*arrow, Box::new(fix(ty))),
233        syn::ReturnType::Default => syn::ReturnType::Default,
234    };
235    let ret = Ret::of(&output);
236    if is_async && matches!(ret, Ret::Iter(_)) {
237        return Err(syn::Error::new(
238            sig.output.span(),
239            "weaveffi: an async function can't return an iterator; return a list, or make the \
240             function synchronous",
241        ));
242    }
243
244    let (core, path, symbol_owner) = match owner {
245        Some(o) => (
246            format!("{}_{name}", o.name),
247            format!("{}.{name}", o.name),
248            Some(o.name.as_str()),
249        ),
250        None => (name.clone(), name.clone(), None),
251    };
252    let symbol = format!("{prefix}_{core}");
253
254    // The metadata frame.
255    let mut fb = FrameBuilder::new();
256    let ir_params = args
257        .iter()
258        .filter(|a| !matches!(a.param, Param::Token))
259        .map(|a| IrParam {
260            name: a.name.clone(),
261            ty: fb.ty(a.param.name()),
262            doc: None,
263        })
264        .collect();
265    let (ret_name, domain) = ret.names();
266    let function = Function {
267        params: ir_params,
268        returns: (kind != Kind::Constructor).then(|| fb.ty(ret_name.clone())),
269        doc: docs(attrs),
270        throws: Some(fb.throws(domain.clone())),
271        r#async: is_async,
272        cancellable,
273        deprecated: deprecated(attrs),
274        ..Function::named(&name)
275    };
276    let item = match owner {
277        Some(o) => Item::Member(Member {
278            interface: o.name.clone(),
279            role: match kind {
280                Kind::Constructor => Role::Constructor,
281                Kind::Method { .. } => Role::Method,
282                _ => Role::Static,
283            },
284            function,
285        }),
286        None => Item::Function(function),
287    };
288    let frame = fb.finish(prefix, location(sig.ident.span()), item);
289
290    // The contract.
291    let keyword = match kind {
292        Kind::Function => "function",
293        Kind::Constructor => "constructor",
294        Kind::Method { .. } => "method",
295        Kind::Static => "static",
296    };
297    let sig_ret = match (kind, owner) {
298        (Kind::Constructor, Some(o)) => name_lit(&o.name),
299        _ => ret_name,
300    };
301    let contract_sig = Sig::callable(
302        keyword,
303        &name,
304        args.iter()
305            .filter(|a| !matches!(a.param, Param::Token))
306            .map(|a| a.param.name())
307            .collect(),
308        sig_ret,
309        quote!(::weaveffi::__rt::Name::suffix(&#domain)),
310        is_async,
311        cancellable,
312    );
313    let contract = contract(prefix, &path, contract_sig);
314
315    let thunk = thunk(
316        prefix,
317        &symbol,
318        symbol_owner,
319        &name,
320        sig,
321        kind,
322        owner,
323        &args,
324        &ret,
325    )?;
326    Ok(scoped(&cfgs(attrs), quote!(#thunk #frame #contract)))
327}

329/// The C entry point (and, for an iterator, its `_next` and `_destroy`).
330#[allow(clippy::too_many_arguments)]
331fn thunk(
332    prefix: &str,
333    symbol: &str,
334    symbol_owner: Option<&str>,
335    name: &str,
336    sig: &syn::Signature,
337    kind: Kind,
338    owner: Option<&Owner<'_>>,
339    args: &[Arg],
340    ret: &Ret,
341) -> syn::Result<TokenStream> {
342    let tag = tag();
343    let fn_ident = &sig.ident;
344    let callee = match owner {
345        Some(o) => {
346            let ty = o.ty;
347            quote!(<#ty>::#fn_ident)
348        }
349        None => quote!(#fn_ident),
350    };
351    let is_async = sig.asyncness.is_some();
352    let locals: Vec<syn::Ident> = (0..args.len())
353        .map(|i| format_ident!("__wv_a{i}"))
354        .collect();
355    let abi_names: Vec<syn::Ident> = args
356        .iter()
357        .map(|a| ident(&format!("__wv_{}", a.name)))
358        .collect();
359    let mut c_params: Vec<TokenStream> = Vec::new();
360    let mut lifts: Vec<TokenStream> = Vec::new();
361    let mut unwraps: Vec<TokenStream> = Vec::new();
362    let mut call_args: Vec<TokenStream> = Vec::new();
363
364    // The receiver.
365    if let (Kind::Method { arc }, Some(o)) = (kind, owner) {
366        let ty = o.ty;
367        c_params.push(quote!(__wv_self: <#ty as ::weaveffi::__rt::FfiRef<#tag>>::Abi));
368        let lift = if arc || is_async {
369            quote!(<#ty as ::weaveffi::__rt::FfiRef<#tag>>::lift_owned(__wv_self))
370        } else {
371            quote!(<#ty as ::weaveffi::__rt::FfiRef<#tag>>::lift_ref(__wv_self))
372        };
373        lifts.push(quote!(let __wv_this = unsafe { #lift };));
374        unwraps.push(quote!(let __wv_this = ::weaveffi::__rt::in_param(__wv_this, "self")?;));
375        call_args.push(if arc {
376            quote!(__wv_this)
377        } else {
378            quote!(&*__wv_this)
379        });
380    }
381
382    let mut token_arg = None;
383    for ((arg, local), abi) in args.iter().zip(&locals).zip(&abi_names) {
384        if matches!(arg.param, Param::Token) {
385            token_arg = Some((local.clone(), abi.clone()));
386            call_args.push(quote!(__wv_cancel));
387            continue;
388        }
389        let abi_ty = arg.param.abi();
390        c_params.push(quote!(#abi: #abi_ty));
391        let lift = if is_async {
392            arg.param.lift_owned(abi, arg.span)?
393        } else {
394            arg.param.lift(abi)
395        };
396        lifts.push(quote!(let #local = unsafe { #lift };));
397        let param_name = &arg.name;
398        unwraps.push(quote!(let #local = ::weaveffi::__rt::in_param(#local, #param_name)?;));
399        call_args.push(if is_async {
400            arg.param.owned_arg(local)
401        } else {
402            arg.param.arg(local)
403        });
404    }
405
406    let r = ret.ty();
407    let rs = r.span();
408    let ret_abi = quote_spanned!(rs=> <#r as ::weaveffi::__rt::FfiReturn<#tag>>::Abi);
409    let attrs = quote! {
410        #[allow(unsafe_code, unused_unsafe, clippy::missing_safety_doc, clippy::let_unit_value)]
411    };
412    if is_async {
413        let (token_param, token_lift, token_out) = match &token_arg {
414            Some((local, abi)) => (
415                quote!(#abi: *const ::weaveffi::__rt::FfiCancelToken,),
416                quote! {
417                    let #local = unsafe { ::weaveffi::__rt::token(#abi) };
418                    let __wv_cancel = ::core::clone::Clone::clone(&#local);
419                },
420                quote!(::core::option::Option::Some(#local)),
421            ),
422            None => (quote!(), quote!(), quote!(::core::option::Option::None)),
423        };
424        // The future owns every input; the token is cloned into the call so
425        // the launcher keeps its own for cancellation.
426        let fut_args = call_args.iter();
427        let (callback_ty, launch) = match ret {
428            Ret::Unit(_) => (
429                quote!(
430                    ::core::option::Option<
431                        unsafe extern "C" fn(
432                            *mut ::core::ffi::c_void,
433                            *mut ::weaveffi::__rt::FfiError,
434                        ),
435                    >
436                ),
437                quote_spanned!(rs=> ::weaveffi::__rt::launch_unit::<#tag, #r, _>),
438            ),
439            _ => (
440                quote!(::core::option::Option<unsafe extern "C" fn(
441                    *mut ::core::ffi::c_void,
442                    *mut ::weaveffi::__rt::FfiError,
443                    #ret_abi,
444                )>),
445                quote_spanned!(rs=> ::weaveffi::__rt::launch::<#tag, #r, _>),
446            ),
447        };
448        return Ok(quote! {
449            #[doc(hidden)]
450            #attrs
451            #[unsafe(export_name = #symbol)]
452            pub unsafe extern "C" fn __weaveffi_thunk(
453                #(#c_params,)*
454                #token_param
455                __wv_callback: #callback_ty,
456                __wv_context: *mut ::core::ffi::c_void,
457            ) {
458                #launch(
459                    move || {
460                        #(#lifts)*
461                        #token_lift
462                        #(#unwraps)*
463                        ::core::result::Result::Ok((
464                            async move { #callee(#(#fut_args),*).await },
465                            #token_out,
466                        ))
467                    },
468                    __wv_callback,
469                    __wv_context,
470                )
471            }
472        });
473    }
474
475    let call = quote!(#callee(#(#call_args),*));
476    if let Ret::Iter(_) = ret {
477        let iter_core = match symbol_owner {
478            Some(o) => format!("{o}_{}Iterator", name.to_upper_camel_case()),
479            None => format!("{}Iterator", name.to_upper_camel_case()),
480        };
481        let next = format!("{prefix}_{iter_core}_next");
482        let destroy = format!("{prefix}_{iter_core}_destroy");
483        let item = quote_spanned!(rs=> <#r as ::weaveffi::__rt::IterReturn<#tag>>::Item);
484        let into_handle =
485            quote_spanned!(rs=> <#r as ::weaveffi::__rt::IterReturn<#tag>>::into_handle);
486        return Ok(quote! {
487            #[doc(hidden)]
488            #attrs
489            #[unsafe(export_name = #symbol)]
490            pub unsafe extern "C" fn __weaveffi_thunk(
491                #(#c_params,)*
492                __wv_out_err: *mut ::weaveffi::__rt::FfiError,
493            ) -> *mut ::weaveffi::__rt::IterHandle<#item> {
494                unsafe {
495                    ::weaveffi::__rt::call(__wv_out_err, ::core::ptr::null_mut(), move || {
496                        #(#lifts)*
497                        #(#unwraps)*
498                        #into_handle(#call)
499                    })
500                }
501            }
502
503            #[doc(hidden)]
504            #attrs
505            #[unsafe(export_name = #next)]
506            pub unsafe extern "C" fn __weaveffi_next(
507                __wv_iter: *const ::weaveffi::__rt::IterHandle<#item>,
508                __wv_out_item: *mut <#item as ::weaveffi::__rt::FfiType<#tag>>::Abi,
509                __wv_out_err: *mut ::weaveffi::__rt::FfiError,
510            ) -> i32 {
511                unsafe { ::weaveffi::__rt::iter_step::<#tag, #item>(__wv_iter, __wv_out_item, __wv_out_err) }
512            }
513
514            #[doc(hidden)]
515            #attrs
516            #[unsafe(export_name = #destroy)]
517            pub unsafe extern "C" fn __weaveffi_destroy(
518                __wv_iter: *mut ::weaveffi::__rt::IterHandle<#item>,
519            ) {
520                unsafe { ::weaveffi::__rt::iter_destroy(__wv_iter) }
521            }
522        });
523    }
524
525    let ret_zero = quote_spanned!(rs=> <#r as ::weaveffi::__rt::FfiReturn<#tag>>::ZERO);
526    let ret_fn = quote_spanned!(rs=> ::weaveffi::__rt::ret::<#tag, #r>);
527    Ok(quote! {
528        #[doc(hidden)]
529        #attrs
530        #[unsafe(export_name = #symbol)]
531        pub unsafe extern "C" fn __weaveffi_thunk(
532            #(#c_params,)*
533            __wv_out_err: *mut ::weaveffi::__rt::FfiError,
534        ) -> #ret_abi {
535            unsafe {
536                ::weaveffi::__rt::call(
537                    __wv_out_err,
538                    #ret_zero,
539                    move || {
540                        #(#lifts)*
541                        #(#unwraps)*
542                        #ret_fn(#call)
543                    },
544                )
545            }
546        }
547    })
