8/// The codec functions the C value-buffer helper header declares for each
9/// record, rich enum, and error payload struct `T`: `T_write`, `T_read`,
10/// `T_decode`, and `T_free`.
11pub const VALUE_CODECS: &[&str] = &["write", "read", "decode", "free"];
12
13/// One C identifier the library's ABI declares.
14#[derive(Debug, Clone, PartialEq, Eq)]
15pub struct Symbol {
16    /// The full, prefixed C identifier.
17    pub name: String,
18    /// The declaration that owns it.
19    pub owner: SymbolOwner,
20}
21
22/// The declaration that owns a C [`Symbol`].
23///
24/// Every `path` is a dotted declaration path: the declaration name, then a
25/// member, variant, or code name (`open`, `Store.get`,
26/// `KvError.NotFound`). `Display` renders the human-readable description
27/// diagnostics quote.
28#[derive(Debug, Clone, PartialEq, Eq)]
29pub enum SymbolOwner {
30    /// The runtime surface every producer exports.
31    Runtime,
32    /// A callable's entry point: the sync symbol, async launcher, or
33    /// iterator launcher.
34    Callable {
35        /// The callable's path.
36        path: String,
37    },
38    /// An async callable's completion-callback typedef.
39    AsyncCallback {
40        /// The callable's path.
41        path: String,
42    },
43    /// The opaque iterator type an iterator-returning callable launches.
44    IteratorType {
45        /// The callable's path.
46        path: String,
47    },
48    /// That iterator's `_next`.
49    IteratorNext {
50        /// The callable's path.
51        path: String,
52    },
53    /// That iterator's `_destroy`.
54    IteratorDestroy {
55        /// The callable's path.
56        path: String,
57    },
58    /// An interface's opaque type.
59    Interface {
60        /// The interface's path.
61        path: String,
62    },
63    /// An interface's `_clone`.
64    InterfaceClone {
65        /// The interface's path.
66        path: String,
67    },
68    /// An interface's `_destroy`.
69    InterfaceDestroy {
70        /// The interface's path.
71        path: String,
72    },
73    /// An enum's type (a C-style enum's `int32_t` type, or a rich enum's
74    /// buffer-header struct).
75    Enum {
76        /// The enum's path.
77        path: String,
78    },
79    /// A rich enum's `_Tag` type.
80    EnumTag {
81        /// The enum's path.
82        path: String,
83    },
84    /// An enum variant's constant.
85    EnumVariant {
86        /// The variant's path.
87        path: String,
88    },
89    /// An error domain's code type.
90    ErrorDomain {
91        /// The domain's path.
92        path: String,
93    },
94    /// An error code's constant.
95    ErrorCode {
96        /// The code's path.
97        path: String,
98    },
99    /// The buffer-header struct of an error code's payload.
100    ErrorPayload {
101        /// The code's path.
102        path: String,
103    },
104    /// A record's buffer-header struct.
105    Record {
106        /// The record's path.
107        path: String,
108    },
109    /// A callback interface's argument struct.
110    Callback {
111        /// The callback interface's path.
112        path: String,
113    },
114    /// A callback interface's vtable type.
115    Vtable {
116        /// The callback interface's path.
117        path: String,
118    },
119    /// The C value-buffer header's struct of a composite (an optional,
120    /// list, or map shape), and its codecs.
121    Composite {
122        /// The composite's stem (`list_i32`).
123        stem: String,
124    },
125    /// One of the [`VALUE_CODECS`] the buffer header declares for a struct.
126    Codec {
127        /// The codec (`write`, `read`, `decode`, or `free`).
128        codec: &'static str,
129        /// The struct's owner: a [`Record`](Self::Record),
130        /// [`Enum`](Self::Enum), or [`ErrorPayload`](Self::ErrorPayload).
131        of: Box<SymbolOwner>,
132    },
133}
134
135impl fmt::Display for SymbolOwner {
136    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
137        match self {
138            Self::Runtime => f.write_str("the runtime surface"),
139            Self::Callable { path } => write!(f, "function '{path}'"),
140            Self::AsyncCallback { path } => write!(f, "the completion type of '{path}'"),
141            Self::IteratorType { path } => write!(f, "the iterator type of '{path}'"),
142            Self::IteratorNext { path } => write!(f, "the iterator step of '{path}'"),
143            Self::IteratorDestroy { path } => write!(f, "the iterator destructor of '{path}'"),
144            Self::Interface { path } => write!(f, "interface '{path}'"),
145            Self::InterfaceClone { path } => write!(f, "the clone of '{path}'"),
146            Self::InterfaceDestroy { path } => write!(f, "the destructor of '{path}'"),
147            Self::Enum { path } => write!(f, "enum '{path}'"),
148            Self::EnumTag { path } => write!(f, "the tag type of enum '{path}'"),
149            Self::EnumVariant { path } => write!(f, "enum variant '{path}'"),
150            Self::ErrorDomain { path } => write!(f, "error domain '{path}'"),
151            Self::ErrorCode { path } => write!(f, "error code '{path}'"),
152            Self::ErrorPayload { path } => {
153                write!(f, "the payload struct of error code '{path}'")
154            }
155            Self::Record { path } => write!(f, "record '{path}'"),
156            Self::Composite { stem } => write!(f, "the value-buffer codec of '{stem}'"),
157            Self::Callback { path } => write!(f, "callback interface '{path}'"),
158            Self::Vtable { path } => write!(f, "the vtable of callback interface '{path}'"),
159            Self::Codec { codec, of } => write!(f, "the value-buffer codec '{codec}' of {of}"),
160        }
161    }
162}

164/// Build the symbol table of `model` (see [`Model::c_symbols`]). The
165/// per-declaration contract functions are left out: their names fall in a
166/// family no declaration may use.
167pub(super) fn collect(model: &Model) -> Vec<Symbol> {
168    let p = model.prefix();
169    let mut out: Vec<Symbol> = RUNTIME_SYMBOLS
170        .iter()
171        .map(|s| s.to_string())
172        .chain(
173            crate::ty::Prim::ALL
174                .into_iter()
175                .filter(|p| p.is_scalar())
176                .map(|prim| format!("opt_{}", prim.snake())),
177        )
178        .chain(std::iter::once("contract_check".to_string()))
179        .map(|s| Symbol {
180            name: format!("{p}_{s}"),
181            owner: SymbolOwner::Runtime,
182        })
183        .collect();
184    let push = |out: &mut Vec<Symbol>, name: &str, owner: SymbolOwner| {
185        out.push(Symbol {
186            name: name.to_string(),
187            owner,
188        });
189    };
190    let callable = |out: &mut Vec<Symbol>, f: &FnBinding| {
191        let path = f.path.clone();
192        push(
193            out,
194            &f.abi.symbol,
195            SymbolOwner::Callable { path: path.clone() },
196        );
197        if let Some(a) = f.async_binding() {
198            push(
199                out,
200                &a.callback_type,
201                SymbolOwner::AsyncCallback { path: path.clone() },
202            );
203        }
204        if let Some(it) = f.iterator() {
205            push(
206                out,
207                &it.iter_tag,
208                SymbolOwner::IteratorType { path: path.clone() },
209            );
210            push(
211                out,
212                &it.next.symbol,
213                SymbolOwner::IteratorNext { path: path.clone() },
214            );
215            push(
216                out,
217                &it.destroy_symbol,
218                SymbolOwner::IteratorDestroy { path },
219            );
220        }
221    };
222    let codecs = |out: &mut Vec<Symbol>, tag: &str, of: &SymbolOwner| {
223        for codec in VALUE_CODECS {
224            push(
225                out,
226                &format!("{tag}_{codec}"),
227                SymbolOwner::Codec {
228                    codec,
229                    of: Box::new(of.clone()),
230                },
231            );
232        }
233    };
234    for e in &model.errors {
235        let path = e.name.clone();
236        push(
237            &mut out,
238            &e.c_tag,
239            SymbolOwner::ErrorDomain { path: path.clone() },
240        );
241        for c in &e.codes {
242            let code = format!("{path}.{}", c.name);
243            push(
244                &mut out,
245                &c.c_const,
246                SymbolOwner::ErrorCode { path: code.clone() },
247            );
248            if !c.fields.is_empty() {
249                let owner = SymbolOwner::ErrorPayload { path: code };
250                let tag = c.payload_tag();
251                codecs(&mut out, &tag, &owner);
252                push(&mut out, &tag, owner);
253            }
254        }
255    }
256    for e in &model.enums {
257        let path = e.name.clone();
258        let owner = SymbolOwner::Enum { path: path.clone() };
259        push(&mut out, &e.c_tag, owner.clone());
260        if e.is_rich() {
261            push(
262                &mut out,
263                &format!("{}_Tag", e.c_tag),
264                SymbolOwner::EnumTag { path: path.clone() },
265            );
266            codecs(&mut out, &e.c_tag, &owner);
267        }
268        for v in &e.variants {
269            push(
270                &mut out,
271                &v.c_const,
272                SymbolOwner::EnumVariant {
273                    path: format!("{path}.{}", v.name),
274                },
275            );
276        }
277    }
278    for r in &model.records {
279        let owner = SymbolOwner::Record {
280            path: r.name.clone(),
281        };
282        codecs(&mut out, &r.c_tag, &owner);
283        push(&mut out, &r.c_tag, owner);
284    }
285    for c in &model.callbacks {
286        let path = c.name.clone();
287        push(
288            &mut out,
289            &c.c_tag,
290            SymbolOwner::Callback { path: path.clone() },
291        );
292        push(&mut out, &c.vtable_tag, SymbolOwner::Vtable { path });
293    }
294    for i in &model.interfaces {
295        let path = i.name.clone();
296        push(
297            &mut out,
298            &i.c_tag,
299            SymbolOwner::Interface { path: path.clone() },
300        );
301        push(
302            &mut out,
303            &i.clone_symbol,
304            SymbolOwner::InterfaceClone { path: path.clone() },
305        );
306        push(
307            &mut out,
308            &i.destroy_symbol,
309            SymbolOwner::InterfaceDestroy { path },
310        );
311        for f in i.members() {
312            callable(&mut out, f);
313        }
314    }
315    for f in &model.functions {
316        callable(&mut out, f);
317    }
318    if model.has_buffers() {
319        for ty in model.composites() {
320            let stem = ty.stem();
321            let owner = SymbolOwner::Composite { stem: stem.clone() };
322            let tag = format!("{p}_buf_{stem}");
323            codecs(&mut out, &tag, &owner);
324            push(&mut out, &tag, owner);
325        }
326    }
327    out
328}
