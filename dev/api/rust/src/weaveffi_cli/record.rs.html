1//! Generation records: what each target's last generation wrote.
2//!
3//! After generating a target, the orchestrator writes
4//! `{out_dir}/.weaveffi/{target}.json`: the path of every file the target
5//! produced. Files recorded by the previous run that the current run no
6//! longer produces are removed, so a renamed type never leaves a stale file
7//! behind, while files the user added (a `node_modules/`, a build directory)
8//! are never touched. Commit the records with the generated bindings, so
9//! `generate --check` on a fresh checkout sees stale files too.
10//!
11//! A record is read back defensively: an entry that isn't a relative path
12//! inside its target's own directory is ignored, so a corrupted or
13//! hand-edited record can never make a run delete anything else.
14
15use std::collections::BTreeSet;
16
17use camino::{Utf8Path, Utf8PathBuf};
18use miette::{IntoDiagnostic, Result, WrapErr};
19use serde::{Deserialize, Serialize};
20
21const RECORD_DIR: &str = ".weaveffi";
22
23/// What one generation of one target produced.
24#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
25pub(crate) struct Record {
26    /// Every file the target produced, relative to the output directory
27    /// (with `/` separators).
28    pub(crate) files: BTreeSet<String>,
29}
30
31fn record_path(out_dir: &Utf8Path, target: &str) -> Utf8PathBuf {
32    out_dir.join(RECORD_DIR).join(format!("{target}.json"))
33}
34
35/// Read the record of the last generation of `target` under `out_dir`, if
36/// any (a missing or unreadable record means "never generated"), keeping
37/// only the entries inside the target's own directory.
38pub(crate) fn read_record(out_dir: &Utf8Path, target: &str) -> Option<Record> {
39    let text = std::fs::read_to_string(record_path(out_dir, target)).ok()?;
40    let mut record: Record = serde_json::from_str(&text).ok()?;
41    record.files.retain(|path| inside(path, target));
42    Some(record)
43}
44
45/// Whether `path` is `{target}/...`, made only of plain components.
46fn inside(path: &str, target: &str) -> bool {
47    let mut parts = path.split('/');
48    parts.next() == Some(target)
49        && path.len() > target.len() + 1
50        && parts.all(|p| {
51            !p.is_empty() && p != "." && p != ".." && !p.contains('\\') && !p.contains(':')
52        })
53}
54
55/// Persist the record of a generation of `target`.
56pub(crate) fn write_record(out_dir: &Utf8Path, target: &str, record: &Record) -> Result<()> {
57    let path = record_path(out_dir, target);
58    let dir = out_dir.join(RECORD_DIR);
59    std::fs::create_dir_all(dir.as_std_path())
60        .into_diagnostic()
61        .wrap_err_with(|| format!("failed to create the record directory {dir}"))?;
62    let json = serde_json::to_string_pretty(record).into_diagnostic()?;
63    std::fs::write(path.as_std_path(), json)
64        .into_diagnostic()
65        .wrap_err_with(|| format!("failed to write the generation record {path}"))
66}

68#[cfg(test)]
69mod tests {
70    use super::*;
71
72    #[test]
73    fn records_round_trip() {
74        let dir = tempfile::tempdir().unwrap();
75        let out = Utf8Path::from_path(dir.path()).unwrap();
76        assert_eq!(read_record(out, "swift"), None);
77        let record = Record {
78            files: ["swift/a.swift".to_string()].into(),
79        };
80        write_record(out, "swift", &record).unwrap();
81        assert_eq!(read_record(out, "swift"), Some(record));
82    }
83
84    #[test]
85    fn entries_outside_the_target_are_ignored() {
86        let dir = tempfile::tempdir().unwrap();
87        let out = Utf8Path::from_path(dir.path()).unwrap();
88        let record = Record {
89            files: [
90                "swift/ok.swift",
91                "swift/../../etc/passwd",
92                "c/other.h",
93                "/abs",
94                "swift/a\\..\\b",
95                "swift",
96            ]
97            .map(String::from)
98            .into(),
99        };
100        write_record(out, "swift", &record).unwrap();
101        let read = read_record(out, "swift").unwrap();
102        assert_eq!(
103            read.files.into_iter().collect::<Vec<_>>(),
104            ["swift/ok.swift"]
105        );
106    }
107}
