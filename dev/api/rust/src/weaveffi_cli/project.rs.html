1//! A WeaveFFI project: where its API comes from, what its library is
2//! called, and how to turn it into bindings.
3//!
4//! A project's input is either an IDL document or a Rust producer crate.
5//! A crate's API is read from the metadata frames in its built library:
6//! [`Project`] builds the crate with Cargo, or reads a library built
7//! earlier ([`Project::library`]). Every `weaveffi` subcommand starts here,
8//! and so can a `build.rs`:
9//!
10//! ```no_run
11//! // build.rs of a crate that ships bindings for an IDL-defined library.
12//! use weaveffi_cli::codegen::Orchestrator;
13//! use weaveffi_cli::project::Project;
14//!
15//! fn main() -> miette::Result<()> {
16//!     println!("cargo::rerun-if-changed=weaveffi.toml");
17//!     println!("cargo::rerun-if-changed=api.yml");
18//!     let project = Project::discover(env!("CARGO_MANIFEST_DIR"))?;
19//!     let targets = project.config.targets(None)?;
20//!     Orchestrator::new()
21//!         .with_targets(targets.iter().map(AsRef::as_ref))
22//!         .run(&project.model()?, &project.config.out_dir(None))?;
23//!     Ok(())
24//! }
25//! ```
26//!
27//! A build script can't read the library of the crate it's building, which
28//! doesn't exist yet, so from a `build.rs` the project's input must be an
29//! IDL, or a library built beforehand that [`Project::library`] names.
30

47/// Where a project's API comes from.
48#[derive(Debug, Clone)]
49pub(crate) enum Source {
50    /// An IDL document (`.yml`, `.yaml`, or `.json`).
51    Idl(Utf8PathBuf),
52    /// A Rust producer crate, whose API is read from its built library.
53    Crate(Box<CargoCrate>),
54    /// A built library with no crate or IDL beside it.
55    Library(Utf8PathBuf),
56}
57
58/// A located project: its configuration and the source of its API.
59#[derive(Debug, Clone)]
60pub struct Project {
61    /// The `weaveffi.toml` settings (the defaults when there is none).
62    pub config: ProjectConfig,
63    /// Where the API comes from.
64    pub(crate) source: Source,
65    /// A built library to read a crate's API from instead of building it.
66    library: Option<Utf8PathBuf>,
67    /// The Cargo profile a crate's library is built with.
68    profile: String,
69    /// The Cargo features it's built with (default: `[build]`'s).
70    features: CargoFeatures,
71    quiet: bool,
72}

74/// A project's API before validation, with the identity it's validated
75/// against.
76#[derive(Debug, Clone)]
77pub struct Definition {
78    /// The API as written in the IDL or embedded in the library.
79    pub api: Api,
80    /// The library's identity.
81    pub identity: Identity,
82    /// The IDL files' names and texts, for diagnostics that quote them.
83    sources: Option<Sources>,

103impl Project {
104    /// The project of the nearest `weaveffi.toml` at or above `dir`, whose
105    /// `[project] input` names the API; without one, the crate at `dir`.
106    ///
107    /// # Errors
108    ///
109    /// Returns an error when there is neither, when the config doesn't
110    /// parse, or when its input can't be resolved.
111    pub fn discover(dir: impl AsRef<Utf8Path>) -> Result<Self> {
112        let dir = dir.as_ref();
113        if let Some(config) = discover(&dir.join(CONFIG_FILE_NAME)) {
114            return Self::locate(Some(config.as_str()), None, None);
115        }
116        if dir.join("Cargo.toml").is_file() {
117            return Self::locate(None, Some(dir.as_str()), None);
118        }
119        Err(miette!(
120            "no {CONFIG_FILE_NAME} at or above {dir} and no Cargo.toml in it; run `weaveffi init`"
121        ))
122    }
123
124    /// Locate a project the way the CLI does: from `input` (a crate
125    /// directory, a `Cargo.toml`, or an IDL file) and the nearest
126    /// `weaveffi.toml` at or above it, or from the config named by `config`
127    /// (else found from the current directory) and its `[project] input`.
128    /// `library` is a built library to read the API from: the crate's own,
129    /// or, with no input at all, a library standing alone.
130    ///
131    /// # Errors
132    ///
133    /// Returns an error when no input can be found, a config file doesn't
134    /// parse, the input isn't a crate or an IDL, or `library` is given for
135    /// an IDL input.
136    pub fn locate(
137        config: Option<&str>,
138        input: Option<&str>,
139        library: Option<&str>,
140    ) -> Result<Self> {
141        let (config, input) = match (input, library) {
142            (Some(input), _) => {
143                let path = Utf8PathBuf::from(input);
144                let start = if path.is_dir() {
145                    path.join(CONFIG_FILE_NAME)
146                } else {
147                    path.clone()
148                };
149                (ProjectConfig::load(config, &start)?, Some(path))
150            }
151            (None, Some(_)) => {
152                let config = match config {
153                    Some(p) => ProjectConfig::from_file(Utf8Path::new(p))?,
154                    None => current_dir()
155                        .ok()
156                        .and_then(|cwd| discover(&cwd.join(CONFIG_FILE_NAME)))
157                        .map(|p| ProjectConfig::from_file(&p))
158                        .transpose()?
159                        .unwrap_or_default(),
160                };
161                let input = config
162                    .project
163                    .input
164                    .as_deref()
165                    .map(|i| config.relative_to_config(i));
166                (config, input)
167            }
168            (None, None) => {
169                let (config, input) = ProjectConfig::locate(config, None)?;
170                (config, Some(input))
171            }
172        };
173        let source = match input {
174            Some(input) => source_of(&input)?,
175            None => Source::Library(Utf8PathBuf::from(library.unwrap_or_default())),
176        };
177        let features = config.build.cargo_features();
178        let mut project = Self {
179            config,
180            source,
181            library: None,
182            profile: "dev".into(),
183            features,
184            quiet: false,
185        };
186        if let Some(library) = library {
187            project = project.library(library)?;
188        }
189        Ok(project)
190    }

192    /// Read a Rust producer's API from the library at `path` instead of
193    /// building the crate.
194    ///
195    /// # Errors
196    ///
197    /// Returns an error when the project's input is an IDL.
198    pub fn library(mut self, path: impl Into<Utf8PathBuf>) -> Result<Self> {
199        let path = path.into();
200        match &mut self.source {
201            Source::Idl(idl) => {
202                return Err(miette!(
203                    "--library reads a Rust producer's API from its built library, but {idl} is \
204                     an IDL; pass the producer crate instead"
205                ))
206            }
207            Source::Crate(_) => self.library = Some(path),
208            Source::Library(lib) => *lib = path,
209        }
210        Ok(self)
211    }

254    /// The library a Rust producer's API is read from: the one named with
255    /// [`library`](Self::library), else the crate's, built now.
256    ///
257    pub(crate) fn library_path(&self) -> Result<Utf8PathBuf> {
258        match &self.source {
259            Source::Idl(idl) => Err(miette!("{idl} is an IDL; it has no library to read")),
260            Source::Library(lib) => Ok(lib.clone()),
261            Source::Crate(krate) => {
262                if let Some(lib) = &self.library {
263                    return Ok(lib.clone());
264                }
265                if in_own_build_script(krate) {
266                    bail!(
267                        "`{}`'s build script can't read the API of the library it's building: \
268                         point `[project] input` at an IDL, or read a library built earlier \
269                         with `Project::library`",
270                        krate.name
271                    );
272                }
273                krate.build_library(&self.profile, &self.features, self.quiet)
274            }
275        }
276    }

278    /// Read the API and resolve the library's identity.
279    ///
280    /// # Errors
281    ///
282    /// Returns an error when the IDL can't be read or parsed, the library
283    /// can't be built or read or holds no metadata for the crate, or the
284    /// `[package]` table sets a key that doesn't apply to the input.
285    pub fn definition(&self) -> Result<Definition> {
286        let package = &self.config.package.identity;
287        match &self.source {
288            Source::Idl(path) => {
289                idl_format(path)?;
290                let (api, sources) = load_api(path.as_std_path()).map_err(|e| match &e {
291                    ParseError::Yaml { src, .. } | ParseError::Json { src, .. } => {
292                        let src = src.clone();
293                        with_named_source(e, path.as_str(), &src)
294                    }
295                    _ => Report::new(e),
296                })?;
297                Ok(Definition {
298                    api,
299                    identity: Identity::new(&name_from_basename(path.as_str()), package),
300                    sources: Some(sources),
301                })
302            }
303            Source::Crate(krate) => {
304                rust_only_keys(package)?;
305                let path = self.library_path()?;
306                let (api, _) = read_api(&path, Some(&krate.lib_name))?;
307                Ok(Definition {
308                    api,
309                    identity: crate_identity(krate, package),
310                    sources: None,
311                })
312            }
313            Source::Library(path) => {
314                rust_only_keys(package)?;
315                let (api, prefix) = read_api(path, None)?;
316                let mut merged = package.clone();
317                merged.c_prefix = Some(prefix.clone());
318                merged.library = Some(library::library_name(path));
319                Ok(Definition {
320                    api,
321                    identity: Identity::new(&prefix, &merged),
322                    sources: None,
323                })
324            }
325        }
326    }

335        self.definition()?.validate().map_err(Report::new)

339/// What `input` is: a crate (a directory with a `Cargo.toml`, or the
340/// manifest itself) or an IDL file.
341fn source_of(input: &Utf8Path) -> Result<Source> {
342    let manifest = if input.is_dir() {
343        let manifest = input.join("Cargo.toml");
344        if !manifest.is_file() {
345            bail!(
346                "{input} is a directory without a Cargo.toml; the input is a Rust producer crate \
347                 or an IDL file (yml|yaml|json)"
348            );
349        }
350        Some(manifest)
351    } else if input.file_name() == Some("Cargo.toml") {
352        Some(input.to_path_buf())
353    } else {
354        None
355    };
356    match manifest {
357        Some(m) => Ok(Source::Crate(Box::new(CargoCrate::resolve(&m)?))),
358        None => {
359            idl_format(input)?;
360            Ok(Source::Idl(input.to_path_buf()))
361        }
362    }
363}

365/// The parser's format token for an IDL file's extension.
366fn idl_format(path: &Utf8Path) -> Result<&'static str> {
367    match path.extension().unwrap_or("") {
368        "yml" | "yaml" => Ok("yaml"),
369        "json" => Ok("json"),
370        "rs" => Err(miette!(
371            "{path} is Rust source; WeaveFFI reads a Rust producer's API from its built \
372             library, so pass the crate (its directory or Cargo.toml) instead"
373        )),
374        other => Err(miette!(
375            "unsupported input {path}{}: expected a Rust producer crate (its directory or \
376             Cargo.toml) or an IDL file (yml|yaml|json)",
377            if other.is_empty() {
378                String::new()
379            } else {
380                format!(" (.{other})")
381            }
382        )),
383    }
384}

386/// Reject `[package]` keys a Rust producer can't override.
387fn rust_only_keys(package: &Package) -> Result<()> {
388    if package.c_prefix.is_some() || package.library.is_some() {
389        bail!(
390            "`[package] c_prefix` and `[package] library` apply only to IDL inputs: a Rust \
391             producer's C prefix and library name are its crate's library name"
392        );
393    }
394    Ok(())
395}

397/// A Rust producer's identity: named after its package, with its library
398/// name as both the C prefix (what the producer macros use) and the
399/// library, and `[package]` metadata over the crate's.
400fn crate_identity(krate: &CargoCrate, package: &Package) -> Identity {
401    let mut merged = package.clone();
402    merged.version = merged.version.or_else(|| Some(krate.version.clone()));
403    merged.description = merged.description.or_else(|| krate.description.clone());
404    merged.license = merged.license.or_else(|| krate.license.clone());
405    merged.homepage = merged.homepage.or_else(|| krate.homepage.clone());
406    merged.repository = merged.repository.or_else(|| krate.repository.clone());
407    if merged.authors.is_empty() {
408        merged.authors = krate.authors.clone();
409    }
410    merged.c_prefix = Some(krate.lib_name.clone());
411    merged.library = Some(krate.lib_name.clone());
412    Identity::new(&krate.name, &merged)
413}

415/// The API the library at `path` embeds, and its crate's C prefix: the
416/// crate with `prefix`, or, without one, the only crate whose metadata the
417/// library holds.
418fn read_api(path: &Utf8Path, prefix: Option<&str>) -> Result<(Api, String)> {
419    let frames = library::read_frames(path)?;
420    let prefixes: std::collections::BTreeSet<&str> =
421        frames.iter().map(|f| f.prefix.as_str()).collect();
422    let prefix = match (prefix, prefixes.len()) {
423        (Some(prefix), _) => prefix.to_string(),
424        (None, 0) => return Err(no_metadata(path, None)),
425        (None, 1) => prefixes
426            .iter()
427            .next()
428            .copied()
429            .unwrap_or_default()
430            .to_string(),
431        (None, _) => {
432            return Err(miette::miette!(
433            "{path} holds the APIs of several crates ({}); pass the producer crate so WeaveFFI \
434             knows which one to bind",
435            prefixes.into_iter().collect::<Vec<_>>().join(", ")
436        ))
437        }
438    };
439    let api = meta::assemble(&frames, &prefix).map_err(|e| miette!("{path}: {e}"))?;
440    if api.declaration_names().next().is_none() {
441        return Err(no_metadata(
442            path,
443            Some((&prefix, prefixes.into_iter().collect())),
444        ));
445    }
446    Ok((api, prefix))
447}

449/// The error for a library without (the crate's) metadata.
450fn no_metadata(path: &Utf8Path, crate_prefix: Option<(&str, Vec<&str>)>) -> Report {
451    let whose = match &crate_prefix {
452        Some((prefix, _)) => format!(" for `{prefix}`"),
453        None => String::new(),
454    };
455    let others = match &crate_prefix {
456        Some((_, others)) if !others.is_empty() => {
457            format!(" (it has metadata for {})", others.join(", "))
458        }
459        _ => String::new(),
460    };
461    miette!(
462        "{path} has no WeaveFFI metadata{whose}{others}. The API is read from the library the \
463         producer builds: tag the items it exports (`#[weaveffi::export]` and the other \
464         attributes), call `weaveffi::export_runtime!();` once at the crate root, and build it \
465         again"
466    )
467}

469/// Whether this process is `krate`'s own build script.
470fn in_own_build_script(krate: &CargoCrate) -> bool {
471    std::env::var_os("OUT_DIR").is_some()
472        && std::env::var("CARGO_MANIFEST_DIR").is_ok_and(|dir| {
473            Utf8Path::new(&dir).canonicalize_utf8().ok() == krate.dir().canonicalize_utf8().ok()
474        })
475}

485    #[test]
486    fn crates_and_idls_are_told_apart() {
487        let root = repo_root();
488        for input in [
489            root.join("samples/kvstore"),
490            root.join("samples/kvstore/Cargo.toml"),
491        ] {
492            let project = Project::locate(None, Some(input.as_str()), None).unwrap();
493            let krate = project.krate().expect("a crate");
494            assert_eq!(krate.lib_name, "kvstore");
495            // The sample's weaveffi.toml sits beside its Cargo.toml.
496            assert_eq!(
497                project.config.package.identity.version.as_deref(),
498                Some("1.0.0")
499            );
500        }
501        let idl = root.join("crates/weaveffi-cli/tests/fixtures/kitchen_sink.yml");
502        let project = Project::locate(None, Some(idl.as_str()), None).unwrap();
503        assert!(matches!(project.source, Source::Idl(_)));
504        let def = project.definition().unwrap();
505        assert_eq!(def.identity.prefix, "kitchen_sink");
506        let err = project.library("libx.so").unwrap_err();
507        assert!(format!("{err}").contains("is an IDL"), "{err}");

509        let err = Project::locate(None, Some("src/lib.rs"), None).unwrap_err();

511        let err = Project::locate(None, Some(root.join("docs").as_str()), None).unwrap_err();
512        assert!(format!("{err}").contains("without a Cargo.toml"), "{err}");
