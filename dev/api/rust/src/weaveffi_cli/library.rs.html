1//! Reading a Rust producer's API out of its built library.
2//!
3//! The producer macros embed one metadata [`Frame`] per declaration in the
4//! producer's library. [`read_frames`] finds them with the `object` crate: in a native
5//! library (ELF, Mach-O, or PE, or every member of a static archive) by
6//! walking the symbol tables for the `{PREFIX}_META_{HASH16}` statics and
7//! reading each one's bytes from the section that holds it; in a `.wasm`
8//! module from the `weaveffi_meta` custom section, which the linker fills
9//! with every frame back to back.
10
11use std::collections::BTreeSet;

13use camino::Utf8Path;
14use miette::{bail, IntoDiagnostic, Result, WrapErr};
15use object::read::archive::ArchiveFile;
16use object::{BinaryFormat, Object, ObjectSection, ObjectSymbol, SymbolSection};
17use weaveffi_model::meta::{self, Frame};

19/// Every metadata frame in the library at `path`: a shared library
20/// (`.so`, `.dylib`, `.dll`), a static archive (`.a`, `.lib`), or a `.wasm`
21/// module. The frames of every crate linked into the library are returned;
22/// [`meta::assemble`] keeps one crate's.
23///
24/// # Errors
25///
26/// Returns an error when the file can't be read or isn't an object file
27/// `object` understands, or when a frame is malformed.
28pub(crate) fn read_frames(path: &Utf8Path) -> Result<Vec<Frame>> {
29    let data = std::fs::read(path)
30        .into_diagnostic()
31        .wrap_err_with(|| format!("failed to read {path}"))?;
32    let mut frames = Vec::new();
33    if let Ok(archive) = ArchiveFile::parse(&*data) {
34        for member in archive.members() {
35            let member = member
36                .into_diagnostic()
37                .wrap_err_with(|| format!("{path} is a malformed archive"))?;
38            let bytes = member
39                .data(&*data)
40                .into_diagnostic()
41                .wrap_err_with(|| format!("{path} is a malformed archive"))?;
42            // Archives also carry symbol tables and other non-object members.
43            if let Ok(file) = object::File::parse(bytes) {
44                frames.extend(object_frames(&file).wrap_err_with(|| {
45                    format!(
46                        "{path}({}) has malformed WeaveFFI metadata",
47                        String::from_utf8_lossy(member.name())
48                    )
49                })?);
50            }
51        }
52        return Ok(frames);
53    }
54    let file = object::File::parse(&*data)
55        .into_diagnostic()
56        .wrap_err_with(|| format!("{path} isn't a library WeaveFFI can read"))?;
57    object_frames(&file).wrap_err_with(|| format!("{path} has malformed WeaveFFI metadata"))
58}

60/// The frames in one object file.
61fn object_frames(file: &object::File<'_>) -> Result<Vec<Frame>> {
62    if file.format() == BinaryFormat::Wasm {
63        return match file.section_by_name(meta::SECTION) {
64            Some(section) => meta::decode(section.data().into_diagnostic()?).into_diagnostic(),
65            None => Ok(Vec::new()),
66        };
67    }
68    // Statically linked and stripped libraries keep the statics in different
69    // tables (the symbol table, the dynamic symbol table, a PE export
70    // table), so look in all of them and read each symbol once.
71    let mut candidates: Vec<(String, u64, Option<object::SectionIndex>)> = Vec::new();
72    for symbol in file.symbols().chain(file.dynamic_symbols()) {
73        let Ok(name) = symbol.name() else { continue };
74        if !meta::is_symbol(name) {
75            continue;
76        }
77        let section = match symbol.section() {
78            SymbolSection::Section(index) => Some(index),
79            _ => continue,
80        };
81        candidates.push((name.to_string(), symbol.address(), section));
82    }
83    for export in file.exports().unwrap_or_default() {
84        let name = String::from_utf8_lossy(export.name()).into_owned();
85        if meta::is_symbol(&name) {
86            candidates.push((name, export.address(), None));
87        }
88    }
89    let mut seen = BTreeSet::new();
90    let mut frames = Vec::new();
91    for (name, address, section) in candidates {
92        if !seen.insert(name.trim_start_matches('_').to_string()) {
93            continue;
94        }
95        let Some(bytes) = data_at(file, address, section) else {
96            bail!("the data of `{name}` isn't in the file");
97        };
98        let len = meta::frame_len(bytes).into_diagnostic()?;
99        let Some(frame) = bytes.get(..len) else {
100            bail!("`{name}` is truncated");
101        };
102        frames.extend(meta::decode(frame).into_diagnostic()?);
103    }
104    Ok(frames)
105}

107/// The file's bytes from `address` to the end of the section holding it
108/// (`section`, when the symbol names it).
109fn data_at<'d>(
110    file: &object::File<'d>,
111    address: u64,
112    section: Option<object::SectionIndex>,
113) -> Option<&'d [u8]> {
114    let section = match section {
115        Some(index) => file.section_by_index(index).ok()?,
116        None => file
117            .sections()
118            .find(|s| s.address() <= address && address < s.address().saturating_add(s.size()))?,
119    };
120    let data = section.data().ok()?;
121    let offset = usize::try_from(address.checked_sub(section.address())?).ok()?;
122    data.get(offset..)
123}

125/// The base name a library file is loaded by: `libkv.so`, `libkv.dylib`,
126/// `libkv.a`, `kv.dll`, `kv.lib`, and `kv.wasm` are all `kv`.
127#[must_use]
128pub(crate) fn library_name(path: &Utf8Path) -> String {
129    let stem = path.file_stem().unwrap_or_default();
130    match path.extension() {
131        Some("so" | "dylib" | "a") => stem.strip_prefix("lib").unwrap_or(stem).to_string(),
132        _ => stem.to_string(),
133    }
134}

140    #[test]
141    fn library_names_drop_platform_decoration() {
142        for (file, name) in [
143            ("target/debug/libkv_store.so", "kv_store"),
144            ("libkv.dylib", "kv"),
145            ("libkv.a", "kv"),
146            ("kv.dll", "kv"),
147            ("kv.lib", "kv"),
148            ("out/kv.wasm", "kv"),
149        ] {
150            assert_eq!(library_name(Utf8Path::new(file)), name, "{file}");
151        }
152    }
153
154    #[test]
155    fn files_without_metadata_have_no_frames_and_junk_is_an_error() {
156        let dir = tempfile::tempdir().unwrap();
157        let junk = Utf8Path::from_path(dir.path()).unwrap().join("libjunk.so");
158        std::fs::write(&junk, b"\x00fake-native\x01").unwrap();
159        let err = read_frames(&junk).unwrap_err();
160        assert!(format!("{err:#}").contains("isn't a library"), "{err:#}");
161        // A real object file with no frames: this test binary.
162        let me = std::env::current_exe().unwrap();
163        let me = Utf8Path::from_path(&me).unwrap();
164        assert!(read_frames(me).unwrap().is_empty());
165    }
