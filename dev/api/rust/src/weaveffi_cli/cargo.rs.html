1//! Resolving a Rust producer crate with `cargo metadata`.
2//!
3//! Cargo is the authority on a crate's name, version (including
4//! `version.workspace = true`), library target name, and target directory, so
5//! WeaveFFI asks it instead of parsing `Cargo.toml` itself. Every command
6//! that reads a Rust producer's API uses [`CargoCrate`] to locate the crate
7//! and [`CargoCrate::build_library`] to build the library the API is read
8//! from, and every build output lands under its
9//! [`target_dir`](CargoCrate::target_dir).
10
11use std::process::{Command, Stdio};
12
13use camino::{Utf8Path, Utf8PathBuf};
14use miette::{bail, IntoDiagnostic, Result, WrapErr};
15use serde::Deserialize;

17/// The Cargo features a producer is built with: `[build] features`,
18/// `all_features`, and `no_default_features`, or the `--features`,
19/// `--all-features`, and `--no-default-features` flags.
20///
21/// Every build of the producer uses the same set, the host build library
22/// mode reads the API from as much as each platform's build, so a
23/// declaration behind `#[cfg(feature = "...")]` is in the bindings exactly
24/// when it's in the libraries they load.
25#[derive(Debug, Clone, Default, PartialEq, Eq, Deserialize)]
26#[serde(default, deny_unknown_fields)]
27pub struct CargoFeatures {
28    /// Features to enable (`--features`).
29    pub features: Vec<String>,
30    /// Enable every feature (`--all-features`).
31    pub all_features: bool,
32    /// Disable the default features (`--no-default-features`).
33    pub no_default_features: bool,
34}
35
36impl CargoFeatures {
37    /// Whether this is Cargo's default (no flag to pass).
38    #[must_use]
39    pub fn is_default(&self) -> bool {
40        *self == Self::default()
41    }
42
43    /// Append the Cargo flags for this set to `cmd`.
44    pub(crate) fn apply(&self, cmd: &mut Command) {
45        if !self.features.is_empty() {
46            cmd.arg("--features").arg(self.features.join(","));
47        }
48        if self.all_features {
49            cmd.arg("--all-features");
50        }
51        if self.no_default_features {
52            cmd.arg("--no-default-features");
53        }
54    }
55}

57/// The facts about a Rust producer crate that building and packaging need,
58/// as `cargo metadata` reports them.
59#[derive(Debug, Clone, PartialEq, Eq)]
60pub struct CargoCrate {
61    /// The Cargo package name (`my-kv`).
62    pub name: String,
63    /// The resolved package version, with workspace inheritance applied.
64    pub version: String,
65    /// The library target's name with `-` mapped to `_` (`my_kv`): the base
66    /// name of the `lib{lib_name}.so` Cargo produces.
67    pub lib_name: String,
68    /// The absolute path of the crate's `Cargo.toml`.
69    pub manifest_path: Utf8PathBuf,
70    /// The Cargo target directory (respecting `CARGO_TARGET_DIR`,
71    /// `build.target-dir`, and workspaces).
72    pub target_dir: Utf8PathBuf,
73    /// The package description, if declared.
74    pub description: Option<String>,
75    /// The package license expression, if declared.
76    pub license: Option<String>,
77    /// The package authors.
78    pub authors: Vec<String>,
79    /// The package homepage, if declared.
80    pub homepage: Option<String>,
81    /// The package repository, if declared.
82    pub repository: Option<String>,
83}
84
85impl CargoCrate {
86    /// Resolve the crate whose manifest is `manifest` by running
87    /// `cargo metadata --no-deps`.
88    ///
89    /// # Errors
90    ///
91    /// Returns an error when `manifest` doesn't exist, when Cargo can't be
92    /// run or rejects the manifest, when the manifest is a virtual workspace
93    /// manifest with no package of its own, or when the package has no
94    /// library target.
95    pub fn resolve(manifest: &Utf8Path) -> Result<Self> {
96        if !manifest.is_file() {
97            bail!("no Cargo.toml at {manifest}");
98        }
99        let output = Command::new(cargo_bin())
100            .args(["metadata", "--format-version", "1", "--no-deps"])
101            .arg("--manifest-path")
102            .arg(manifest.as_str())
103            .output()
104            .into_diagnostic()
105            .wrap_err("failed to run `cargo metadata`; is Cargo installed and on PATH?")?;
106        if !output.status.success() {
107            bail!(
108                "`cargo metadata` failed for {manifest}:\n{}",
109                String::from_utf8_lossy(&output.stderr).trim_end()
110            );
111        }
112        let metadata: Metadata = serde_json::from_slice(&output.stdout)
113            .into_diagnostic()
114            .wrap_err("failed to parse `cargo metadata` output")?;
115        let wanted = canonical(manifest);
116        let Some(package) = metadata
117            .packages
118            .into_iter()
119            .find(|p| canonical(&p.manifest_path) == wanted)
120        else {
121            bail!(
122                "{manifest} is a virtual workspace manifest; point at the producer crate's own \
123                 Cargo.toml instead"
124            );
125        };
126        let Some(lib) = package.targets.iter().find(|t| t.is_lib()) else {
127            bail!(
128                "the crate `{}` has no library target; a WeaveFFI producer is a library \
129                 (src/lib.rs)",
130                package.name
131            );
132        };
133        Ok(Self {
134            lib_name: lib.name.replace('-', "_"),
135            name: package.name,
136            version: package.version,
137            manifest_path: package.manifest_path,
138            target_dir: metadata.target_directory,
139            description: package.description,
140            license: package.license,
141            authors: package.authors,
142            homepage: package.homepage,
143            repository: package.repository,
144        })
145    }
146
147    /// The directory `weaveffi build` lays its per-platform outputs out in:
148    /// `{target_dir}/weaveffi`.
149    #[must_use]
150    pub fn weaveffi_dir(&self) -> Utf8PathBuf {
151        self.target_dir.join("weaveffi")
152    }
153
154    /// The directory holding the crate's `Cargo.toml`.
155    #[must_use]
156    pub fn dir(&self) -> &Utf8Path {
157        self.manifest_path.parent().unwrap_or(Utf8Path::new("."))
158    }
159
160    /// Build the crate's library for the host as a `cdylib` with `cargo
161    /// rustc --lib --crate-type cdylib --profile {profile}` and `features`
162    /// (so the crate needs no particular `crate-type`), forwarding Cargo's
163    /// diagnostics to stderr, and return the shared library it produced.
164    pub(crate) fn build_library(
165        &self,
166        profile: &str,
167        features: &CargoFeatures,
168        quiet: bool,
169    ) -> Result<Utf8PathBuf> {
170        let mut cmd = Command::new(cargo_bin());
171        cmd.args(["rustc", "--lib", "--crate-type", "cdylib"])
172            .args(["--profile", profile])
173            .arg("--message-format=json-render-diagnostics")
174            .arg("--manifest-path")
175            .arg(self.manifest_path.as_str());
176        features.apply(&mut cmd);
177        if quiet {
178            cmd.arg("--quiet");
179        }
180        let output = cmd
181            .stdout(Stdio::piped())
182            .stderr(Stdio::inherit())
183            .output()
184            .into_diagnostic()
185            .wrap_err("failed to run `cargo rustc`; is Cargo installed and on PATH?")?;
186        if !output.status.success() {
187            bail!(
188                "cargo failed to build `{}`; see its errors above",
189                self.name
190            );
191        }
192        let files = self.lib_artifacts(&output.stdout);
193        files
194            .iter()
195            .find(|f| {
196                [".dylib", ".so", ".dll"]
197                    .iter()
198                    .any(|s| f.as_str().ends_with(s))
199            })
200            .cloned()
201            .ok_or_else(|| {
202                miette::miette!("cargo built `{}` but reported no shared library", self.name)
203            })
204    }
205
206    /// The files Cargo reported for this crate's library target, from the
207    /// JSON messages (`--message-format=json`) it printed to stdout.
208    ///
209    /// Each is the file rustc wrote into the profile's `deps` directory
210    /// rather than the copy Cargo reports, which it uplifts into the profile
211    /// directory: Cargo deletes and re-creates that copy on every run, even
212    /// one with nothing to rebuild, so another `cargo` run on the same
213    /// target directory (a concurrent `weaveffi` command, an editor) can
214    /// remove it while this process reads it. The `deps` file changes only
215    /// when the crate is rebuilt. A file with no `deps` original (a name
216    /// Cargo decorated with a hash) is returned as reported.
217    #[must_use]
218    pub fn lib_artifacts(&self, stdout: &[u8]) -> Vec<Utf8PathBuf> {
219        let manifest = canonical(&self.manifest_path);
220        let mut out = Vec::new();
221        for line in String::from_utf8_lossy(stdout).lines() {
222            let Ok(msg) = serde_json::from_str::<serde_json::Value>(line) else {
223                continue;
224            };
225            if msg["reason"] != "compiler-artifact" {
226                continue;
227            }
228            let ours = msg["manifest_path"]
229                .as_str()
230                .is_some_and(|p| canonical(Utf8Path::new(p)) == manifest);
231            let lib = msg["target"]["kind"]
232                .as_array()
233                .is_some_and(|kinds| kinds.iter().any(|k| k != "custom-build"));
234            if !ours || !lib {
235                continue;
236            }
237            if let Some(files) = msg["filenames"].as_array() {
238                out.extend(
239                    files
240                        .iter()
241                        .filter_map(|f| f.as_str())
242                        .map(|f| unuplifted(Utf8Path::new(f))),
243                );
244            }
245        }
246        out
247    }
248}

250/// The `cargo` executable: `$CARGO` when running under Cargo, else `cargo`.
251#[must_use]
252pub fn cargo_bin() -> String {
253    std::env::var("CARGO").unwrap_or_else(|_| "cargo".to_string())
254}

256/// The original in `deps/` of a file Cargo uplifted to `path`, if there is
257/// one (see [`CargoCrate::lib_artifacts`]), else `path`.
258fn unuplifted(path: &Utf8Path) -> Utf8PathBuf {
259    if let (Some(dir), Some(name)) = (path.parent(), path.file_name()) {
260        let original = dir.join("deps").join(name);
261        if original.is_file() {
262            return original;
263        }
264    }
265    path.to_path_buf()
266}

268/// `path` with symlinks and `..` resolved where possible, for comparing
269/// manifest paths.
270fn canonical(path: &Utf8Path) -> Utf8PathBuf {
271    path.canonicalize_utf8()
272        .unwrap_or_else(|_| path.to_path_buf())
273}

275#[derive(Deserialize)]
276struct Metadata {
277    packages: Vec<Package>,
278    target_directory: Utf8PathBuf,
279}

281#[derive(Deserialize)]
282struct Package {
283    name: String,
284    version: String,
285    manifest_path: Utf8PathBuf,
286    targets: Vec<Target>,
287    description: Option<String>,
288    license: Option<String>,
289    #[serde(default)]
290    authors: Vec<String>,
291    homepage: Option<String>,
292    repository: Option<String>,
293}
294
295#[derive(Deserialize)]
296struct Target {
297    name: String,
298    kind: Vec<String>,
299}
300
301impl Target {
302    fn is_lib(&self) -> bool {
303        self.kind.iter().any(|k| {
304            matches!(
305                k.as_str(),
306                "lib" | "rlib" | "dylib" | "cdylib" | "staticlib" | "proc-macro"
307            )
308        })
309    }
310}

312#[cfg(test)]
313mod tests {
314    use super::*;
315
316    fn repo_root() -> Utf8PathBuf {
317        Utf8Path::new(env!("CARGO_MANIFEST_DIR")).join("../..")
318    }
319
320    #[test]
321    fn resolves_a_workspace_member() {
322        let manifest = repo_root().join("samples/calculator/Cargo.toml");
323        let krate = CargoCrate::resolve(&manifest).unwrap();
324        assert_eq!(krate.name, "calculator");
325        assert_eq!(krate.lib_name, "calculator");
326        assert!(krate.target_dir.ends_with("target"), "{}", krate.target_dir);
327        assert_eq!(krate.weaveffi_dir(), krate.target_dir.join("weaveffi"));
328    }
329
330    #[test]
331    fn inherited_versions_are_resolved() {
332        let krate = CargoCrate::resolve(&repo_root().join("crates/weaveffi/Cargo.toml")).unwrap();
333        assert_eq!(krate.version, env!("CARGO_PKG_VERSION"));
334    }
335
336    #[test]
337    fn virtual_and_missing_manifests_are_errors() {
338        let err = CargoCrate::resolve(&repo_root().join("Cargo.toml")).unwrap_err();
339        assert!(err.to_string().contains("virtual workspace"), "{err}");
340        let err = CargoCrate::resolve(Utf8Path::new("/nonexistent/Cargo.toml")).unwrap_err();
341        assert!(err.to_string().contains("no Cargo.toml"), "{err}");
342    }
