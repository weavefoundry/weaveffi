1//! Cross-compiling a Rust producer for each platform: `weaveffi build`.
2//!
3//! [`Builder`] runs `cargo rustc --lib` once per [`Platform`] with the
4//! artifact kinds that platform's packages need (a `cdylib` for desktop and
5//! Android, plus a `staticlib` for an `XCFramework` on Apple platforms, a
6//! `staticlib` alone for iOS, and a `.wasm` module for `wasm32`), so a
7//! producer's `Cargo.toml` needs no particular `crate-type`. It fixes what a
8//! plain `cargo build` leaves machine-specific (a macOS library's install
9//! name becomes `@rpath/lib{library}.dylib`, a Linux or Android library's
10//! soname `lib{library}.so`), sets Apple deployment targets and the Android
11//! NDK toolchain, and lays the results out in
12//! `{target_dir}/weaveffi/<platform>/`, where `weaveffi package` reads them
13//! back as a [`BinarySet`](crate::platform::BinarySet).
14
15pub(crate) mod ndk;
16

28/// Build settings shared by every platform (the `[build]` table).
29#[derive(Debug, Clone, PartialEq, Eq)]
30pub(crate) struct BuildSettings {
31    /// The Cargo profile to build with (`--profile`).
32    pub(crate) profile: String,
33    /// The Cargo features to build with.
34    pub(crate) features: CargoFeatures,
35    /// The minimum macOS version (`MACOSX_DEPLOYMENT_TARGET`).
36    pub(crate) macos_deployment_target: String,
37    /// The minimum iOS version (`IPHONEOS_DEPLOYMENT_TARGET`).
38    pub(crate) ios_deployment_target: String,
39    /// The minimum Android API level, which selects the NDK compiler.
40    pub(crate) android_api: u32,
41}
42
43impl Default for BuildSettings {
44    /// Release builds for macOS 11.0, iOS 13.0, and Android API 21.
45    fn default() -> Self {
46        Self {
47            profile: "release".into(),
48            features: CargoFeatures::default(),
49            macos_deployment_target: "11.0".into(),
50            ios_deployment_target: "13.0".into(),
51            android_api: 21,
52        }
53    }
54}

56/// Cross-compiles one producer crate into `{target_dir}/weaveffi/<platform>/`.
57pub(crate) struct Builder<'a> {
58    krate: &'a CargoCrate,
59    library: &'a str,
60    settings: &'a BuildSettings,
61    quiet: bool,
62    installed_targets: OnceLock<Option<Vec<String>>>,
63    ndk: OnceLock<Result<Ndk, String>>,
64}
65
66impl<'a> Builder<'a> {
67    /// A builder for `krate` whose outputs are named after `library` (the
68    /// identity's library, which the generated bindings load).
69    pub(crate) fn new(
70        krate: &'a CargoCrate,
71        library: &'a str,
72        settings: &'a BuildSettings,
73    ) -> Self {
74        Self {
75            krate,
76            library,
77            settings,
78            quiet: false,
79            installed_targets: OnceLock::new(),
80            ndk: OnceLock::new(),
81        }
82    }
83
84    /// Pass `--quiet` to Cargo.
85    #[must_use]
86    pub(crate) fn quiet(mut self, quiet: bool) -> Self {
87        self.quiet = quiet;
88        self
89    }
90
91    /// The directory a platform's outputs are laid out in:
92    /// `{target_dir}/weaveffi/<platform-id>`.
93    #[must_use]
94    pub(crate) fn output_dir(&self, platform: Platform) -> Utf8PathBuf {
95        self.krate.weaveffi_dir().join(platform.id())
96    }
97
98    /// The Android NDK, located once.
99    ///
100    /// # Errors
101    ///
102    /// Returns an error when no NDK is found.
103    pub(crate) fn ndk(&self) -> Result<&Ndk> {
104        self.ndk
105            .get_or_init(|| Ndk::locate().map_err(|e| e.to_string()))
106            .as_ref()
107            .map_err(|e| miette::miette!("{e}"))
108    }
109
110    /// Check, before compiling anything, that `platform` can be built here:
111    /// a suitable host, an installed rustup target, and (for Android) an
112    /// NDK.
113    ///
114    /// # Errors
115    ///
116    /// Returns an error naming the missing piece and how to add it.
117    pub(crate) fn check(&self, platform: Platform) -> Result<()> {
118        platform.check_host()?;
119        let triple = platform.rust_target();
120        if let Some(installed) = self.installed_targets() {
121            if !installed.iter().any(|t| t == triple) {
122                bail!(
123                    "the Rust target `{triple}` ({}) isn't installed; run `rustup target add \
124                     {triple}`",
125                    platform.id()
126                );
127            }
128        }
129        if platform.os() == Os::Android {
130            self.ndk()?.clang(platform, self.settings.android_api)?;
131        }
132        Ok(())
133    }
134
135    /// The rustup targets installed for the crate's toolchain, or `None`
136    /// when rustup isn't available (a distribution's Rust, for example).
137    fn installed_targets(&self) -> Option<&Vec<String>> {
138        self.installed_targets
139            .get_or_init(|| {
140                let dir = self.krate.manifest_path.parent()?;
141                let out = Command::new("rustup")
142                    .args(["target", "list", "--installed"])
143                    .current_dir(dir)
144                    .output()
145                    .ok()?;
146                out.status.success().then(|| {
147                    String::from_utf8_lossy(&out.stdout)
148                        .lines()
149                        .map(|l| l.trim().to_string())
150                        .collect()
151                })
152            })
153            .as_ref()
154    }
155
156    /// Cross-compile the crate for `platform` and lay the results out in
157    /// [`output_dir`](Self::output_dir), replacing anything there.
158    ///
159    /// # Errors
160    ///
161    /// Returns an error when [`check`](Self::check) fails, Cargo fails, or
162    /// Cargo produces none of the expected libraries.
163    pub(crate) fn build(&self, platform: Platform) -> Result<NativeBinary> {
164        self.check(platform)?;
165        let triple = platform.rust_target();
166        let library = self.library;
167        let crate_types = match platform.os() {
168            Os::Ios => "staticlib",
169            Os::MacOs => "cdylib,staticlib",
170            _ => "cdylib",
171        };
172        let mut link_args: Vec<String> = Vec::new();
173        match platform.os() {
174            Os::MacOs => {
175                link_args.push(format!("-Wl,-install_name,@rpath/lib{library}.dylib"));
176            }
177            Os::Linux => link_args.push(format!("-Wl,-soname,lib{library}.so")),
178            Os::Android => {
179                link_args.push(format!("-Wl,-soname,lib{library}.so"));
180                link_args.push("-Wl,-z,max-page-size=16384".into());
181            }
182            Os::Wasm => {
183                link_args.push("--export-table".into());
184                link_args.push("--growable-table".into());
185            }
186            Os::Windows | Os::Ios => {}
187        }
188
189        let mut cmd = Command::new(cargo_bin());
190        cmd.args(["rustc", "--lib", "--message-format=json-render-diagnostics"])
191            .arg("--manifest-path")
192            .arg(self.krate.manifest_path.as_str())
193            .args(["--target", triple, "--crate-type", crate_types]);
194        cmd.args(["--profile", &self.settings.profile]);
195        self.settings.features.apply(&mut cmd);
196        if self.quiet {
197            cmd.arg("--quiet");
198        }
199        if !link_args.is_empty() {
200            cmd.arg("--");
201            for arg in &link_args {
202                cmd.arg("-C").arg(format!("link-arg={arg}"));
203            }
204        }
205        match platform.os() {
206            Os::MacOs => {
207                cmd.env(
208                    "MACOSX_DEPLOYMENT_TARGET",
209                    &self.settings.macos_deployment_target,
210                );
211            }
212            Os::Ios => {
213                cmd.env(
214                    "IPHONEOS_DEPLOYMENT_TARGET",
215                    &self.settings.ios_deployment_target,
216                );
217            }
218            Os::Android => {
219                let ndk = self.ndk()?;
220                let clang = ndk.clang(platform, self.settings.android_api)?;
221                let env_triple = triple.replace('-', "_");
222                cmd.env(
223                    format!("CARGO_TARGET_{}_LINKER", env_triple.to_uppercase()),
224                    clang.as_str(),
225                )
226                .env(format!("CC_{env_triple}"), clang.as_str())
227                .env(format!("AR_{env_triple}"), ndk.ar()?.as_str());
228            }
229            Os::Linux | Os::Windows | Os::Wasm => {}
230        }
231        let output = cmd
232            .stdout(Stdio::piped())
233            .stderr(Stdio::inherit())
234            .output()
235            .into_diagnostic()
236            .wrap_err("failed to run cargo")?;
237        if !output.status.success() {
238            bail!(
239                "cargo failed to build `{}` for {} ({triple}); see its errors above",
240                self.krate.name,
241                platform.id()
242            );
243        }
244        let produced = self.krate.lib_artifacts(&output.stdout);
245
246        let dir = self.output_dir(platform);
247        if dir.exists() {
248            std::fs::remove_dir_all(dir.as_std_path())
249                .into_diagnostic()
250                .wrap_err_with(|| format!("failed to clear {dir}"))?;
251        }
252        std::fs::create_dir_all(dir.as_std_path())
253            .into_diagnostic()
254            .wrap_err_with(|| format!("failed to create {dir}"))?;
255        let place = |suffix: &str, name: String| -> Result<Option<Utf8PathBuf>> {
256            let Some(source) = produced.iter().find(|p| p.as_str().ends_with(suffix)) else {
257                return Ok(None);
258            };
259            let dest = dir.join(name);
260            std::fs::copy(source.as_std_path(), dest.as_std_path())
261                .into_diagnostic()
262                .wrap_err_with(|| format!("failed to copy {source} to {dest}"))?;
263            Ok(Some(dest))
264        };
265        let primary = platform.lib_filename(library);
266        let (suffix, name) = match platform.os() {
267            Os::MacOs => (".dylib", primary),
268            Os::Linux | Os::Android => (".so", primary),
269            Os::Windows => (".dll", primary),
270            Os::Wasm => (".wasm", primary),
271            Os::Ios => (".a", primary),
272        };
273        let Some(lib) = place(suffix, name)? else {
274            bail!(
275                "cargo built `{}` for {triple} but produced no {suffix} library",
276                self.krate.name
277            );
278        };
279        let mut binary = NativeBinary::new(platform, lib);
280        match platform.os() {
281            Os::Ios => binary.staticlib = Some(binary.library.clone()),
282            Os::MacOs => binary.staticlib = place(".a", format!("lib{library}.a"))?,
283            Os::Windows => binary.import_library = place(".dll.lib", format!("{library}.dll.lib"))?,
284            _ => {}
285        }
286        Ok(binary)
287    }
288}

290/// An error and its causes on one line, for a skip warning.
291pub(crate) fn one_line(e: &miette::Report) -> String {
292    e.chain()
293        .map(ToString::to_string)
294        .collect::<Vec<_>>()
295        .join(": ")
