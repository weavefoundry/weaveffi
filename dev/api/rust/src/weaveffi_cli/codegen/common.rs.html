1//! Shared codegen primitives that every language generator can reuse: the
2//! doc-comment emitter, prose wrapping, and `PascalCase` conversion.
3//!
4//! Specialised flavours that exist in only one generator (Go's
5//! godoc-style first-line symbol prefix, .NET's `<summary>` XML tags,
6//! Python's triple-quoted docstring) stay generator-local because
7//! their behaviour is non-uniform; this module deliberately covers
8//! only the common 80%.
9
10/// Doc-comment flavour used by [`emit_doc`].
11///
12/// Specialised flavours like Go's godoc-symbol prefix or .NET's
13/// `<summary>` element are intentionally absent and remain in their
14/// own generators because their first-line behaviour is non-uniform.
15#[derive(Debug, Clone, Copy, PartialEq, Eq)]
16pub enum DocCommentStyle {
17    /// `/// ...` per line (Swift, Dart, Rust).
18    TripleSlash,
19    /// `# ...` per line (Python `#` comments, Ruby).
20    Hash,
21    /// `/** ... */` block; single-line collapses to `/** text */`
22    /// (C, C++, Kotlin/KDoc, JSDoc, TypeScript .d.ts).
23    Javadoc,
24}
25
26/// Emit a doc comment for `doc` at the given `indent`, using the given
27/// `style`. No-op when `doc` is `None` or trims to empty.
28///
29/// The output always ends with a trailing newline so a generator can
30/// follow it directly with the symbol declaration on the next line.
31pub fn emit_doc(out: &mut String, doc: &Option<String>, indent: &str, style: DocCommentStyle) {
32    let Some(doc) = doc else {
33        return;
34    };
35    let doc = doc.trim();
36    if doc.is_empty() {
37        return;
38    }
39    match style {
40        DocCommentStyle::TripleSlash => emit_line_doc(out, doc, indent, "///"),
41        DocCommentStyle::Hash => emit_line_doc(out, doc, indent, "#"),
42        DocCommentStyle::Javadoc => emit_javadoc(out, doc, indent),
43    }
44}
45
46fn emit_line_doc(out: &mut String, doc: &str, indent: &str, marker: &str) {
47    for line in doc.lines() {
48        out.push_str(indent);
49        if line.is_empty() {
50            out.push_str(marker);
51            out.push('\n');
52        } else {
53            out.push_str(marker);
54            out.push(' ');
55            out.push_str(line);
56            out.push('\n');
57        }
58    }
59}

61fn emit_javadoc(out: &mut String, doc: &str, indent: &str) {
62    // A `*/` in the text would end the comment early.
63    let escaped;
64    let doc = if doc.contains("*/") {
65        escaped = doc.replace("*/", "*\\/");
66        escaped.as_str()
67    } else {
68        doc
69    };
70    if doc.contains('\n') {
71        out.push_str(indent);
72        out.push_str("/**\n");
73        for line in doc.lines() {
74            out.push_str(indent);
75            if line.is_empty() {
76                out.push_str(" *\n");
77            } else {
78                out.push_str(" * ");
79                out.push_str(line);
80                out.push('\n');
81            }
82        }
83        out.push_str(indent);
84        out.push_str(" */\n");
85    } else {
86        out.push_str(indent);
87        out.push_str("/** ");
88        out.push_str(doc);
89        out.push_str(" */\n");
90    }
91}

93/// Word-wrap generated prose into lines of at most `width` columns (a
94/// longer word gets a line of its own), keeping blank-line paragraph
95/// breaks. For text the generator writes; IDL docs keep their own breaks.
96#[must_use]
97pub fn wrap(text: &str, width: usize) -> String {
98    let mut lines: Vec<String> = Vec::new();
99    for paragraph in text.split("\n\n") {
100        if !lines.is_empty() {
101            lines.push(String::new());
102        }
103        let mut line = String::new();
104        for word in paragraph.split_whitespace() {
105            if !line.is_empty() && line.len() + 1 + word.len() > width {
106                lines.push(std::mem::take(&mut line));
107            }
108            if !line.is_empty() {
109                line.push(' ');
110            }
111            line.push_str(word);
112        }
113        lines.push(line);
114    }
115    lines.join("\n")
116}

118/// Convert a `snake_case` identifier to `PascalCase` by uppercasing the
119/// first character of each `_`-separated segment and preserving the rest.
120///
121/// This deliberately splits on `_` only; it does **not** re-case interior
122/// letters the way `heck::ToUpperCamelCase` does, so an acronym-bearing
123/// name like `get_HTTP` becomes `GetHTTP`, not `GetHttp`. It is the single
124/// source of truth for the `snake_to_pascal` / `to_pascal_case` helpers
125/// that the Python, Android, and Wasm generators each defined locally.
126pub fn pascal_case(s: &str) -> String {
127    s.split('_')
128        .map(|part| {
129            let mut chars = part.chars();
130            match chars.next() {
131                None => String::new(),
132                Some(first) => first.to_uppercase().chain(chars).collect::<String>(),
133            }
134        })
135        .collect()
136}
