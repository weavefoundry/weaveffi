1//! How each value crosses a call boundary: the one dispatch every target's
2//! call layer matches on.
3//!
4//! In ABI revision 6 a value is one C value in every position, so a target
5//! marshals a parameter, a return, an async result, an iterator item, or a
6//! callback argument or return with the same per-[`Crossing`] code, varying
7//! only in who owns what (see the C ABI contract):
8//!
9//! | Crossing | C value | Into the producer (parameter, callback return) | Out of the producer (return, result, item, callback argument) |
10//! |---|---|---|---|
11//! | [`Scalar`](Crossing::Scalar) | the scalar | range-check, pass | use |
12//! | [`Enum`](Crossing::Enum) | `int32_t` | pass the discriminant | wrap in the enum |
13//! | [`OptScalar`](Crossing::OptScalar) | `{p}_opt_{prim}` | `{has, value}` | `has ? value : none` |
14//! | [`Utf8`](Crossing::Utf8), [`Bytes`](Crossing::Bytes), [`Slice`](Crossing::Slice), [`Buffer`](Crossing::Buffer) | `{p}_run` | encode into a run: borrowed for a parameter, from `{p}_alloc` for a callback return | decode the run; release it with `{p}_run_free` unless it's a callback argument (borrowed) |
15//! | [`Object`](Crossing::Object) | object pointer | the wrapper's pointer (borrowed); a fresh `_clone` for a callback return | adopt the reference |
16//!
17//! The model already classified every value ([`Ty::family`]); this module
18//! only names the cases a target writes code for.
19
20use weaveffi_model::ty::{Family, Prim, RunKind, Ty};
21
22/// How one value crosses.
23#[derive(Debug, Clone, Copy, PartialEq, Eq)]
24pub(crate) enum Crossing<'a> {
25    /// An integer, float, or `bool` by value.
26    Scalar(Prim),
27    /// A C-style enum, as its `int32_t` discriminant.
28    Enum(&'a str),
29    /// An optional scalar, as a `{p}_opt_{prim}`.
30    OptScalar(Prim),
31    /// An optional C-style enum, as a `{p}_opt_i32`.
32    OptEnum(&'a str),
33    /// A string, as a run of UTF-8.
34    Utf8,
35    /// Bytes, as a raw run.
36    Bytes,
37    /// A typed array of a numeric primitive, as a run of packed elements.
38    Slice(Prim),
39    /// A value buffer of type `ty` (a record, a rich enum, or an optional,
40    /// list, or map of any other family).
41    Buffer(&'a Ty),
42    /// An object pointer to `interface`, null for none when `nullable`.
43    Object {
44        /// The interface.
45        interface: &'a str,
46        /// `true` for `Interface?`.
47        nullable: bool,
48    },
49}
50
51impl Crossing<'_> {
52    /// `true` when the value crosses as a `{p}_run`.
53    #[must_use]
54    pub(crate) fn is_run(&self) -> bool {
55        matches!(
56            self,
57            Crossing::Utf8 | Crossing::Bytes | Crossing::Slice(_) | Crossing::Buffer(_)
58        )
59    }
60}
61
62/// How a value of type `ty` crosses.
63#[must_use]
64pub(crate) fn crossing(ty: &Ty) -> Crossing<'_> {
65    match (ty.family(), ty) {
66        (Family::Direct, Ty::Enum(name)) => Crossing::Enum(name),
67        (Family::Direct, _) => Crossing::Scalar(ty.scalar().expect("a Direct value is a scalar")),
68        (Family::OptDirect, Ty::Optional(inner)) => match inner.as_ref() {
69            Ty::Enum(name) => Crossing::OptEnum(name),
70            other => Crossing::OptScalar(other.scalar().expect("an OptDirect payload is a scalar")),
71        },
72        (Family::Run(RunKind::Utf8), _) => Crossing::Utf8,
73        (Family::Run(RunKind::Bytes), _) => Crossing::Bytes,
74        (Family::Run(RunKind::Slice(p)), _) => Crossing::Slice(p),
75        (Family::Run(RunKind::Buffer), _) => Crossing::Buffer(ty),
76        (Family::Object { nullable }, _) => Crossing::Object {
77            interface: ty
78                .interface_name()
79                .expect("an Object value names an interface"),
80            nullable,
81        },
82        (Family::OptDirect, _) => unreachable!("only an optional is OptDirect"),
83    }
84}

86#[cfg(test)]
87mod tests {
88    use super::*;
89
90    #[test]
91    fn every_family_has_a_crossing() {
92        let opt = |t| Ty::Optional(Box::new(t));
93        let list = |t| Ty::List(Box::new(t));
94        assert_eq!(
95            crossing(&Ty::Prim(Prim::Bool)),
96            Crossing::Scalar(Prim::Bool)
97        );
98        assert_eq!(crossing(&Ty::Enum("E".into())), Crossing::Enum("E"));
99        assert_eq!(
100            crossing(&opt(Ty::Prim(Prim::U8))),
101            Crossing::OptScalar(Prim::U8)
102        );
103        assert_eq!(crossing(&opt(Ty::Enum("E".into()))), Crossing::OptEnum("E"));
104        assert_eq!(crossing(&Ty::Prim(Prim::String)), Crossing::Utf8);
105        assert_eq!(crossing(&Ty::Prim(Prim::Bytes)), Crossing::Bytes);
106        assert_eq!(
107            crossing(&list(Ty::Prim(Prim::F64))),
108            Crossing::Slice(Prim::F64)
109        );
110        let rec = Ty::Record("R".into());
111        assert_eq!(crossing(&rec), Crossing::Buffer(&rec));
112        assert!(crossing(&rec).is_run());
113        assert_eq!(
114            crossing(&opt(Ty::Interface("I".into()))),
115            Crossing::Object {
116                interface: "I",
117                nullable: true
118            }
119        );
120    }
