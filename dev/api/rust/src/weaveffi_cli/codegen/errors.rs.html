1//! The **error tables**: every error domain with its codes and the type
2//! names a target gives them.
3//!
4//! A function declared `throws: KvError` fails with one of `KvError`'s
5//! positive codes (plus its fields as a value-buffer payload when the code
6//! declares any), a runtime code (negative), or, when the producer is newer
7//! than the bindings, a positive code the bindings don't know. Domains are
8//! open: an unknown positive code maps to the domain's base error type with
9//! its code and message preserved, never to a crash. `throws: any` fails
10//! with code -1 and a message only.
11//!
12//! Type names come from [`weaveffi_model::errors::type_name`] (one suffix,
13//! never doubled: `KitchenErrors` with `Error` is `KitchenError`, `KvError`
14//! stays `KvError`, `Failure` becomes `FailureError`), so no target names a
15//! domain or code by hand. A target picks the suffix its ecosystem uses
16//! (`"Error"`, `"Exception"`).
17//!
18//! # Example
19//!
20//! A target that raises one exception class per code, each deriving from
21//! its domain's base class, and maps codes back when decoding a failure:
22//!
23//! ```ignore
24//! use crate::codegen::errors;
25//!
26//! for table in errors::tables(model, "Exception") {
27//!     w.line(format!("class {}(LibraryError): ...", table.type_name));
28//!     for row in &table.codes {
29//!         w.line(format!(
30//!             "class {}({}): code = {}",
31//!             row.type_name, table.type_name, row.code.value
32//!         ));
33//!     }
34//!     // An unknown positive code: raise `table.type_name` itself.
35//! }
36//! ```
37
38use weaveffi_model::errors::type_name;
39use weaveffi_model::model::{ErrorBinding, ErrorCodeBinding, Model};
40
41/// One error domain and its codes, named for a target.
42#[derive(Debug, Clone)]
43pub(crate) struct ErrorTable<'m> {
44    /// The domain as lowered (raw `name`, `c_tag`, ...).
45    pub(crate) domain: &'m ErrorBinding,
46    /// The domain's base type name: `type_name(domain.name, suffix)`.
47    pub(crate) type_name: String,
48    /// The codes in declaration order.
49    pub(crate) codes: Vec<CodeRow<'m>>,
50}
51
52/// One code of an [`ErrorTable`].
53#[derive(Debug, Clone)]
54pub(crate) struct CodeRow<'m> {
55    /// The code as lowered (`name`, `value`, `message`, `c_const`, fields).
56    pub(crate) code: &'m ErrorCodeBinding,
57    /// The code's own type name: `type_name(code.name, suffix)`
58    /// (`NotFound` -> `NotFoundError`), for targets with one type per code.
59    /// Targets that nest codes as cases of the domain type use
60    /// [`weaveffi_model::errors::pascal`] instead.
61    pub(crate) type_name: String,
62}
63
64/// Every error domain in `model`, in declaration order, with type names
65/// built with `suffix`.
66#[must_use]
67pub(crate) fn tables<'m>(model: &'m Model, suffix: &str) -> Vec<ErrorTable<'m>> {
68    model
69        .errors
70        .iter()
71        .map(|domain| ErrorTable {
72            domain,
73            type_name: type_name(&domain.name, suffix),
74            codes: domain
75                .codes
76                .iter()
77                .map(|code| CodeRow {
78                    code,
79                    type_name: type_name(&code.name, suffix),
80                })
81                .collect(),
82        })
83        .collect()
84}

86#[cfg(test)]
87mod tests {
88    use super::*;
89    use crate::codegen::test_model;
90
91    #[test]
92    fn names_never_double_the_suffix() {
93        let model = test_model(
94            r#"
95version: "0.13.0"
96errors:
97  - name: KitchenErrors
98    codes: [{ name: NOT_FOUND, code: 1, message: missing }]
99  - name: Failure
100    codes: [{ name: Broken, code: 1, message: broken }]
101  - name: KvError
102    codes: [{ name: Full, code: 1, message: full }]
103"#,
104        );
105        let names: Vec<(String, Vec<String>)> = tables(&model, "Error")
106            .into_iter()
107            .map(|t| {
108                (
109                    t.type_name.clone(),
110                    t.codes.iter().map(|c| c.type_name.clone()).collect(),
111                )
112            })
113            .collect();
114        assert_eq!(
115            names,
116            [
117                ("KitchenError".into(), vec!["NotFoundError".into()]),
118                ("FailureError".into(), vec!["BrokenError".into()]),
119                ("KvError".into(), vec!["FullError".into()]),
120            ]
121        );
122        let exceptions = tables(&model, "Exception");
123        assert_eq!(exceptions[0].type_name, "KitchenException");
124        assert_eq!(exceptions[0].codes[0].type_name, "NotFoundException");
125    }
126}
