1//! Orchestration: rendering targets and turning their output into a
2//! [`Changeset`] against an output directory.
3//!
4//! Rendering is pure: a [`Target`] returns its files in memory, with paths
5//! relative to its own directory (`{out_dir}/{target}/`). The
6//! [`Orchestrator`] renders every target, compares the result with the
7//! output directory ([`Orchestrator::plan`]), and only then touches disk
8//! ([`Changeset::apply`]). That one plan is what `weaveffi generate` writes,
9//! what `generate --check` reports, and what `generate --diff` prints. A
10//! plan rewrites only files whose contents changed and removes files a
11//! previous run wrote that the current run no longer produces (see the
12//! generation records in `.weaveffi/`).
13//!
14//! The submodules are the **shared emitters** every target renders through
15//! instead of keeping its own copy:
16//!
17//! - `codecs`: the value-buffer composites an API uses and their one
18//!   canonical stem (`list_i32`, `opt_Item`).
19//! - `contract`: the contract rows a consumer checks at load.
20//! - `errors`: every error domain with its codes and target type names.
21//! - `docs`: doc and deprecation text with backticked identifiers in the
22//!   target's spelling.
23//! - `common`: doc-comment emission, prose wrapping, and `PascalCase`.
24//! - [`CodeWriter`]: the indentation-aware writer.
25//!
26//! In ABI revision 6 every value crosses as one C value chosen by its
27//! [`Family`](weaveffi_model::ty::Family), so a target marshals each value
28//! by its type alone.
29

51/// A single generated file: its path relative to the target's directory and
52/// its rendered contents.
53#[derive(Debug, Clone, PartialEq, Eq)]
54pub struct OutputFile {
55    /// Path relative to the target's directory (`{out_dir}/{target}/`),
56    /// with `/` separators.
57    pub path: Utf8PathBuf,
58    /// The rendered file contents.
59    pub contents: String,
60}
61
62impl OutputFile {
63    /// Pair a path, relative to the target's directory, with its rendered
64    /// contents.
65    ///
66    /// The path is normalized to `/` separators, which every platform's file
67    /// APIs accept, so listings, generation records, and tests see the same
68    /// path on Windows as elsewhere.
69    pub fn new(path: impl Into<Utf8PathBuf>, contents: impl Into<String>) -> Self {
70        let path: Utf8PathBuf = path.into();
71        let path = if path.as_str().contains('\\') {
72            Utf8PathBuf::from(path.as_str().replace('\\', "/"))
73        } else {
74            path
75        };
76        Self {
77            path,
78            contents: contents.into(),
79        }
80    }
81}

83/// The path of a target's `file` relative to the output directory
84/// (`{target}/{file}`), with `/` separators on every platform.
85fn output_path(target: &str, file: &Utf8Path) -> Result<String> {
86    let mut parts = vec![target];
87    for component in file.components() {
88        match component {
89            Utf8Component::Normal(part) => parts.push(part),
90            Utf8Component::CurDir => {}
91            _ => {
92                return Err(miette!(
93                    "the {target} target rendered {file}, which isn't a path inside its directory"
94                ))
95            }
96        }
97    }
98    Ok(parts.join("/"))
99}

101/// How applying a [`Changeset`] changes one file.
102#[derive(Debug, Clone, Copy, PartialEq, Eq)]
103pub enum ChangeKind {
104    /// The file doesn't exist yet.
105    Added,
106    /// The file exists with different contents.
107    Modified,
108    /// A previous generation wrote the file and this one doesn't.
109    Removed,
110}
111
112impl ChangeKind {
113    /// The one-character marker `generate --check` prints before a path.
114    #[must_use]
115    pub fn marker(self) -> char {
116        match self {
117            Self::Added => '+',
118            Self::Modified => '~',
119            Self::Removed => '-',
120        }
121    }
122}

124/// One planned file of one target.
125#[derive(Debug, Clone)]
126struct PlannedFile {
127    /// Path relative to the output directory.
128    path: String,
129    contents: String,
130    /// What's on disk now, when the file exists.
131    old: Option<Vec<u8>>,
132}
133
134impl PlannedFile {
135    fn change(&self) -> Option<ChangeKind> {
136        match &self.old {
137            None => Some(ChangeKind::Added),
138            Some(old) if old != self.contents.as_bytes() => Some(ChangeKind::Modified),
139            Some(_) => None,
140        }
141    }
142}

144/// One target's part of a [`Changeset`].
145#[derive(Debug, Clone)]
146struct TargetPlan {
147    name: &'static str,
148    files: Vec<PlannedFile>,
149    /// Files the previous generation wrote that this one doesn't, still on
150    /// disk, with their current contents.
151    removed: Vec<(String, Vec<u8>)>,
152    record: Record,
153    previous: Option<Record>,
154}
155
156impl TargetPlan {
157    fn changes_files(&self) -> bool {
158        !self.removed.is_empty() || self.files.iter().any(|f| f.change().is_some())
159    }
160
161    fn is_current(&self) -> bool {
162        !self.changes_files() && self.previous.as_ref() == Some(&self.record)
163    }
164}

166/// Everything regenerating would do to an output directory, computed
167/// without writing: which files would be added, modified, or removed.
168#[derive(Debug, Clone)]
169pub struct Changeset {
170    out_dir: Utf8PathBuf,
171    targets: Vec<TargetPlan>,
172}

174/// One file a [`Changeset`] would add, modify, or remove.
175#[derive(Debug, Clone, Copy, PartialEq, Eq)]
176pub struct Change<'a> {
177    /// What happens to the file.
178    pub kind: ChangeKind,
179    /// The file's path relative to the output directory.
180    pub path: &'a str,
181}
182
183impl Changeset {
184    /// The output directory the plan was made against.
185    #[must_use]
186    pub fn out_dir(&self) -> &Utf8Path {
187        &self.out_dir
188    }
189
190    /// Every file the targets render, relative to the output directory, in
191    /// target order (what `generate --dry-run` lists).
192    pub fn rendered(&self) -> impl Iterator<Item = &str> {
193        self.targets
194            .iter()
195            .flat_map(|t| t.files.iter().map(|f| f.path.as_str()))
196    }
197
198    /// Every file applying the plan would add, modify, or remove, in target
199    /// order.
200    pub fn changes(&self) -> impl Iterator<Item = Change<'_>> {
201        self.targets.iter().flat_map(|t| {
202            let written = t.files.iter().filter_map(|f| {
203                f.change().map(|kind| Change {
204                    kind,
205                    path: &f.path,
206                })
207            });
208            let removed = t.removed.iter().map(|(path, _)| Change {
209                kind: ChangeKind::Removed,
210                path,
211            });
212            written.chain(removed)
213        })
214    }
215
216    /// Whether applying the plan would leave every generated file as it is.
217    /// (A missing generation record alone doesn't count as a change.)
218    #[must_use]
219    pub fn is_empty(&self) -> bool {
220        self.changes().next().is_none()
221    }
222
223    /// A unified diff (`git diff` style, with `a/` and `b/` prefixes) of
224    /// every change, with `/dev/null` standing in for added and removed
225    /// files.
226    #[must_use]
227    pub fn unified_diff(&self) -> String {
228        let mut out = String::new();
229        for t in &self.targets {
230            for f in &t.files {
231                let old = match (&f.old, f.change()) {
232                    (_, None) => continue,
233                    (None, _) => None,
234                    (Some(old), _) => Some(String::from_utf8_lossy(old)),
235                };
236                push_diff(&mut out, &f.path, old.as_deref(), Some(&f.contents));
237            }
238            for (path, old) in &t.removed {
239                push_diff(&mut out, path, Some(&String::from_utf8_lossy(old)), None);
240            }
241        }
242        out
243    }
244
245    /// Apply the plan: write every added or modified file, remove stale
246    /// files, and update the generation records.
247    ///
248    /// # Errors
249    ///
250    /// Returns an error when a directory can't be created or a file can't
251    /// be written or removed.
252    pub fn apply(&self) -> Result<GenerateReport> {
253        let mut report = GenerateReport::default();
254        if self.targets.iter().all(TargetPlan::is_current) {
255            report.up_to_date = self.targets.iter().map(|t| t.name).collect();
256            return Ok(report);
257        }
258        std::fs::create_dir_all(self.out_dir.as_std_path())
259            .into_diagnostic()
260            .wrap_err_with(|| format!("failed to create the output directory {}", self.out_dir))?;
261        for t in &self.targets {
262            let record_changed = t.previous.as_ref() != Some(&t.record);
263            let mut changed = record_changed;
264            for file in &t.files {
265                if file.change().is_none() {
266                    report.unchanged += 1;
267                    continue;
268                }
269                let path = self.out_dir.join(&file.path);
270                if let Some(parent) = path.parent() {
271                    std::fs::create_dir_all(parent.as_std_path())
272                        .into_diagnostic()
273                        .wrap_err_with(|| format!("failed to create {parent}"))?;
274                }
275                std::fs::write(path.as_std_path(), &file.contents)
276                    .into_diagnostic()
277                    .wrap_err_with(|| format!("failed to write {path}"))?;
278                report.written += 1;
279                changed = true;
280            }
281            for (rel, _) in &t.removed {
282                let path = self.out_dir.join(rel);
283                std::fs::remove_file(path.as_std_path())
284                    .into_diagnostic()
285                    .wrap_err_with(|| format!("failed to remove the stale file {path}"))?;
286                report.removed.push(rel.clone());
287                changed = true;
288            }
289            if record_changed {
290                record::write_record(&self.out_dir, t.name, &t.record)?;
291            }
292            if changed {
293                report.generated.push(t.name);
294            } else {
295                report.up_to_date.push(t.name);
296            }
297        }
298        Ok(report)
299    }
300}

302/// Append one file's unified diff to `out`.
303fn push_diff(out: &mut String, path: &str, old: Option<&str>, new: Option<&str>) {
304    let old_name = if old.is_some() {
305        format!("a/{path}")
306    } else {
307        "/dev/null".to_string()
308    };
309    let new_name = if new.is_some() {
310        format!("b/{path}")
311    } else {
312        "/dev/null".to_string()
313    };
314    let diff = TextDiff::from_lines(old.unwrap_or(""), new.unwrap_or(""));
315    let _ = write!(
316        out,
317        "{}",
318        diff.unified_diff()
319            .context_radius(3)
320            .header(&old_name, &new_name)
321    );
322}

324/// What a generation run did.
325#[derive(Debug, Default, Clone, PartialEq, Eq)]
326pub struct GenerateReport {
327    /// Targets that wrote or removed a file.
328    pub generated: Vec<&'static str>,
329    /// Targets whose output already matched the output directory.
330    pub up_to_date: Vec<&'static str>,
331    /// Files written because they were new or their contents changed.
332    pub written: usize,
333    /// Files rendered with the contents already on disk.
334    pub unchanged: usize,
335    /// Files removed because the previous run wrote them and this run did
336    /// not (paths relative to the output directory).
337    pub removed: Vec<String>,
338}
339
340impl GenerateReport {
341    /// One line describing the run, for `weaveffi generate`'s output.
342    #[must_use]
343    pub fn summary(&self, out_dir: &Utf8Path) -> String {
344        if self.generated.is_empty() {
345            return format!(
346                "{out_dir} is up to date ({} targets)",
347                self.up_to_date.len()
348            );
349        }
350        let mut line = format!(
351            "Generated {} in {out_dir}: {} written, {} unchanged",
352            self.generated.join(", "),
353            self.written,
354            self.unchanged
355        );
356        if !self.removed.is_empty() {
357            let _ = write!(line, ", {} stale removed", self.removed.len());
358        }
359        if !self.up_to_date.is_empty() {
360            let _ = write!(line, " ({} up to date)", self.up_to_date.join(", "));
361        }
362        line
363    }
364}

366/// Runs a set of targets against one model and output directory.
367#[derive(Default)]
368pub struct Orchestrator<'a> {
369    targets: Vec<&'a dyn Target>,
370}
371
372impl<'a> Orchestrator<'a> {
373    /// An orchestrator with no targets.
374    #[must_use]
375    pub fn new() -> Self {
376        Self::default()
377    }
378
379    /// Add a target.
380    #[must_use]
381    pub fn with_target(mut self, target: &'a dyn Target) -> Self {
382        self.targets.push(target);
383        self
384    }
385
386    /// Add several targets.
387    #[must_use]
388    pub fn with_targets(mut self, targets: impl IntoIterator<Item = &'a dyn Target>) -> Self {
389        self.targets.extend(targets);
390        self
391    }
392
393    /// Render every target (in parallel) and compare the result with
394    /// `out_dir`, without writing anything.
395    ///
396    /// # Errors
397    ///
398    /// Returns an error when a target renders a path outside its directory.
399    pub fn plan(&self, model: &Model, out_dir: &Utf8Path) -> Result<Changeset> {
400        let targets = self
401            .targets
402            .par_iter()
403            .map(|&target| plan_target(target, model, out_dir))
404            .collect::<Result<Vec<_>>>()?;
405        Ok(Changeset {
406            out_dir: out_dir.to_path_buf(),
407            targets,
408        })
409    }
410
411    /// Generate every target under `out_dir`: [`plan`](Self::plan), then
412    /// [`apply`](Changeset::apply).
413    ///
414    /// # Errors
415    ///
416    /// Returns the errors of [`plan`](Self::plan) and
417    /// [`apply`](Changeset::apply).
418    pub fn run(&self, model: &Model, out_dir: &Utf8Path) -> Result<GenerateReport> {
419        self.plan(model, out_dir)?.apply()
420    }
421}
422
423/// Render `target` and compare its files with `out_dir`.
424fn plan_target(target: &dyn Target, model: &Model, out_dir: &Utf8Path) -> Result<TargetPlan> {
425    let name = target.name();
426    let files = target
427        .render(model)
428        .into_iter()
429        .map(|file| {
430            let path = output_path(name, &file.path)?;
431            let old = std::fs::read(out_dir.join(&path).as_std_path()).ok();
432            Ok(PlannedFile {
433                path,
434                contents: file.contents,
435                old,
436            })
437        })
438        .collect::<Result<Vec<_>>>()?;
439    let record = Record {
440        files: files.iter().map(|f| f.path.clone()).collect(),
441    };
442    let previous = record::read_record(out_dir, name);
443    let removed = previous
444        .iter()
445        .flat_map(|p| p.files.iter())
446        .filter(|rel| !record.files.contains(*rel))
447        .filter_map(|rel| {
448            let bytes = std::fs::read(out_dir.join(rel).as_std_path()).ok()?;
449            Some((rel.clone(), bytes))
450        })
451        .collect();
452    Ok(TargetPlan {
453        name,
454        files,
455        removed,
456        record,
457        previous,
458    })
459}

477    struct Counting {
478        name: &'static str,
479        calls: Arc<AtomicUsize>,
480        files: Arc<Mutex<Vec<&'static str>>>,
481    }
482
483    impl Target for Counting {
484        fn name(&self) -> &'static str {
485            self.name
486        }
487
488        fn render(&self, model: &Model) -> Vec<OutputFile> {
489            self.calls.fetch_add(1, Ordering::SeqCst);
490            self.files
491                .lock()
492                .unwrap()
493                .iter()
494                .map(|f| {
495                    OutputFile::new(
496                        *f,
497                        format!("{} {}", model.prefix(), model.functions[0].name),
498                    )
499                })
500                .collect()
501        }
502    }
503
504    fn model_named(name: &str) -> Model {
505        let mut api = Api::new();
506        api.functions.push(Function::named(name));
507        let identity = weaveffi_model::pkg::Identity::named("api");
508        weaveffi_model::validate::validate(&api, &identity, None).unwrap()
509    }
510
511    fn counting(name: &'static str, calls: &Arc<AtomicUsize>) -> Counting {
512        Counting {
513            name,
514            calls: Arc::clone(calls),
515            files: Arc::new(Mutex::new(vec!["out.txt"])),
516        }
517    }

519    #[test]
520    fn every_run_renders_but_only_changed_files_are_written() {
521        let dir = tempfile::tempdir().unwrap();
522        let out = Utf8Path::from_path(dir.path()).unwrap();
523        let calls = Arc::new(AtomicUsize::new(0));
524        let c = counting("c", &calls);
525        let orch = Orchestrator::new().with_target(&c);
526        let model = model_named("math");
527
528        let first = orch.run(&model, out).unwrap();
529        assert_eq!((first.generated.len(), first.written), (1, 1));
530        let rendered = std::fs::read_to_string(out.join("c/out.txt")).unwrap();
531        let second = orch.run(&model, out).unwrap();
532        assert_eq!(second.up_to_date, ["c"]);
533        assert_eq!(second.written, 0);
534        assert_eq!(calls.load(Ordering::SeqCst), 2);
535
536        std::fs::write(out.join("c/out.txt"), "edited").unwrap();
537        let plan = orch.plan(&model, out).unwrap();
538        let changes: Vec<_> = plan.changes().collect();
539        assert_eq!(
540            changes,
541            [Change {
542                kind: ChangeKind::Modified,
543                path: "c/out.txt"
544            }]
545        );
546        assert!(
547            plan.unified_diff().contains("-edited\n"),
548            "{}",
549            plan.unified_diff()
550        );
551        let third = plan.apply().unwrap();
552        assert_eq!((third.generated.len(), third.written), (1, 1));
553        assert_eq!(
554            std::fs::read_to_string(out.join("c/out.txt")).unwrap(),
555            rendered
556        );
557        assert!(orch.plan(&model, out).unwrap().is_empty());
558    }

561    fn stale_files_are_removed_but_user_files_are_kept() {
562        let dir = tempfile::tempdir().unwrap();
563        let out = Utf8Path::from_path(dir.path()).unwrap();
564        let calls = Arc::new(AtomicUsize::new(0));
565        let c = counting("c", &calls);
566        let files = Arc::clone(&c.files);

568        let orch = Orchestrator::new().with_target(&c);
569        orch.run(&model_named("math"), out).unwrap();
570        std::fs::write(out.join("c/user.txt"), "mine").unwrap();

573        let plan = orch.plan(&model_named("math"), out).unwrap();
574        assert_eq!(

577                kind: ChangeKind::Removed,
