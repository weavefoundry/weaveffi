1//! Call-path renderers: the wrapper methods for every call shape (sync,
2//! async, iterator), the marshalling of every value by its [`Crossing`],
3//! and the library class holding the free functions.
4//!
5//! Every value is one C value in every position, so four expressions cover
6//! every crossing:
7//!
8//! * [`Marshal`] lends a parameter to the producer: a scalar as is, a
9//!   `bool` as one byte, an optional scalar as its `FfiOpt*` struct, a
10//!   string, bytes, typed array, or value buffer as an `FfiRun` over pinned
11//!   memory, an object as its `SafeHandle`, and a callback interface as an
12//!   `FfiCallback` of a pinned implementation and its static vtable;
13//! * [`take`] receives a value the producer hands over (a return, an async
14//!   result, an iterator item), releasing its run with `{p}_run_free`;
15//! * [`borrow`] receives a callback method's argument, whose run stays the
16//!   producer's;
17//! * [`give`] hands a callback method's return to the producer, as a run
18//!   from `{p}_alloc` or a fresh object reference.
19//!
20//! Every wrapper follows one frame: encode arguments (pooled UTF-8, pooled
21//! value buffers, callback registrations), pin them with `fixed`, call the
22//! import with a stack `FfiError`, check the error, and receive the result.
23

37/// The `[UnmanagedCallersOnly]` attribute every native entry point carries.
38pub(crate) const UNMANAGED_CALLERS_ONLY: &str =
39    "[UnmanagedCallersOnly(CallConvs = new[] { typeof(CallConvCdecl) })]";

41/// The receiver of a wrapper: what the method is a member of.
42#[derive(Clone, Copy)]
43pub(crate) enum Receiver<'a> {
44    /// A free function or interface static: a `static` method.
45    Static,
46    /// An interface instance method: passes `Handle` as `self`.
47    Instance,
48    /// The interface's `new` constructor: a real C# constructor of the
49    /// named class.
50    Constructor(&'a str),
51}

53/// The expression converting `v`, a C value of type `ty` the producer
54/// handed over, to its C# value: a run is decoded and released, an object
55/// reference adopted into a wrapper.
56pub(crate) fn take(cx: Cx<'_>, ty: &Ty, v: &str) -> String {
57    match crossing(ty) {
58        Crossing::Utf8 => format!("Ffi.TakeString({v})"),
59        Crossing::Bytes => format!("Ffi.TakeBytes({v})"),
60        Crossing::Slice(p) => format!("Ffi.TakeArray<{}>({v})", prim_cs(p)),
61        Crossing::Buffer(ty) => format!("Ffi.TakeBuffer({v}, {})", read_lambda(cx, ty, 0)),
62        _ => direct_out(cx, ty, v),
63    }
64}

66/// The expression converting `v`, a callback method's argument of type
67/// `ty`, to its C# value: a run is borrowed for the call (a typed array or
68/// bytes as a span over the producer's memory), and an object reference,
69/// which moves to the consumer, is adopted.
70pub(crate) fn borrow(cx: Cx<'_>, ty: &Ty, v: &str) -> String {
71    match crossing(ty) {
72        Crossing::Utf8 => format!("Ffi.ReadString({v})"),
73        Crossing::Bytes => format!("Ffi.Borrow<byte>({v})"),
74        Crossing::Slice(p) => format!("Ffi.Borrow<{}>({v})", prim_cs(p)),
75        Crossing::Buffer(ty) => format!("Ffi.ReadBuffer({v}, {})", read_lambda(cx, ty, 0)),
76        _ => direct_out(cx, ty, v),
77    }
78}

80/// The C# value of a C value that isn't a run.
81fn direct_out(cx: Cx<'_>, ty: &Ty, v: &str) -> String {
82    match crossing(ty) {
83        Crossing::Scalar(Prim::Bool) => format!("{v} != 0"),
84        Crossing::Scalar(_) | Crossing::Enum(_) => v.to_string(),
85        Crossing::OptScalar(_) => format!("{v}.ToNullable()"),
86        Crossing::OptEnum(name) => format!("({}?){v}.ToNullable()", cx.ty(name)),
87        Crossing::Object {
88            interface,
89            nullable,
90        } => adopt(cx, v, nullable, interface),
91        Crossing::Utf8 | Crossing::Bytes | Crossing::Slice(_) | Crossing::Buffer(_) => {
92            unreachable!("a run is taken or borrowed")
93        }
94    }
95}

97/// The expression handing `v`, a callback method's C# return of type `ty`,
98/// to the producer, which adopts it: a run allocated with `{p}_alloc`, or a
99/// fresh strong reference. A null for a required object passes through as
100/// null, which the producer rejects as a marshalling failure (-3).
101pub(crate) fn give(ty: &Ty, v: &str) -> String {
102    match crossing(ty) {
103        Crossing::Scalar(Prim::Bool) => format!("Ffi.Bool({v})"),
104        Crossing::Scalar(_) | Crossing::Enum(_) => v.to_string(),
105        Crossing::OptScalar(p) => format!("{}.Of({v})", opt_struct(p)),
106        Crossing::OptEnum(_) => format!("FfiOptI32.Of((int?){v})"),
107        Crossing::Utf8 => format!("Ffi.ReturnString({v})"),
108        Crossing::Bytes => format!("Ffi.ReturnBytes({v})"),
109        Crossing::Slice(_) => format!("Ffi.ReturnArray({v})"),
110        Crossing::Buffer(ty) => format!("Ffi.ReturnBuffer({v}, {})", write_lambda(ty, 0)),
111        Crossing::Object { .. } => format!("{v}?.CloneHandle() ?? IntPtr.Zero"),
112    }
113}

115/// The argument marshalling for one parameter list: statements run before
116/// the call, the `fixed` pins it needs, one expression per ABI slot, and
117/// the callback registrations to release if the call never reaches the
118/// producer.
119struct Marshal {
120    setup: Vec<String>,
121    pins: Vec<String>,
122    args: Vec<String>,
123    callbacks: Vec<String>,
124}
125
126impl Marshal {
127    /// Plan the marshalling of `params`.
128    fn new(cx: Cx<'_>, params: &[ParamBinding]) -> Self {
129        let mut m = Marshal {
130            setup: Vec::new(),
131            pins: Vec::new(),
132            args: Vec::new(),
133            callbacks: Vec::new(),
134        };
135        let mut registrations = Vec::new();
136        for p in params {
137            let n = camel(&p.name);
138            let local = p.name.to_lower_camel_case();
139            let ty = match &p.ty {
140                ParamTy::Value(ty) => ty,
141                // The producer owns the registration once the call reaches
142                // it, even when another argument fails: its vtable
143                // `free(ctx)` releases it. An absent optional
144                // implementation passes a null vtable.
145                ParamTy::Callback { name, nullable } => {
146                    let ctx = format!("{local}Ctx");
147                    let pointer = format!("{}.Pointer", vtable_class_cs(name));
148                    let vtable = if *nullable {
149                        format!("{n} == null ? IntPtr.Zero : {pointer}")
150                    } else {
151                        pointer
152                    };
153                    m.args.push(format!("new FfiCallback({ctx}, {vtable})"));
154                    registrations.push(format!("var {ctx} = Ffi.Register({n});"));
155                    m.callbacks.push(ctx);
156                    continue;
157                }
158            };
159            match crossing(ty) {
160                Crossing::Scalar(Prim::Bool) => m.args.push(format!("Ffi.Bool({n})")),
161                Crossing::Scalar(_) | Crossing::Enum(_) => m.args.push(n),
162                Crossing::OptScalar(p) => m.args.push(format!("{}.Of({n})", opt_struct(p))),
163                Crossing::OptEnum(_) => m.args.push(format!("FfiOptI32.Of((int?){n})")),
164                Crossing::Utf8 => {
165                    m.setup
166                        .push(format!("using var {local}Utf8 = new FfiUtf8({n});"));
167                    m.pins.push(format!("byte* {local}Ptr = {local}Utf8"));
168                    m.args
169                        .push(format!("new FfiRun({local}Ptr, {local}Utf8.Length)"));
170                }
171                Crossing::Bytes => {
172                    m.pins.push(format!("byte* {local}Ptr = {n}"));
173                    m.args
174                        .push(format!("new FfiRun({local}Ptr, (nuint){n}.Length)"));
175                }
176                // `len` counts bytes: the element count times its width.
177                Crossing::Slice(p) => {
178                    let size = p.size().expect("a typed array's element is a scalar");
179                    m.pins.push(format!("{}* {local}Ptr = {n}", prim_cs(p)));
180                    m.args.push(format!(
181                        "new FfiRun((byte*){local}Ptr, (nuint){n}.Length * {size})"
182                    ));
183                }
184                Crossing::Buffer(ty) => {
185                    let writer = format!("{local}Writer");
186                    m.setup
187                        .push(format!("using var {writer} = new FfiBufferWriter();"));
188                    m.setup.push(format!("{};", write_call(ty, &writer, &n, 0)));
189                    m.pins.push(format!("byte* {local}Ptr = {writer}.Written"));
190                    m.args
191                        .push(format!("new FfiRun({local}Ptr, (nuint){writer}.Length)"));
192                }
193                Crossing::Object {
194                    interface,
195                    nullable,
196                } => {
197                    if nullable {
198                        m.args.push(format!(
199                            "{n}?.Handle ?? {}.NativeHandle.Null",
200                            cx.ty(interface)
201                        ));
202                    } else {
203                        m.args.push(format!("{n}.Handle"));
204                    }
205                }
206            }
207        }
208        // Register last, so a failing encode above can't strand a
209        // registration.
210        m.setup.extend(registrations);
211        m
212    }
213
214    /// Emit the setup statements.
215    fn write_setup(&self, w: &mut CodeWriter) {
216        for line in &self.setup {
217            w.line(line);
218        }
219    }
220
221    /// Emit `body` inside one stacked `fixed` statement per pin, or bare
222    /// when nothing needs pinning.
223    fn pinned(&self, w: &mut CodeWriter, body: impl FnOnce(&mut CodeWriter)) {
224        if self.pins.is_empty() {
225            body(w);
226            return;
227        }
228        for pin in &self.pins {
229            w.line(format!("fixed ({pin})"));
230        }
231        w.line("{");
232        w.scope(body);
233        w.line("}");
234    }
235
236    /// Emit `call` (a statement), releasing every callback registration if
237    /// it throws before reaching the producer. `declare` is the declaration
238    /// of the variable the statement assigns, when it assigns one.
239    fn guarded(&self, w: &mut CodeWriter, declare: Option<String>, call: &str) {
240        if self.callbacks.is_empty() {
241            match declare {
242                Some(_) => w.line(format!("var {call}")),
243                None => w.line(call),
244            };
245            return;
246        }
247        if let Some(decl) = declare {
248            w.line(decl);
249        }
250        w.line("try");
251        w.block("{", "}", |w| {
252            w.line(call);
253        });
254        w.line("catch");
255        w.block("{", "}", |w| {
256            self.unregister(w);
257            w.line("throw;");
258        });
259    }
260
261    /// The statements releasing every callback registration.
262    fn unregister(&self, w: &mut CodeWriter) {
263        for ctx in &self.callbacks {
264            w.line(format!("Ffi.Unregister({ctx});"));
265        }
266    }
267}

269/// The public signature's parameter list.
270fn params_sig(params: &[ParamBinding]) -> Vec<String> {
271    params
272        .iter()
273        .map(|p| format!("{} {}", param_cs(&p.ty), camel(&p.name)))
274        .collect()
275}

277/// The slot arguments of a call: the receiver (when any), every marshalled
278/// parameter slot, then `extra`.
279fn call_args(receiver: Receiver<'_>, m: &Marshal, extra: &[String]) -> String {
280    let mut args = Vec::new();
281    if matches!(receiver, Receiver::Instance) {
282        args.push("Handle".to_string());
283    }
284    args.extend(m.args.iter().cloned());
285    args.extend(extra.iter().cloned());
286    args.join(", ")
287}

289/// The modifiers, return type, and name opening a method (or the bare class
290/// name of a constructor).
291fn opener(receiver: Receiver<'_>, ret: &str, name: &str) -> String {
292    match receiver {
293        Receiver::Static => format!("public static {ret} {name}"),
294        Receiver::Instance => format!("public {ret} {name}"),
295        Receiver::Constructor(class) => format!("public {class}"),
296    }
297}

299/// The expression adopting an object pointer into a wrapper (`null` for a
300/// null pointer of a nullable object).
301fn adopt(cx: Cx<'_>, ptr: &str, nullable: bool, interface: &str) -> String {
302    let class = cx.ty(interface);
303    if nullable {
304        format!("{ptr} == IntPtr.Zero ? null : {class}.Adopt({ptr})")
305    } else {
306        format!("{class}.Adopt({ptr})")
307    }
308}

310/// Render one wrapper (any shape) named `name`.
311pub(crate) fn render_callable(
312    w: &mut CodeWriter,
313    docs: &Docs,
314    f: &FnBinding,
315    name: &str,
316    receiver: Receiver<'_>,
317    err: &ErrCtx,
318    cx: Cx<'_>,
319) {
320    if let Some(a) = f.async_binding() {
321        render_async(w, docs, f, a, name, receiver, err, cx);
322    } else if let Some(it) = f.iterator() {
323        render_iterator(w, docs, f, it, name, receiver, err, cx);
324    } else {
325        render_sync(w, docs, f, name, receiver, err, cx);
326    }
327}

329/// The documentation, `<exception>`, and `[Obsolete]` lines of a wrapper.
330fn write_header(w: &mut CodeWriter, docs: &Docs, f: &FnBinding, err: &ErrCtx) {
331    docs.summary(w, &f.doc);
332    for p in &f.params {
333        docs.param(w, &camel(&p.name), &p.doc);
334    }
335    err.write_doc(w);
336    docs.obsolete(w, &f.deprecated);
337}

339/// A synchronous wrapper: one call, one check, one received result. A
340/// constructor stores the adopted pointer as its handle instead of
341/// returning it.
342fn render_sync(
343    w: &mut CodeWriter,
344    docs: &Docs,
345    f: &FnBinding,
346    name: &str,
347    receiver: Receiver<'_>,
348    err: &ErrCtx,
349    cx: Cx<'_>,
350) {
351    let value = f.ret_value();
352    let ret = match (receiver, value) {
353        (Receiver::Constructor(_), _) => String::new(),
354        (_, Some(ty)) => owned_cs(ty),
355        (_, None) => "void".into(),
356    };
357    let m = Marshal::new(cx, &f.params);
358    let call = format!(
359        "NativeMethods.{}({})",
360        f.abi.symbol,
361        call_args(receiver, &m, &["&ffiErr".to_string()])
362    );
363
364    write_header(w, docs, f, err);
365    w.line(format!(
366        "{}({})",
367        opener(receiver, &ret, name),
368        params_sig(&f.params).join(", ")
369    ));
370    w.block("{", "}", |w| {
371        m.write_setup(w);
372        w.line("var ffiErr = default(FfiError);");
373        m.pinned(w, |w| {
374            if value.is_some() {
375                let decl = format!("{} ffiResult;", cs_ctype(cx.ns, &f.abi.ret));
376                m.guarded(w, Some(decl), &format!("ffiResult = {call};"));
377            } else {
378                m.guarded(w, None, &format!("{call};"));
379            }
380            w.line(err.check("ffiErr"));
381            match (value, receiver) {
382                (Some(_), Receiver::Constructor(_)) => {
383                    w.line("Handle = new NativeHandle(ffiResult);");
384                }
385                (Some(ty), _) => {
386                    w.line(format!("return {};", take(cx, ty, "ffiResult")));
387                }
388                (None, _) => {}
389            }
390        });
391    });
392    w.blank();
393}

395/// An async wrapper: a `Task`-returning method launching the call with a
396/// pending `FfiCall` as `context`, plus the `[UnmanagedCallersOnly]`
397/// completion that resolves it. Every async method takes a
398/// `CancellationToken`: a cancellable function links it to the native
399/// cancel token; any other stops waiting when it fires.
400#[allow(clippy::too_many_arguments)]
401fn render_async(
402    w: &mut CodeWriter,
403    docs: &Docs,
404    f: &FnBinding,
405    a: &AsyncBinding,
406    name: &str,
407    receiver: Receiver<'_>,
408    err: &ErrCtx,
409    cx: Cx<'_>,
410) {
411    let value = f.ret_value();
412    let result = value.map(owned_cs);
413    let task = match &result {
414        Some(t) => format!("Task<{t}>"),
415        None => "Task".into(),
416    };
417    let call_ty = result.clone().unwrap_or_else(|| "FfiVoid".into());
418    let complete = format!("FfiComplete{name}");
419    let m = Marshal::new(cx, &f.params);
420    let mut extra = Vec::new();
421    if a.cancellable {
422        extra.push("ffiCall.CancelToken()".to_string());
423    }
424    extra.push(format!("&{complete}"));
425    extra.push("ffiCall.Context".to_string());
426    let call = format!(
427        "NativeMethods.{}({});",
428        f.abi.symbol,
429        call_args(receiver, &m, &extra)
430    );
431
432    write_header(w, docs, f, err);
433    if a.cancellable {
434        w.line("/// <param name=\"cancellationToken\">Cancels the native call; the task then");
435        w.line("/// completes as canceled.</param>");
436    } else {
437        w.line("/// <param name=\"cancellationToken\">Stops waiting: the task completes as");
438        w.line("/// canceled at once, and the native call, which can't be cancelled, finishes");
439        w.line("/// in the background and releases its result.</param>");
440    }
441    let mut sig = params_sig(&f.params);
442    sig.push("CancellationToken cancellationToken = default".into());
443    let opener = match receiver {
444        Receiver::Instance => format!("public {task} {name}"),
445        _ => format!("public static {task} {name}"),
446    };
447    w.line(format!("{opener}({})", sig.join(", ")));
448    w.block("{", "}", |w| {
449        w.line("if (cancellationToken.IsCancellationRequested)");
450        w.block("{", "}", |w| {
451            let from = match &result {
452                Some(t) => format!("Task.FromCanceled<{t}>(cancellationToken)"),
453                None => "Task.FromCanceled(cancellationToken)".into(),
454            };
455            w.line(format!("return {from};"));
456        });
457        m.write_setup(w);
458        w.line(format!(
459            "var ffiCall = new FfiCall<{call_ty}>(cancellationToken);"
460        ));
461        w.line("try");
462        w.block("{", "}", |w| {
463            m.pinned(w, |w| {
464                w.line(&call);
465            });
466        });
467        w.line("catch");
468        w.block("{", "}", |w| {
469            w.line("ffiCall.Abandon();");
470            m.unregister(w);
471            w.line("throw;");
472        });
473        w.line("return ffiCall.Launched();");
474    });
475    w.blank();
476
477    // The completion: fires exactly once, from any thread.
478    let slots: Vec<String> = a
479        .callback
480        .params
481        .iter()
482        .map(|s| format!("{} {}", cs_ctype(cx.ns, &s.ty), s.name))
483        .collect();
484    w.line(UNMANAGED_CALLERS_ONLY);
485    w.line(format!(
486        "private static void {complete}({})",
487        slots.join(", ")
488    ));
489    w.block("{", "}", |w| {
490        w.line(format!(
491            "var ffiCall = FfiCall<{call_ty}>.Complete(context);"
492        ));
493        w.line("try");
494        w.block("{", "}", |w| {
495            w.line("if (err != null)");
496            w.block("{", "}", |w| {
497                w.line(format!("ffiCall.SetError(err, {});", err.map));
498                w.line("return;");
499            });
500            let received = match value {
501                Some(ty) => take(cx, ty, "result"),
502                None => "default".into(),
503            };
504            w.line(format!("ffiCall.SetResult({received});"));
505        });
506        w.line("catch (Exception e)");
507        w.block("{", "}", |w| {
508            w.line("ffiCall.SetException(e);");
509        });
510    });
511    w.blank();
512}

514/// An iterator wrapper: a re-enumerable `IEnumerable<T>` whose every
515/// enumeration launches a new native iterator through an `FfiLaunch{Name}`
516/// helper and pulls one item per `MoveNext` through a static
517/// `FfiNext{Name}`. Span parameters are copied once, since the sequence
518/// outlives the call.
519#[allow(clippy::too_many_arguments)]
520fn render_iterator(
521    w: &mut CodeWriter,
522    docs: &Docs,
523    f: &FnBinding,
524    it: &IteratorBinding,
525    name: &str,
526    receiver: Receiver<'_>,
527    err: &ErrCtx,
528    cx: Cx<'_>,
529) {
530    let elem = owned_cs(&it.elem);
531    let launch = format!("FfiLaunch{name}");
532    let next = format!("FfiNext{name}");
533    let m = Marshal::new(cx, &f.params);
534
535    write_header(w, docs, f, err);
536    w.line("/// <remarks>Streams lazily, one native call per item. Each enumeration launches a");
537    w.line("/// new native iterator, so a failure to start throws from");
538    w.line("/// <c>GetEnumerator</c> (the start of a <c>foreach</c>); disposing the enumerator");
539    w.line("/// (as <c>foreach</c> does) releases it.</remarks>");
540    let ret = format!("IEnumerable<{elem}>");
541    w.line(format!(
542        "{}({})",
543        opener(receiver, &ret, name),
544        params_sig(&f.params).join(", ")
545    ));
546    w.block("{", "}", |w| {
547        let mut args = Vec::new();
548        for p in &f.params {
549            let n = camel(&p.name);
550            let lent =
551                p.ty.value()
552                    .is_some_and(|ty| matches!(crossing(ty), Crossing::Slice(_) | Crossing::Bytes));
553            if lent {
554                let copy = format!("{}Copy", p.name.to_lower_camel_case());
555                w.line(format!("var {copy} = {n}.ToArray();"));
556                args.push(copy);
557            } else {
558                args.push(n);
559            }
560        }
561        w.line(format!(
562            "return new FfiSequence<{elem}>(() => {launch}({}), &{next});",
563            args.join(", ")
564        ));
565    });
566    w.blank();
567
568    // The launcher: a sync call returning the native iterator.
569    let call = format!(
570        "NativeMethods.{}({})",
571        f.abi.symbol,
572        call_args(receiver, &m, &["&ffiErr".to_string()])
573    );
574    let modifier = match receiver {
575        Receiver::Instance => "private",
576        _ => "private static",
577    };
578    w.line(format!(
579        "{modifier} FfiIteratorHandle {launch}({})",
580        params_sig(&f.params).join(", ")
581    ));
582    w.block("{", "}", |w| {
583        m.write_setup(w);
584        w.line("var ffiErr = default(FfiError);");
585        m.pinned(w, |w| {
586            m.guarded(
587                w,
588                Some("IntPtr ffiResult;".into()),
589                &format!("ffiResult = {call};"),
590            );
591            w.line(err.check("ffiErr"));
592            w.line(format!(
593                "return new FfiIteratorHandle(ffiResult, &NativeMethods.{});",
594                it.destroy_symbol
595            ));
596        });
597    });
598    w.blank();
599
600    // `_next`: one item per call; 0 means exhausted.
601    let item_ty = cs_ctype(
602        cx.ns,
603        &weaveffi_model::abi::CType::of(&it.elem, weaveffi_model::abi::Position::Transfer),
604    );
605    w.line(format!(
606        "private static bool {next}(FfiIteratorHandle iter, out {elem} item)"
607    ));
608    w.block("{", "}", |w| {
609        w.line(format!("var ffiItem = default({item_ty});"));
610        w.line("var ffiErr = default(FfiError);");
611        w.line(format!(
612            "var ffiMore = NativeMethods.{}(iter, &ffiItem, &ffiErr);",
613            it.next.symbol
614        ));
615        w.line(err.check("ffiErr"));
616        w.line("if (ffiMore == 0)");
617        w.block("{", "}", |w| {
618            w.line("item = default!;");
619            w.line("return false;");
620        });
621        w.line(format!("item = {};", take(cx, &it.elem, "ffiItem")));
622        w.line("return true;");
623    });
624    w.blank();
625}

627/// Render the free functions as static methods of the library class (the
628/// generated half of the `partial` class the runtime declares with
629/// `Check()` and `AbiVersion`).
630pub(crate) fn render_functions(
631    w: &mut CodeWriter,
632    model: &Model,
633    docs: &Docs,
634    class: &str,
635    cx: Cx<'_>,
636) {
637    if model.functions.is_empty() {
638        return;
639    }
640    w.line(format!("public static unsafe partial class {class}"));
641    w.block("{", "}", |w| {
642        for f in &model.functions {
643            let name = function_cs(&f.name, class);
644            let err = ErrCtx::new(model, &f.error, cx);
645            render_callable(w, docs, f, &name, Receiver::Static, &err, cx);
646        }
647    });
648    w.blank();
649}
