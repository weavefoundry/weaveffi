1//! XML doc-comment emission, with backticked API identifiers in IDL docs
2//! rewritten to their C# spelling (`` `new_op` `` becomes `<c>NewOp</c>`).
3
4use std::collections::{HashMap, HashSet};
5
6use heck::{ToLowerCamelCase, ToUpperCamelCase};
7use weaveffi_model::errors;
8use weaveffi_model::model::Model;
9
10use crate::codegen::docs::{self, ApiNames, Doc, IdentKind};
11use crate::codegen::CodeWriter;
12use crate::targets::dotnet::types::{callback_interface_cs, cs_str, xml_text};
13
14/// Renders IDL doc and deprecation text in C# terms.
15pub(crate) struct Docs {
16    names: ApiNames,
17    callbacks: HashSet<String>,
18    codes: HashMap<String, String>,
19}
20
21impl Docs {
22    /// Index every identifier `model` declares.
23    pub(crate) fn new(model: &Model) -> Self {
24        let callbacks = model.callbacks.iter().map(|cb| cb.name.clone()).collect();
25        let codes = model
26            .errors
27            .iter()
28            .flat_map(|d| {
29                let exc = errors::exception_type_name(&d.name);
30                d.codes
31                    .iter()
32                    .map(move |c| (c.name.clone(), format!("{exc}.{}", errors::pascal(&c.name))))
33            })
34            .collect();
35        Self {
36            names: ApiNames::new(model),
37            callbacks,
38            codes,
39        }
40    }
41
42    /// The C# spelling of an API identifier, or `None` to keep it as
43    /// written (an ambiguous name).
44    fn spell(&self, ident: &str) -> Option<String> {
45        Some(match self.names.kind(ident)? {
46            IdentKind::Function
47            | IdentKind::Member
48            | IdentKind::CallbackMethod
49            | IdentKind::Field => ident.to_upper_camel_case(),
50            IdentKind::Param => ident.to_lower_camel_case(),
51            IdentKind::Type if self.callbacks.contains(ident) => callback_interface_cs(ident),
52            IdentKind::ErrorDomain => errors::exception_type_name(ident),
53            IdentKind::ErrorCode => self.codes.get(ident)?.clone(),
54            IdentKind::Type | IdentKind::Variant => return None,
55        })
56    }
57
58    /// `text` as XML doc content: escaped, with every identifier code span
59    /// rendered as `<c>` in C# spelling.
60    pub(crate) fn xml(&self, text: &str) -> String {
61        docs::map_code_spans(&xml_text(text), |ident| {
62            Some(format!(
63                "<c>{}</c>",
64                self.spell(ident).unwrap_or_else(|| ident.to_string())
65            ))
66        })
67    }
68
69    /// The `[Obsolete("...")]` line for a deprecated item.
70    pub(crate) fn obsolete(&self, w: &mut CodeWriter, deprecated: &Option<String>) {
71        let none: Option<String> = None;
72        if let Some(msg) = Doc::new(&none, deprecated).deprecation(|i| self.spell(i)) {
73            w.line(format!("[Obsolete(\"{}\")]", cs_str(&msg)));
74        }
75    }
76
77    /// Emit a `<summary>` for an item's doc comment, if it has one.
78    pub(crate) fn summary(&self, w: &mut CodeWriter, doc: &Option<String>) {
79        if let Some(d) = trimmed(doc) {
80            element(w, "<summary>", "</summary>", &self.xml(d));
81        }
82    }
83
84    /// Emit one `<param>` (or any named element) for a documented item.
85    /// `name` must already be the C# spelling (without the `@` escape, which
86    /// XML doc references omit).
87    pub(crate) fn param(&self, w: &mut CodeWriter, name: &str, doc: &Option<String>) {
88        if let Some(d) = trimmed(doc) {
89            let open = format!("<param name=\"{}\">", name.trim_start_matches('@'));
90            element(w, &open, "</param>", &self.xml(d));
91        }
92    }
93}
94
95/// The trimmed, non-empty text of an optional doc comment.
96fn trimmed(doc: &Option<String>) -> Option<&str> {
97    doc.as_deref().map(str::trim).filter(|d| !d.is_empty())
98}
99
100/// Emit one XML element (`<summary>`, `<param name="x">`) holding `body`,
101/// which is already XML: on one line when it's a single line, else one
102/// `///` line each.
103pub(crate) fn element(w: &mut CodeWriter, open: &str, close: &str, body: &str) {
104    if body.contains('\n') {
105        w.line(format!("/// {open}"));
106        for line in body.lines() {
107            w.line(format!("/// {line}").trim_end());
108        }
109        w.line(format!("/// {close}"));
110    } else {
111        w.line(format!("/// {open}{body}{close}"));
112    }
113}

115#[cfg(test)]
116mod tests {
117    use super::*;
118    use crate::codegen::test_model;
119
120    #[test]
121    fn identifiers_take_their_csharp_spelling() {
122        let model = test_model(
123            r#"
124version: "0.13.0"
125errors:
126  - name: KvError
127    codes: [{ name: not_found, code: 1, message: missing }]
128callbacks:
129  - name: Listener
130    methods: [{ name: on_put, params: [{ name: new_key, type: string }] }]
131functions:
132  - name: new_op
133    params: [{ name: the_key, type: string }]
134    return: i32
135"#,
136        );
137        let docs = Docs::new(&model);
138        assert_eq!(
139            docs.xml("Use `new_op` with `the_key`, a `Listener`, `on_put`, & `KvError`."),
140            "Use <c>NewOp</c> with <c>theKey</c>, a <c>IListener</c>, <c>OnPut</c>, &amp; <c>KvException</c>."
141        );
142        assert_eq!(
143            docs.xml("Fails with `not_found` or `a + b`."),
144            "Fails with <c>KvException.NotFound</c> or `a + b`."
145        );
146    }
