20/// A type's doc comment, with a trailing `@deprecated` paragraph when it's
21/// deprecated. Types carry the deprecation in their docs rather than as an
22/// attribute, so the generated code that marshals them compiles
23/// warning-free.
24pub(crate) fn type_doc(
25    ctx: &Ctx<'_>,
26    doc: &Option<String>,
27    deprecated: &Option<String>,
28) -> Option<String> {
29    let doc = Doc::new(doc, deprecated);
30    let spell = |s: &str| ctx.spell(s);
31    let note = doc.deprecation(spell).map(|m| format!("@deprecated {m}"));
32    match (doc.text(spell), note) {
33        (Some(d), Some(n)) => Some(format!("{d}\n\n{n}")),
34        (d, n) => d.or(n),
35    }
36}
37
38/// A field's or variant's doc comment, rewritten.
39fn field_doc(ctx: &Ctx<'_>, doc: &Option<String>) -> Option<String> {
40    Doc::new(doc, &None).text(|s| ctx.spell(s))
41}

45/// Append every C-style enum as `enum class {Name} : int32_t`. Rich enums
46/// are value types, rendered with the records.
47pub(crate) fn render_cpp_enums(w: &mut CodeWriter, ctx: &Ctx<'_>) {
48    for e in ctx.model.enums.iter().filter(|e| !e.is_rich()) {
49        w.doc(
50            &type_doc(ctx, &e.doc, &e.deprecated),
51            DocCommentStyle::Javadoc,
52        );
53        w.block(format!("enum class {} : int32_t {{", e.name), "};", |w| {
54            for v in &e.variants {
55                w.doc(&field_doc(ctx, &v.doc), DocCommentStyle::Javadoc);
56                w.line(format!("{} = {},", cpp_ident(&v.name), v.value));
57            }
58        });
59        w.blank();
60    }
61}

65/// A value type emitted as a plain C++ struct: a record or a rich enum. Both
66/// cross the ABI in value buffers, may nest one another, and are ordered
67/// together so a by-value member's type is complete before its holder.
68pub(crate) enum ValueDef<'a> {
69    /// A record: a plain struct with typed members.
70    Record(&'a RecordBinding),
71    /// A rich enum: a `std::variant`-backed sum type.
72    Rich(&'a EnumBinding),
73}
74
75impl ValueDef<'_> {
76    /// The value type's C++ name.
77    pub(crate) fn name(&self) -> &str {
78        match self {
79            ValueDef::Record(s) => &s.name,
80            ValueDef::Rich(e) => &e.name,
81        }
82    }
83
84    /// Names of other value types this one holds by value.
85    fn deps(&self) -> Vec<String> {
86        let mut deps = Vec::new();
87        let fields: Vec<&Ty> = match self {
88            ValueDef::Record(s) => s.fields.iter().map(|f| &f.ty).collect(),
89            ValueDef::Rich(e) => e
90                .variants
91                .iter()
92                .flat_map(|v| &v.fields)
93                .map(|f| &f.ty)
94                .collect(),
95        };
96        for ty in fields {
97            collect_value_deps(ty, &mut deps);
98        }
99        deps
100    }
101
102    /// Append the type's definition. `complete` names the value types
103    /// defined so far (see [`member_decl`]).
104    pub(crate) fn render(&self, w: &mut CodeWriter, ctx: &Ctx<'_>, complete: &HashSet<&str>) {
105        match self {
106            ValueDef::Record(s) => render_cpp_record(w, ctx, s, complete),
107            ValueDef::Rich(e) => render_cpp_rich_enum(w, ctx, e, complete),
108        }
109    }
110
111    /// Append the definitions of the type's `operator==` (and its variant
112    /// payloads'), which need every value type complete.
113    pub(crate) fn render_equality(&self, w: &mut CodeWriter) {
114        match self {
115            ValueDef::Record(s) => render_equality_definition(w, &s.name, &s.fields),
116            ValueDef::Rich(e) => {
117                let name = &e.name;
118                for v in &e.variants {
119                    let payload = format!("{name}::{}", cpp_ident(&v.name));
120                    render_equality_definition(w, &payload, &v.fields);
121                }
122                w.line(format!(
123                    "inline bool operator==(const {name}& a, const {name}& b) {{ return a.value == b.value; }}"
124                ));
125            }
126        }
127        w.blank();
128    }
129}

131/// Collect the names of value types reachable from `ty` through optionals,
132/// lists, and maps. Interfaces aren't collected: every interface class is
133/// complete before any value type.
134fn collect_value_deps(ty: &Ty, deps: &mut Vec<String>) {
135    match ty {
136        Ty::Record(n) | Ty::RichEnum(n) => deps.push(n.clone()),
137        Ty::Optional(inner) | Ty::List(inner) => collect_value_deps(inner, deps),
138        Ty::Map(k, v) => {
139            collect_value_deps(k, deps);
140            collect_value_deps(v, deps);
141        }
142        _ => {}
143    }
144}

146/// Every record and rich enum, ordered so anything a type holds is defined
147/// before it (depth-first post-order, declaration order breaking ties).
148pub(crate) fn value_types_in_order(model: &Model) -> Vec<ValueDef<'_>> {
149    let records = model.records.iter().map(ValueDef::Record);
150    let rich = model
151        .enums
152        .iter()
153        .filter(|e| e.is_rich())
154        .map(ValueDef::Rich);
155    let mut defs: Vec<Option<ValueDef<'_>>> = records.chain(rich).map(Some).collect();
156    let deps: Vec<Vec<String>> = defs.iter().flatten().map(ValueDef::deps).collect();
157    let index: HashMap<String, usize> = defs
158        .iter()
159        .flatten()
160        .enumerate()
161        .map(|(i, d)| (d.name().to_string(), i))
162        .collect();
163
164    fn visit(
165        i: usize,
166        deps: &[Vec<String>],
167        index: &HashMap<String, usize>,
168        state: &mut [u8],
169        order: &mut Vec<usize>,
170    ) {
171        // 0 unvisited, 1 on the stack (a cycle stops here), 2 emitted.
172        if state[i] != 0 {
173            return;
174        }
175        state[i] = 1;
176        for d in &deps[i] {
177            if let Some(&j) = index.get(d) {
178                visit(j, deps, index, state, order);
179            }
180        }
181        state[i] = 2;
182        order.push(i);
183    }
184
185    let mut state = vec![0u8; defs.len()];
186    let mut order = Vec::with_capacity(defs.len());
187    for i in 0..defs.len() {
188        visit(i, &deps, &index, &mut state, &mut order);
189    }
190    order
191        .into_iter()
192        .map(|i| defs[i].take().expect("each type is emitted once"))
193        .collect()
194}

196/// One member declaration of a record or variant payload, with a default
197/// member initializer (zero for scalars and enums, empty otherwise) when its
198/// type can be value-initialized. A member holding an interface wrapper, or
199/// a record that holds one, has none, and an initializer must provide it.
200/// Every other member has a default, so a braced initializer may stop after
201/// the members it needs without `-Wmissing-field-initializers` warnings.
202///
203/// The exception is an optional holding a value type not yet in `complete`
204/// (two types can hold each other): from C++20, `{}` instantiates
205/// `std::optional`'s `constexpr` constructor, which needs the type complete.
206/// Without it the member still starts empty.
207fn member_decl(ctx: &Ctx<'_>, f: &FieldBinding, complete: &HashSet<&str>) -> String {
208    let incomplete_optional = matches!(f.ty, Ty::Optional(_)) && {
209        let mut deps = Vec::new();
210        collect_value_deps(&f.ty, &mut deps);
211        deps.iter().any(|d| !complete.contains(d.as_str()))
212    };
213    let init = if defaultable(ctx.model, &f.ty) && !incomplete_optional {
214        "{}"
215    } else {
216        ""
217    };
218    format!("{} {}{init};", cpp_type(&f.ty), cpp_ident(&f.name))
219}

221/// Whether `ty` is a scalar, `bool`, or C-style enum: trivially copied, and
222/// zero when value-initialized.
223fn is_scalar(ty: &Ty) -> bool {
224    match ty {
225        Ty::Prim(p) => p.is_scalar(),
226        Ty::Enum(_) => true,
227        _ => false,
228    }
229}

231/// Append the friend declaration of a value struct's `operator==` and its
232/// inline `operator!=`. [`render_equality_definition`] defines `==` once
233/// every value type is complete: comparing a list of a type that's still
234/// incomplete (two types can hold each other) doesn't compile from C++20,
235/// where `std::vector`'s comparison is `constexpr`.
236fn render_equality(w: &mut CodeWriter, name: &str, fields: &[FieldBinding]) {
237    w.line("/** Memberwise equality; interface members compare by identity. */");
238    if fields.is_empty() {
239        w.line(format!(
240            "friend bool operator==(const {name}&, const {name}&) noexcept;"
241        ));
242    } else {
243        w.line(format!(
244            "friend bool operator==(const {name}& a, const {name}& b);"
245        ));
246    }
247    w.line(format!(
248        "friend bool operator!=(const {name}& a, const {name}& b) {{ return !(a == b); }}"
249    ));
250}

252/// Append the definition of a value struct's `operator==`, for `name` as
253/// spelled at namespace scope: memberwise, in declaration order.
254/// Floating-point members compare as IEEE values, and interface members by
255/// identity.
256fn render_equality_definition(w: &mut CodeWriter, name: &str, fields: &[FieldBinding]) {
257    if fields.is_empty() {
258        w.line(format!(
259            "inline bool operator==(const {name}&, const {name}&) noexcept {{ return true; }}"
260        ));
261        return;
262    }
263    let cmp: Vec<String> = fields
264        .iter()
265        .map(|f| {
266            let n = cpp_ident(&f.name);
267            format!("a.{n} == b.{n}")
268        })
269        .collect();
270    let head = format!("inline bool operator==(const {name}& a, const {name}& b) {{");
271    if cmp.len() <= 2 {
272        w.line(format!("{head} return {}; }}", cmp.join(" && ")));
273    } else {
274        w.block(head, "}", |w| {
275            w.line(format!("return {}", cmp[0]));
276            w.scope(|w| {
277                for (i, c) in cmp.iter().enumerate().skip(1) {
278                    let end = if i + 1 == cmp.len() { ";" } else { "" };
279                    w.line(format!("&& {c}{end}"));
280                }
281            });
282        });
283    }
284}

286/// Render a record as a plain C++ aggregate: typed members in declaration
287/// (and wire) order with default member initializers, and memberwise
288/// equality. An interface-typed member is the RAII wrapper held by value,
289/// so copying the record shares the object and destroying it releases one
290/// reference.
291fn render_cpp_record(
292    w: &mut CodeWriter,
293    ctx: &Ctx<'_>,
294    s: &RecordBinding,
295    complete: &HashSet<&str>,
296) {
297    w.doc(
298        &type_doc(ctx, &s.doc, &s.deprecated),
299        DocCommentStyle::Javadoc,
300    );
301    w.block(format!("struct {} {{", s.name), "};", |w| {
302        for f in &s.fields {
303            w.doc(&field_doc(ctx, &f.doc), DocCommentStyle::Javadoc);
304            w.line(member_decl(ctx, f, complete));
305        }
306        if !s.fields.is_empty() {
307            w.blank();
308        }
309        render_equality(w, &s.name, &s.fields);
310    });
311    w.blank();
312}

314/// Render a rich enum as a `std::variant`-backed sum type: one payload
315/// struct per variant, a `value` member holding the active payload, a nested
316/// `Tag` enum mirroring the wire discriminants, a `tag()` reader, and
317/// equality. Construct one as `Shape{Shape::Circle{2.0}}`.
318fn render_cpp_rich_enum(
319    w: &mut CodeWriter,
320    ctx: &Ctx<'_>,
321    e: &EnumBinding,
322    complete: &HashSet<&str>,
323) {
324    let name = &e.name;
325    w.doc(
326        &type_doc(ctx, &e.doc, &e.deprecated),
327        DocCommentStyle::Javadoc,
328    );
329    w.block(format!("struct {name} {{"), "};", |w| {
330        w.line(format!(
331            "/** Discriminant identifying the active variant of `{name}`. */"
332        ));
333        w.block("enum class Tag : int32_t {", "};", |w| {
334            for v in &e.variants {
335                w.line(format!("{} = {},", cpp_ident(&v.name), v.value));
336            }
337        });
338        w.blank();
339        for v in &e.variants {
340            let vn = cpp_ident(&v.name);
341            w.doc(&field_doc(ctx, &v.doc), DocCommentStyle::Javadoc);
342            w.block(format!("struct {vn} {{"), "};", |w| {
343                for f in &v.fields {
344                    w.doc(&field_doc(ctx, &f.doc), DocCommentStyle::Javadoc);
345                    w.line(member_decl(ctx, f, complete));
346                }
347                if !v.fields.is_empty() {
348                    w.blank();
349                }
350                render_equality(w, &vn, &v.fields);
351            });
352            w.blank();
353        }
354        let alts: Vec<String> = e.variants.iter().map(|v| cpp_ident(&v.name)).collect();
355        w.line("/** The active variant's payload. */");
356        w.line(format!("std::variant<{}> value;", alts.join(", ")));
357        w.blank();
358        w.line("/** The tag of the active variant. */");
359        w.block("Tag tag() const noexcept {", "}", |w| {
360            let tags: Vec<String> = alts.iter().map(|v| format!("Tag::{v}")).collect();
361            w.line(format!(
362                "static constexpr Tag tags[] = {{{}}};",
363                tags.join(", ")
364            ));
365            w.line("return tags[value.index()];");
366        });
367        w.blank();
368        w.line("/** Equal when the same variant is active with equal payloads. */");
369        w.line(format!(
370            "friend bool operator==(const {name}& a, const {name}& b);"
371        ));
372        w.line(format!(
373            "friend bool operator!=(const {name}& a, const {name}& b) {{ return !(a == b); }}"
374        ));
375    });
376    w.blank();
377}

381/// The member name of an error code's field: [`cpp_ident`], with a trailing
382/// underscore when it would hide a member every exception has (`code()`,
383/// `what()`).
384pub(crate) fn error_field(name: &str) -> String {
385    lang::escape_member(&cpp_ident(name), &["code", "what"])
386}
387
388/// Append one error domain: its class derived from `Error`, one subclass
389/// per declared code (with typed members for the code's payload fields),
390/// and its `detail::Errors` policy.
391///
392/// The policy's `raise` throws `Cancelled` for -5 and the root `Error` for
393/// any other negative (runtime) code, the code's class for a declared code,
394/// and the domain class itself for a positive code these bindings don't
395/// know (domains are open: a newer producer may add codes). Its `report`
396/// turns a domain exception a callback implementation threw into the
397/// producer's error slot, payload included; any other exception is a plain
398/// failure (-1).
399pub(crate) fn render_domain_error(w: &mut CodeWriter, ctx: &Ctx<'_>, table: &ErrorTable<'_>) {
400    let domain = &table.type_name;
401    let prefix = ctx.prefix;
402    w.line(format!(
403        "/** The `{}` error domain: catch a code's class or this. */",
404        table.domain.name
405    ));
406    w.line(format!("class {domain} : public Error {{"));
407    w.line("public:");
408    w.scope(|w| {
409        w.line("/** Builds an error carrying one of the domain's `code`s and its `message`. */");
410        w.line(format!(
411            "{domain}(int32_t code, const std::string& message) : Error(code, message) {{}}"
412        ));
413    });
414    w.line("};");
415    w.blank();
416
417    for row in &table.codes {
418        let code = row.code;
419        let class = &row.type_name;
420        let doc = Doc::new(&code.doc, &None)
421            .text(|s| ctx.spell(s))
422            .unwrap_or_else(|| code.message.clone());
423        w.doc(
424            &Some(format!("{doc}\n\nCode {}.", code.value)),
425            DocCommentStyle::Javadoc,
426        );
427        w.line(format!("class {class} : public {domain} {{"));
428        w.line("public:");
429        w.scope(|w| {
430            for f in &code.fields {
431                w.doc(&field_doc(ctx, &f.doc), DocCommentStyle::Javadoc);
432                w.line(format!("{} {};", cpp_type(&f.ty), error_field(&f.name)));
433            }
434            if !code.fields.is_empty() {
435                w.blank();
436            }
437            // The message parameter steps aside for a field named `message`.
438            let mut message = "message".to_string();
439            while code.fields.iter().any(|f| error_field(&f.name) == message) {
440                message.push('_');
441            }
442            let mut params = vec![format!("const std::string& {message}")];
443            let mut inits = vec![format!("{domain}({}, {message})", code.value)];
444            for f in &code.fields {
445                let name = error_field(&f.name);
446                params.push(format!("{} {name}", cpp_type(&f.ty)));
447                if is_scalar(&f.ty) {
448                    inits.push(format!("{name}({name})"));
449                } else {
450                    inits.push(format!("{name}(std::move({name}))"));
451                }
452            }
453            w.line("/** Builds the error with the producer's `message` and the code's fields. */");
454            let explicit = if code.fields.is_empty() {
455                "explicit "
456            } else {
457                ""
458            };
459            w.line(format!(
460                "{explicit}{class}({}) : {} {{}}",
461                params.join(", "),
462                inits.join(", ")
463            ));
464        });
465        w.line("};");
466        w.blank();
467    }
468
469    w.line("namespace detail {");
470    w.blank();
471    w.line(format!(
472        "/** Raises and reports `{domain}` and its codes' classes. */"
473    ));
474    w.line("template <>");
475    w.block(format!("struct Errors<{domain}> {{"), "};", |w| {
476        w.block(
477            format!("[[noreturn]] static void raise(const {prefix}_error& err) {{"),
478            "}",
479            |w| {
480                w.line("if (err.code < 0) raise_runtime<Error>(err);");
481                w.block("switch (err.code) {", "}", |w| {
482                    for row in &table.codes {
483                        let code = row.code;
484                        let class = &row.type_name;
485                        if code.fields.is_empty() {
486                            w.line(format!("case {}: throw {class}(message(err));", code.value));
487                        } else {
488                            let fields: Vec<String> =
489                                code.fields.iter().map(|f| cpp_type(&f.ty)).collect();
490                            w.line(format!(
491                                "case {}: raise_with_fields<{class}, {}>(err);",
492                                code.value,
493                                fields.join(", ")
494                            ));
495                        }
496                    }
497                    w.line(format!("default: throw {domain}(err.code, message(err));"));
498                });
499            },
500        );
501        w.blank();
502        w.block(
503            format!("static void report({prefix}_error* out_err) noexcept {{"),
504            "}",
505            |w| {
506                w.line("try {");
507                w.scope(|w| {
508                    w.line("throw;");
509                });
510                for row in table.codes.iter().filter(|r| !r.code.fields.is_empty()) {
511                    let fields: Vec<String> = row
512                        .code
513                        .fields
514                        .iter()
515                        .map(|f| format!("e.{}", error_field(&f.name)))
516                        .collect();
517                    w.line(format!("}} catch (const {}& e) {{", row.type_name));
518                    w.scope(|w| {
519                        w.line(format!(
520                            "report_with_fields(out_err, e, {});",
521                            fields.join(", ")
522                        ));
523                    });
524                }
525                w.line(format!("}} catch (const {domain}& e) {{"));
526                w.scope(|w| {
527                    w.line("set_error(out_err, e.code(), e.what());");
528                });
529                w.line("} catch (...) {");
530                w.scope(|w| {
531                    w.line("report_current(out_err, -1);");
532                });
533                w.line("}");
534            },
535        );
536    });
537    w.blank();
538    w.line("} // namespace detail");
539    w.blank();
540}

544/// The C++ name and declaration kind of each interface member. The
545/// synchronous constructor named `new` becomes the C++ constructor; every
546/// other constructor becomes a static factory named after it.
547fn member_kinds(i: &InterfaceBinding) -> Vec<(&FnBinding, String, FnKind<'_>)> {
548    let class = i.name.as_str();
549    let mut members = Vec::new();
550    for c in &i.constructors {
551        if c.name == "new" && matches!(c.shape, Shape::Sync) {
552            members.push((c, class.to_string(), FnKind::Ctor { class }));
553        } else {
554            members.push((c, cpp_member_name(&c.name), FnKind::Static { class }));
555        }
556    }
557    for m in &i.methods {
558        members.push((m, cpp_member_name(&m.name), FnKind::Method { class }));
559    }
560    for s in &i.statics {
561        members.push((s, cpp_member_name(&s.name), FnKind::Static { class }));
562    }
563    members
564}

566/// Append an interface's class: the reference-counting core (a
567/// `detail::Handle` with the class's `traits`, so the class itself follows
568/// the Rule of Zero), the `adopt` constructor, the `handle()` and
569/// `clone_handle()` readers, identity equality, and the *declarations* of
570/// its members. The member bodies follow every value type and codec
571/// ([`render_cpp_interface_members`]); the class comes first so records can
572/// hold it by value.
573pub(crate) fn render_cpp_interface_class(w: &mut CodeWriter, ctx: &Ctx<'_>, i: &InterfaceBinding) {
574    let name = &i.name;
575    let tag = &i.c_tag;
576    w.doc(
577        &type_doc(ctx, &i.doc, &i.deprecated),
578        DocCommentStyle::Javadoc,
579    );
580    w.line(format!("class {name} {{"));
581    w.line("public:");
582    w.scope(|w| {
583        w.line("/** The C type this class wraps. */");
584        w.line(format!("using raw_type = {tag};"));
585        w.blank();
586        w.line(format!(
587            "/** Adopts one strong reference to a producer object: `{name}(adopt, raw)`. */"
588        ));
589        w.line(format!(
590            "explicit {name}(adopt_t, raw_type* raw) noexcept : raw_(raw) {{}}"
591        ));
592        w.blank();
593        for (f, cpp_name, kind) in member_kinds(i) {
594            render_member_decl(w, ctx, f, &cpp_name, kind);
595        }
596        w.line("/** The wrapped pointer, borrowed (null after a move): this wrapper keeps its reference. */");
597        w.line("const raw_type* handle() const noexcept { return raw_.get(); }");
598        w.blank();
599        w.line("/** A new strong reference to the object, which the caller owns. */");
600        w.line("raw_type* clone_handle() const noexcept { return raw_.clone(); }");
601        w.blank();
602        w.line("/** Whether both wrap the same producer object. */");
603        w.line(format!(
604            "friend bool operator==(const {name}& a, const {name}& b) noexcept {{ return a.raw_.get() == b.raw_.get(); }}"
605        ));
606        w.line(format!(
607            "friend bool operator!=(const {name}& a, const {name}& b) noexcept {{ return !(a == b); }}"
608        ));
609    });
610    w.blank();
611    w.line("private:");
612    w.scope(|w| {
613        w.block("struct traits {", "};", |w| {
614            w.line(format!("using raw_type = {tag};"));
615            w.line(format!(
616                "static raw_type* clone(const raw_type* raw) noexcept {{ return {}(raw); }}",
617                i.clone_symbol
618            ));
619            w.line(format!(
620                "static void destroy(raw_type* raw) noexcept {{ {}(raw); }}",
621                i.destroy_symbol
622            ));
623        });
624        w.blank();
625        w.line("detail::Handle<traits> raw_;");
626    });
627    w.line("};");
628    w.blank();
629}
