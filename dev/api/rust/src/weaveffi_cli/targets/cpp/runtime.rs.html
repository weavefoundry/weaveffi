1//! The fixed runtime of the generated header, written from real C++ sources
2//! under `runtime/` with `{{PLACEHOLDER}}` substitution:
3//!
4//! * `prelude.hpp`: the exception hierarchy, the adopt tag, the generic
5//!   machinery every wrapper delegates to (`detail::Errors<E>` and
6//!   `detail::check<E>`, `detail::settle` for async completions,
7//!   `detail::callback` for trampolines, `detail::Handle<Traits>` for
8//!   objects, and the run and optional-scalar helpers), `CancelToken`, the
9//!   generic `Range<T>` every iterator returns, and `check_library()` with
10//!   the contract rows;
11//! * `buffer.hpp` (when any value crosses as a value buffer): the
12//!   value-buffer reader and writer and the overloaded `detail::write` and
13//!   `detail::read` templates for primitives, enums, objects, optionals,
14//!   vectors, and maps.
15//!
16//! The library check embeds every declaration's contract row (from the
17//! shared [`contract`] emitter) as the library's contract function, the
18//! expected hash, and the path, so a mismatch names the declaration.
19
20use crate::codegen::contract;
21use crate::codegen::CodeWriter;
22use weaveffi_model::model::{Model, ABI_VERSION};
23
24/// The runtime prelude. `{{CONTRACTS}}` marks where the contract rows go
25/// and `{{CHECKS}}` where `check_library()` compares them.
26const PRELUDE: &str = include_str!("runtime/prelude.hpp");
27
28/// The value-buffer runtime.
29const BUFFER: &str = include_str!("runtime/buffer.hpp");
30
31/// Append the runtime prelude. `check_library()` compares the producer's
32/// ABI revision and every declaration's contract hash against the ones
33/// this header was generated with.
34pub(crate) fn render_prelude_runtime(w: &mut CodeWriter, model: &Model) {
35    let library = &model.identity.library;
36    let text = PRELUDE
37        .replace("{{PREFIX}}", model.prefix())
38        .replace("{{MACRO}}", &model.identity.macro_prefix())
39        .replace("{{LIBRARY}}", library)
40        .replace("{{ABI}}", &ABI_VERSION.to_string());
41    let (head, rest) = text
42        .split_once("{{CONTRACTS}}")
43        .expect("the prelude marks the contract rows");
44    let (middle, tail) = rest
45        .split_once("{{CHECKS}}")
46        .expect("the prelude marks the contract check");
47    let rows = contract::rows(model);
48
49    w.raw(head);
50    // C++ has no empty arrays, so an API without declarations has no rows.
51    if !rows.is_empty() {
52        w.line("/** The declarations this header was generated with. */");
53        w.block("inline const ContractRow contract_rows[] = {", "};", |w| {
54            for row in &rows {
55                w.line(format!(
56                    "{{&{}, {}ull, \"{}\"}}, // {}",
57                    row.symbol,
58                    contract::hex(row.hash),
59                    row.path,
60                    row.signature
61                ));
62            }
63        });
64        w.blank();
65    }
66    w.raw(middle);
67    if !rows.is_empty() {
68        // The check sits inside `check_library()`'s initializing lambda.
69        w.indent().indent();
70        w.block(
71            "if (std::string why = detail::contract_mismatch(detail::contract_rows); !why.empty()) {",
72            "}",
73            |w| {
74                w.line(format!("return \"{library}: \" + why;"));
75            },
76        );
77        w.dedent().dedent();
78    }
79    w.raw(tail);
80}
81
82/// Append the value-buffer runtime.
83pub(crate) fn render_buffer_runtime(w: &mut CodeWriter, prefix: &str) {
84    w.raw(BUFFER.replace("{{PREFIX}}", prefix));
