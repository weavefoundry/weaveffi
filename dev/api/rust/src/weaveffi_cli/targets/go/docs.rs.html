1//! Go doc comments, rendered from the shared [`Doc`] text.
2//!
3//! IDL docs are copied as written, with backticked API identifiers turned
4//! into Go doc links (`` `new_op` `` is `[NewOp]`, a method `[Store.Get]`)
5//! or their Go spelling (a field `ExpiresAt`, a parameter `ttlSeconds`).
6//! Go convention starts the doc of an exported package-level declaration
7//! with its name, so a summary that doesn't is prefixed with `Name: `;
8//! fields, constants, and interface methods keep their text as is. A
9//! deprecation becomes the final `Deprecated:` paragraph that Go tools
10//! recognize.
11

21/// The width generated comment text wraps at (IDL docs keep their own line
22/// breaks).
23const WRAP: usize = 76;
24
25/// Every API identifier a doc may name in backticks, with its Go spelling.
26pub(crate) struct DocNames<'a> {
27    api: ApiNames,
28    go: &'a GoNames,
29    /// Enum variants (C-style constants and rich-enum structs) by C
30    /// constant.
31    variants: BTreeMap<String, String>,
32}
33
34impl<'a> DocNames<'a> {
35    /// Index the identifiers of `model`, spelled per `go`.
36    pub(crate) fn new(model: &Model, go: &'a GoNames) -> Self {
37        let variants = model
38            .enums
39            .iter()
40            .flat_map(|e| {
41                e.variants.iter().map(move |v| {
42                    (
43                        v.c_const.clone(),
44                        format!("{}{}", pascal(&e.name), pascal(&v.name)),
45                    )
46                })
47            })
48            .collect();
49        Self {
50            api: ApiNames::new(model),
51            go,
52            variants,
53        }
54    }
55
56    /// The Go replacement for a backticked identifier, backticks included:
57    /// a doc link for a package-level name or method, the plain Go spelling
58    /// for a field or parameter, and `None` (kept as written) for an
59    /// identifier that names declarations of different kinds.
60    fn replace(&self, ident: &str) -> Option<String> {
61        let link = |s: &str| Some(format!("[{s}]"));
62        match self.api.kind(ident)? {
63            IdentKind::Function | IdentKind::Member => {
64                link(self.go.callable(self.api.c_name(ident)?)?)
65            }
66            IdentKind::Type => link(&pascal(ident)),
67            IdentKind::ErrorDomain => link(&self.go.domain(ident).iface),
68            IdentKind::ErrorCode => link(self.go.code_by_const(self.api.c_name(ident)?)?),
69            IdentKind::Variant => link(self.variants.get(self.api.c_name(ident)?)?),
70            IdentKind::Field | IdentKind::CallbackMethod => Some(pascal(ident)),
71            IdentKind::Param => Some(method_param(ident)),
72        }
73    }
74
75    /// `text` with its backticked identifiers in Go spelling.
76    pub(crate) fn render(&self, text: &str) -> String {
77        map_code_spans(text, |ident| self.replace(ident))
78    }
79
80    /// The doc text and deprecation message of a declaration, rendered.
81    pub(crate) fn of(
82        &self,
83        doc: &Option<String>,
84        deprecated: &Option<String>,
85    ) -> (Option<String>, Option<String>) {
86        let d = Doc::new(doc, deprecated);
87        let keep = |_: &str| None;
88        (
89            d.text(keep).map(|t| self.render(&t)),
90            d.deprecation(keep).map(|t| self.render(&t)),
91        )
92    }
93}

95/// A Go doc comment under construction: paragraphs, then `Deprecated:`.
96#[derive(Default)]
97pub(crate) struct GoDoc {
98    paragraphs: Vec<String>,
99    deprecated: Option<String>,
100}
101
102impl GoDoc {
103    /// The doc of the exported package-level declaration `name`: `text`,
104    /// prefixed with `Name: ` unless its first word is already the name.
105    pub(crate) fn decl(name: &str, text: Option<String>) -> Self {
106        let mut doc = Self::default();
107        if let Some(text) = text {
108            let first = text.split_whitespace().next().unwrap_or_default();
109            if first.trim_end_matches(['.', ',', ':', ';', '\'']) == name {
110                doc.paragraphs.push(text);
111            } else {
112                doc.paragraphs.push(format!("{name}: {text}"));
113            }
114        }
115        doc
116    }
117
118    /// A doc whose text needs no name (a field, constant, or interface
119    /// method).
120    pub(crate) fn plain(text: Option<String>) -> Self {
121        Self {
122            paragraphs: text.into_iter().collect(),
123            deprecated: None,
124        }
125    }
126
127    /// Append a paragraph of generated text, rewrapped.
128    pub(crate) fn para(mut self, text: &str) -> Self {
129        self.paragraphs.push(wrap(text, WRAP));
130        self
131    }
132
133    /// Append a list of the documented parameters, labelled with their Go
134    /// spellings: `(name, doc)` pairs.
135    pub(crate) fn params(mut self, params: impl IntoIterator<Item = (String, String)>) -> Self {
136        let mut list = Vec::new();
137        for (name, doc) in params {
138            let mut lines = doc.lines();
139            list.push(format!("  - {name}: {}", lines.next().unwrap_or_default()));
140            list.extend(lines.map(|l| {
141                if l.trim().is_empty() {
142                    String::new()
143                } else {
144                    format!("    {l}")
145                }
146            }));
147        }
148        if !list.is_empty() {
149            self.paragraphs.push("Parameters:".into());
150            self.paragraphs.push(list.join("\n"));
151        }
152        self
153    }
154
155    /// Set the `Deprecated:` paragraph.
156    pub(crate) fn deprecated(mut self, msg: Option<String>) -> Self {
157        self.deprecated = msg;
158        self
159    }
160
161    /// Write the comment at the writer's indentation.
162    pub(crate) fn emit(self, w: &mut CodeWriter) {
163        let mut paragraphs = self.paragraphs;
164        if let Some(msg) = self.deprecated {
165            paragraphs.push(wrap(&format!("Deprecated: {msg}"), WRAP));
166        }
167        for (i, p) in paragraphs.iter().enumerate() {
168            if i > 0 {
169                w.line("//");
170            }
171            for line in p.lines() {
172                let line = line.trim_end();
173                if line.is_empty() {
174                    w.line("//");
175                } else {
176                    w.line(format!("// {line}"));
177                }
178            }
179        }
180    }
181}

183#[cfg(test)]
184mod tests {
185    use super::*;
186    use crate::codegen::test_model;
187
188    fn render(doc: GoDoc) -> String {
189        let mut w = CodeWriter::tabs();
190        doc.emit(&mut w);
191        w.finish()
192    }
193
194    #[test]
195    fn package_level_docs_start_with_the_name() {
196        let doc = |name: &str, text: &str| render(GoDoc::decl(name, Some(text.into())));
197        assert_eq!(
198            doc("Sum", "Sum returns the sum."),
199            "// Sum returns the sum.\n"
200        );
201        assert_eq!(doc("Ping", "Trivial helper"), "// Ping: Trivial helper\n");
202        assert_eq!(doc("Open", "Open a store."), "// Open a store.\n");
203        assert_eq!(
204            render(GoDoc::plain(Some("A field.\n\nMore.".into()))),
205            "// A field.\n//\n// More.\n"
206        );
207        assert_eq!(render(GoDoc::decl("X", None)), "");
208    }
209
210    #[test]
211    fn deprecation_comes_last_and_long_text_wraps() {
212        let doc = GoDoc::decl("Old", Some("Old returns it.".into()))
213            .para("Extra.")
214            .deprecated(Some("Use [New] instead.".into()));
215        assert_eq!(
216            render(doc),
217            "// Old returns it.\n//\n// Extra.\n//\n// Deprecated: Use [New] instead.\n"
218        );
219        let long = render(GoDoc::plain(None).para(&"word ".repeat(30)));
220        assert!(long.lines().all(|l| l.len() <= WRAP + 3), "{long}");
221    }

223    #[test]
224    fn backticked_identifiers_become_links_or_go_spellings() {
225        let model = test_model(
226            r#"
227version: "0.13.0"
228errors:
229  - name: KvErrors
230    codes: [{ name: not_found, code: 1, message: missing }]
231enums:
232  - name: Mode
233    variants: [{ name: fast, value: 0 }]
234records:
235  - name: Entry
236    fields: [{ name: expires_at, type: i64 }]
237interfaces:
238  - name: Store
239    constructors: [{ name: open, params: [] }]
240    methods:
241      - { name: get_all, params: [{ name: ttl_seconds, type: i64 }] }
242functions:
243  - { name: new_op, params: [] }
244"#,
245        );
246        let go = GoNames::new(&model);
247        let names = DocNames::new(&model, &go);
248        assert_eq!(
249            names.render(
250                "Use `new_op`, `open`, `get_all`, `Entry`, `KvErrors`, `not_found`, \
251                 `fast`, `expires_at`, `ttl_seconds`, `kv`, or `x + 1`."
252            ),
253            "Use [NewOp], [OpenStore], [Store.GetAll], [Entry], [KvError], \
254             [NotFoundError], [ModeFast], ExpiresAt, ttlSeconds, `kv`, or `x + 1`."
255        );
256    }
