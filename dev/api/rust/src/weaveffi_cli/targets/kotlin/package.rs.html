13/// The Kotlin Gradle plugin version the build files pin.
14const KOTLIN_VERSION: &str = "2.0.21";
15/// The Android Gradle plugin version the Android flavor pins.
16const AGP_VERSION: &str = "8.7.3";
17/// The coroutines runtime async callables suspend through.
18const COROUTINES: &str = "org.jetbrains.kotlinx:kotlinx-coroutines-core:1.9.0";
19/// The JNA release the bindings call the C ABI through.

22/// Escape `s` for a double-quoted Kotlin string literal.
23pub(crate) fn kts_quote(s: &str) -> String {
24    s.replace('\\', "\\\\")
25        .replace('"', "\\\"")
26        .replace('$', "\\$")
27}

29/// Wrap `body` in the generated-file header and trailer.
30fn wrap(style: CommentStyle, file: &str, body: &str) -> String {
31    format!(
32        "{}{body}\n{}",
33        render_prelude(style),
34        render_trailer(style, file)
35    )
36}

38/// The classpath resource directory JNA extracts a desktop platform's
39/// library from (its `Platform.RESOURCE_PREFIX`), or `None` off the
40/// desktop.
41pub(crate) fn jna_resource_prefix(platform: Platform) -> Option<&'static str> {
42    Some(match platform {
43        Platform::MacosArm64 => "darwin-aarch64",
44        Platform::MacosX64 => "darwin-x86-64",
45        Platform::LinuxX64 => "linux-x86-64",
46        Platform::LinuxArm64 => "linux-aarch64",
47        Platform::WindowsX64 => "win32-x86-64",
48        _ => return None,
49    })
50}

52/// `settings.gradle.kts`: plugin and dependency repositories, so the module
53/// builds on its own, and the root project named after the package.
54pub(crate) fn settings_gradle(id: &Identity, flavor: KotlinFlavor) -> String {
55    let google = match flavor {
56        KotlinFlavor::Android => "        google()\n",
57        KotlinFlavor::Jvm => "",
58    };
59    let body = format!(
60        r#"pluginManagement {{
61    repositories {{
62{google}        mavenCentral()
63        gradlePluginPortal()
64    }}
65}}
66
67dependencyResolutionManagement {{
68    repositories {{
69{google}        mavenCentral()
70    }}
71}}
72
73rootProject.name = "{}"
74"#,
75        kts_quote(&id.name)
76    );
77    wrap(CommentStyle::DoubleSlash, "settings.gradle.kts", &body)
78}

80/// `build.gradle.kts` for the configured flavor.
81pub(crate) fn build_gradle(
82    id: &Identity,
83    package: &str,
84    config: &KotlinConfig,
85    has_async: bool,
86) -> String {
87    let (min_sdk, compile_sdk) = (config.min_sdk, config.compile_sdk);
88    let group = kts_quote(package);
89    let version = kts_quote(&id.version);
90    let artifact = kts_quote(&id.name);
91    let jna = match config.flavor {
92        // The AAR carries JNA's own native dispatch library per ABI.
93        KotlinFlavor::Android => format!("net.java.dev.jna:jna:{JNA_VERSION}@aar"),
94        KotlinFlavor::Jvm => format!("net.java.dev.jna:jna:{JNA_VERSION}"),
95    };
96    let coroutines = if has_async {
97        format!("\n    implementation(\"{COROUTINES}\")")
98    } else {
99        String::new()
100    };
101    let deps = format!("dependencies {{\n    implementation(\"{jna}\"){coroutines}\n}}\n");
102    let lib = &id.library;
103    let body = match config.flavor {
104        KotlinFlavor::Android => format!(
105            r#"plugins {{
106    id("com.android.library") version "{AGP_VERSION}"
107    id("org.jetbrains.kotlin.android") version "{KOTLIN_VERSION}"
108    `maven-publish`
109}}
110
111group = "{group}"
112version = "{version}"
113
114// The bindings call the producer library through JNA. A packaged build
115// ships it as src/main/jniLibs/<abi>/lib{lib}.so, which Gradle packages into
116// the AAR.
117android {{
118    namespace = "{group}"
119    compileSdk = {compile_sdk}
120    defaultConfig {{
121        minSdk = {min_sdk}
122        consumerProguardFiles("consumer-rules.pro")
123    }}
124    compileOptions {{
125        sourceCompatibility = JavaVersion.VERSION_1_8
126        targetCompatibility = JavaVersion.VERSION_1_8
127    }}
128    publishing {{
129        singleVariant("release") {{
130            withSourcesJar()
131        }}
132    }}
133}}
134
135kotlin {{
136    compilerOptions {{
137        jvmTarget.set(org.jetbrains.kotlin.gradle.dsl.JvmTarget.JVM_1_8)
138    }}
139}}
140
141{deps}
142publishing {{
143    publications {{
144        register<MavenPublication>("release") {{
145            artifactId = "{artifact}"
146            afterEvaluate {{
147                from(components["release"])
148            }}
149        }}
150    }}
151}}
152"#,
153        ),
154        KotlinFlavor::Jvm => format!(
155            r#"plugins {{
156    kotlin("jvm") version "{KOTLIN_VERSION}"
157    `java-library`
158    `maven-publish`
159}}
160
161group = "{group}"
162version = "{version}"
163
164// The bindings call the producer library through JNA, which loads it from
165// the path in the {env} environment variable, the system's library path, or
166// the classpath's <os>-<arch>/ resources (which a packaged build ships under
167// src/main/resources/).
168
169{deps}
170java {{
171    withSourcesJar()
172}}
173
174publishing {{
175    publications {{
176        create<MavenPublication>("maven") {{
177            artifactId = "{artifact}"
178            from(components["java"])
179        }}
180    }}
181}}
182"#,
183            env = id.library_env_var(),
184        ),
185    };
186    wrap(CommentStyle::DoubleSlash, "build.gradle.kts", &body)
187}

189/// `consumer-rules.pro`: JNA binds the C functions by method name and reads
190/// structure fields and callback methods reflectively, so R8 must keep the
191/// package and JNA intact in consuming apps.
192pub(crate) fn consumer_rules(package: &str) -> String {
193    let body = format!(
194        "# JNA binds C functions by method name and reads structures reflectively.\n\
195         -keep class {package}.** {{ *; }}\n\
196         -dontwarn java.awt.*\n\
197         -keep class com.sun.jna.* {{ *; }}\n\
198         -keepclassmembers class * extends com.sun.jna.* {{ public *; }}\n"
199    );
200    wrap(CommentStyle::Hash, "consumer-rules.pro", &body)
201}

203/// The packaged module's README: the layout, the bundled libraries, and how
204/// each runtime loads them.
205pub(crate) fn packaged_readme(id: &Identity, package: &str, ctx: &PackageContext) -> String {
206    let mut android = Vec::new();
207    let mut desktop = Vec::new();
208    for nb in &ctx.binaries.binaries {
209        let filename = ctx.binaries.bundled_filename(nb.platform);
210        if let Some(abi) = nb.platform.android_abi() {
211            android.push(format!("- `src/main/jniLibs/{abi}/{filename}`"));
212        } else if let Some(prefix) = jna_resource_prefix(nb.platform) {
213            desktop.push(format!("- `src/main/resources/{prefix}/{filename}`"));
214        }
215    }
216    let list = |v: Vec<String>| {
217        if v.is_empty() {
218            "- (none bundled)".to_string()
219        } else {
220            v.join("\n")
221        }
222    };
223    let body = format!(
224        r#"# {name} (Kotlin)
225
226Kotlin bindings for `{name}` in the `{package}` package: the Kotlin sources
227under `src/main/kotlin/`, which call the C ABI through JNA
228(`net.java.dev.jna:jna:{JNA_VERSION}`), and the prebuilt producer library for each
229bundled platform. There's no native glue to build.
230
231## Android
232
233The producer library sits in `src/main/jniLibs/<abi>/`, which Gradle packages
234into the AAR; JNA's own native library comes with its AAR. Build and publish
235with Gradle (`gradle assembleRelease`, `gradle publishToMavenLocal`).
236
237## Desktop JVM
238
239The producer library for each desktop platform sits in
240`src/main/resources/<os>-<arch>/`, where JNA finds it on the classpath and
241extracts it at run time. Set `{env}` to load a library from a path instead.
242
243## Bundled Android ABIs
244
245{android}
246
247## Bundled desktop platforms
248
249{desktop}
250"#,
251        name = id.name,
252        env = id.library_env_var(),
253        android = list(android),
254        desktop = list(desktop),
255    );
256    format!(
257        "{}{body}\n{}",
258        render_prelude(CommentStyle::Xml),
259        render_trailer(CommentStyle::Xml, "README.md")
260    )
