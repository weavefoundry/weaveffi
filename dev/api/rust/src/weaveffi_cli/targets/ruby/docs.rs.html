1//! YARD documentation: doc and deprecation text in Ruby spelling, and the
2//! `@param`, `@return`, `@raise`, and `@deprecated` tags of callables.
3
4use heck::ToSnakeCase;
5use weaveffi_model::errors::pascal;

7use crate::codegen::common::{wrap, DocCommentStyle};
8use crate::codegen::docs::{ApiNames, Doc, IdentKind};
9use crate::codegen::CodeWriter;
10use crate::targets::ruby::types::{rb_const, rb_domain};

12/// How the bindings spell an identifier an IDL doc names in backticks:
13/// callables, parameters, and fields in snake_case, error domains and
14/// codes by their class names, and types by their (escaped) constants.
15/// Variants keep their IDL spelling.
16pub(crate) fn spell(names: &ApiNames, ident: &str) -> Option<String> {
17    match names.kind(ident)? {
18        IdentKind::Function
19        | IdentKind::Member
20        | IdentKind::CallbackMethod
21        | IdentKind::Param
22        | IdentKind::Field => Some(ident.to_snake_case()),
23        IdentKind::ErrorDomain => Some(rb_domain(ident)),
24        IdentKind::ErrorCode => Some(pascal(ident)),
25        IdentKind::Type => Some(rb_const(ident)),
26        IdentKind::Variant => None,
27    }
28}

30/// One `@param` tag: the Ruby name, its YARD type, and its doc.
31pub(crate) struct ParamDoc {
32    /// The parameter's Ruby name.
33    pub(crate) name: String,
34    /// Its YARD type (`Integer`, `Store, nil`).
35    pub(crate) ty: String,
36    /// Its IDL doc, if any.
37    pub(crate) doc: Option<String>,
38}

40/// Everything one callable's YARD comment says.
41#[derive(Default)]
42pub(crate) struct CallableDoc {
43    /// Notes appended to the doc text (async and iterator behavior).
44    pub(crate) notes: Vec<String>,
45    /// The `@param` tags, in order.
46    pub(crate) params: Vec<ParamDoc>,
47    /// The `@return` type, if the callable returns a value.
48    pub(crate) ret: Option<String>,
49    /// The `@raise` tags: `(class, when)`.
50    pub(crate) raises: Vec<(String, String)>,
51}

53/// Emit `doc`'s text (rewritten through [`spell`]) followed by `extra`'s
54/// notes and tags, then `@deprecated` with the rewritten message.
55pub(crate) fn emit(w: &mut CodeWriter, names: &ApiNames, doc: &Doc<'_>, extra: &CallableDoc) {
56    let spelling = |ident: &str| spell(names, ident);
57    let text = doc.text(spelling);
58    let has_text = text.is_some();
59    w.doc(&text, DocCommentStyle::Hash);
60    for (i, note) in extra.notes.iter().enumerate() {
61        if i == 0 && has_text {
62            w.line("#");
63        }
64        let width = 78usize.saturating_sub(w.indent_str().len() + 2).max(40);
65        w.doc(&Some(wrap(note, width)), DocCommentStyle::Hash);
66    }
67    let has_tags = !extra.params.is_empty()
68        || extra.ret.is_some()
69        || !extra.raises.is_empty()
70        || doc.deprecation(spelling).is_some();
71    if has_tags && (has_text || !extra.notes.is_empty()) {
72        w.line("#");
73    }
74    for p in &extra.params {
75        let mut lines = p
76            .doc
77            .as_deref()
78            .map(|d| crate::codegen::docs::rewrite(d.trim(), spelling))
79            .unwrap_or_default()
80            .lines()
81            .map(str::to_string)
82            .collect::<Vec<_>>()
83            .into_iter();
84        match lines.next() {
85            Some(first) if !first.is_empty() => {
86                w.line(format!("# @param {} [{}] {first}", p.name, p.ty));
87            }
88            _ => {
89                w.line(format!("# @param {} [{}]", p.name, p.ty));
90            }
91        }
92        for line in lines {
93            if line.is_empty() {
94                w.line("#");
95            } else {
96                w.line(format!("#   {line}"));
97            }
98        }
99    }
100    if let Some(ret) = &extra.ret {
101        w.line(format!("# @return [{ret}]"));
102    }
103    for (class, when) in &extra.raises {
104        w.line(format!("# @raise [{class}] {when}"));
105    }
106    if let Some(msg) = doc.deprecation(spelling) {
107        w.line(format!("# @deprecated {msg}"));
108    }
109}

111/// Emit a declaration's doc text and `@deprecated` tag (records, enums,
112/// interfaces, callback interfaces).
113pub(crate) fn emit_plain(w: &mut CodeWriter, names: &ApiNames, doc: &Doc<'_>) {
114    emit(w, names, doc, &CallableDoc::default());
