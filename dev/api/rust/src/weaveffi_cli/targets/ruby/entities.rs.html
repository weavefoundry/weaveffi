22/// The longest line kept on one line before wrapping a member list.
23const MAX_LINE: usize = 100;

25/// Render one error domain: a class under the library's `Error` with one
26/// nested class per code, each carrying its `CODE`, its documented
27/// `MESSAGE`, and readers for its payload fields (which the runtime's
28/// `Error#initialize` takes as required keywords). A code these bindings
29/// don't know (the library is newer) raises the domain class itself, with
30/// the code and message.
31pub(crate) fn render_error(w: &mut CodeWriter, ctx: &RbCtx, table: &ErrorTable<'_>) {
32    let domain = rb_const(&table.type_name);
33    w.blank();
34    let doc = Doc::new(&table.domain.doc, &None).text(|i| docs::spell(&ctx.names, i));
35    if doc.is_some() {
36        w.doc(&doc, DocCommentStyle::Hash);
37        w.line("#");
38    }
39    let about = format!(
40        "The `{}` error domain. A code these bindings don't know (from a newer \
41         library) raises this class itself, with its code and message.",
42        table.domain.name
43    );
44    w.doc(&Some(wrap(&about, 76)), DocCommentStyle::Hash);
45    w.block(format!("class {domain} < Error"), "end", |w| {
46        for (idx, row) in table.codes.iter().enumerate() {
47            let c = row.code;
48            if idx > 0 {
49                w.blank();
50            }
51            let doc = c.doc.clone().unwrap_or_else(|| c.message.clone());
52            w.doc(&Some(doc), DocCommentStyle::Hash);
53            w.block(
54                format!("class {} < {domain}", pascal(&c.name)),
55                "end",
56                |w| {
57                    w.line(format!("CODE = {}", c.value));
58                    w.line(format!("MESSAGE = '{}'", rb_str_literal(&c.message)));
59                    for f in &c.fields {
60                        w.blank();
61                        w.doc(&f.doc, DocCommentStyle::Hash);
62                        w.line(format!("# @return [{}]", rb_doc_type(&f.ty)));
63                        w.line(format!("attr_reader :{}", rb_error_field_name(&f.name)));
64                    }
65                },
66            );
67        }
68    });
69}

71/// Render one C-style enum as a module of integer constants, one
72/// `SHOUTY_SNAKE` constant per variant.
73pub(crate) fn render_enum(w: &mut CodeWriter, ctx: &RbCtx, e: &EnumBinding) {
74    w.blank();
75    docs::emit_plain(w, &ctx.names, &Doc::new(&e.doc, &e.deprecated));
76    w.block(format!("module {}", rb_const(&e.name)), "end", |w| {
77        for v in &e.variants {
78            w.doc(&v.doc, DocCommentStyle::Hash);
79            w.line(format!("{} = {}", v.name.to_shouty_snake_case(), v.value));
80        }
81    });
82}

84/// The YARD `@!attribute` tags documenting a `Data` class's members.
85fn emit_attributes(w: &mut CodeWriter, fields: &[FieldBinding]) {
86    for f in fields {
87        w.line(format!("# @!attribute [r] {}", rb_field_name(&f.name)));
88        let doc = f.doc.as_deref().map(str::trim).filter(|d| !d.is_empty());
89        match doc {
90            Some(doc) => {
91                let mut lines = doc.lines();
92                w.line(format!(
93                    "#   @return [{}] {}",
94                    rb_doc_type(&f.ty),
95                    lines.next().unwrap_or_default()
96                ));
97                for line in lines {
98                    w.line(format!("#     {line}").trim_end());
99                }
100            }
101            None => {
102                w.line(format!("#   @return [{}]", rb_doc_type(&f.ty)));
103            }
104        }
105    }
106}

108/// `Data.define(:a, :b)`, wrapped one member per line when long. `lead`
109/// precedes it on the first line (`Item = `, `class Circle < `).
110fn emit_data_define(w: &mut CodeWriter, lead: &str, fields: &[FieldBinding]) {
111    let members: Vec<String> = fields
112        .iter()
113        .map(|f| format!(":{}", rb_field_name(&f.name)))
114        .collect();
115    let args = if members.is_empty() {
116        String::new()
117    } else {
118        format!("({})", members.join(", "))
119    };
120    let one_line = format!("{lead}Data.define{args}");
121    if w.indent_str().len() + one_line.len() <= MAX_LINE {
122        w.line(one_line);
123        return;
124    }
125    w.line(format!("{lead}Data.define("));
126    w.scope(|w| {
127        for m in &members {
128            w.line(format!("{m},"));
129        }
130    });
131    w.line(")");
132}

134/// Render one record as a `Data` class: immutable, constructed with
135/// keywords (or positionally), with value equality and hashing, `with`,
136/// `to_h`, and pattern matching.
137pub(crate) fn render_record(w: &mut CodeWriter, ctx: &RbCtx, r: &RecordBinding) {
138    w.blank();
139    let doc = Doc::new(&r.doc, &r.deprecated);
140    docs::emit_plain(w, &ctx.names, &doc);
141    if !r.fields.is_empty() && (r.doc.is_some() || r.deprecated.is_some()) {
142        w.line("#");
143    }
144    emit_attributes(w, &r.fields);
145    emit_data_define(w, &format!("{} = ", rb_const(&r.name)), &r.fields);
146}

148/// Render one rich enum as a module of `Data` variant classes. Each variant
149/// includes the module (so `shape.is_a?(Shape)` holds) and carries its wire
150/// `TAG`, which `#tag` returns.
151pub(crate) fn render_rich_enum(w: &mut CodeWriter, ctx: &RbCtx, e: &EnumBinding) {
152    let name = rb_const(&e.name);
153    w.blank();
154    docs::emit_plain(w, &ctx.names, &Doc::new(&e.doc, &e.deprecated));
155    w.block(format!("module {name}"), "end", |w| {
156        w.line("# The active variant's wire tag.");
157        w.line("# @return [Integer]");
158        w.block("def tag", "end", |w| {
159            w.line("self.class::TAG");
160        });
161        for v in &e.variants {
162            w.blank();
163            w.doc(&v.doc, DocCommentStyle::Hash);
164            if !v.fields.is_empty() && v.doc.is_some() {
165                w.line("#");
166            }
167            emit_attributes(w, &v.fields);
168            emit_data_define(w, &format!("class {} < ", rb_const(&v.name)), &v.fields);
169            w.scope(|w| {
170                w.line(format!("include {name}"));
171                w.blank();
172                w.line(format!("TAG = {}", v.value));
173            });
174            w.line("end");
175        }
176    });
177}

179/// Render one interface as a wrapper class on the runtime's `Handle`
180/// (which supplies `close`, `closed?`, `==`/`hash` by native object,
181/// `dup`/`clone`, `inspect`, and the GC backstop). A constructor named
182/// `new` becomes `initialize`; every other constructor is a class-method
183/// factory, and an interface without a `new` hides it; methods pin the
184/// wrapper's pointer for the call; statics are class methods.
185pub(crate) fn render_interface(w: &mut CodeWriter, ctx: &RbCtx, i: &InterfaceBinding) {
186    let class = rb_const(&i.name);
187    w.blank();
188    docs::emit_plain(w, &ctx.names, &Doc::new(&i.doc, &i.deprecated));
189    w.block(format!("class {class} < Bridge::Handle"), "end", |w| {
190        let mut first = true;
191        if !i.constructors.iter().any(|c| c.name == "new") {
192            w.line("private_class_method :new");
193            first = false;
194        }
195        let mut gap = |w: &mut CodeWriter| {
196            if !std::mem::take(&mut first) {
197                w.blank();
198            }
199        };
200        for c in &i.constructors {
201            gap(w);
202            let kind = if c.name == "new" {
203                ScopeKind::Init
204            } else {
205                ScopeKind::Factory
206            };
207            let scope = RbScope {
208                kind,
209                class: Some(&class),
210            };
211            render_callable(w, ctx, c, &scope);
212        }
213        for f in i.methods.iter().chain(&i.statics) {
214            gap(w);
215            let kind = if f.has_self() {
216                ScopeKind::Method
217            } else {
218                ScopeKind::Static
219            };
220            let scope = RbScope {
221                kind,
222                class: Some(&class),
223            };
224            render_callable(w, ctx, f, &scope);
225        }
226    });
227}

229/// `Bridge.record(Class, field: type, ...)`, wrapped one field per line
230/// when long.
231fn emit_layout(w: &mut CodeWriter, class: &str, fields: &[(String, String)]) {
232    let parts: Vec<String> = std::iter::once(class.to_string())
233        .chain(fields.iter().map(|(n, t)| format!("{n}: {t}")))
234        .collect();
235    let one_line = format!("Bridge.record({})", parts.join(", "));
236    if w.indent_str().len() + one_line.len() <= MAX_LINE {
237        w.line(one_line);
238        return;
239    }
240    w.line("Bridge.record(");
241    w.scope(|w| {
242        for p in &parts {
243            w.line(format!("{p},"));
244        }
245    });
246    w.line(")");
247}

249/// The private registrations closing the bindings: each interface's
250/// clone and destroy functions, then the wire layout of every record,
251/// rich-enum variant (and the variants of each rich enum), and error code
252/// with a payload.
253pub(crate) fn render_layouts(w: &mut CodeWriter, model: &Model, tables: &[ErrorTable<'_>]) {
254    w.blank();
255    w.line("# How the library clones and releases each interface's objects, and how");
256    w.line("# each record, variant, and error payload is laid out in a value buffer.");
257    for i in &model.interfaces {
258        w.line(format!(
259            "Bridge.interface({}, :{}, :{})",
260            rb_const(&i.name),
261            i.clone_symbol,
262            i.destroy_symbol
263        ));
264    }
265    let record_fields = |fields: &[FieldBinding]| -> Vec<(String, String)> {
266        fields
267            .iter()
268            .map(|f| (rb_field_name(&f.name), rb_wire(&f.ty)))
269            .collect()
270    };
271    for r in &model.records {
272        emit_layout(w, &rb_const(&r.name), &record_fields(&r.fields));
273    }
274    for e in model.enums.iter().filter(|e| e.is_rich()) {
275        let name = rb_const(&e.name);
276        let variants: Vec<String> = e
277            .variants
278            .iter()
279            .map(|v| format!("{name}::{}", rb_const(&v.name)))
280            .collect();
281        for (v, class) in e.variants.iter().zip(&variants) {
282            emit_layout(w, class, &record_fields(&v.fields));
283        }
284        let one_line = format!("Bridge.union({name}, {})", variants.join(", "));
285        if w.indent_str().len() + one_line.len() <= MAX_LINE {
286            w.line(one_line);
287        } else {
288            w.line("Bridge.union(");
289            w.scope(|w| {
290                w.line(format!("{name},"));
291                for v in &variants {
292                    w.line(format!("{v},"));
293                }
294            });
295            w.line(")");
296        }
297    }
298    for table in tables {
299        let domain = rb_const(&table.type_name);
300        for row in table.codes.iter().filter(|r| !r.code.fields.is_empty()) {
301            let fields: Vec<(String, String)> = row
302                .code
303                .fields
304                .iter()
305                .map(|f| (rb_error_field_name(&f.name), rb_wire(&f.ty)))
306                .collect();
307            emit_layout(w, &format!("{domain}::{}", pascal(&row.code.name)), &fields);
308        }
309    }
