1//! Packaging scaffolds for a distributable C artifact: the `CMakeLists.txt`
2//! that exposes the bundled per-platform library as an imported target, and
3//! the README describing the layout.
4
5use crate::package::PackageContext;
6use crate::utils::{render_prelude, render_trailer, CommentStyle};
7
8/// Render a `CMakeLists.txt` that exposes the bundled per-platform library as
9/// an `IMPORTED` target (`<lib>::<lib>`) with the header's include directory,
10/// selecting the right library for the host platform and architecture.
11pub(crate) fn render_packaged_cmake(lib: &str) -> String {
12    let prelude = render_prelude(CommentStyle::Hash);
13    let trailer = render_trailer(CommentStyle::Hash, "CMakeLists.txt");
14    let body = r#"cmake_minimum_required(VERSION 3.15)
15project(@LIB@ C)
16
17# Select the prebuilt library bundled for the host platform and architecture.
18if(APPLE)
19  if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
20    set(_@LIB@_platform "darwin-arm64")
21  else()
22    set(_@LIB@_platform "darwin-x64")
23  endif()
24  set(_@LIB@_libfile "lib@LIB@.dylib")
25elseif(WIN32)
26  set(_@LIB@_platform "windows-x64")
27  set(_@LIB@_libfile "@LIB@.dll")
28else()
29  if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
30    set(_@LIB@_platform "linux-arm64")
31  else()
32    set(_@LIB@_platform "linux-x64")
33  endif()
34  set(_@LIB@_libfile "lib@LIB@.so")
35endif()
36
37add_library(@LIB@ SHARED IMPORTED GLOBAL)
38set_target_properties(@LIB@ PROPERTIES
39  IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/lib/${_@LIB@_platform}/${_@LIB@_libfile}"
40  INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_LIST_DIR}/include")
41if(WIN32)
42  set_target_properties(@LIB@ PROPERTIES
43    IMPORTED_IMPLIB "${CMAKE_CURRENT_LIST_DIR}/lib/${_@LIB@_platform}/@LIB@.dll.lib")
44endif()
45add_library(@LIB@::@LIB@ ALIAS @LIB@)
46"#
47    .replace("@LIB@", lib);
48    format!("{prelude}{body}\n{trailer}")
49}

51/// README for a packaged C artifact bundling the header and per-platform libs.
52pub(crate) fn render_packaged_readme(lib: &str, header_name: &str, ctx: &PackageContext) -> String {
53    let prelude = render_prelude(CommentStyle::Xml);
54    let trailer = render_trailer(CommentStyle::Xml, "README.md");
55    let platforms: Vec<String> = ctx
56        .binaries
57        .platforms()
58        .map(|p| format!("- `lib/{}/`", p.id()))
59        .collect();
60    let platform_list = platforms.join("\n");
61    format!(
62        r#"{prelude}# {lib} (C)
63
64The stable C ABI header (`include/{header_name}`) plus a prebuilt shared library
65for each supported platform under `lib/<platform>/`.
66
67## Use with CMake
68
69```cmake
70add_subdirectory(path/to/{lib})
71target_link_libraries(your_app PRIVATE {lib}::{lib})
72```
73
74`CMakeLists.txt` selects the right library for the host platform and exposes the
75include directory automatically.
76
77## Bundled platforms
78
79{platform_list}
80
81{trailer}"#,
82    )
83}
