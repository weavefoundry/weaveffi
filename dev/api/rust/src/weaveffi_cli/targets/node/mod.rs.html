1//! Node.js binding generator.
2//!
3//! Emits an npm package: the ES module (`index.js`, `index.d.ts`,
4//! `runtime.js`, and the `./debug` export) shared with the WebAssembly
5//! target through the [shared JavaScript layer](crate::targets::js), and
6//! its transport, `native.js`, which loads the producer's shared library
7//! with [koffi](https://koffi.dev) (a pinned dependency of the package) and
8//! binds each C symbol from its signature. There is no generated C and
9//! nothing to compile at install time: `weaveffi package` ships the
10//! library in one npm package per platform, which the main package lists
11//! as optional dependencies.
12

39/// Per-target configuration for [`NodeGenerator`].
40#[derive(Debug, Clone, Serialize, Deserialize)]
41#[serde(default, deny_unknown_fields)]
42pub struct NodeConfig {
43    /// npm package name (default: the package identity's name).
44    pub name: Option<String>,
45    /// The `engines.node` range of `package.json` (default `>=18`).
46    pub node_engine: String,
47}
48
49impl Default for NodeConfig {
50    fn default() -> Self {
51        Self {
52            name: None,
53            node_engine: ">=18".into(),
54        }
55    }
56}
57
58impl NodeConfig {
59    fn names<'a>(&'a self, identity: &'a Identity) -> NodeNames<'a> {
60        NodeNames {
61            identity,
62            package: self.name.clone().unwrap_or_else(|| identity.name.clone()),
63            node_engine: &self.node_engine,
64        }
65    }
66}

79/// The generated `index.js`: the imports, the C signatures and contract,
80/// loading the library with its load-time checks, and the shared API.
81fn render_index(model: &Model, names: &NodeNames<'_>, errors: &ErrorNames) -> String {
82    let id = names.identity;
83    let string = js::names::js_string;
84    let mut w = CodeWriter::two_space();
85    w.raw(render_prelude(CommentStyle::DoubleSlash));
86    js::render_imports(&mut w, errors);
87    w.line("import { $loadNative } from './native.js';");
88    w.blank();
89    js::render_tables(&mut w, model);
90    w.blank();
91    w.line("// Load the native library, check that it matches these bindings, and bind");
92    w.line("// the functions they call.");
93    w.block("const $m = $loadNative({", "});", |w| {
94        w.line(format!("name: {},", string(&id.name)));
95        w.line(format!("library: {},", string(&id.library)));
96        w.line(format!("prefix: {},", string(model.prefix())));
97        w.line(format!("env: {},", string(&id.library_env_var())));
98        w.line(format!("pkg: {},", string(&names.package)));
99        w.line("base: import.meta.url,");
100    });
101    w.line(js::verify_call(model));
102    w.line("const $f = $m.bind($sigs);");
103    w.blank();
104    js::render_api(&mut w, model, errors);
105    w.blank();
106    w.raw(render_trailer(CommentStyle::DoubleSlash, "index.js"));
107    w.finish()
108}

110/// Render `native.js`.
111fn render_native() -> String {
112    let mut out = render_prelude(CommentStyle::DoubleSlash);
113    out.push_str(NATIVE_JS);
114    out.push('\n');
115    out.push_str(&render_trailer(CommentStyle::DoubleSlash, "native.js"));
116    out
117}

119/// Every file of the generated package, as `(name, contents)` pairs.
120/// `platforms` lists the per-platform packages of the packaged layout
121/// (empty for plain generation).
122fn render_files(
123    model: &Model,
124    names: &NodeNames<'_>,
125    platforms: &[String],
126) -> Vec<(String, String)> {
127    let errors = ErrorNames::new(model);
128    vec![
129        ("index.js".into(), render_index(model, names, &errors)),
130        ("index.d.ts".into(), js::render_dts(model, &errors, "")),
131        ("runtime.js".into(), js::render_runtime(&errors)),
132        ("native.js".into(), render_native()),
133        ("debug.js".into(), js::render_debug_js()),
134        ("debug.d.ts".into(), js::render_debug_dts()),
135        ("package.json".into(), render_package_json(names, platforms)),
136        ("README.md".into(), render_readme(names, platforms)),
137    ]
138}

140impl Target for NodeGenerator {
141    fn name(&self) -> &'static str {
142        "node"
143    }
144
145    /// The package root, next to `index.js`, where the loader looks before
146    /// the system's library search.
147    fn dev_bundle_dir(&self, _model: &Model) -> Option<Utf8PathBuf> {
148        Some(Utf8PathBuf::new())
149    }
150
151    fn fixed_files(&self) -> &'static [&'static str] {
152        &[
153            "README.md",
154            "debug.d.ts",
155            "debug.js",
156            "native.js",
157            "package.json",
158            "runtime.js",
159        ]
160    }
161
162    fn render(&self, model: &Model) -> Vec<OutputFile> {
163        let names = self.config.names(&model.identity);
164        render_files(model, &names, &[])
165            .into_iter()
166            .map(|(name, contents)| OutputFile::new(name, contents))
167            .collect()
168    }
169
170    /// The packaged layout follows the esbuild convention: one npm tarball
171    /// per platform, `{package}-{os}-{cpu}`, gated by npm `os`/`cpu` and
172    /// carrying the producer library, plus the main package, which lists
173    /// them in `optionalDependencies` and loads the installed one.
174    fn package(&self, model: &Model, ctx: &PackageContext<'_>) -> Result<Vec<Artifact>> {
175        let names = self.config.names(&model.identity);
176        let id = names.identity;
177        let mut artifacts = Vec::new();
178        let mut platform_names = Vec::new();
179        for nb in &ctx.binaries.binaries {
180            let Some(tokens) = node_platform_tokens(nb.platform) else {
181                continue;
182            };
183            let name = format!("{}-{}-{}", names.package, tokens.0, tokens.1);
184            let library = nb.platform.lib_filename(&id.library);
185            let files = vec![
186                PackagedFile::text(
187                    "package.json",
188                    render_platform_package_json(&name, id, tokens, std::slice::from_ref(&library)),
189                ),
190                PackagedFile::copy(library, nb.library.clone()),
191            ];
192            artifacts.push(Artifact::tar_gz(
193                format!("node/{}", npm_tarball_name(&name, &id.version)),
194                "package",
195                files,
196            ));
197            platform_names.push(name);
198        }
199        if platform_names.is_empty() {
200            return Ok(Vec::new());
201        }
202        let files = render_files(model, &names, &platform_names)
203            .into_iter()
204            .map(|(name, contents)| PackagedFile::text(name, contents))
205            .collect();
206        artifacts.push(Artifact::tar_gz(
207            format!("node/{}", npm_tarball_name(&names.package, &id.version)),
208            "package",
209            files,
210        ));
211        Ok(artifacts)
212    }
213}
