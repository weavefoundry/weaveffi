11/// Which way a value travels, which decides how its containers are
12/// annotated: what the bindings accept is abstract (any `Sequence` or
13/// `Mapping`), and what they hand out is concrete (`list`, `dict`).
14#[derive(Clone, Copy, PartialEq, Eq)]
15pub(crate) enum Dir {
16    /// Into the native library: function parameters, callback returns, and
17    /// value-buffer writers.
18    In,
19    /// Out of the native library: function returns, callback parameters,
20    /// record and error fields, and value-buffer readers.
21    Out,
22}
23
24/// The Python typing hint for `ty` travelling in direction `dir`. The module
25/// starts with `from __future__ import annotations`, so user types are
26/// written bare even when declared later in the file.
27pub(crate) fn py_type_hint(ty: &Ty, dir: Dir) -> String {
28    match ty {
29        Ty::Prim(Prim::F32 | Prim::F64) => "float".into(),
30        Ty::Prim(Prim::Bool) => "bool".into(),
31        Ty::Prim(Prim::String) => "str".into(),
32        Ty::Prim(Prim::Bytes) => "bytes".into(),
33        Ty::Prim(_) => "int".into(),
34        Ty::Enum(name) | Ty::Record(name) | Ty::RichEnum(name) | Ty::Interface(name) => {
35            name.clone()
36        }
37        Ty::Optional(inner) => format!("{} | None", py_type_hint(inner, dir)),
38        Ty::List(inner) => match dir {
39            Dir::In => format!("Sequence[{}]", py_type_hint(inner, dir)),
40            Dir::Out => format!("list[{}]", py_type_hint(inner, dir)),
41        },
42        Ty::Map(k, v) => match dir {
43            Dir::In => format!(
44                "Mapping[{}, {}]",
45                py_type_hint(k, dir),
46                py_type_hint(v, dir)
47            ),
48            Dir::Out => format!("dict[{}, {}]", py_type_hint(k, dir), py_type_hint(v, dir)),
49        },
50    }
51}

53/// The hint for a parameter: its value type (accepted abstractly), or a
54/// callback interface's class (`| None` when nullable).
55pub(crate) fn py_param_hint(ty: &ParamTy) -> String {
56    match ty {
57        ParamTy::Value(ty) => py_type_hint(ty, Dir::In),
58        ParamTy::Callback { name, nullable } => {
59            if *nullable {
60                format!("{name} | None")
61            } else {
62                name.clone()
63            }
64        }
65    }
66}

68/// The hint for a callable's return: `None` for a void callable, the
69/// iterator's `NativeIterator[T]`, or the value type.
70pub(crate) fn py_return_hint(ret: Option<&RetTy>) -> String {
71    match ret {
72        None => "None".into(),
73        Some(RetTy::Value(ty)) => py_type_hint(ty, Dir::Out),
74        Some(RetTy::Iterator(elem)) => format!("NativeIterator[{}]", py_type_hint(elem, Dir::Out)),
75    }
76}

78/// The builtin types an annotation names.
79const BUILTIN_HINTS: &[&str] = &[
80    "bool", "bytes", "dict", "float", "int", "list", "str", "type",
81];
82
83/// The annotation for `ty` inside a class declaring `fields`. A field named
84/// like a builtin type (`int: int`) shadows it for every later annotation in
85/// the class body, so such classes spell the builtins `builtins.int`.
86pub(crate) fn py_field_hint(fields: &[FieldBinding], ty: &Ty) -> String {
87    let hint = py_type_hint(ty, Dir::Out);
88    if !fields
89        .iter()
90        .any(|f| BUILTIN_HINTS.contains(&f.name.as_str()))
91    {
92        return hint;
93    }
94    let mut out = String::with_capacity(hint.len());
95    let mut word = String::new();
96    let flush = |word: &mut String, out: &mut String| {
97        if BUILTIN_HINTS.contains(&word.as_str()) {
98            out.push_str("builtins.");
99        }
100        out.push_str(word);
101        word.clear();
102    };
103    for c in hint.chars() {
104        if c.is_ascii_alphanumeric() || c == '_' {
105            word.push(c);
106            continue;
107        }
108        flush(&mut word, &mut out);
109        out.push(c);
110    }
111    flush(&mut word, &mut out);
112    out
113}

115/// The IDL spelling of a slice element or range-checked integer kind, which
116/// names the runtime's per-kind helpers (`_i32`, `_take_array(.., "f64")`).
117pub(crate) fn prim_kind(p: Prim) -> &'static str {
118    p.snake()
119}

121/// The runtime checker an integer of primitive `p` goes through before
122/// `ctypes` sees it (`ctypes` truncates an out-of-range integer silently):
123/// `_i32` for `i32` (and every C-style enum), `_u64` for `u64`, and so on.
124/// `None` for floats and `bool`, which need no range check.
125pub(crate) fn int_checker(p: Prim) -> Option<&'static str> {
126    Some(match p {
127        Prim::I8 => "_i8",
128        Prim::I16 => "_i16",
129        Prim::I32 => "_i32",
130        Prim::I64 => "_i64",
131        Prim::U8 => "_u8",
132        Prim::U16 => "_u16",
133        Prim::U32 => "_u32",
134        Prim::U64 => "_u64",
135        Prim::Bool | Prim::F32 | Prim::F64 | Prim::String | Prim::Bytes => return None,
136    })
137}

139/// The runtime's `ctypes.Structure` mirroring the C `{prefix}_opt_{prim}`
140/// for the scalar `p`: `_OptI32`, `_OptF64`, `_OptBool`.
141pub(crate) fn opt_struct(p: Prim) -> String {
142    format!("_Opt{}", p.pascal())
143}

145/// The `ctypes.Structure` mirroring a callback interface's C argument

147pub(crate) fn callback_arg_struct(name: &str) -> String {

151/// The `ctypes` spelling of a scalar primitive (`ctypes.c_int32`).
152fn scalar_ctype(p: Prim) -> &'static str {
153    match p {
154        Prim::Bool => "ctypes.c_bool",
155        Prim::I8 => "ctypes.c_int8",
156        Prim::I16 => "ctypes.c_int16",
157        Prim::I32 => "ctypes.c_int32",
158        Prim::I64 => "ctypes.c_int64",
159        Prim::U8 => "ctypes.c_uint8",
160        Prim::U16 => "ctypes.c_uint16",
161        Prim::U32 => "ctypes.c_uint32",
162        Prim::U64 => "ctypes.c_uint64",
163        Prim::F32 => "ctypes.c_float",
164        Prim::F64 => "ctypes.c_double",
165        Prim::String | Prim::Bytes => unreachable!("{p} is not a scalar"),
166    }
167}

169/// Who calls a C signature, which decides how its error pointer is
170/// spelled.
171#[derive(Clone, Copy, PartialEq, Eq)]
172pub(crate) enum Caller {
173    /// Python calls the producer (an `argtypes` entry): the error slot is a
174    /// `POINTER(_ErrorStruct)` the wrapper passes `byref`.
175    Python,
176    /// The producer calls Python (a callback method or an async
177    /// completion): every pointer, the error slot included, arrives as a
178    /// plain `c_void_p` address.
179    Producer,
180}
181
182/// Maps a shared ABI [`CType`] onto its `ctypes` spelling. Every value is
183/// one C value: a scalar is its `ctypes` type (an enum is `c_int32`), an
184/// optional scalar the runtime's `_Opt{Prim}` structure, a run the
185/// runtime's `_Run` structure (both by value), a callback interface its
186/// `_{Name}Arg` structure, and every object, iterator, cancel-token, or
187/// context pointer a `c_void_p`. An out slot (an iterator's `T* out_item`)
188/// is `POINTER(T)`.
189pub(crate) fn py_ctype(ty: &CType, caller: Caller) -> String {
190    match ty {
191        CType::Scalar(p) => scalar_ctype(*p).into(),
192        CType::Size => "ctypes.c_size_t".into(),
193        CType::Void => "None".into(),
194        CType::Enum(_) => "ctypes.c_int32".into(),
195        CType::Opt(p) => opt_struct(*p),
196        CType::Run => "_Run".into(),
197        CType::Callback(name) => callback_arg_struct(name),
198        CType::Error => "_ErrorStruct".into(),
199        CType::Object(_) | CType::Vtable(_) | CType::Named(_) | CType::CancelToken => {
200            "ctypes.c_void_p".into()
201        }
202        CType::Ptr { pointee, .. } => match pointee.as_ref() {
203            CType::Error if caller == Caller::Python => "ctypes.POINTER(_ErrorStruct)".into(),
204            CType::Scalar(_) | CType::Enum(_) | CType::Opt(_) | CType::Run => {
205                format!("ctypes.POINTER({})", py_ctype(pointee, caller))
206            }
207            CType::Ptr { .. } => "ctypes.POINTER(ctypes.c_void_p)".into(),
208            _ => "ctypes.c_void_p".into(),
209        },
210    }
211}

213/// The `ctypes` type of the local a wrapper allocates for an out slot
214/// (`T* out_item`) and passes by reference: the pointee, with an object
215/// pointer received as a raw `c_void_p` address.
216pub(crate) fn py_out_local(slot: &CType) -> String {
217    match slot {
218        CType::Ptr { pointee, .. } => match pointee.as_ref() {
219            CType::Ptr { .. } => "ctypes.c_void_p".into(),
220            other => py_ctype(other, Caller::Python),
221        },
222        other => unreachable!("an out slot is a pointer, not {other:?}"),
223    }
224}

226/// The Python type `ctypes` hands a callback for one C slot (a callback
227/// method's or an async completion's parameter), as an annotation: integers
228/// and enums are `int`, floats `float`, `bool` `bool`, an optional scalar or
229/// a run its structure, and every pointer an address, `int | None` (`None`
230/// for null).
231pub(crate) fn py_slot_hint(ty: &CType) -> String {
232    match ty {
233        CType::Scalar(Prim::F32 | Prim::F64) => "float".into(),
234        CType::Scalar(Prim::Bool) => "bool".into(),
235        CType::Void => "None".into(),
236        CType::Opt(p) => opt_struct(*p),
237        CType::Run => "_Run".into(),
238        CType::Callback(name) => callback_arg_struct(name),
239        CType::Error => "_ErrorStruct".into(),
240        CType::Ptr { .. }
241        | CType::Object(_)
242        | CType::Vtable(_)
243        | CType::Named(_)
244        | CType::CancelToken => "int | None".into(),
245        CType::Scalar(_) | CType::Size | CType::Enum(_) => "int".into(),
246    }
247}

249/// The value a callback method returns after it failed, which the producer
250/// discards: `0`, `0.0`, `False`, an absent optional, the empty run, or a
251/// null pointer. `None` for a `void` method.
252pub(crate) fn zero_value(ty: &CType) -> Option<String> {
253    Some(match ty {
254        CType::Void => return None,
255        CType::Scalar(Prim::Bool) => "False".into(),
256        CType::Scalar(Prim::F32 | Prim::F64) => "0.0".into(),
257        CType::Scalar(_) | CType::Size | CType::Enum(_) => "0".into(),
258        CType::Opt(p) => format!("{}()", opt_struct(*p)),
259        CType::Run => "_Run()".into(),
260        _ => "None".into(),
261    })
262}

283/// The Python spelling of an IDL value identifier (parameter name):
284/// snake_case via heck, then escaped with a trailing `_` when it's a
285/// keyword (`class`) or a name the generated body relies on (`self`,
286/// `len`). IDL names are usually already snake, so the case conversion is a
287/// safety net for camelCase inputs.
288pub(crate) fn py_name(name: &str) -> String {
289    py_local(&name.to_snake_case())
290}
291
292/// Escape `name` for use as a local (a parameter or a trampoline slot)
293/// without changing its case.
294pub(crate) fn py_local(name: &str) -> String {
295    let escaped = lang::escape_ident(name, lang::PYTHON_KEYWORDS);
296    if PY_BODY_NAMES.binary_search(&escaped.as_str()).is_ok() {
297        format!("{escaped}_")
298    } else {
299        escaped
300    }
301}

309/// Names an error code's payload field must not take: the attributes every
310/// exception already has (or the generated error classes define), and the
311/// constructor's receiver.
312const EXCEPTION_ATTRS: &[&str] = &[
313    "CODE",
314    "add_note",
315    "args",
316    "code",
317    "message",
318    "self",
319    "with_traceback",
320];

322/// The Python spelling of an error code's payload field: [`py_field`],
323/// with a trailing `_` when it would shadow an exception attribute or the
324/// receiver.
325pub(crate) fn py_error_field(name: &str) -> String {
326    let field = py_field(name);
327    if EXCEPTION_ATTRS.contains(&field.as_str()) {
328        format!("{field}_")
329    } else {
330        field
331    }
332}

334/// The Python spelling of an enum variant, used both as the `IntEnum` member
335/// and as the suffix of a rich enum's per-variant dataclass. Variants are
336/// PascalCase, so only the capitalized keywords (`None`, `True`, `False`)
337/// can collide; `None = 0` inside a class body is a `SyntaxError`.
338pub(crate) fn py_variant(name: &str) -> String {
339    lang::escape_ident(name, lang::PYTHON_KEYWORDS)
340}

348/// Names every object wrapper class already defines: `close` (from the
349/// runtime's `_Handle`). The class's private helpers start with `_`, which
350/// a snake-cased IDL name never does.
351const OBJECT_MEMBERS: &[&str] = &["close"];

353/// The Python spelling of an interface member (a factory constructor,
354/// method, or static): [`py_member_name`], with a trailing `_` when it

356pub(crate) fn py_object_member(name: &str) -> String {
357    lang::escape_member(&py_member_name(name), OBJECT_MEMBERS)

360/// The module-level name a C symbol's bound `ctypes` function is stored
361/// under: `_c_` plus the symbol without its `{prefix}_`.
362pub(crate) fn py_binding_name(symbol: &str, prefix: &str) -> String {
363    let core = symbol
364        .strip_prefix(prefix)
365        .and_then(|s| s.strip_prefix('_'))
366        .unwrap_or(symbol);
367    format!("_c_{core}")
368}

370/// Escape a string for embedding in a double-quoted Python literal.
371pub(crate) fn py_str_literal(s: &str) -> String {
372    let mut out = String::with_capacity(s.len());
373    for c in s.chars() {
374        match c {
375            '\\' => out.push_str("\\\\"),
376            '"' => out.push_str("\\\""),
377            '\n' => out.push_str("\\n"),
378            '\r' => out.push_str("\\r"),
379            '\t' => out.push_str("\\t"),
380            c => out.push(c),
381        }
382    }
383    out
