1//! The fixed Python runtime every generated module starts with.
2//!
3//! The runtime is ordinary Python kept in `runtime/*.py` and spliced in with
4//! `{{PLACEHOLDER}}` substitution: `core.py` (the root and trap exceptions,
5//! the run and optional-scalar structures, library loading, the load-time
6//! ABI and contract check, the run helpers, the handle base classes, and the
7//! value-buffer codec), `aio.py` (the async completion registry),
8//! `cancel.py` (cancel tokens), `callbacks.py` (the callback-interface
9//! handle table, vtable entries, return hand-off, and failure reporting),
10//! and `ffi.py` (vtable entries returning a structure, as libffi closures).
11//! All but the core are emitted only when the API uses the feature. The
12//! contract rows are written into the writer where `core.py` says
13//! `{{CONTRACT}}`.
14
15use crate::codegen::contract;
16use crate::codegen::CodeWriter;
17use heck::ToSnakeCase;
18use weaveffi_model::model::ABI_VERSION;

20use crate::targets::python::callbacks::returns_struct;
21use crate::targets::python::types::py_str_literal;
22use crate::targets::python::Gen;

24const CORE: &str = include_str!("runtime/core.py");
25const AIO: &str = include_str!("runtime/aio.py");
26const CANCEL: &str = include_str!("runtime/cancel.py");
27const CALLBACKS: &str = include_str!("runtime/callbacks.py");
28const FFI: &str = include_str!("runtime/ffi.py");
29
30/// Write the body of the `_CONTRACT` list: one `(symbol, hash, path)` row per
31/// declaration, sorted by id, each with its canonical signature as a
32/// comment.
33fn render_contract(w: &mut CodeWriter, g: &Gen<'_>) {
34    w.scope(|w| {
35        for row in contract::rows(g.model) {
36            w.line(format!(
37                "(\"{}\", {}, \"{}\"),  # {}",
38                row.symbol,
39                contract::hex(row.hash),
40                py_str_literal(&row.path),
41                row.signature
42            ));
43        }
44    });
45}

47/// Substitute every scalar `{{PLACEHOLDER}}` a runtime template uses.
48fn fill(template: &str, g: &Gen<'_>) -> String {
49    let id = &g.model.identity;
50    let (darwin, linux, windows) = id.library_files();
51    template
52        .replace("{{NAME}}", &id.name)
53        .replace("{{ERROR}}", &g.root_error)
54        .replace("{{TRAP}}", &g.trap_error)
55        .replace(
56            "{{ERROR_FROM}}",
57            &format!("_{}_from", g.root_error.to_snake_case()),
58        )
59        .replace("{{PREFIX}}", g.prefix)
60        .replace("{{LIBRARY_ENV}}", &id.library_env_var())
61        .replace("{{LIB_DARWIN}}", &darwin)
62        .replace("{{LIB_LINUX}}", &linux)
63        .replace("{{LIB_WINDOWS}}", &windows)
64        .replace("{{ABI_VERSION}}", &ABI_VERSION.to_string())
65}

67/// Append the runtime: the core, then the feature parts the API needs.
68pub(crate) fn render_runtime(w: &mut CodeWriter, g: &Gen<'_>) {
69    let model = g.model;
70    let callbacks = !model.callbacks.is_empty();
71    let is_async = model.has_async();
72    let cancellable = model.has_cancellable();
73    let structs = model
74        .callbacks
75        .iter()
76        .any(|cb| cb.methods.iter().any(returns_struct));
77    let mut imports = String::new();
78    if callbacks {
79        imports.push_str("import abc\n");
80    }
81    if is_async {
82        imports.push_str("import asyncio\n");
83    }
84    let core = fill(CORE, g).replace("{{IMPORTS}}", &imports);
85    let (head, tail) = core
86        .split_once("{{CONTRACT}}\n")
87        .expect("core.py has a {{CONTRACT}} line");
88    w.raw(head);
89    render_contract(w, g);
90    w.raw(tail);
91    for (wanted, template) in [
92        (is_async, AIO),
93        (cancellable, CANCEL),
94        (callbacks, CALLBACKS),
95        (structs, FFI),
96    ] {
97        if wanted {
98            w.blank().blank();
99            w.raw(fill(template, g));
100        }
101    }
102}
