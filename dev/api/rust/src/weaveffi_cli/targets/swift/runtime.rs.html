22/// Replace every `{{KEY}}` in `template` with its value.
23///
24/// # Panics
25///
26/// Panics (in debug builds) when a placeholder is left unfilled, which is a
27/// bug in the caller's substitution list.
28fn fill(template: &str, vars: &[(&str, &str)]) -> String {
29    let mut out = template.to_string();
30    for (key, value) in vars {
31        out = out.replace(&format!("{{{{{key}}}}}"), value);
32    }
33    debug_assert!(!out.contains("{{"), "unfilled placeholder in:\n{out}");
34    out
35}

37/// The names the fixed sources are spliced with.
38pub(crate) struct Names<'a> {
39    /// The Swift module (and package and product) name.
40    pub(crate) module: &'a str,
41    /// The C module name, `C{module}`.
42    pub(crate) c_module: &'a str,
43    /// The native library base name.
44    pub(crate) library: &'a str,
45    /// The C header file name.
46    pub(crate) header: &'a str,
47}

49/// The runtime source (without the generated-file prelude): the library's
50/// load check, error helpers, the value-buffer codec, the generic sequence,
51/// and async and callback support.
52pub(crate) fn render_runtime(model: &Model, names: &Names) -> String {
53    fill(
54        RUNTIME,
55        &[
56            ("RUNTIME_ERROR", &runtime_error_name(names.module)),
57            ("LIBRARY_TYPE", &library_type_name(names.module)),
58            ("PREFIX", model.prefix()),
59            ("MODULE", names.module),
60            ("C_MODULE", names.c_module),
61            ("LIBRARY", names.library),
62            ("ABI_VERSION", &ABI_VERSION.to_string()),
63        ],
64    )
65}

67/// Render `wvCheckContracts()`, which the runtime's load check calls after
68/// the ABI revision: every declaration's contract function against the hash
69/// these bindings were generated with. The functions are linked, so a
70/// declaration the library lacks fails to link instead.
71pub(crate) fn render_contract_checks(w: &mut CodeWriter, model: &Model, library_type: &str) {
72    let rows = contract::rows(model);
73    w.line("// MARK: - Contract");
74    w.blank();
75    w.line("/// The first declaration these bindings were generated with that the library");
76    w.line(format!(
77        "/// declares differently, checked once by `{library_type}.check()`."
78    ));
79    w.line(format!(
80        "func wvCheckContracts() -> {library_type}.LoadError? {{"
81    ));
82    w.indent();
83    if rows.is_empty() {
84        w.line("nil");
85    } else {
86        w.line("wvCheckContract([");
87        w.scope(|w| {
88            for row in &rows {
89                w.line(format!(
90                    "({}, {}, \"{}\"),",
91                    row.symbol,
92                    hex(row.hash),
93                    swift_str(&row.path)
94                ));
95            }
96        });
97        w.line("])");
98    }
99    w.dedent();
100    w.line("}");
101    w.blank();
102}

104/// The `Package.swift` body (after the tools-version line and prelude):
105/// `fallback` is the C target used when no `XCFramework` sits next to the
106/// manifest, explained by `source_comment`, and `platforms` lists the
107/// minimum OS versions.
108pub(crate) fn render_package(
109    names: &Names,
110    source_comment: &str,
111    fallback: &str,
112    platforms: &str,
113) -> String {
114    fill(
115        PACKAGE,
116        &[
117            ("SOURCE_COMMENT", source_comment),
118            ("FALLBACK", fallback),
119            ("PLATFORMS", platforms),
120            ("MODULE", names.module),
121            ("C_MODULE", names.c_module),
122        ],
123    )
124}

126/// The system-library module map declaring the bundled header.
127pub(crate) fn render_module_map(names: &Names) -> String {
128    fill(
129        MODULE_MAP,
130        &[
131            ("C_MODULE", names.c_module),
132            ("HEADER", names.header),
133            ("LIBRARY", names.library),
134        ],
135    )
136}

138/// The packaged README: the `XCFramework`'s `slices`, and the `url` and

142        README,

144            ("MODULE", names.module),
