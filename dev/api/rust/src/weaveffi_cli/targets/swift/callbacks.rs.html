1//! Callback-interface rendering: the Swift `protocol` a consumer implements
2//! and the process-wide vtable of `@convention(c)` trampolines the library
3//! calls through.
4//!
5//! A callback argument is a `{p}_{Cb}` struct of a context (which retains the
6//! implementation, released by the vtable's `free`) and a pointer to one
7//! static vtable per interface. The vtable carries the `{size, flags, free}`
8//! header (`size` is the vtable's as these bindings compiled it, so a newer
9//! producer can tell which methods they lack; `flags` is 0: any thread may
10//! call) and one trampoline per method. Arguments arrive as one C value each
11//! (borrowed runs are copied; objects are adopted), results go back as one C
12//! value (a run from `{p}_alloc` and an object as a fresh reference, both
13//! adopted by the library), and failures are reported through `out_err`
14//! instead of unwinding through the C frame, per the method's
15//! [`ErrorStrategy`].
16

27/// The `- Throws:` callout of a protocol requirement.
28fn throws_doc(m: &CallbackMethodBinding, ctx: &SwiftCtx) -> Option<String> {
29    match &m.error {
30        ErrorStrategy::Trap => None,
31        ErrorStrategy::Untyped => Some(
32            "- Throws: Any error, which fails the library's call with the error's description."
33                .to_string(),
34        ),
35        ErrorStrategy::Domain(name) => Some(format!(
36            "- Throws: ``{}`` to fail the library's call with that error and its fields; any \
37             other error fails it with the error's description.",
38            ctx.domain_type(name)
39        )),
40    }
41}

43/// Render the `public protocol` the consumer conforms to: one requirement
44/// per method with idiomatic labels and types, `throws` when the IDL method
45/// declares errors. Implementations are classes (`AnyObject`) the library
46/// may call from any thread (`Sendable`).
47fn render_protocol(w: &mut CodeWriter, cb: &CallbackBinding, ctx: &SwiftCtx) {
48    let name = &cb.name;
49    if ctx.doc(&cb.doc).is_some() {
50        emit_decl_doc(w, ctx, &cb.doc, &None);
51        w.line("///");
52    }
53    w.line("/// Implement this protocol with a class and pass an instance where the API");
54    w.line("/// expects it. The library keeps the instance until it no longer needs it and");
55    w.line("/// may call any method from any thread.");
56    if let Some(attr) = ctx.deprecated_attr(&cb.deprecated) {
57        w.line(attr);
58    }
59    w.line(format!("public protocol {name}: AnyObject, Sendable {{"));
60    w.indent();
61    for (i, m) in cb.methods.iter().enumerate() {
62        if i > 0 {
63            w.blank();
64        }
65        let labels: Vec<_> = m.params.iter().map(|p| swift_ident(&p.name)).collect();
66        let param_docs: Vec<_> = labels
67            .iter()
68            .cloned()
69            .zip(m.params.iter().map(|p| &p.doc))
70            .collect();
71        emit_fn_doc(
72            w,
73            ctx,
74            &m.doc,
75            &param_docs,
76            &throws_doc(m, ctx).into_iter().collect::<Vec<_>>(),
77        );
78        if let Some(attr) = ctx.deprecated_attr(&m.deprecated) {
79            w.line(attr);
80        }
81        let params = labels
82            .iter()
83            .zip(&m.params)
84            .map(|(label, p)| format!("{label}: {}", ctx.swift_type(&p.ty)))
85            .collect::<Vec<_>>()
86            .join(", ");
87        let throws = if m.error.throws() { " throws" } else { "" };
88        let ret = m
89            .ret
90            .as_ref()
91            .map(|t| format!(" -> {}", ctx.swift_type(t)))
92            .unwrap_or_default();
93        w.line(format!(
94            "func {}({params}){throws}{ret}",
95            swift_ident(&m.name)
96        ));
97    }
98    w.dedent();
99    w.line("}");
100    w.blank();
101}

103/// Render one trampoline closure literal (the vtable entry for `m`) as a
104/// labeled argument of the vtable struct initializer; `last` drops the
105/// trailing comma.
106///
107/// The closure's formals are positional (`ctx`, `a0`, `a1`, ..., `outErr`)
108/// rather than the IDL's parameter names, so no parameter can shadow a name
109/// the body uses.
110fn render_trampoline(
111    w: &mut CodeWriter,
112    cb: &CallbackBinding,
113    m: &CallbackMethodBinding,
114    last: bool,
115    ctx: &SwiftCtx,
116) {
117    // No trailing comma: argument lists accept one only from Swift 6.1.
118    let sep = if last { "" } else { "," };
119    let args: Vec<String> = (0..m.params.len()).map(|i| format!("a{i}")).collect();
120    let mut formals = vec!["ctx".to_string()];
121    formals.extend(args.iter().cloned());
122    formals.push("outErr".to_string());
123    let call_args = m
124        .params
125        .iter()
126        .zip(&args)
127        .map(|(p, a)| {
128            format!(
129                "{}: {}",
130                swift_ident(&p.name),
131                receive(&p.ty, a, false, ctx)
132            )
133        })
134        .collect::<Vec<_>>()
135        .join(", ");
136    let call = format!("wvImpl.{}({call_args})", swift_ident(&m.name));
137    let (body, fallback) = match &m.ret {
138        None => (call, "()".to_string()),
139        Some(ty) => (hand_back(ty, &call, ctx), ctx.c_zero(&crossing(ty))),
140    };
141    let body = if m.error.throws() {
142        format!("try {body}")
143    } else {
144        body
145    };
146    let throwing = match &m.error {
147        ErrorStrategy::Domain(name) => format!(", throwing: {}.self", ctx.domain_type(name)),
148        ErrorStrategy::Trap | ErrorStrategy::Untyped => String::new(),
149    };
150    // The vtable field is spelled exactly as the C header declares it, so a
151    // method whose name is a C keyword carries the header's escape.
152    w.line(format!(
153        "{}: {{ {} in",
154        c_param_name(&m.abi.symbol),
155        formals.join(", ")
156    ));
157    w.scope(|w| {
158        w.line(format!(
159            "wvInvoke(ctx, outErr, as: (any {}).self{throwing}, fallback: {fallback}) {{ wvImpl in",
160            cb.name
161        ));
162        w.scope(|w| {
163            w.line(body);
164        });
165        w.line("}");
166    });
167    w.line(format!("}}{sep}"));
168}

170/// Render the process-wide vtable namespace: `shared` holds the C struct
171/// (its header, then capture-free trampolines, implicitly `@convention(c)`)
172/// at a stable address the library can hold for the process lifetime.
173fn render_vtable(w: &mut CodeWriter, cb: &CallbackBinding, ctx: &SwiftCtx) {
174    let proto = &cb.name;
175    let vtable_name = callback_vtable_name(&cb.name);
176    let vtable_tag = &cb.vtable_tag;
177    let doc = format!(
178        "The process-wide vtable the library calls `{proto}` implementations through \
179         (a `{vtable_tag}`). Every entry recovers the implementation from `ctx`, copies \
180         or adopts its arguments, hands its result back, and reports a thrown error \
181         through `out_err` instead of unwinding."
182    );
183    w.doc(&Some(wrap(&doc, 76)), DocCommentStyle::TripleSlash);
184    w.line(format!("enum {vtable_name} {{"));
185    w.indent();
186    w.line(format!("static let shared = WvVtable({vtable_tag}("));
187    w.indent();
188    w.line(format!("size: UInt32(MemoryLayout<{vtable_tag}>.size),"));
189    w.line("flags: 0,");
190    w.line(format!(
191        "free: {{ ctx in wvRelease(ctx, as: (any {proto}).self) }}{}",
192        if cb.methods.is_empty() { "" } else { "," }
193    ));
194    for (i, m) in cb.methods.iter().enumerate() {
195        render_trampoline(w, cb, m, i + 1 == cb.methods.len(), ctx);
196    }
197    w.dedent();
198    w.line("))");
199    w.dedent();
200    w.line("}");
201    w.blank();
202}

204/// Render everything one callback interface contributes: the public
205/// protocol and the vtable namespace whose `shared.pointer` every call site
206/// passing this interface hands to the library.
207pub(crate) fn render_swift_callback(w: &mut CodeWriter, cb: &CallbackBinding, ctx: &SwiftCtx) {
208    render_protocol(w, cb, ctx);
209    render_vtable(w, cb, ctx);
