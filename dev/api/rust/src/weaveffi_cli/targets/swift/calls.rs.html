1//! Callable rendering: sync, async, and iterator-returning wrapper
2//! functions, and the per-[`Crossing`] marshalling every position shares.
3//!
4//! In C ABI revision 6 every value crosses as one C value, so a value is
5//! marshalled the same way whether it's a parameter, a return, an async
6//! result, an iterator item, or a callback argument or return; only who owns
7//! a run differs (see [`crate::codegen::values`]). How a failure surfaces
8//! comes from the callable's [`ErrorStrategy`]; this module only renders the
9//! Swift spelling.
10

22/// Locals a wrapper body declares next to its parameters. A parameter
23/// spelled like one keeps its argument label and gets a `_`-suffixed local
24/// name (`err err_: Int32`), so the body never confuses the two. (The
25/// `{local}_run` and `{local}_cb` locals can't collide with another
26/// parameter's: a lowerCamelCase local never contains an underscore but a
27/// trailing escape.)
28const BODY_LOCALS: &[&str] = &["err", "rv", "token", "context", "continuation"];
29
30/// One parameter of a wrapper: its argument label and the local name the
31/// body uses (they differ only when the label is a body local).
32pub(crate) struct Param<'a> {
33    /// The argument label (lowerCamelCase, keyword-escaped).
34    pub(crate) label: String,
35    /// The name the body refers to the argument by.
36    pub(crate) local: String,
37    /// The parameter as lowered.
38    pub(crate) binding: &'a ParamBinding,
39}
40
41impl Param<'_> {
42    /// The declaration in a signature: `label: Type`, or `label local: Type`.
43    fn decl(&self, ctx: &SwiftCtx) -> String {
44        let ty = ctx.param_type(&self.binding.ty);
45        if self.label == self.local {
46            format!("{}: {ty}", self.label)
47        } else {
48            format!("{} {}: {ty}", self.label, self.local)
49        }
50    }
51}
52
53/// The wrapper parameters of `f`.
54pub(crate) fn swift_params(f: &FnBinding) -> Vec<Param<'_>> {
55    f.params
56        .iter()
57        .map(|binding| {
58            let label = swift_ident(&binding.name);
59            let local = lang::escape_member(&label, BODY_LOCALS);
60            Param {
61                label,
62                local,
63                binding,
64            }
65        })
66        .collect()
67}

71/// The runtime helper that lends a run-crossing value of `crossing` for one
72/// call, binding the run to the closure's parameter, or `None` for a value
73/// that crosses without one.
74fn lender(crossing: &Crossing) -> Option<&'static str> {
75    Some(match crossing {
76        Crossing::Utf8 => "wvWithUTF8",
77        Crossing::Bytes => "wvWithBytes",
78        Crossing::Slice(_) => "wvWithSlice",
79        Crossing::Buffer(_) => "wvWithEncoded",
80        _ => return None,
81    })
82}

84/// The C value of `value` (a Swift expression of the value's Swift type)
85/// for a position that doesn't take a run: a scalar as itself, an enum as
86/// its raw value, an optional scalar as its `{p}_opt_{prim}`, and an object
87/// as its pointer, `fresh` (a fresh strong reference, for a callback return)
88/// or borrowed (the wrapper's own, for a parameter).
89fn pass(crossing: &Crossing, value: &str, fresh: bool, ctx: &SwiftCtx) -> String {
90    match crossing {
91        Crossing::Scalar(_) => value.to_string(),
92        Crossing::Enum(_) => format!("{value}.rawValue"),
93        Crossing::OptScalar(_) => format!("{}(wv: {value})", ctx.c_value_type(crossing)),
94        Crossing::OptEnum(_) => {
95            format!("{}(wv: {value}?.rawValue)", ctx.c_value_type(crossing))
96        }
97        Crossing::Object { nullable, .. } => {
98            let q = if *nullable { "?" } else { "" };
99            let member = if fresh { "clonePtr()" } else { "ptr" };
100            format!("{value}{q}.{member}")
101        }
102        Crossing::Utf8 | Crossing::Bytes | Crossing::Slice(_) | Crossing::Buffer(_) => {
103            unreachable!("a run crosses through its lender")
104        }
105    }
106}

108/// The C value of `value` handed to the library as a callback method's
109/// return: runs are allocated with `{p}_alloc` and adopted by the library,
110/// and an object is a fresh strong reference.
111pub(crate) fn hand_back(ty: &Ty, value: &str, ctx: &SwiftCtx) -> String {
112    let c = crossing(ty);
113    let helper = match c {
114        Crossing::Utf8 => "wvHandOverString",
115        Crossing::Bytes => "wvHandOverBytes",
116        Crossing::Slice(_) => "wvHandOverSlice",
117        Crossing::Buffer(_) => "wvHandOverBuffer",
118        _ => return pass(&c, value, true, ctx),
119    };
120    format!("{helper}({value})")
121}

123/// The Swift value of type `ty` that the library handed over as the C value
124/// `raw`. An `owned` run (a return, async result, or iterator item) is
125/// released after it's copied; a borrowed one (a callback argument) is only
126/// copied. An object always moves one strong reference, which the new
127/// wrapper adopts.
128pub(crate) fn receive(ty: &Ty, raw: &str, owned: bool, ctx: &SwiftCtx) -> String {
129    let take = if owned { "wvTake" } else { "wvBorrow" };
130    match crossing(ty) {
131        Crossing::Scalar(_) => raw.to_string(),
132        Crossing::Enum(name) => format!("wvEnumCase({name}.self, {raw})"),
133        Crossing::OptScalar(_) => format!("{raw}.wvValue"),
134        Crossing::OptEnum(name) => format!("{raw}.wvValue.map {{ wvEnumCase({name}.self, $0) }}"),
135        Crossing::Utf8 => format!("{take}String({raw})"),
136        Crossing::Bytes => format!("{take}Bytes({raw})"),
137        Crossing::Slice(p) => format!("{take}Slice({raw}, as: {}.self)", prim_swift_type(p)),
138        Crossing::Buffer(ty) => format!("{take}Buffer({raw}, as: {}.self)", ctx.swift_type(ty)),
139        Crossing::Object {
140            interface,
141            nullable: true,
142        } => format!("{raw}.map {{ {interface}(ptr: $0) }}"),
143        Crossing::Object {
144            interface,
145            nullable: false,
146        } => format!("{interface}(ptr: wvNonNull({raw}))"),
147    }
148}

152/// How a wrapper reports a failure, from the callable's [`ErrorStrategy`]
153/// and cancellability.
154///
155/// A callable declared `throws: Domain` throws the domain's enum for its
156/// positive codes; one declared `throws: any` throws the runtime error; a
157/// cancellable async callable also throws (`CancellationError` for code
158/// `-5`). Any other callable has a plain signature and stops the process
159/// with `fatalError`, naming the code and message, since a reported error
160/// can only be a producer bug.
161pub(crate) struct ErrCtx {
162    error: ErrorStrategy,
163    cancellable: bool,
164    /// The Swift type of the domain, for `throws: Domain`.
165    domain_type: Option<String>,
166}
167
168impl ErrCtx {
169    /// The error context of `f`.
170    pub(crate) fn for_fn(f: &FnBinding, ctx: &SwiftCtx) -> Self {
171        Self {
172            domain_type: f.error.domain().map(|d| ctx.domain_type(d)),
173            error: f.error.clone(),
174            cancellable: f.cancellable(),
175        }
176    }
177
178    /// `true` when the Swift signature `throws`.
179    pub(crate) fn throws(&self) -> bool {
180        self.error.throws() || self.cancellable
181    }
182
183    /// The statement checking the error slot of a synchronous call.
184    fn check_stmt(&self) -> String {
185        match (&self.error, &self.domain_type) {
186            (ErrorStrategy::Domain(_), Some(ty)) => format!("try wvCheck(&err, {ty}.self)"),
187            (ErrorStrategy::Trap, _) => "wvTrap(&err)".to_string(),
188            _ => "try wvCheck(&err)".to_string(),
189        }
190    }
191
192    /// The statements a completion runs for its boxed error `err`: resume
193    /// the continuation by throwing, or stop the process.
194    fn completion_failure(&self) -> Vec<String> {
195        let take = match (&self.error, &self.domain_type) {
196            (ErrorStrategy::Domain(_), Some(ty)) => format!("wvTakeError(err, {ty}.self)"),
197            (ErrorStrategy::Trap, _) if self.cancellable => "wvTakeCancellation(err)".to_string(),
198            (ErrorStrategy::Trap, _) => return vec!["wvFatal(err.pointee)".to_string()],
199            _ => "wvTakeError(err)".to_string(),
200        };
201        vec![
202            format!("cont.resume(throwing: {take})"),
203            "return".to_string(),
204        ]
205    }
206
207    /// The closure a sequence maps a mid-stream failure through.
208    fn sequence_failure(&self) -> String {
209        match (&self.error, &self.domain_type) {
210            (ErrorStrategy::Domain(_), Some(ty)) => format!("{{ wvDomainError($0, {ty}.self) }}"),
211            (ErrorStrategy::Trap, _) => "{ wvFatal($0) }".to_string(),
212            _ => "wvRuntimeError".to_string(),
213        }
214    }
215
216    /// The `- Throws:` doc callout of a throwing wrapper.
217    fn throws_doc(&self, runtime_error: &str) -> Option<String> {
218        let mut doc = match (&self.error, &self.domain_type) {
219            (ErrorStrategy::Domain(_), Some(ty)) => {
220                format!("- Throws: ``{ty}`` for a declared failure, ``{runtime_error}`` otherwise")
221            }
222            (ErrorStrategy::Trap, _) if self.cancellable => {
223                "- Throws: `CancellationError` when the calling task is cancelled, which \
224                 cancels the native call."
225                    .to_string()
226            }
227            (ErrorStrategy::Trap, _) => return None,
228            _ => format!("- Throws: ``{runtime_error}`` with the library's message"),
229        };
230        if self.error.throws() {
231            if self.cancellable {
232                doc.push_str(
233                    ", and `CancellationError` when the calling task is cancelled, which \
234                     cancels the native call",
235                );
236            }
237            doc.push('.');
238        }
239        Some(doc)
240    }
241}

245/// Where a wrapper sits, which decides how it reaches its object and whether
246/// it runs the load-time checks.
247#[derive(Clone, Copy, PartialEq, Eq)]
248pub(crate) enum Receiver {
249    /// A free function or static member: runs the load-time checks first.
250    Static,
251    /// An instance method passing its own object as the leading argument.
252    Instance,
253    /// The `new` constructor, rendered as `public init`.
254    Init,
255}
256
257/// Render one wrapper for `f` (sync, async, or iterator-returning) named
258/// `swift_name`. A free function is a top-level `func`; an interface
259/// member, a member of its class.
260pub(crate) fn render_callable(
261    w: &mut CodeWriter,
262    f: &FnBinding,
263    swift_name: &str,
264    receiver: Receiver,
265    member: bool,
266    ctx: &SwiftCtx,
267) {
268    let err = ErrCtx::for_fn(f, ctx);
269    let params = swift_params(f);
270    let mut callouts = Vec::new();
271    if f.iterator().is_some() {
272        callouts.push(
273            "- Returns: A lazy sequence that pulls one element per step from the library."
274                .to_string(),
275        );
276    }
277    callouts.extend(err.throws_doc(&ctx.runtime_error));
278    let param_docs: Vec<_> = params
279        .iter()
280        .map(|p| (p.label.clone(), &p.binding.doc))
281        .collect();
282    emit_fn_doc(w, ctx, &f.doc, &param_docs, &callouts);
283    if let Some(attr) = ctx.deprecated_attr(&f.deprecated) {
284        w.line(attr);
285    }
286    let sig = params
287        .iter()
288        .map(|p| p.decl(ctx))
289        .collect::<Vec<_>>()
290        .join(", ");
291    let effects = match (f.is_async(), err.throws()) {
292        (true, true) => " async throws",
293        (true, false) => " async",
294        (false, true) => " throws",
295        (false, false) => "",
296    };
297    if receiver == Receiver::Init {
298        w.line(format!("public init({sig}){effects} {{"));
299    } else {
300        let ret = f
301            .ret
302            .as_ref()
303            .map(|t| format!(" -> {}", ctx.ret_type(t)))
304            .unwrap_or_default();
305        let static_kw = if member && receiver == Receiver::Static {
306            "static "
307        } else {
308            ""
309        };
310        w.line(format!(
311            "public {static_kw}func {swift_name}({sig}){effects}{ret} {{"
312        ));
313    }
314    w.indent();
315    if receiver != Receiver::Instance {
316        w.line("wvLoad()");
317    }
318    match f.async_binding() {
319        Some(binding) => render_async_body(w, f, binding, &params, receiver, &err, ctx),
320        None => render_sync_body(w, f, &params, receiver, &err, ctx),
321    }
322    w.dedent();
323    w.line("}");
324}

326/// The C arguments for `params`, after the receiver's object for an
327/// instance method: a run-crossing value as the run its lending closure
328/// binds (`{local}_run`), a callback interface as its staged argument struct
329/// (`{local}_cb`), and anything else as its C value.
330fn c_call_args(params: &[Param], receiver: Receiver, ctx: &SwiftCtx) -> Vec<String> {
331    let mut args = Vec::new();
332    if receiver == Receiver::Instance {
333        args.push("self.ptr".to_string());
334    }
335    for p in params {
336        let n = &p.local;
337        args.push(match &p.binding.ty {
338            ParamTy::Callback { .. } => format!("{n}_cb"),
339            ParamTy::Value(ty) => {
340                let c = crossing(ty);
341                if c.is_run() {
342                    format!("{n}_run")
343                } else {
344                    // The wrapper borrows its own reference for the call;
345                    // Swift keeps every argument alive until it returns.
346                    pass(&c, n, false, ctx)
347                }
348            }
349        });
350    }
351    args
352}

354/// Emit the pre-call staging of callback parameters: each implementation is
355/// retained for the library behind the `{p}_{Cb}` argument struct, whose
356/// vtable's `free` entry releases it (also when the call fails). A `nil`
357/// optional callback is the zero struct, whose null vtable means none.
358fn render_arg_staging(w: &mut CodeWriter, params: &[Param], ctx: &SwiftCtx) {
359    for p in params {
360        let ParamTy::Callback { name, nullable } = &p.binding.ty else {
361            continue;
362        };
363        let n = &p.local;
364        let tag = ctx.model.callback(name).c_tag.clone();
365        let vtable = format!("{}.shared.pointer", callback_vtable_name(name));
366        if *nullable {
367            w.line(format!(
368                "let {n}_cb = {n}.map {{ {tag}(ctx: wvRetain($0), vtable: {vtable}) }} ?? {tag}()"
369            ));
370        } else {
371            w.line(format!(
372                "let {n}_cb = {tag}(ctx: wvRetain({n}), vtable: {vtable})"
373            ));
374        }
375    }
376}

378/// Open the closures lending every run-crossing parameter (a string, bytes,
379/// typed array, or value buffer), binding `{local}_run`. `bind` prefixes the
380/// first opening line (`let rv: T = ` or nothing); nested ones return their
381/// inner value implicitly. Returns how many closures were opened.
382fn open_lending_closures(w: &mut CodeWriter, params: &[Param], bind: &str) -> usize {
383    let mut opened = 0;
384    for p in params {
385        let Some(ty) = p.binding.ty.value() else {
386            continue;
387        };
388        let Some(lend) = lender(&crossing(ty)) else {
389            continue;
390        };
391        let n = &p.local;
392        let prefix = if opened == 0 { bind } else { "" };
393        w.line(format!("{prefix}{lend}({n}) {{ {n}_run in"));
394        w.indent();
395        opened += 1;
396    }
397    opened
398}
399
400/// Close `n` closures opened by [`open_lending_closures`].
401fn close_closures(w: &mut CodeWriter, n: usize) {
402    for _ in 0..n {
403        w.dedent();
404        w.line("}");
405    }
406}

408/// The Swift type of the raw C result of a synchronous call (the one C
409/// value of its return, or the iterator handle), which annotates its
410/// binding when the call sits inside lending closures.
411fn raw_return_swift(f: &FnBinding, ctx: &SwiftCtx) -> Option<String> {
412    if f.iterator().is_some() {
413        return Some("OpaquePointer?".to_string());
414    }
415    f.ret_value().map(|ty| ctx.c_value_type(&crossing(ty)))
416}

418/// Render the body of a synchronous (or iterator-returning) callable: the
419/// error slot, callback staging, the C call wrapped in whatever lending
420/// closures the inputs need, the error check, and the return conversion.
421fn render_sync_body(
422    w: &mut CodeWriter,
423    f: &FnBinding,
424    params: &[Param],
425    receiver: Receiver,
426    err: &ErrCtx,
427    ctx: &SwiftCtx,
428) {
429    w.line("var err = WvError()");
430    render_arg_staging(w, params, ctx);
431    let mut args = c_call_args(params, receiver, ctx);
432    args.push("&err".to_string());
433    let call = format!("{}({})", f.abi.symbol, args.join(", "));
434
435    let raw = raw_return_swift(f, ctx);
436    let bind = raw
437        .as_ref()
438        .map(|t| format!("let rv: {t} = "))
439        .unwrap_or_default();
440    let opened = open_lending_closures(w, params, &bind);
441    if opened == 0 {
442        if raw.is_some() {
443            w.line(format!("let rv = {call}"));
444        } else {
445            w.line(call);
446        }
447    } else {
448        w.line(call);
449        close_closures(w, opened);
450    }
451    w.line(err.check_stmt());
452
453    if let Some(it) = f.iterator() {
454        render_sequence(w, it, err, ctx);
455        return;
456    }
457    let Some(ty) = f.ret_value() else {
458        return;
459    };
460    if receiver == Receiver::Init {
461        // The constructor adopts the returned reference as its own.
462        w.line("self.ptr = wvNonNull(rv)");
463        return;
464    }
465    w.line(format!("return {}", receive(ty, "rv", true, ctx)));
466}

468/// Return the `NativeSequence` over the iterator handle in `rv`: its pull
469/// closure calls `_next` once per element and receives the element like a
470/// return of the same type.
471fn render_sequence(w: &mut CodeWriter, it: &IteratorBinding, err: &ErrCtx, ctx: &SwiftCtx) {
472    let elem = &it.elem;
473    let c = crossing(elem);
474    let mut value = receive(elem, "item", true, ctx);
475    // An optional element must stay distinguishable from the end of the
476    // stream (`nil`).
477    if matches!(elem, Ty::Optional(_)) {
478        value = format!(".some({value})");
479    }
480    w.line(format!(
481        "return NativeSequence(wvNonNull(rv), release: {{ {}($0) }}, failure: {}) {{ handle, err in",
482        it.destroy_symbol,
483        err.sequence_failure()
484    ));
485    w.scope(|w| {
486        w.line(format!(
487            "var item: {} = {}",
488            ctx.c_value_type(&c),
489            ctx.c_zero(&c)
490        ));
491        w.line(format!(
492            "guard {}(handle, &item, err) != 0 else {{ return nil }}",
493            it.next.symbol
494        ));
495        w.line(format!("return {value}"));
496    });
497    w.line("}");
498}

500/// Render the body of an async callable: a checked continuation resumed
501/// exactly once from the completion callback, wrapped in a cancellation
502/// handler that cancels the native token for a cancellable callable.
503fn render_async_body(
504    w: &mut CodeWriter,
505    f: &FnBinding,
506    binding: &AsyncBinding,
507    params: &[Param],
508    receiver: Receiver,
509    err: &ErrCtx,
510    ctx: &SwiftCtx,
511) {
512    let ret = f
513        .ret
514        .as_ref()
515        .map_or_else(|| "Void".to_string(), |t| ctx.ret_type(t));
516    let err_ty = if err.throws() { "Error" } else { "Never" };
517    let cancellable = binding.cancellable;
518
519    if cancellable {
520        w.line("let token = WvCancelToken()");
521        w.line("return try await withTaskCancellationHandler {");
522        w.indent();
523    }
524    let ret_kw = if cancellable { "" } else { "return " };
525    if err.throws() {
526        w.line(format!(
527            "{ret_kw}try await withCheckedThrowingContinuation {{ (continuation: CheckedContinuation<{ret}, Error>) in"
528        ));
529    } else {
530        w.line(format!(
531            "{ret_kw}await withCheckedContinuation {{ (continuation: CheckedContinuation<{ret}, Never>) in"
532        ));
533    }
534    w.indent();
535
536    // The library copies or retains every input during the launch, so
537    // lending them for the launch call's duration is enough.
538    render_arg_staging(w, params, ctx);
539    w.line("let context = Unmanaged.passRetained(WvContinuation(continuation)).toOpaque()");
540    let opened = open_lending_closures(w, params, "");
541
542    let mut args = c_call_args(params, receiver, ctx);
543    if cancellable {
544        args.push("token.raw".to_string());
545    }
546    let (formals, value) = match f.ret_value() {
547        Some(ty) => ("context, err, result", receive(ty, "result", true, ctx)),
548        None => ("context, err", "()".to_string()),
549    };
550    let mut launch = args.join(", ");
551    if !launch.is_empty() {
552        launch.push_str(", ");
553    }
554    w.line(format!("{}({launch}{{ {formals} in", f.abi.symbol));
555    w.indent();
556    w.line(format!(
557        "let cont = Unmanaged<WvContinuation<{ret}, {err_ty}>>.fromOpaque(context!).takeRetainedValue().value"
558    ));
559    w.line("if let err = err {");
560    w.scope(|w| {
561        for line in err.completion_failure() {
562            w.line(line);
563        }
564    });
565    w.line("}");
566    w.line(format!("cont.resume(returning: {value})"));
567    w.dedent();
568    w.line("}, context)");
569    close_closures(w, opened);
570    w.dedent();
571    w.line("}");
572    if cancellable {
573        w.dedent();
574        w.line("} onCancel: {");
575        w.scope(|w| {
576            w.line("token.cancel()");
577        });
578        w.line("}");
579    }
