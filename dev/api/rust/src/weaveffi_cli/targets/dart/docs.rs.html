1//! Doc-comment emission: `///` comments carried from the IDL, with every
2//! backticked API identifier respelled for Dart (through the shared
3//! [`codegen::docs`](crate::codegen::docs) rewriting), plus the generated
4//! notes (streaming contract, ownership, thrown exception, cancellation) a
5//! wrapper's declaration carries and its `@Deprecated` annotation.
6//!
7//! In doc comments, functions, types, and exceptions become dartdoc
8//! references (`` `new_op` `` is `[newOp]`); members, parameters, and
9//! fields keep their backticks in Dart spelling. Deprecation messages are
10//! plain strings, so every identifier keeps its backticks there.
11
12use crate::codegen::common::DocCommentStyle;
13use crate::codegen::docs::{map_code_spans, rewrite, ApiNames, Doc, IdentKind};
14use crate::codegen::CodeWriter;
15use weaveffi_model::model::{FnBinding, Model};
16use weaveffi_model::ty::RetTy;
17
18use crate::targets::dart::calls::Throws;
19use crate::targets::dart::types::{
20    dart_class, dart_ident, dart_member, dart_str_literal, exception_class,
21};
22
23/// Spells the API identifiers IDL docs and deprecation messages mention.
24pub(crate) struct Docs {
25    names: ApiNames,
26}
27
28impl Docs {
29    /// Index every identifier `model` declares.
30    pub(crate) fn new(model: &Model) -> Self {
31        Self {
32            names: ApiNames::new(model),
33        }
34    }
35
36    /// The Dart spelling of `ident` and whether dartdoc can link to it from
37    /// anywhere in the library, or `None` to keep it as written.
38    fn spell(&self, ident: &str) -> Option<(String, bool)> {
39        if ident.contains('.') {
40            return None;
41        }
42        Some(match self.names.kind(ident)? {
43            IdentKind::Function => (dart_ident(ident), true),
44            IdentKind::Type => (dart_class(ident), true),
45            IdentKind::ErrorDomain | IdentKind::ErrorCode => (exception_class(ident), true),
46            IdentKind::Member => (dart_member(ident), false),
47            IdentKind::CallbackMethod | IdentKind::Field | IdentKind::Param => {
48                (dart_ident(ident), false)
49            }
50            IdentKind::Variant => return None,
51        })
52    }
53
54    /// `text` respelled for a doc comment: linkable names become `[name]`,
55    /// the rest keep their backticks.
56    fn doc_text(&self, text: &str) -> String {
57        map_code_spans(text, |ident| {
58            self.spell(ident).map(|(name, link)| {
59                if link {
60                    format!("[{name}]")
61                } else {
62                    format!("`{name}`")
63                }
64            })
65        })
66    }
67
68    /// Emit `doc` (respelled) at the writer's current depth.
69    pub(crate) fn write(&self, w: &mut CodeWriter, doc: &Option<String>) {
70        let none = None;
71        let text = Doc::new(doc, &none).text(|_| None);
72        w.doc(
73            &text.map(|t| self.doc_text(&t)),
74            DocCommentStyle::TripleSlash,
75        );
76    }
77
78    /// Emit `@Deprecated(...)` when `deprecated` carries a message.
79    pub(crate) fn write_deprecated(&self, w: &mut CodeWriter, deprecated: &Option<String>) {
80        let none = None;
81        if let Some(msg) = Doc::new(&none, deprecated).deprecation(|_| None) {
82            let msg = rewrite(&msg, |ident| self.spell(ident).map(|(name, _)| name));
83            w.line(format!("@Deprecated('{}')", dart_str_literal(&msg)));
84        }
85    }
86
87    /// Emit a wrapper's doc comment followed by the generated notes that
88    /// apply to it, then its `@Deprecated` annotation.
89    pub(crate) fn write_wrapper(&self, w: &mut CodeWriter, f: &FnBinding, throws: &Throws) {
90        self.write(w, &f.doc);
91        let mut notes: Vec<String> = Vec::new();
92        let para = |notes: &mut Vec<String>, lines: &[&str]| {
93            if !notes.is_empty() {
94                notes.push(String::new());
95            }
96            notes.extend(lines.iter().map(|l| (*l).to_string()));
97        };
98        match &f.ret {
99            Some(RetTy::Iterator(elem)) => {
100                para(
101                    &mut notes,
102                    &[
103                        "Returns a lazy [Iterable]: each step pulls one element from a native",
104                        "iterator, and iterating again starts a new one. The native iterator",
105                        "is released when the iteration ends, fails, or is abandoned and",
106                        "collected.",
107                    ],
108                );
109                if elem.interface_name().is_some() {
110                    para(
111                        &mut notes,
112                        &[
113                            "Each element is owned by the caller: call its `dispose()` when",
114                            "you're done with it.",
115                        ],
116                    );
117                }
118            }
119            Some(RetTy::Value(ret)) if ret.interface_name().is_some() => {
120                if matches!(ret, weaveffi_model::ty::Ty::Optional(_)) {
121                    para(
122                        &mut notes,
123                        &[
124                            "Returns null when the producer reports no object. A returned object",
125                            "is owned by the caller: call its `dispose()` when you're done with it.",
126                        ],
127                    );
128                } else {
129                    para(
130                        &mut notes,
131                        &[
132                            "The returned object is owned by the caller: call its `dispose()`",
133                            "when you're done with it.",
134                        ],
135                    );
136                }
137            }
138            _ => {}
139        }
140        if f.cancellable() {
141            para(
142                &mut notes,
143                &[
144                    "Cancelling [cancelToken] completes the returned future with a",
145                    "[CancelledException].",
146                ],
147            );
148        }
149        match throws {
150            Throws::Domain(exc) => {
151                let line =
152                    format!("Throws [{exc}] (or a subclass for one of its codes) on failure.");
153                para(&mut notes, &[line.as_str()]);
154            }
155            Throws::Untyped => para(
156                &mut notes,
157                &["Throws a [NativeException] with [NativeException.genericCode] on failure."],
158            ),
159            Throws::Trap => {}
160        }
161        if !notes.is_empty() {
162            if f.doc.as_deref().is_some_and(|d| !d.trim().is_empty()) {
163                w.line("///");
164            }
165            for n in notes {
166                w.line(if n.is_empty() {
167                    "///".to_string()
168                } else {
169                    format!("/// {n}")
170                });
171            }
172        }
173        self.write_deprecated(w, &f.deprecated);
174    }
175}
