82/// The Dart spelling of an IDL value identifier (parameter, field, method):
83/// lowerCamelCase, then keyword-escaped (`class` becomes `class_`). The
84/// result never starts with `_`, so generated locals that do can't collide
85/// with it.
86pub(crate) fn dart_ident(name: &str) -> String {
87    lang::escape_ident(&name.to_lower_camel_case(), lang::DART_KEYWORDS)
88}
89
90/// Members every interface wrapper class declares or inherits:
91/// `_NativeObject`'s `dispose` and `Object`'s members. A named constructor
92/// or static can't share a name with an instance member either, so every
93/// interface member is escaped against the same set. The wrapper's private
94/// members start with `_`, which [`dart_ident`] never does.
95const OBJECT_MEMBERS: &[&str] = &[
96    "dispose",
97    "hashCode",
98    "noSuchMethod",
99    "runtimeType",
100    "toString",
101];
102
103/// The Dart spelling of an interface member (a named constructor, method,
104/// or static): [`dart_ident`], with a trailing `_` when it would collide
105/// with a member the wrapper class declares or inherits (`dispose` is
106/// `dispose_`).
107pub(crate) fn dart_member(name: &str) -> String {
108    lang::escape_member(&dart_ident(name), OBJECT_MEMBERS)
109}
110
111/// Members every value class (a record or rich-enum variant) declares or
112/// inherits: its `==`, `hashCode`, and `toString`, and `Object`'s.
113const VALUE_MEMBERS: &[&str] = &["hashCode", "noSuchMethod", "runtimeType", "toString"];
114
115/// Members every error-code exception declares or inherits:
116/// `NativeException`'s `code` and `message`, and `Object`'s.
117const ERROR_MEMBERS: &[&str] = &[
118    "code",
119    "hashCode",
120    "message",
121    "noSuchMethod",
122    "runtimeType",
123    "toString",
124];
125
126/// The Dart spelling of a record or rich-enum variant field:
127/// [`dart_ident`], with a trailing `_` when it would collide with a member
128/// the value class declares or inherits (`to_string` is `toString_`).
129pub(crate) fn dart_field(name: &str) -> String {
130    lang::escape_member(&dart_ident(name), VALUE_MEMBERS)
131}
132
133/// The Dart spelling of an error code's field: like [`dart_field`], and
134/// also escaped against `NativeException`'s `code` and `message` (a field
135/// `message` is `message_`).
136pub(crate) fn error_field(name: &str) -> String {
137    lang::escape_member(&dart_ident(name), ERROR_MEMBERS)
138}

140/// The Dart class of a user type: its name in UpperCamelCase, escaped when
141/// it would collide with a reserved type.
142pub(crate) fn dart_class(name: &str) -> String {
143    let class = name.to_upper_camel_case();
144    if RESERVED_TYPES.binary_search(&class.as_str()).is_ok() {
145        format!("{class}_")
146    } else {
147        class
148    }
149}

151/// The exception class of an error domain or code (`KvError` is
152/// `KvException`, `KitchenErrors` is `KitchenException`, a code `NotFound`
153/// is `NotFoundException`), named by the shared
154/// [`type_name`](weaveffi_model::errors::type_name) and escaped like any
155/// class.
156pub(crate) fn exception_class(raw: &str) -> String {
157    dart_class(&type_name(raw, "Exception"))
158}

160/// The surface Dart type of a value the bindings hand to the caller: a
161/// return, a field, a callback argument. `bytes` is a `Uint8List`, and a
162/// `u64` is carried as its two's-complement bit pattern in a Dart `int`.
163pub(crate) fn dart_type(ty: &Ty) -> String {
164    surface(ty, "Uint8List")
165}
166
167/// The surface Dart type of a value the caller hands to the bindings: a
168/// parameter or a callback's return. Like [`dart_type`], except that
169/// `bytes` (at any depth) accepts any `List<int>`, so a `Uint8List` and a
170/// list literal both fit; every output type is assignable to it.
171pub(crate) fn dart_in_type(ty: &Ty) -> String {
172    surface(ty, "List<int>")
173}
174
175fn surface(ty: &Ty, bytes: &str) -> String {
176    match ty {
177        Ty::Prim(Prim::F32 | Prim::F64) => "double".into(),
178        Ty::Prim(Prim::Bool) => "bool".into(),
179        Ty::Prim(Prim::String) => "String".into(),
180        Ty::Prim(Prim::Bytes) => bytes.into(),
181        Ty::Prim(_) => "int".into(),
182        Ty::Enum(n) | Ty::Record(n) | Ty::RichEnum(n) | Ty::Interface(n) => dart_class(n),
183        Ty::Optional(inner) => format!("{}?", surface(inner, bytes)),
184        Ty::List(inner) => format!("List<{}>", surface(inner, bytes)),
185        Ty::Map(k, v) => format!("Map<{}, {}>", surface(k, bytes), surface(v, bytes)),
186    }
187}

189/// The Dart type of a wrapper parameter.
190pub(crate) fn param_type(ty: &ParamTy) -> String {
191    match ty {
192        ParamTy::Value(t) => dart_in_type(t),
193        ParamTy::Callback { name, nullable } => {
194            let class = dart_class(name);
195            if *nullable {
196                format!("{class}?")
197            } else {
198                class
199            }
200        }
201    }
202}

204/// The Dart return type of a wrapper (before any `Future<...>`).
205pub(crate) fn return_type(f: &FnBinding) -> String {
206    match &f.ret {
207        None => "void".into(),
208        Some(RetTy::Value(t)) => dart_type(t),
209        Some(RetTy::Iterator(t)) => format!("Iterable<{}>", dart_type(t)),
210    }
211}

213/// The `dart:ffi` (native, Dart) type pair of one C ABI slot or return.
214///
215/// A run is the runtime's `_Run` struct, an optional scalar its
216/// `_Opt{Prim}` struct, and a callback-interface argument the shared
217/// `_CallbackArg` struct, each by value. Opaque pointees (objects,
218/// iterators, cancel tokens, `void*`) are `Pointer<Void>`, and the error
219/// slot is `Pointer<_Error>`. A bare [`CType::Named`] (an async completion
220/// callback) has no generic spelling, so callers substitute the completion's
221/// typedef.
222pub(crate) fn ffi_type(ct: &CType) -> (String, String) {
223    let same = |t: String| (t.clone(), t);
224    match ct {
225        CType::Scalar(p) => (scalar_native(*p).into(), scalar_dart(*p).into()),
226        CType::Enum(_) => ("Int32".into(), "int".into()),
227        CType::Size => ("Size".into(), "int".into()),
228        CType::Void => ("Void".into(), "void".into()),
229        CType::Opt(p) => same(opt_struct(*p)),
230        CType::Run => same("_Run".into()),
231        CType::Callback(_) => same("_CallbackArg".into()),
232        CType::Ptr { pointee, .. } => same(format!("Pointer<{}>", pointee_type(pointee))),
233        CType::Object(_)
234        | CType::Vtable(_)
235        | CType::Named(_)
236        | CType::Error
237        | CType::CancelToken => {
238            unreachable!("{ct:?} only appears behind a pointer")
239        }
240    }
241}

243/// The `dart:ffi` type a pointer slot points at.
244fn pointee_type(ct: &CType) -> String {
245    match ct {
246        CType::Error => "_Error".into(),
247        CType::Void
248        | CType::CancelToken
249        | CType::Object(_)
250        | CType::Vtable(_)
251        | CType::Named(_) => "Void".into(),
252        other => ffi_type(other).0,
253    }
254}

256/// The `dart:ffi` native type of a scalar primitive.
257fn scalar_native(p: Prim) -> &'static str {
258    match p {
259        Prim::Bool => "Bool",
260        Prim::I8 => "Int8",
261        Prim::I16 => "Int16",
262        Prim::I32 => "Int32",
263        Prim::I64 => "Int64",
264        Prim::U8 => "Uint8",
265        Prim::U16 => "Uint16",
266        Prim::U32 => "Uint32",
267        Prim::U64 => "Uint64",
268        Prim::F32 => "Float",
269        Prim::F64 => "Double",
270        Prim::String | Prim::Bytes => unreachable!("{p} is not a scalar"),
271    }
272}

274/// The Dart type of a scalar primitive.
275fn scalar_dart(p: Prim) -> &'static str {
276    match p {
277        Prim::Bool => "bool",
278        Prim::F32 | Prim::F64 => "double",
279        _ => "int",
280    }
281}

283/// The runtime's `_Opt{Prim}` struct mirroring `{p}_opt_{prim}`.
284pub(crate) fn opt_struct(p: Prim) -> String {
285    format!("_Opt{}", p.pascal())
286}

288/// The runtime's `_opt{Prim}` function building an `_Opt{Prim}` from a
289/// nullable Dart value.
290pub(crate) fn opt_fn(p: Prim) -> String {
291    format!("_opt{}", p.pascal())
292}

294/// The runtime's typed-array helper for `elem` and operation `op`
295/// (`stage`, `copy`, `take`, `handOver`): `_takeI32s`, `_stageF64s`, ...
296pub(crate) fn slice_fn(op: &str, elem: Prim) -> String {
297    format!("_{op}{}s", elem.pascal())
298}

300/// The `_CallbackMessage` typed-data constant of a typed-array element.
301pub(crate) fn typed_data_kind(elem: Prim) -> &'static str {
302    match elem {
303        Prim::I8 => "typedInt8",
304        Prim::I16 => "typedInt16",
305        Prim::I32 => "typedInt32",
306        Prim::I64 => "typedInt64",
307        Prim::U16 => "typedUint16",
308        Prim::U32 => "typedUint32",
309        Prim::U64 => "typedUint64",
310        Prim::F32 => "typedFloat32",
311        Prim::F64 => "typedFloat64",
312        other => unreachable!("{other:?} isn't a typed-array element"),
313    }
314}

316/// The private Dart variable holding the bound function of C symbol `sym`.
317pub(crate) fn ffi_var(sym: &str) -> String {

321/// The private Dart typedef name derived from C name `c_name`.
322pub(crate) fn ffi_typedef(c_name: &str) -> String {

324}

326/// The zero value of a scalar: what a callback trampoline returns when its
327/// implementation threw.
328pub(crate) fn zero_literal(p: Prim) -> &'static str {
329    match p {
330        Prim::Bool => "false",
331        Prim::F32 | Prim::F64 => "0.0",
332        _ => "0",
333    }
334}

336/// Escape a string for a single-quoted Dart literal.
337pub(crate) fn dart_str_literal(s: &str) -> String {
338    s.replace('\\', "\\\\")
339        .replace('\'', "\\'")
340        .replace('$', "\\$")
341        .replace('\n', "\\n")
342}

344#[cfg(test)]
345mod tests {
346    use super::*;
347
348    #[test]
349    fn reserved_types_are_sorted_and_escaped() {
350        assert!(RESERVED_TYPES.windows(2).all(|w| w[0] < w[1]));
351        assert_eq!(dart_class("Store"), "Store");
352        assert_eq!(dart_class("CancelToken"), "CancelToken_");
353        assert_eq!(dart_class("String"), "String_");
354        assert_eq!(dart_class("Int32List"), "Int32List_");
355    }
356
357    #[test]
358    fn exceptions_never_double_the_suffix() {
359        assert_eq!(exception_class("KvError"), "KvException");
360        assert_eq!(exception_class("KitchenErrors"), "KitchenException");
361        assert_eq!(exception_class("Failure"), "FailureException");
362        assert_eq!(exception_class("NOT_FOUND"), "NotFoundException");
363        assert_eq!(exception_class("Native"), "NativeException_");
364        assert_eq!(exception_class("Error"), "Exception_");
365    }
366
367    #[test]
368    fn interface_members_avoid_the_wrapper_members() {
369        assert_eq!(dart_member("dispose"), "dispose_");
370        assert_eq!(dart_member("to_string"), "toString_");
371        assert_eq!(dart_member("hash_code"), "hashCode_");
372        assert_eq!(dart_member("close"), "close");
373        assert_eq!(dart_member("class"), "class_");
374    }
375
376    #[test]
377    fn fields_avoid_the_class_members() {
378        assert_eq!(dart_field("to_string"), "toString_");
379        assert_eq!(dart_field("message"), "message");
380        assert_eq!(error_field("message"), "message_");
381        assert_eq!(error_field("code"), "code_");
382        assert_eq!(error_field("key"), "key");
383    }
384
385    #[test]
386    fn bytes_are_typed_data_out_and_any_list_in() {
387        let bytes = Ty::Prim(Prim::Bytes);
388        let nested = Ty::Map(
389            Box::new(Ty::Prim(Prim::String)),
390            Box::new(Ty::Optional(Box::new(bytes.clone()))),
391        );
392        assert_eq!(dart_type(&bytes), "Uint8List");
393        assert_eq!(dart_in_type(&bytes), "List<int>");
394        assert_eq!(dart_type(&nested), "Map<String, Uint8List?>");
395        assert_eq!(dart_in_type(&nested), "Map<String, List<int>?>");
396    }
397
398    #[test]
399    fn slots_map_to_runtime_structs_and_pointers() {
400        let ptr = CType::ptr;
401        assert_eq!(ffi_type(&ptr(CType::Error)).0, "Pointer<_Error>");
402        assert_eq!(ffi_type(&CType::Run).0, "_Run");
403        assert_eq!(ffi_type(&ptr(CType::Run)).0, "Pointer<_Run>");
404        assert_eq!(ffi_type(&CType::Opt(Prim::U32)).0, "_OptU32");
405        assert_eq!(
406            ffi_type(&CType::Callback("Listener".into())).0,
407            "_CallbackArg"
408        );
409        assert_eq!(
410            ffi_type(&ptr(CType::Named("ScanIterator".into()))).0,
411            "Pointer<Void>"
412        );
413        assert_eq!(
414            ffi_type(&ptr(ptr(CType::Object("Store".into())))).0,
415            "Pointer<Pointer<Void>>"
416        );
417        assert_eq!(
418            ffi_type(&CType::Scalar(Prim::F32)),
419            ("Float".into(), "double".into())
420        );
421        assert_eq!(
422            ffi_type(&CType::Enum("Kind".into())),
423            ("Int32".into(), "int".into())
424        );
425        assert_eq!(slice_fn("take", Prim::U64), "_takeU64s");
426        assert_eq!(opt_fn(Prim::Bool), "_optBool");
427    }
