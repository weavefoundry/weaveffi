1//! The little syntax the macros still read from a signature: whether a
2//! parameter is borrowed (`&T`, `Option<&T>`), a callback interface
3//! (`Arc<dyn Trait>`, `Option<Arc<dyn Trait>>`), or the cancel token, and
4//! whether a return is nothing, a value, or an iterator. Everything else
5//! about a type, including how it crosses, comes from its traits.

13/// How a parameter is lifted.
14pub(crate) enum Param {
15    /// An owned value `T: FfiType`.
16    Value(syn::Type),
17    /// A borrowed `&X`, `X: FfiRef`.
18    Ref(syn::Type),
19    /// A borrowed optional `Option<&X>`, `X: FfiRef`.
20    OptRef(syn::Type),
21    /// A callback interface `Arc<dyn Trait>`; the field is `dyn Trait`.
22    Callback(syn::Type),
23    /// An optional callback interface `Option<Arc<dyn Trait>>`.
24    OptCallback(syn::Type),
25    /// The `weaveffi::CancelToken` of a cancellable async function.
26    Token,
27}

29/// `ty` without the invisible groups (and parentheses) around it: a
30/// `macro_rules!` fragment like `$t:ty` reaches an attribute macro wrapped
31/// in a `syn::Type::Group`, which every syntactic check here looks through.
32pub(crate) fn ungroup(mut ty: &syn::Type) -> &syn::Type {
33    loop {
34        match ty {
35            syn::Type::Group(g) => ty = &g.elem,
36            syn::Type::Paren(p) => ty = &p.elem,
37            _ => return ty,
38        }
39    }
40}

42/// The last segment of a type path.
43pub(crate) fn last_segment(ty: &syn::Type) -> Option<&syn::PathSegment> {
44    match ungroup(ty) {
45        syn::Type::Path(p) if p.qself.is_none() => p.path.segments.last(),
46        _ => None,
47    }
48}
49
50/// The generic type arguments of a path segment.
51fn type_args(seg: &syn::PathSegment) -> Vec<&syn::Type> {
52    match &seg.arguments {
53        syn::PathArguments::AngleBracketed(args) => args
54            .args
55            .iter()
56            .filter_map(|a| match a {
57                syn::GenericArgument::Type(t) => Some(t),
58                _ => None,
59            })
60            .collect(),
61        _ => Vec::new(),
62    }
63}
64
65/// `inner` when `ty` is `Name<inner>` (by its last segment).
66fn single_arg<'a>(ty: &'a syn::Type, name: &str) -> Option<&'a syn::Type> {
67    let seg = last_segment(ty)?;
68    if seg.ident != name {
69        return None;
70    }
71    match type_args(seg).as_slice() {
72        [inner] => Some(inner),
73        _ => None,
74    }
75}

77/// `dyn Trait` when `ty` is `Arc<dyn Trait>`.
78fn callback_object(ty: &syn::Type) -> Option<syn::Type> {
79    let inner = ungroup(single_arg(ty, "Arc")?);
80    match inner {
81        syn::Type::TraitObject(_) => Some(inner.clone()),
82        _ => None,
83    }
84}

86impl Param {
87    /// Classify a parameter's type.
88    ///
89    /// # Errors
90    ///
91    /// Returns a spanned error for a `&mut` parameter, a raw pointer, or a
92    /// boxed callback.
93    pub(crate) fn of(ty: &syn::Type) -> syn::Result<Param> {
94        let ty = ungroup(ty);
95        match ty {
96            syn::Type::Reference(r) => {
97                if r.mutability.is_some() {
98                    return Err(syn::Error::new(
99                        ty.span(),
100                        "weaveffi: a `&mut` parameter can't cross the FFI boundary; take the \
101                         value (or `&T`) and return the result",
102                    ));
103                }
104                return Ok(Param::Ref((*r.elem).clone()));
105            }
106            syn::Type::Ptr(_) => {
107                return Err(syn::Error::new(
108                    ty.span(),
109                    "weaveffi: raw pointers can't cross the FFI boundary; declare the pointee \
110                     a #[weaveffi::interface] and pass it as `&T` or `Arc<T>`",
111                ))
112            }
113            _ => {}
114        }
115        if last_segment(ty).is_some_and(|s| s.ident == "CancelToken") {
116            return Ok(Param::Token);
117        }
118        if let Some(object) = callback_object(ty) {
119            return Ok(Param::Callback(object));
120        }
121        if let Some(inner) = single_arg(ty, "Option") {
122            if let syn::Type::Reference(r) = ungroup(inner) {
123                if r.mutability.is_none() {
124                    return Ok(Param::OptRef((*r.elem).clone()));
125                }
126            }
127            if let Some(object) = callback_object(inner) {
128                return Ok(Param::OptCallback(object));
129            }
130        }
131        if let Some(inner) = single_arg(ty, "Box") {
132            if matches!(ungroup(inner), syn::Type::TraitObject(_)) {
133                return Err(syn::Error::new(
134                    ty.span(),
135                    "weaveffi: take a callback interface as `Arc<dyn Trait>`",
136                ));
137            }
138        }
139        Ok(Param::Value(ty.clone()))
140    }

142    /// The parameter's C type.
143    pub(crate) fn abi(&self) -> TokenStream {
144        let tag = tag();
145        match self {
146            Param::Value(t) => {
147                quote_spanned!(t.span()=> <#t as ::weaveffi::__rt::FfiType<#tag>>::Abi)
148            }
149            Param::Ref(t) => quote_spanned!(t.span()=> <#t as ::weaveffi::__rt::FfiRef<#tag>>::Abi),
150            Param::OptRef(t) => {
151                quote_spanned!(t.span()=> <#t as ::weaveffi::__rt::FfiRef<#tag>>::OptAbi)
152            }
153            Param::Callback(t) | Param::OptCallback(t) => quote_spanned! {t.span()=>
154                ::weaveffi::__rt::CallbackArg<<#t as ::weaveffi::__rt::CallbackType<#tag>>::Vtable>
155            },
156            Param::Token => quote!(*const ::weaveffi::__rt::FfiCancelToken),
157        }
158    }
159
160    /// The parameter's IDL spelling, as a `Name` expression. The cancel
161    /// token has none.
162    pub(crate) fn name(&self) -> TokenStream {
163        let tag = tag();
164        match self {
165            Param::Value(t) => {
166                quote_spanned!(t.span()=> <#t as ::weaveffi::__rt::FfiType<#tag>>::NAME)
167            }
168            Param::Ref(t) => {
169                quote_spanned!(t.span()=> <#t as ::weaveffi::__rt::FfiRef<#tag>>::NAME)
170            }
171            Param::OptRef(t) => quote_spanned! {t.span()=>
172                ::weaveffi::__rt::Name::optional(&<#t as ::weaveffi::__rt::FfiRef<#tag>>::NAME)
173            },
174            Param::Callback(t) => {
175                quote_spanned!(t.span()=> <#t as ::weaveffi::__rt::CallbackType<#tag>>::NAME)
176            }
177            Param::OptCallback(t) => quote_spanned! {t.span()=>
178                ::weaveffi::__rt::Name::optional(&<#t as ::weaveffi::__rt::CallbackType<#tag>>::NAME)
179            },
180            Param::Token => quote!(::weaveffi::__rt::Name::EMPTY),
181        }
182    }

184    /// Lift the C value `abi` for a synchronous call: a `Result` whose `Ok`
185    /// is held (for a borrow) or owned.
186    pub(crate) fn lift(&self, abi: &syn::Ident) -> TokenStream {
187        let tag = tag();
188        match self {
189            Param::Value(t) => {
190                quote_spanned!(t.span()=> <#t as ::weaveffi::__rt::FfiType<#tag>>::lift(#abi))
191            }
192            Param::Ref(t) => {
193                quote_spanned!(t.span()=> <#t as ::weaveffi::__rt::FfiRef<#tag>>::lift_ref(#abi))
194            }
195            Param::OptRef(t) => {
196                quote_spanned!(t.span()=> <#t as ::weaveffi::__rt::FfiRef<#tag>>::lift_opt_ref(#abi))
197            }
198            Param::Callback(t) => {
199                quote_spanned!(t.span()=> ::weaveffi::__rt::lift_callback::<#tag, #t>(#abi))
200            }
201            Param::OptCallback(t) => {
202                quote_spanned!(t.span()=> ::weaveffi::__rt::lift_callback_opt::<#tag, #t>(#abi))
203            }
204            Param::Token => quote!(::core::result::Result::<_, ::weaveffi::__rt::FfiError>::Ok(
205                ::weaveffi::__rt::token(#abi)
206            )),
207        }
208    }

210    /// Lift the C value `abi` for an async call, whose future owns every
211    /// input.
212    ///
213    /// # Errors
214    ///
215    /// Returns a spanned error for an `Option<&T>` parameter, which an async
216    /// function can't take.
217    pub(crate) fn lift_owned(
218        &self,
219        abi: &syn::Ident,
220        span: proc_macro2::Span,
221    ) -> syn::Result<TokenStream> {
222        let tag = tag();
223        Ok(match self {
224            Param::Ref(t) => {
225                quote_spanned!(t.span()=> <#t as ::weaveffi::__rt::FfiRef<#tag>>::lift_owned(#abi))
226            }
227            Param::OptRef(_) => {
228                return Err(syn::Error::new(
229                    span,
230                    "weaveffi: an async function can't borrow an optional parameter; take \
231                     `Option<T>` by value",
232                ))
233            }
234            _ => self.lift(abi),
235        })
236    }

238    /// The argument the producer function receives, from the lifted local
239    /// `local`.
240    pub(crate) fn arg(&self, local: &syn::Ident) -> TokenStream {
241        match self {
242            Param::Ref(_) => quote!(&*#local),
243            Param::OptRef(_) => quote!(#local.as_deref()),
244            _ => quote!(#local),
245        }
246    }
247
248    /// The argument an async function's future passes, from the owned local
249    /// `local`.
250    pub(crate) fn owned_arg(&self, local: &syn::Ident) -> TokenStream {
251        match self {
252            Param::Ref(_) => quote!(::core::borrow::Borrow::borrow(&#local)),
253            _ => quote!(#local),
254        }
255    }

258/// What a callable returns.
259pub(crate) enum Ret {
260    /// Nothing, `()`, or a `Result<(), E>`: an async completion has no
261    /// result slot. Carries the written type (`()` when omitted).
262    Unit(syn::Type),
263    /// A value: `R: FfiReturn`.
264    Value(syn::Type),
265    /// An iterator: `Iter<T>` or `Result<Iter<T>, E>`, `R: IterReturn`.
266    Iter(syn::Type),
267}
268
269/// Whether `ty` is `()`.
270fn is_unit(ty: &syn::Type) -> bool {
271    matches!(ungroup(ty), syn::Type::Tuple(t) if t.elems.is_empty())
272}
273
274impl Ret {
275    /// Classify a return type.
276    pub(crate) fn of(output: &syn::ReturnType) -> Ret {
277        let ty = match output {
278            syn::ReturnType::Default => return Ret::Unit(syn::parse_quote!(())),
279            syn::ReturnType::Type(_, ty) => (**ty).clone(),
280        };
281        if is_unit(&ty) {
282            return Ret::Unit(ty);
283        }
284        let ok = last_segment(&ty)
285            .filter(|s| s.ident == "Result")
286            .and_then(|s| type_args(s).first().copied());
287        if let Some(ok) = ok {
288            if is_unit(ok) {
289                return Ret::Unit(ty);
290            }
291            if single_arg(ok, "Iter").is_some() {
292                return Ret::Iter(ty);
293            }
294        }
295        if single_arg(&ty, "Iter").is_some() {
296            return Ret::Iter(ty);
297        }
298        Ret::Value(ty)
299    }
300
301    /// The written type.
302    pub(crate) fn ty(&self) -> &syn::Type {
303        match self {
304            Ret::Unit(t) | Ret::Value(t) | Ret::Iter(t) => t,
305        }
306    }
307
308    /// The return's IDL spelling and its `throws` domain, as `Name`
309    /// expressions.
310    pub(crate) fn names(&self) -> (TokenStream, TokenStream) {
311        let tag = tag();
312        let t = self.ty();
313        match self {
314            Ret::Unit(_) | Ret::Value(_) => (
315                quote_spanned!(t.span()=> <#t as ::weaveffi::__rt::FfiReturn<#tag>>::NAME),
316                quote_spanned!(t.span()=> <#t as ::weaveffi::__rt::FfiReturn<#tag>>::DOMAIN),
317            ),
318            Ret::Iter(_) => (
319                quote_spanned!(t.span()=> <#t as ::weaveffi::__rt::IterReturn<#tag>>::NAME),
320                quote_spanned!(t.span()=> <#t as ::weaveffi::__rt::IterReturn<#tag>>::DOMAIN),
321            ),
322        }
323    }
324}

326/// The parts of a callback method's `Result<T, E>` return: `T` (or `None`
327/// for `()`) and `E`.
328///
329/// # Errors
330///
331/// Returns a spanned error for any other return.
332pub(crate) fn callback_result(
333    output: &syn::ReturnType,
334    span: proc_macro2::Span,
335) -> syn::Result<(Option<syn::Type>, syn::Type)> {
336    let span = match output {
337        syn::ReturnType::Type(_, ty) => ty.span(),
338        syn::ReturnType::Default => span,
339    };
340    let err = || {
341        syn::Error::new(
342            span,
343            "weaveffi: a callback interface method returns `Result<T, E>` (`Result<(), E>` for \
344             no value): the consumer's implementation can fail, and the `Err` is how its \
345             failure reaches you; use `weaveffi::ForeignError` as `E`, or a \
346             `#[weaveffi::error]` domain that implements `From<weaveffi::ForeignError>`",
347        )
348    };
349    let syn::ReturnType::Type(_, ty) = output else {
350        return Err(err());
351    };
352    let seg = last_segment(ty)
353        .filter(|s| s.ident == "Result")
354        .ok_or_else(err)?;
355    match type_args(seg).as_slice() {
356        [ok, e] => Ok(((!is_unit(ok)).then(|| (*ok).clone()), (*e).clone())),
357        _ => Err(err()),
358    }
