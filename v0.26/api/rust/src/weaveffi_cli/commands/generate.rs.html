1//! `weaveffi generate`: parse, validate, plan the selected targets against
2//! the output directory, then write the plan, or with `--dry-run`,
3//! `--check`, or `--diff`, report it without writing.
4//!
5//! With `--dev`, the inner loop for a Rust producer: after writing, the
6//! library the API was read from is copied into each generated package that
7//! looks for a bundled copy (each target's
8//! [`dev_bundle_dir`](crate::targets::Target::dev_bundle_dir)), and for the
9//! other targets the one environment variable to set (or, for targets that
10//! link at build time, the directory to link from) is printed.
11

22/// Options for [`cmd_generate`].
23pub struct GenerateArgs<'a> {
24    /// Where the project is.
25    pub locate: Locate<'a>,
26    /// `--out`.
27    pub out: Option<&'a str>,
28    /// `--target`.
29    pub targets: Option<&'a [String]>,
30    /// `--warn`.
31    pub warn: bool,
32    /// `--dry-run`: list the files the targets render.
33    pub dry_run: bool,
34    /// `--check`: exit 1 when anything would change.
35    pub check: bool,
36    /// `--diff`: print the unified diff of what would change.
37    pub diff: bool,
38    /// `--dev`: point the generated packages at the library.
39    pub dev: bool,
40}
41
42/// Run `weaveffi generate`.
43pub fn cmd_generate(args: &GenerateArgs<'_>) -> Result<ExitCode> {
44    let quiet = args.locate.quiet;
45    let mut project = args.locate.project()?;
46    let library = if args.dev {
47        if project.krate().is_none() && !project.is_library() {
48            bail!(
49                "`--dev` points the bindings at a Rust producer's library, but the project's \
50                 input is an IDL"
51            );
52        }
53        let library = absolute(project.library_path()?);
54        project = project.library(library.clone())?;
55        Some(library)
56    } else {
57        None
58    };
59    let model = super::load_model(&project, args.warn)?;
60    let out_dir = project.config.out_dir(args.out);
61    let targets = project.config.targets(args.targets)?;
62    let plan = Orchestrator::new()
63        .with_targets(targets.iter().map(AsRef::as_ref))
64        .plan(&model, &out_dir)?;
65
66    if args.dry_run {
67        for path in plan.rendered() {
68            println!("{path}");
69        }
70        return Ok(ExitCode::SUCCESS);
71    }
72    if args.diff {
73        print!("{}", plan.unified_diff());
74    }
75    if args.check {
76        let changes: Vec<_> = plan.changes().collect();
77        if !args.diff {
78            for change in &changes {
79                println!("{} {}", change.kind.marker(), change.path);
80            }
81        }
82        if changes.is_empty() {
83            if !quiet {
84                eprintln!("{out_dir} is up to date");
85            }
86            return Ok(ExitCode::SUCCESS);
87        }
88        eprintln!(
89            "{} generated file{} in {out_dir} would change; run `weaveffi generate`",
90            changes.len(),
91            if changes.len() == 1 { "" } else { "s" }
92        );
93        return Ok(ExitCode::FAILURE);
94    }
95    if args.diff {
96        return Ok(ExitCode::SUCCESS);
97    }
98    let report = plan.apply()?;
99    if !quiet {
100        println!("{}", report.summary(&out_dir));
101    }
102    if let Some(library) = library {
103        point_at(&library, &model, &targets, &out_dir, quiet)?;
104    }
105    Ok(ExitCode::SUCCESS)
106}

108/// `path`, made absolute when it can be.
109fn absolute(path: Utf8PathBuf) -> Utf8PathBuf {
110    std::path::absolute(path.as_std_path())
111        .ok()
112        .and_then(|p| Utf8PathBuf::from_path_buf(p).ok())
113        .unwrap_or(path)
114}

116/// Point every target's generated package in `out_dir` at `library`: copy
117/// it into the packages that bundle one, and print how the others find it.
118fn point_at(
119    library: &Utf8Path,
120    model: &Model,
121    targets: &[Box<dyn Target>],
122    out_dir: &Utf8Path,
123    quiet: bool,
124) -> Result<()> {
125    let shared = matches!(library.extension(), Some("so" | "dylib" | "dll"));
126    let mut placed = Vec::new();
127    let mut by_env = Vec::new();
128    let mut by_link = Vec::new();
129    let mut wasm = false;
130    for target in targets {
131        if let Some(dir) = target.dev_bundle_dir(model).filter(|_| shared) {
132            let dir = out_dir.join(target.name()).join(dir);
133            copy_into(library, &dir)?;
134            placed.push(dir);
135            continue;
136        }
137        match target.linkage() {
138            Linkage::Runtime => by_env.push(target.name()),
139            Linkage::Link => by_link.push(target.name()),
140            Linkage::Wasm => wasm = true,
141        }
142    }
143    if quiet {
144        return Ok(());
145    }
146    println!("Bindings point at {library}");
147    for dir in &placed {
148        println!("  copied into {dir}");
149    }
150    let env = model.identity.library_env_var();
151    if !by_env.is_empty() {
152        println!("  {}: export {env}={library}", by_env.join(", "));
153    }
154    if !by_link.is_empty() {
155        let dir = library.parent().unwrap_or(Utf8Path::new("."));
156        let path_var = match std::env::consts::OS {
157            "macos" => "DYLD_LIBRARY_PATH",
158            "windows" => "PATH",
159            _ => "LD_LIBRARY_PATH",
160        };
161        println!(
162            "  {}: link with -L{dir} -l{} and run with {path_var}={dir}",
163            by_link.join(", "),
164            model.identity.library
165        );
166    }
167    if wasm {
168        println!("  wasm: needs a wasm32 build: `weaveffi build --platforms wasm32`");
169    }
170    Ok(())
171}

173/// Copy `library` into `dir`, replacing an older copy.
174fn copy_into(library: &Utf8Path, dir: &Utf8Path) -> Result<()> {
175    std::fs::create_dir_all(dir.as_std_path())
176        .into_diagnostic()
177        .wrap_err_with(|| format!("failed to create {dir}"))?;
178    let dest = dir.join(library.file_name().unwrap_or_default());
179    std::fs::copy(library.as_std_path(), dest.as_std_path())
180        .into_diagnostic()
181        .wrap_err_with(|| format!("failed to copy {library} to {dest}"))?;
182    Ok(())
183}

185#[cfg(test)]
186mod tests {
187    use super::*;
188
189    fn args<'a>(input: &'a str, out: &'a str, targets: &'a [String]) -> GenerateArgs<'a> {
190        GenerateArgs {
191            locate: Locate {
192                input: Some(input),
193                ..Locate::default()
194            },
195            out: Some(out),
196            targets: Some(targets),
197            warn: false,
198            dry_run: false,
199            check: false,
200            diff: false,
201            dev: false,
202        }
203    }
204
205    #[test]
206    fn dry_run_lists_files_without_writing() {
207        let dir = tempfile::tempdir().unwrap();
208        let yml = dir.path().join("api.yml");
209        std::fs::write(
210            &yml,
211            "version: \"0.13.0\"\nfunctions:\n  - { name: add, params: [{ name: a, type: i32 }], return: i32 }\n",
212        )
213        .unwrap();
214        let out = dir.path().join("out");
215        let targets = ["c".to_string()];
216        cmd_generate(&GenerateArgs {
217            dry_run: true,
218            ..args(yml.to_str().unwrap(), out.to_str().unwrap(), &targets)
219        })
220        .unwrap();
221        assert!(!out.exists(), "dry-run should not create output directory");
222    }
223
224    #[test]
225    fn dev_needs_a_producer() {
226        let dir = tempfile::tempdir().unwrap();
227        let yml = dir.path().join("api.yml");
228        std::fs::write(&yml, "version: \"0.13.0\"\n").unwrap();
229        let out = dir.path().join("out");
230        let targets = ["c".to_string()];
231        let err = cmd_generate(&GenerateArgs {
232            dev: true,
233            ..args(yml.to_str().unwrap(), out.to_str().unwrap(), &targets)
234        })
235        .unwrap_err();
236        assert!(format!("{err}").contains("--dev"), "{err}");
237    }
