1//! `weaveffi init`: set up a project so a bare `weaveffi generate` works.
2//!
3//! In a Rust crate (a directory with a `Cargo.toml` that has a `[package]`),
4//! it writes a `weaveffi.toml` whose `[project] input` is the crate itself
5//! (`"."`), whose API WeaveFFI reads from the library the crate builds, then
6//! checks the things a producer needs (a `weaveffi` dependency, at least one
7//! `#[weaveffi::export]`, and one `weaveffi::export_runtime!()` call) and
8//! prints any that are missing. It never edits `Cargo.toml` or
9//! source files. Anywhere else, it writes a starter IDL plus a
10//! `weaveffi.toml` that points at it.
11
12use std::process::ExitCode;
13
14use camino::{Utf8Path, Utf8PathBuf};
15use miette::{bail, IntoDiagnostic, Result, WrapErr};
16use weaveffi_model::pkg::c_ident;

18use crate::config::CONFIG_FILE_NAME;

20/// The `targets` line `init` writes: the C header plus one language, which
21/// a project edits to the targets it ships.
22const TARGETS_LINE: &str = "targets = [\"c\", \"python\"]\n";
23
24/// Options for [`cmd_init`].
25pub struct InitArgs<'a> {
26    /// The project directory.
27    pub dir: &'a str,
28    /// `--name`.
29    pub name: Option<&'a str>,
30    /// `--force`.
31    pub force: bool,
32    /// `--quiet`.
33    pub quiet: bool,
34}
35
36/// Run `weaveffi init`.
37pub fn cmd_init(args: &InitArgs<'_>) -> Result<ExitCode> {
38    let dir = Utf8PathBuf::from(args.dir);
39    std::fs::create_dir_all(dir.as_std_path())
40        .into_diagnostic()
41        .wrap_err_with(|| format!("failed to create {dir}"))?;
42    let config_path = dir.join(CONFIG_FILE_NAME);
43    if config_path.exists() && !args.force {
44        bail!("{config_path} already exists; pass --force to overwrite it");
45    }
46    let manifest = dir.join("Cargo.toml");
47    let crate_name = std::fs::read_to_string(manifest.as_std_path())
48        .ok()
49        .and_then(|text| text.parse::<toml::Table>().ok())
50        .and_then(|doc| {
51            doc.get("package")?
52                .get("name")?
53                .as_str()
54                .map(str::to_string)
55        });
56    match crate_name {
57        Some(name) => init_rust(&dir, &config_path, &name, args.quiet)?,
58        None => init_idl(&dir, &config_path, args.name, args.quiet)?,
59    }
60    Ok(ExitCode::SUCCESS)
61}

63fn write(path: &Utf8Path, contents: &str) -> Result<()> {
64    std::fs::write(path.as_std_path(), contents)
65        .into_diagnostic()
66        .wrap_err_with(|| format!("failed to write {path}"))
67}

69fn init_rust(dir: &Utf8Path, config_path: &Utf8Path, crate_name: &str, quiet: bool) -> Result<()> {
70    write(
71        config_path,
72        &format!(
73            "# WeaveFFI project configuration. See https://weaveffi.com/guides/config.html\n\
74             [project]\n\
75             input = \".\"\n\
76             out = \"bindings\"\n\
77             {TARGETS_LINE}"
78        ),
79    )?;
80    if quiet {
81        return Ok(());
82    }
83    println!("Wrote {config_path} for the Rust producer `{crate_name}`.");
84    let manifest =
85        std::fs::read_to_string(dir.join("Cargo.toml").as_std_path()).unwrap_or_default();
86    let lib = std::fs::read_to_string(dir.join("src/lib.rs").as_std_path()).unwrap_or_default();
87    let mut todo = Vec::new();
88    if !manifest.contains("weaveffi") {
89        todo.push("run `cargo add weaveffi`");
90    }
91    if !lib.contains("export_runtime!") {
92        todo.push("call `weaveffi::export_runtime!();` once in src/lib.rs");
93    }
94    if !lib.contains("weaveffi::export") {
95        todo.push("annotate the functions and types to export with `#[weaveffi::export]` and its siblings");
96    }
97    if todo.is_empty() {
98        println!("The crate is ready: run `weaveffi generate`.");
99    } else {
100        println!("Before running `weaveffi generate`:");
101        for step in todo {
102            println!("  - {step}");
103        }
104    }
105    Ok(())
106}

108fn init_idl(dir: &Utf8Path, config_path: &Utf8Path, name: Option<&str>, quiet: bool) -> Result<()> {
109    let name = match name {
110        Some(n) => n.to_string(),
111        None => std::path::absolute(dir.as_std_path())
112            .ok()
113            .and_then(|p| p.file_name().map(|n| n.to_string_lossy().into_owned()))
114            .unwrap_or_else(|| "api".to_string()),
115    };
116    let idl_name = format!("{}.yml", c_ident(&name));
117    let idl_path = dir.join(&idl_name);
118    if !idl_path.exists() {
119        write(
120            &idl_path,
121            &format!(
122                "# yaml-language-server: $schema=https://weaveffi.com/weaveffi.schema.json\n\
123                 version: \"{version}\"\n\
124                 records:\n  \
125                   - name: Greeting\n    \
126                     fields:\n      \
127                       - {{ name: text, type: string }}\n      \
128                       - {{ name: count, type: u32 }}\n\
129                 functions:\n  \
130                   - name: greet\n    \
131                     doc: Build a greeting for `name`.\n    \
132                     params:\n      \
133                       - {{ name: name, type: string }}\n    \
134                     return: Greeting\n",
135                version = weaveffi_model::ir::CURRENT_SCHEMA_VERSION,
136            ),
137        )?;
138    }
139    write(
140        config_path,
141        &format!(
142            "# WeaveFFI project configuration. See https://weaveffi.com/guides/config.html\n\
143             [project]\n\
144             input = \"{idl_name}\"\n\
145             out = \"bindings\"\n\
146             {TARGETS_LINE}\
147             \n\
148             [package]\n\
149             name = \"{name}\"\n\
150             version = \"0.1.0\"\n"
151        ),
152    )?;
153    if !quiet {
154        println!("Wrote {idl_path} and {config_path}.");
155        println!(
156            "Run `weaveffi generate` to produce the C header your native library implements \
157             and the Python package over it; edit `targets` in {config_path} for the other \
158             languages you ship."
159        );
160    }
161    Ok(())
162}

164#[cfg(test)]
165mod tests {
166    use super::*;
167
168    #[test]
169    fn idl_project_validates_and_generates() {
170        let dir = tempfile::tempdir().unwrap();
171        let root = Utf8Path::from_path(dir.path()).unwrap().join("greeter");
172        cmd_init(&InitArgs {
173            dir: root.as_str(),
174            name: None,
175            force: false,
176            quiet: true,
177        })
178        .unwrap();
179        let project =
180            crate::project::Project::locate(Some(root.join(CONFIG_FILE_NAME).as_str()), None, None)
181                .unwrap();
182        assert!(matches!(
183            &project.source,
184            crate::project::Source::Idl(p) if *p == root.join("greeter.yml")
185        ));
186        assert_eq!(
187            project.config.package.identity.name.as_deref(),
188            Some("greeter")
189        );
190        project.model().unwrap();
191        let again = cmd_init(&InitArgs {
192            dir: root.as_str(),
193            name: None,
194            force: false,
195            quiet: true,
196        });
197        assert!(again.is_err(), "init must not overwrite without --force");
198    }
199
200    #[test]
201    fn rust_project_points_at_the_crate() {
202        let dir = tempfile::tempdir().unwrap();
203        let root = Utf8Path::from_path(dir.path()).unwrap();
204        std::fs::write(root.join("Cargo.toml"), "[package]\nname = \"kv\"\n").unwrap();
205        cmd_init(&InitArgs {
206            dir: root.as_str(),
207            name: None,
208            force: false,
209            quiet: true,
210        })
211        .unwrap();
212        let cfg = crate::config::ProjectConfig::from_file(&root.join(CONFIG_FILE_NAME)).unwrap();
213        assert_eq!(cfg.project.input.as_deref(), Some(Utf8Path::new(".")));
214        let names: Vec<&str> = cfg
215            .targets(None)
216            .unwrap()
217            .iter()
218            .map(|t| t.name())
219            .collect();
220        assert_eq!(names, ["c", "python"]);
221    }
