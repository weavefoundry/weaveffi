1//! How each value crosses, in Python: one match on the shared [`Crossing`]
2//! per direction, used by every position (a parameter, a return, an async
3//! result, an iterator item, and a callback method's arguments and return).
4//!
5//! | Crossing | Into the producer | Out of the producer |
6//! |---|---|---|
7//! | `Scalar` | range-checked `int`, a `float`, or a `bool` | the value |
8//! | `Enum` | the range-checked discriminant | wrapped in its `IntEnum` |
9//! | `OptScalar`, `OptEnum` | an `_Opt{Prim}(has, value)` structure | `value if has else None` |
10//! | `Utf8`, `Bytes`, `Slice`, `Buffer` | a `_Run` lent for the call (a parameter) or from `{p}_alloc` (a callback return) | copied, then released with `{p}_run_free` unless borrowed (a callback argument) |
11//! | `Object` | the wrapper's pointer, lent (a parameter) or a fresh `_clone` (a callback return) | adopted into a new wrapper |
12
13use weaveffi_model::ty::{Prim, Ty};

15use crate::codegen::values::{crossing, Crossing};
16use crate::targets::python::codec::{decode_expr, write_fn};
17use crate::targets::python::types::{int_checker, opt_struct, prim_kind};

19/// Whether the consumer owns a run the producer handed over.
20#[derive(Clone, Copy, PartialEq, Eq)]
21pub(crate) enum Ownership {
22    /// A return, async result, or iterator item: copy it, then release it.
23    Owned,
24    /// A callback method's argument: lent for the call, so copy it only.
25    Borrowed,
26}
27
28/// The expression turning `expr` (one C value as `ctypes` hands it out: an
29/// `int`, `float`, or `bool` for a scalar, the structure for an optional
30/// scalar or a run, and an address for an object) into the Python value of
31/// type `ty`. `expr` is evaluated more than once for a nullable object, so
32/// it must be a name.
33pub(crate) fn received(expr: &str, ty: &Ty, own: Ownership) -> String {
34    let owned = own == Ownership::Owned;
35    match crossing(ty) {
36        Crossing::Scalar(_) => expr.to_string(),
37        Crossing::Enum(name) => format!("{name}({expr})"),
38        Crossing::OptScalar(p) => {
39            let kind = match p {
40                Prim::Bool => "bool",
41                Prim::F32 | Prim::F64 => "float",
42                _ => "int",
43            };
44            format!("_opt_{kind}({expr})")
45        }
46        Crossing::OptEnum(name) => format!("_opt_enum({expr}, {name})"),
47        Crossing::Utf8 if owned => format!("_take_str({expr})"),
48        Crossing::Utf8 => format!("_peek_str({expr})"),
49        Crossing::Bytes if owned => format!("_take_bytes({expr})"),
50        Crossing::Bytes => format!("_peek_bytes({expr})"),
51        Crossing::Slice(p) if owned => format!("_take_array({expr}, \"{}\")", prim_kind(p)),
52        Crossing::Slice(p) => format!("_peek_array({expr}, \"{}\")", prim_kind(p)),
53        Crossing::Buffer(ty) if owned => decode_expr(&format!("_take_bytes({expr})"), ty),
54        Crossing::Buffer(ty) => decode_expr(&format!("_peek_bytes({expr})"), ty),
55        Crossing::Object {
56            interface,
57            nullable,
58        } => adopt_expr(interface, expr, nullable),
59    }
60}

62/// The expression adopting the object pointer `p` (one strong reference)
63/// into a new wrapper of `class`; `None` for a null `I?`.
64pub(crate) fn adopt_expr(class: &str, p: &str, nullable: bool) -> String {
65    if nullable {
66        format!("{class}._adopt({p}) if {p} else None")
67    } else {
68        format!("{class}._adopt(_required({p}))")
69    }
70}

72/// The expression checking the scalar `expr` of primitive `p` before it
73/// reaches `ctypes`, named `what` in the error: an integer's range check,
74/// `_float` for a float, and `bool()` for a `bool` when `strict` (a callback
75/// return, where a wrong type must fail inside the trampoline's `try`), or
76/// the value itself when not (a parameter, which `ctypes` checks).
77pub(crate) fn checked_scalar(expr: &str, p: Prim, what: &str, strict: bool) -> String {
78    match int_checker(p) {
79        Some(check) => format!("{check}({expr}, \"{what}\")"),
80        None if !strict => expr.to_string(),
81        None if p == Prim::Bool => format!("bool({expr})"),
82        None => format!("_float({expr}, \"{what}\")"),
83    }
84}
85
86/// The `_Opt{Prim}` structure of the optional scalar `expr`, its value
87/// checked as [`checked_scalar`] does. `expr` is evaluated more than once.
88pub(crate) fn opt_value(expr: &str, p: Prim, what: &str, strict: bool) -> String {
89    let opt = opt_struct(p);
90    format!(
91        "{opt}(True, {}) if {expr} is not None else {opt}()",
92        checked_scalar(expr, p, what, strict)
93    )
94}

96/// How a callback method hands its return `expr` (one value of type `ty`,
97/// held in a name) back to the producer: the statements to run first, and
98/// the expression of the C value. Runs come from `{p}_alloc` and objects are
99/// fresh references, both of which the producer adopts; `what` names the
100/// return in a range-check error.
101pub(crate) fn returned(expr: &str, ty: &Ty, what: &str) -> (Vec<String>, String) {
102    let value = match crossing(ty) {
103        Crossing::Scalar(p) => checked_scalar(expr, p, what, true),
104        Crossing::Enum(_) => checked_scalar(expr, Prim::I32, what, true),
105        Crossing::OptScalar(p) => opt_value(expr, p, what, true),
106        Crossing::OptEnum(_) => opt_value(expr, Prim::I32, what, true),
107        Crossing::Utf8 => format!("_alloc_run({expr}.encode(\"utf-8\"))"),
108        Crossing::Bytes => format!("_alloc_run(bytes({expr}))"),
109        Crossing::Slice(p) => format!("_alloc_array({expr}, \"{}\", \"{what}\")", prim_kind(p)),
110        Crossing::Buffer(ty) => {
111            let stmts = vec![
112                "_w = _Writer()".to_string(),
113                format!("{}(_w, {expr})", write_fn(ty)),
114            ];
115            return (stmts, "_alloc_run(_w.finish())".into());
116        }
117        Crossing::Object {
118            interface,
119            nullable: false,
120        } => format!("_callback_return_object({expr}, {interface})"),
121        Crossing::Object {
122            interface,
123            nullable: true,
124        } => format!("_callback_return_object_opt({expr}, {interface})"),
125    };
126    (Vec::new(), value)
127}
