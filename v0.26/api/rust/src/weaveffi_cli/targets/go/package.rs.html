11/// The `(GOOS, GOARCH)` build-constraint tokens for a [`Platform`], used on
12/// `#cgo` directive lines, or `None` for a platform the cgo package has no
13/// slot for (Android and wasm32 binaries are skipped when packaging).
14fn go_build_tags(p: Platform) -> Option<(&'static str, &'static str)> {
15    match p {
16        Platform::MacosArm64 => Some(("darwin", "arm64")),
17        Platform::MacosX64 => Some(("darwin", "amd64")),
18        Platform::LinuxX64 => Some(("linux", "amd64")),
19        Platform::LinuxArm64 => Some(("linux", "arm64")),
20        Platform::WindowsX64 => Some(("windows", "amd64")),
21        Platform::AndroidArm64
22        | Platform::AndroidX64
23        | Platform::Wasm32
24        | Platform::IosArm64
25        | Platform::IosSimArm64
26        | Platform::IosSimX64 => None,
27    }
28}

30/// The bundled binaries this package has a `#cgo` slot for, in the binary
31/// set's order.
32fn bundled<'a>(
33    ctx: &'a PackageContext<'a>,
34) -> impl Iterator<Item = (&'a NativeBinary, &'static str, &'static str)> {
35    ctx.binaries
36        .binaries
37        .iter()
38        .filter_map(|nb| go_build_tags(nb.platform).map(|(goos, goarch)| (nb, goos, goarch)))
39}

41/// The generated `go.mod` for the emitted module.
42pub(crate) fn render_go_mod(module_path: &str) -> String {
43    let prelude = render_prelude(CommentStyle::DoubleSlash);
44    let trailer = render_trailer(CommentStyle::DoubleSlash, "go.mod");
45    // Go 1.24 is required for `runtime.AddCleanup`, which releases an
46    // object wrapper nobody closed (1.23 brought the `iter` package the lazy
47    // `iter<T>` wrappers return).
48    format!("{prelude}module {module_path}\n\ngo 1.24\n\n{trailer}")
49}

51/// README for the generate-mode output.
52pub(crate) fn render_readme(names: &Names) -> String {
53    let prelude = render_prelude(CommentStyle::Xml);
54    let trailer = render_trailer(CommentStyle::Xml, "README.md");
55    let Names {
56        module_path,
57        package,
58        library,
59        header,
60        ..
61    } = names;
62    format!(
63        r#"{prelude}# {package} (Go)
64
65cgo bindings for the `{library}` native library.
66
67## Build
68
69The module ships its own copy of the C header (`{header}`) and links with
70`-l{library}`. Point the linker at the directory holding `lib{library}.so`,
71`lib{library}.dylib`, or `{library}.dll`, and make the library findable at
72run time:
73
74```sh
75export CGO_LDFLAGS="-L/path/to/lib"
76export LD_LIBRARY_PATH="/path/to/lib"   # DYLD_LIBRARY_PATH on macOS
77go build ./...
78```
79
80cgo links the library when the program is built and the platform loader
81finds it when the program starts, so the library-path environment variable
82the runtime-loading bindings (Python, .NET, and others) honor doesn't apply.
83Add `-Wl,-rpath,/path/to/lib` to `CGO_LDFLAGS` to record the directory in the
84binary instead of setting the loader path.
85
86Depend on the module from your own project with a `require` (and, for a
87local checkout, a `replace`) of `{module_path}`, then import it:
88
89```go
90import "{module_path}"
91```
92
93Go 1.24 or newer is required.
94
95## Usage notes
96
97- Call `Check()` at startup: it returns an error when the linked library
98  isn't the one these bindings were generated for (a different ABI revision,
99  or a declaration that changed; one that's missing fails to link). The
100  first call into the package runs the same checks, and every call after a
101  failed check panics with that error.
102- Records are plain structs and rich enums are sealed interfaces; both cross
103  the boundary by value. Optional scalars are pointers, and strings, byte
104  slices, and numeric slices lend their own storage for the call.
105- Interfaces are reference-counted objects. Call `Close` when you're done
106  with one (an unreachable wrapper is released eventually otherwise); it's
107  safe to call twice or while a call is in flight.
108- A function that declares an error domain returns that domain's code types,
109  which you match with `errors.As`; one that fails with any error returns an
110  `*Error`. Any other call panics with an `*Error` when the library reports
111  a failure, since that's a bug. Async functions always return the error.
112- Async functions take a `context.Context` first. Cancellable ones cancel
113  the native call when the context is done.
114- Callback interfaces are Go interfaces you implement. The library may call
115  them from any thread; a panic is reported to the native caller as a
116  callback failure instead of crashing.
117
118{trailer}"#
119    )
120}

122/// README for a packaged Go module that bundles per-platform libraries.
123fn render_packaged_readme(ctx: &PackageContext, names: &Names) -> String {
124    let prelude = render_prelude(CommentStyle::Xml);
125    let trailer = render_trailer(CommentStyle::Xml, "README.md");
126    let platforms: Vec<String> = bundled(ctx)
127        .map(|(nb, _, _)| format!("- `lib/{}/`", nb.platform.id()))
128        .collect();
129    let platform_list = platforms.join("\n");
130    let package = &names.package;
131    format!(
132        r#"{prelude}# {package} (Go)
133
134cgo bindings with a prebuilt shared library bundled for each platform under
135`lib/<platform>/`. The cgo preamble adds the matching `${{SRCDIR}}`-relative
136library search path and rpath per GOOS/GOARCH, so `go build` links the right
137library with no manual `CGO_LDFLAGS`.
138
139## Bundled platforms
140
141{platform_list}
142
143{trailer}"#,
144    )
145}

147/// The packaged module: the generated module with a self-contained cgo
148/// preamble plus one bundled shared library per desktop platform, or
149/// nothing when no desktop platform was built. Binaries for platforms
150/// without a `GOOS,GOARCH` cgo slot (Android, iOS, `wasm32`) are skipped.
151pub(crate) fn package_files(
152    model: &Model,
153    ctx: &PackageContext,
154    config: &GoConfig,
155) -> Vec<Artifact> {
156    let names = Names::new(model, config);
157    let library = &names.library;
158    if bundled(ctx).next().is_none() {
159        return Vec::new();
160    }
161
162    // Expand the single generate-mode `#cgo LDFLAGS` line into per
163    // GOOS/GOARCH library search and rpath directives (all `${SRCDIR}`
164    // relative). cgo selects the matching line at build time.
165    let original = format!("#cgo LDFLAGS: -l{library}\n");
166    let mut cgo = String::new();
167    for (nb, goos, goarch) in bundled(ctx) {
168        let id = nb.platform.id();
169        if nb.platform == Platform::WindowsX64 {
170            cgo.push_str(&format!(
171                "#cgo {goos},{goarch} LDFLAGS: -L${{SRCDIR}}/lib/{id}\n"
172            ));
173        } else {
174            cgo.push_str(&format!(
175                "#cgo {goos},{goarch} LDFLAGS: -L${{SRCDIR}}/lib/{id} -Wl,-rpath,${{SRCDIR}}/lib/{id}\n"
176            ));
177        }
178    }
179    cgo.push_str(&original);
180
181    let mut files: Vec<PackagedFile> = render_files(model, config)
182        .into_iter()
183        .map(|(name, contents)| {
184            let contents = match name.as_str() {
185                "bindings.go" => contents.replace(&original, &cgo),
186                "README.md" => render_packaged_readme(ctx, &names),
187                _ => contents,
188            };
189            PackagedFile::text(name, contents)
190        })
191        .collect();
192    files.extend(per_platform_libraries(ctx.binaries, "lib", |p| {
193        go_build_tags(p).is_some()
194    }));
195    vec![Artifact::directory(format!("go/{}", names.package), files)]
196}
