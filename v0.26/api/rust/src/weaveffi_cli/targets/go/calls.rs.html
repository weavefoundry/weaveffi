1//! Call rendering: sync, async, and iterator wrappers and the parameter
2//! marshalling they share.
3//!
4//! Each value crosses as one C value (see [`values`](super::values)), so a
5//! wrapper lends each parameter, passes `&cErr` (sync calls and iterators)
6//! or the completion and its context (async calls), and receives the one
7//! result. How a failure surfaces comes from the callable's
8//! [`ErrorStrategy`](weaveffi_model::model::ErrorStrategy).
9

21/// The wrapper type a method is declared on.
22#[derive(Clone, Copy)]
23pub(crate) struct Receiver<'a> {
24    /// The Go type name of the interface wrapper (`Store`).
25    pub(crate) ty: &'a str,
26}
27
28/// Render the Go wrapper of one callable: a package function (a free
29/// function, constructor, or static) or, with `recv`, a method.
30pub(crate) fn render_function(
31    w: &mut CodeWriter,
32    ctx: &Ctx,
33    f: &FnBinding,
34    go_name: &str,
35    recv: Option<Receiver>,
36) {
37    match &f.shape {
38        Shape::Async(ab) => render_async(w, ctx, f, ab, go_name, recv),
39        Shape::Iterator(it) => render_iterator(w, ctx, f, it, go_name, recv),
40        Shape::Sync => render_sync(w, ctx, f, go_name, recv),
41    }
42}

46/// The wrapper's signature line.
47fn header(recv: Option<Receiver>, go_name: &str, params: &[String], ret_sig: &str) -> String {
48    let params = params.join(", ");
49    match recv {
50        Some(r) => format!("func (s *{}) {go_name}({params}){ret_sig} {{", r.ty),
51        None => format!("func {go_name}({params}){ret_sig} {{"),
52    }
53}

55/// The wrapper's Go parameter list.
56fn go_params(f: &FnBinding) -> Vec<String> {
57    let mut params = Vec::new();
58    if f.is_async() {
59        params.push("ctx context.Context".to_string());
60    }
61    params.extend(
62        f.params
63            .iter()
64            .map(|p| format!("{} {}", names::param(&p.name, f), param_type(&p.ty))),
65    );
66    params
67}

69/// The opening of every wrapper body: a package function runs the
70/// load-time checks (see `Check`), and a method borrows its receiver.
71/// `s.native()` panics when the wrapper was already closed, and the
72/// deferred release keeps the object alive (even across a concurrent
73/// `Close`) until the call returns.
74fn emit_prologue(w: &mut CodeWriter, recv: Option<Receiver>, args: &mut Vec<String>) {
75    if recv.is_some() {
76        w.line("cSelf := s.native()");
77        w.line("defer s.release()");
78        args.push("cSelf".into());
79    } else {
80        w.line("wvLoaded()");
81    }
82}

84/// The local a parameter's C value is staged in: `c` and the Go parameter
85/// name, capitalized (`other` is `cOther`, `err_` is `cErr_`).
86fn staging(go_param: &str) -> String {
87    let mut chars = go_param.chars();
88    let first: String = chars
89        .next()
90        .map(|c| c.to_ascii_uppercase())
91        .into_iter()
92        .collect();
93    format!("c{first}{}", chars.as_str())
94}

96/// Emit the staging for every parameter of `f` and push one C argument per
97/// parameter, in order.
98///
99/// Value parameters are lent (see [`lend`]). A callback implementation is
100/// kept in a handle table, passed as the argument struct's context beside
101/// its interface's static vtable; a nil `Cb?` (or a typed nil) passes a
102/// null vtable. Beside a callback, every value is staged first, so a value
103/// that panics while it's lent (a nil required object, a rich-enum value
104/// that's no variant) can't strand a callback's handle.
105fn emit_params(w: &mut CodeWriter, ctx: &Ctx, f: &FnBinding, args: &mut Vec<String>) {
106    let has_callbacks = f.params.iter().any(|p| p.ty.callback().is_some());
107    let mut slots: Vec<Option<String>> = vec![None; f.params.len()];
108    for (i, p) in f.params.iter().enumerate() {
109        if let ParamTy::Value(ty) = &p.ty {
110            let name = names::param(&p.name, f);
111            let c = staging(&name);
112            let mut arg = lend(w, ctx, ty, &name, &c);
113            if has_callbacks && arg != c {
114                w.line(format!("{c} := {arg}"));
115                arg = c;
116            }
117            slots[i] = Some(arg);
118        }
119    }
120    for (i, p) in f.params.iter().enumerate() {
121        let ParamTy::Callback { name: cb, nullable } = &p.ty else {
122            continue;
123        };
124        let name = names::param(&p.name, f);
125        let tag = &ctx.model.callback(cb).c_tag;
126        let vtable = format!(
127            "C.{}()",
128            vtable_accessor(&ctx.model.callback(cb).vtable_tag)
129        );
130        slots[i] = Some(if *nullable {
131            let c = staging(&name);
132            w.line(format!(
133                "{c}Ctx, {c}Vtable := wvOptionalCallback({name}, {vtable})"
134            ));
135            format!("C.{tag}{{ctx: {c}Ctx, vtable: {c}Vtable}}")
136        } else {
137            format!(
138                "C.{tag}{{ctx: wvNewCallback({name}, \"{}\"), vtable: {vtable}}}",
139                pascal(cb)
140            )
141        });
142    }
143    args.extend(
144        slots
145            .into_iter()
146            .map(|s| s.expect("every parameter has a slot")),
147    );
148}

150/// The doc comment of a wrapper: its IDL doc and documented parameters,
151/// then `extra` paragraphs, then `Deprecated:`.
152fn wrapper_doc(ctx: &Ctx, f: &FnBinding, go_name: &str, extra: &[String]) -> GoDoc {
153    let (doc, deprecated) = ctx.docs.of(&f.doc, &f.deprecated);
154    let params = f.params.iter().filter_map(|p| {
155        let (doc, _) = ctx.docs.of(&p.doc, &None);
156        doc.map(|d| (names::param(&p.name, f), d))
157    });
158    let mut doc = GoDoc::decl(go_name, doc).params(params);
159    for p in extra {
160        doc = doc.para(p);
161    }
162    doc.deprecated(deprecated)
163}

167/// Render a sync callable: the Go wrapper lending the parameters, calling
168/// the C symbol, checking the error slot, and taking the result. A
169/// callable that declares errors returns `(T, error)` (its domain's code
170/// types, or an `*Error` when it fails with any error); any other panics
171/// with an `*Error`.
172fn render_sync(
173    w: &mut CodeWriter,
174    ctx: &Ctx,
175    f: &FnBinding,
176    go_name: &str,
177    recv: Option<Receiver>,
178) {
179    let throws = f.error.throws();
180    let ret = f.ret_value();
181    let ret_sig = match (ret, throws) {
182        (Some(r), true) => format!(" ({}, error)", go_type(r)),
183        (Some(r), false) => format!(" {}", go_type(r)),
184        (None, true) => " error".into(),
185        (None, false) => String::new(),
186    };
187    wrapper_doc(ctx, f, go_name, &[]).emit(w);
188    w.block(header(recv, go_name, &go_params(f), &ret_sig), "}", |w| {
189        let mut args = Vec::new();
190        emit_prologue(w, recv, &mut args);
191        emit_params(w, ctx, f, &mut args);
192        w.line(format!("var cErr C.{}_error", ctx.prefix));
193        args.push("&cErr".into());
194        let call = format!("C.{}({})", f.abi.symbol, args.join(", "));
195        match ret {
196            Some(_) => w.line(format!("cRet := {call}")),
197            None => w.line(call),
198        };
199        if throws {
200            let mapped = ctx.map_failure(&f.error, "wvTakeError(&cErr)");
201            w.block("if cErr.code != 0 {", "}", |w| {
202                match ret {
203                    Some(ty) => w.line(format!("return {}, {mapped}", go_zero(ty))),
204                    None => w.line(format!("return {mapped}")),
205                };
206            });
207        } else {
208            w.line("wvTrap(&cErr)");
209        }
210        let ok = if throws { ", nil" } else { "" };
211        match ret {
212            Some(ty) => {
213                w.line(format!("return {}{ok}", take(ty, "cRet")));
214            }
215            None if throws => {
216                w.line("return nil");
217            }
218            None => {}
219        }
220    });
221    w.blank();
222}

226/// Render an `iter<T>` callable as Go's lazy sequence idiom over the
227/// runtime's `wvSeq` (`iter.Seq[T]`) or, when it declares errors, `wvSeq2`
228/// (`iter.Seq2[T, error]`, a failure yielded as a final `(zero, err)`
229/// pair). The wrapper supplies the launch, whose cursor pulls one element
230/// per step through `_next` and destroys the iterator.
231fn render_iterator(
232    w: &mut CodeWriter,
233    ctx: &Ctx,
234    f: &FnBinding,
235    it: &IteratorBinding,
236    go_name: &str,
237    recv: Option<Receiver>,
238) {
239    let throws = f.error.throws();
240    let elem = &it.elem;
241    let elem_go = go_type(elem);
242    let seq = ret_type(f.ret.as_ref().expect("an iterator returns"), throws);
243    let mut note = format!(
244        "{go_name} returns a lazy sequence: each range over it launches the native \
245         iterator, pulls one element per step, and releases the iterator when the \
246         loop ends, early or not."
247    );
248    if throws {
249        note.push_str(" A failure is yielded as a final (zero value, error) pair.");
250    }
251    wrapper_doc(ctx, f, go_name, &[note]).emit(w);
252
253    let p = ctx.prefix;
254    w.block(
255        header(recv, go_name, &go_params(f), &format!(" {seq}")),
256        "}",
257        |w| {
258            if recv.is_none() {
259                w.line("wvLoaded()");
260            }
261            let (seq_fn, tail) = if throws {
262                ("wvSeq2", format!(", {})", ctx.mapper(&f.error)))
263            } else {
264                ("wvSeq", ")".to_string())
265            };
266            w.line(format!(
267                "return {seq_fn}(func(cErr *C.{p}_error) wvCursor[{elem_go}] {{"
268            ));
269            w.scope(|w| {
270                let mut args = Vec::new();
271                if recv.is_some() {
272                    emit_prologue(w, recv, &mut args);
273                }
274                emit_params(w, ctx, f, &mut args);
275                args.push("cErr".into());
276                w.line(format!("it := C.{}({})", f.abi.symbol, args.join(", ")));
277                w.block(format!("return wvCursor[{elem_go}]{{"), "}", |w| {
278                    w.block(
279                        format!("next: func(cErr *C.{p}_error) ({elem_go}, bool) {{"),
280                        "},",
281                        |w| emit_next(w, ctx, it, elem),
282                    );
283                    w.line(format!(
284                        "destroy: func() {{ C.{}(it) }},",
285                        it.destroy_symbol
286                    ));
287                });
288            });
289            w.line(format!("}}{tail}"));
290        },
291    );
292    w.blank();
293}

295/// The body of an iterator cursor's `next`: declare the item, call
296/// `_next`, and take the element.
297fn emit_next(w: &mut CodeWriter, ctx: &Ctx, it: &IteratorBinding, elem: &Ty) {
298    w.line(format!(
299        "var cItem {}",
300        cgo_value(ctx, elem, Position::Transfer)
301    ));
302    w.block(
303        format!("if C.{}(it, &cItem, cErr) == 0 {{", it.next.symbol),
304        "}",
305        |w| {
306            w.line(format!("return {}, false", go_zero(elem)));
307        },
308    );
309    w.line(format!("return {}, true", take(elem, "cItem")));
310}

314/// The C name of the exported Go trampoline behind an async completion
315/// typedef.
316pub(crate) fn completion_trampoline(callback_type: &str) -> String {
317    format!("goWv_{callback_type}")
318}

320/// The C identifier of the static preamble function returning the address
321/// of a callback interface's one vtable (Go can't take the address of a
322/// `static` C variable through cgo, so wrappers call this instead).
323pub(crate) fn vtable_accessor(vtable_tag: &str) -> String {
324    format!("wvVtablePtr_{vtable_tag}")
325}

327/// Render an async callable: the exported completion trampoline, which
328/// takes the result and delivers it on the call's channel, and a blocking
329/// wrapper taking a leading `context.Context`.
330///
331/// Every async wrapper returns an `error` and never panics on a failure:
332/// a function that declares a domain returns its code types, any other an
333/// `*Error`. A cancellable one creates a native cancel token, cancels it
334/// when `ctx` is done, and returns `ctx.Err()` for the cancelled
335/// completion; any other returns `ctx.Err()` as soon as `ctx` is done and
336/// abandons the result (see the runtime's `wvAwait`).
337fn render_async(
338    w: &mut CodeWriter,
339    ctx: &Ctx,
340    f: &FnBinding,
341    ab: &AsyncBinding,
342    go_name: &str,
343    recv: Option<Receiver>,
344) {
345    let tramp = completion_trampoline(&ab.callback_type);
346    let ret = f.ret_value();
347    let val_ty = ret.map_or_else(|| "struct{}".to_string(), go_type);
348
349    let params = &ab.callback.params;
350    let formals: Vec<String> = params
351        .iter()
352        .map(|s| format!("{} {}", names::slot(&s.name), cgo_type(&s.ty, ctx.prefix)))
353        .collect();
354    let context = names::slot(&params[0].name);
355    let err_slot = names::slot(&params[1].name);
356    let value = match (params.get(2), ret) {
357        (Some(result), Some(ty)) => take(ty, &names::slot(&result.name)),
358        _ => "struct{}{}".to_string(),
359    };
360    w.line(format!("//export {tramp}"));
361    w.block(
362        format!("func {tramp}({}) {{", formals.join(", ")),
363        "}",
364        |w| {
365            w.block(
366                format!("wvComplete({context}, {err_slot}, func() {val_ty} {{"),
367                "})",
368                |w| {
369                    w.line(format!("return {value}"));
370                },
371            );
372        },
373    );
374    w.blank();
375
376    let note = if f.cancellable() {
377        format!(
378            "{go_name} blocks until the call completes. Cancelling ctx cancels the \
379             native call, which then returns ctx.Err()."
380        )
381    } else {
382        format!(
383            "{go_name} blocks until the call completes or ctx is done; in the latter \
384             case it returns ctx.Err() at once and the native result is discarded."
385        )
386    };
387    wrapper_doc(ctx, f, go_name, &[note]).emit(w);
388
389    let (ret_sig, zero) = match ret {
390        Some(r) => (
391            format!(" ({}, error)", go_type(r)),
392            format!("{}, ", go_zero(r)),
393        ),
394        None => (" error".to_string(), String::new()),
395    };
396    w.block(header(recv, go_name, &go_params(f), &ret_sig), "}", |w| {
397        if recv.is_none() {
398            w.line("wvLoaded()");
399        }
400        w.block("if err := ctx.Err(); err != nil {", "}", |w| {
401            w.line(format!("return {zero}err"));
402        });
403        let mut args = Vec::new();
404        if recv.is_some() {
405            emit_prologue(w, recv, &mut args);
406        }
407        emit_params(w, ctx, f, &mut args);
408        let token = if ab.cancellable {
409            w.line(format!("token := C.{}_cancel_token_create()", ctx.prefix));
410            w.line(format!(
411                "defer C.{}_cancel_token_destroy(token)",
412                ctx.prefix
413            ));
414            args.push("token".into());
415            "token"
416        } else {
417            "nil"
418        };
419        w.line(format!("call := wvNewAsyncCall[{val_ty}]()"));
420        args.push(format!("C.{}(unsafe.Pointer(C.{tramp}))", ab.callback_type));
421        args.push("call.context()".into());
422        w.line(format!("C.{}({})", f.abi.symbol, args.join(", ")));
423        let await_call = format!("wvAwait(ctx, call, {token}, {})", ctx.mapper(&f.error));
424        match ret {
425            Some(_) => w.line(format!("return {await_call}")),
426            None => {
427                w.line(format!("_, err := {await_call}"));
428                w.line("return err")
429            }
430        };
431    });
432    w.blank();
