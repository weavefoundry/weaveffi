1//! Value marshalling: the Go expression that moves one value across the
2//! boundary in each direction, per its [`Crossing`].
3//!
4//! Every value is one C value in every position, so the four directions
5//! below cover parameters, returns, async results, iterator items, and
6//! callback arguments and returns alike:
7//!
8//! | Crossing | [`lend`] (parameter) | [`take`] (return, result, item) | [`borrow`] (callback argument) | [`hand_over`] (callback return) |
9//! |---|---|---|---|---|
10//! | scalar, enum | converted | converted | converted | converted |
11//! | optional scalar or enum | `wvOptIn*` | `wvOptOut*` | `wvOptOut*` | `wvOptIn*` |
12//! | string, bytes, typed array | a view of the Go memory | copied, then `run_free` | copied | copied into `{p}_alloc` |
13//! | value buffer | encoded, then viewed | decoded, then `run_free` | decoded | encoded into `{p}_alloc` |
14//! | object | the wrapper's pointer, borrowed | adopted | adopted | a fresh reference (`share`) |
15

26/// The C type of `ty` in `position`, in cgo spelling.
27pub(crate) fn cgo_value(ctx: &Ctx, ty: &Ty, position: Position) -> String {
28    cgo_type(&CType::of(ty, position), ctx.prefix)
29}
30
31/// The runtime's optional-scalar converters for `p` (`wvOptInI64`,
32/// `wvOptOutI64`) take this stem.
33fn opt_stem(p: Prim) -> &'static str {
34    p.pascal()
35}
36
37/// Emit the staging a parameter `name` of type `ty` needs and return its C
38/// argument: a scalar converted, an optional joined into its C struct, a
39/// string, bytes, or typed array lent as a view of the Go memory (cgo
40/// keeps it in place for the call), a buffered value encoded first, and an
41/// object's pointer borrowed for the call like a receiver's (a nil
42/// `Interface?` passes null).
43pub(crate) fn lend(w: &mut CodeWriter, ctx: &Ctx, ty: &Ty, name: &str, c: &str) -> String {
44    match crossing(ty) {
45        Crossing::Scalar(_) | Crossing::Enum(_) => {
46            format!("{}({name})", cgo_value(ctx, ty, Position::Param))
47        }
48        Crossing::OptScalar(p) => format!("wvOptIn{}({name})", opt_stem(p)),
49        Crossing::OptEnum(_) => format!("wvOptInI32((*int32)({name}))"),
50        Crossing::Utf8 => format!("wvStrRun({name})"),
51        Crossing::Bytes => format!("wvBytesRun({name})"),
52        Crossing::Slice(_) => format!("wvSliceRun({name})"),
53        Crossing::Buffer(ty) => format!("wvBytesRun(wvEncode({name}, {}))", write_fn(ty)),
54        Crossing::Object { nullable, .. } => {
55            if nullable {
56                w.line(format!("var {c} {}", cgo_value(ctx, ty, Position::Param)));
57                w.block(format!("if {name} != nil {{"), "}", |w| {
58                    w.line(format!("{c} = {name}.native()"));
59                    w.line(format!("defer {name}.release()"));
60                });
61            } else {
62                w.line(format!("{c} := {name}.native()"));
63                w.line(format!("defer {name}.release()"));
64            }
65            c.to_string()
66        }
67    }
68}

70/// The Go expression receiving the C value `c` of type `ty` that the
71/// library hands over (a return, an async result, or an iterator item),
72/// taking ownership: a run is copied or decoded and then released, and an
73/// object is adopted (a null `Interface?` adopts to nil).
74pub(crate) fn take(ty: &Ty, c: &str) -> String {
75    receive(ty, c, true)
76}
77
78/// The Go expression receiving the C value `c` of type `ty` that the
79/// library lends a callback method: like [`take`], but a run is only
80/// copied or decoded, since the library keeps it. An object still moves
81/// one strong reference, which the new wrapper adopts.
82pub(crate) fn borrow(ty: &Ty, c: &str) -> String {
83    receive(ty, c, false)
84}
85
86fn receive(ty: &Ty, c: &str, owned: bool) -> String {
87    let how = if owned { "Take" } else { "Borrow" };
88    match crossing(ty) {
89        Crossing::Scalar(_) | Crossing::Enum(_) => format!("{}({c})", go_type(ty)),
90        Crossing::OptScalar(p) => format!("wvOptOut{}({c})", opt_stem(p)),
91        Crossing::OptEnum(e) => format!("(*{})(wvOptOutI32({c}))", pascal(e)),
92        Crossing::Utf8 => format!("wv{how}String({c})"),
93        Crossing::Bytes => format!("wv{how}Bytes({c})"),
94        Crossing::Slice(p) => format!("wv{how}Slice[{}]({c})", prim_type(p)),
95        Crossing::Buffer(ty) if owned => format!("wvDecode({c}, {})", read_fn(ty)),
96        Crossing::Buffer(ty) => format!("wvDecodeBorrowed({c}, {})", read_fn(ty)),
97        Crossing::Object { interface, .. } => format!("wvAdopt{}({c})", pascal(interface)),
98    }
99}

101/// The C value a callback method returns for the Go value `v` of type
102/// `ty`, which the library adopts: a scalar converted, an optional joined,
103/// a string, bytes, typed array, or encoded buffer copied into a run from
104/// the library's allocator, and an object as a fresh strong reference (nil
105/// stays null).
106pub(crate) fn hand_over(ctx: &Ctx, ty: &Ty, v: &str) -> String {
107    match crossing(ty) {
108        Crossing::Scalar(_) | Crossing::Enum(_) => {
109            format!("{}({v})", cgo_value(ctx, ty, Position::Transfer))
110        }
111        Crossing::OptScalar(p) => format!("wvOptIn{}({v})", opt_stem(p)),
112        Crossing::OptEnum(_) => format!("wvOptInI32((*int32)({v}))"),
113        Crossing::Utf8 => format!("wvAllocString({v})"),
114        Crossing::Bytes | Crossing::Slice(_) => format!("wvAllocSlice({v})"),
115        Crossing::Buffer(ty) => format!("wvAllocSlice(wvEncode({v}, {}))", write_fn(ty)),
116        Crossing::Object { .. } => format!("{v}.share()"),
117    }
118}
