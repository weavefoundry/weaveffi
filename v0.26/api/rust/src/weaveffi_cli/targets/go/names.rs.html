1//! Go identifiers: case conversion with Go's initialisms, keyword escaping,
2//! and the names the flat package gives every declaration.
3//!
4//! A library is one namespace and renders into one Go package, so every
5//! exported name is computed here once ([`GoNames`]) and looked up by the
6//! renderers: types keep their IDL name (`Store`), constructors become
7//! factories (`OpenStore`), statics are prefixed with their type
8//! (`StoreOpenMany`), error domains and codes take the shared error type
9//! names (`KvError`, `*KeyNotFoundError`), and a free function whose name
10//! would clash with any of those gains a trailing underscore
11//! (`OpenStore_`).
12

20/// The initialisms Go spells in one case (`UserID`, `ParseURL`,
21/// `HTTPServer`), from the standard Go style guide's list.
22const INITIALISMS: &[&str] = &[
23    "ACL", "API", "ASCII", "CPU", "CSS", "DNS", "EOF", "GUID", "HTML", "HTTP", "HTTPS", "ID", "IP",
24    "JSON", "LHS", "QPS", "RAM", "RHS", "RPC", "SLA", "SMTP", "SQL", "SSH", "TCP", "TLS", "TTL",
25    "UDP", "UI", "UID", "URI", "URL", "UTF8", "UUID", "VM", "XML", "XMPP", "XSRF", "XSS",
26];
27
28/// Exported names the runtime files declare.
29const RUNTIME_NAMES: &[&str] = &["Check", "DebugLive", "Error"];
30
31/// Names a wrapper or trampoline body uses besides its parameters: the
32/// packages it calls into, the receiver, and its locals. A parameter or
33/// slot spelled like one of them gains a trailing underscore.
34const RESERVED: &[&str] = &[
35    "C", "cErr", "cItem", "cRet", "cSelf", "call", "context", "err", "iter", "it", "ret",
36    "runtime", "token", "unsafe",
37];
38
39/// The words of an identifier in any IDL spelling (`snake_case`,
40/// `PascalCase`, `SCREAMING_CASE`), lowercased.
41fn words(name: &str) -> Vec<String> {
42    name.to_snake_case()
43        .split('_')
44        .filter(|w| !w.is_empty())
45        .map(str::to_string)
46        .collect()
47}
48
49/// One word in exported position: an initialism in capitals, anything else
50/// capitalized.
51fn capitalize(word: &str) -> String {
52    let upper = word.to_ascii_uppercase();
53    if INITIALISMS.contains(&upper.as_str()) {
54        return upper;
55    }
56    let mut chars = word.chars();
57    match chars.next() {
58        Some(first) => first.to_uppercase().chain(chars).collect(),
59        None => String::new(),
60    }
61}

63/// The exported (PascalCase) Go spelling of an IDL name, with Go's
64/// initialisms: `user_id` is `UserID`, `ttl_seconds` is `TTLSeconds`.
65pub(crate) fn pascal(name: &str) -> String {
66    words(name).iter().map(|w| capitalize(w)).collect()
67}
68
69/// The unexported (lowerCamelCase) Go spelling of an IDL name, with Go's
70/// initialisms after the first word: `user_id` is `userID`, `url` is `url`.
71pub(crate) fn camel(name: &str) -> String {
72    let mut out = String::new();
73    for (i, w) in words(name).iter().enumerate() {
74        if i == 0 {
75            out.push_str(w);
76        } else {
77            out.push_str(&capitalize(w));
78        }
79    }
80    out
81}

83/// The default Go package name for a C prefix: the prefix without its
84/// underscores (`kitchen_sink` is `kitchensink`), lowercased, and escaped
85/// if it's a Go keyword.
86pub(crate) fn package(prefix: &str) -> String {
87    let name: String = prefix
88        .chars()
89        .filter(|c| *c != '_')
90        .flat_map(char::to_lowercase)
91        .collect();
92    lang::escape_ident(&name, lang::GO_KEYWORDS)
93}

95/// Methods every object wrapper declares: `Close` releases the wrapper's
96/// reference (its unexported helpers can't collide with an exported name).
97const OBJECT_METHODS: &[&str] = &["Close"];
98
99/// The Go name of an interface method: [`pascal`], with a trailing
100/// underscore when it would redeclare a method the wrapper already has
101/// (`close` is `Close_`). Constructors and statics are package functions
102/// prefixed or suffixed with the type name, so they can't collide.
103pub(crate) fn method(name: &str) -> String {
104    lang::escape_member(&pascal(name), OBJECT_METHODS)
105}
106
107/// The Go spelling of a parameter of the wrapper for `f`: lowerCamelCase,
108/// escaped against Go's keywords and the names wrapper bodies use. A
109/// method's receiver is `s`, and an async wrapper's leading
110/// `context.Context` is `ctx`, so a parameter spelled like either is
111/// escaped there too.
112pub(crate) fn param(name: &str, f: &FnBinding) -> String {
113    let ident = lang::escape_ident(&camel(name), lang::GO_KEYWORDS);
114    let taken = RESERVED.contains(&ident.as_str())
115        || (f.has_self() && ident == "s")
116        || (f.is_async() && ident == "ctx");
117    if taken {
118        format!("{ident}_")
119    } else {
120        ident
121    }
122}
123
124/// The Go spelling of a callback method's parameter in the interface the
125/// consumer implements: lowerCamelCase, escaped against Go's keywords (a
126/// method signature has no body to collide with).
127pub(crate) fn method_param(name: &str) -> String {
128    lang::escape_ident(&camel(name), lang::GO_KEYWORDS)
129}
130
131/// The Go spelling of one C ABI slot inside an exported trampoline: the C
132/// name, escaped against Go's keywords and the names bodies use.
133pub(crate) fn slot(name: &str) -> String {
134    let ident = lang::escape_ident(name, lang::GO_KEYWORDS);
135    if RESERVED.contains(&ident.as_str()) {
136        format!("{ident}_")
137    } else {
138        ident
139    }
140}

142/// The exported Go name of a struct field or variant field.
143pub(crate) fn field(name: &str) -> String {
144    pascal(name)
145}
146
147/// The Go name of an error code's payload field: [`field`], with a
148/// trailing underscore when it would collide with the error type's
149/// `Message` field or its `Code` and `Error` methods.
150pub(crate) fn error_field(name: &str) -> String {
151    let f = field(name);
152    if matches!(f.as_str(), "Message" | "Code" | "Error") {
153        format!("{f}_")
154    } else {
155        f
156    }
157}

159/// The Go names of one error domain.
160#[derive(Debug, Clone)]
161pub(crate) struct DomainNames {
162    /// The sealed interface every code type implements (`KvError`): the
163    /// domain's shared type name in Go's casing.
164    pub(crate) iface: String,
165    /// The concrete type of a code these bindings don't declare
166    /// (`UnknownKvError`).
167    pub(crate) unknown: String,
168    /// The unexported helper mapping a failure onto the domain
169    /// (`wvKvError`).
170    pub(crate) mapper: String,
171}
172
173/// Every exported name the generated package declares for the API, keyed by
174/// the declaration it names.
175pub(crate) struct GoNames {
176    /// Every callable's Go name by C symbol: a free function's package
177    /// function, a constructor's factory, a static's prefixed function, and
178    /// a method's `Type.Method` (the spelling a doc link uses).
179    callables: BTreeMap<String, String>,
180    /// Error domains by IDL name.
181    domains: BTreeMap<String, DomainNames>,
182    /// Error-code types by C constant.
183    codes: BTreeMap<String, String>,
184}
185
186impl GoNames {
187    /// Name every declaration of `model`, resolving clashes in the flat
188    /// package (see the module docs).
189    pub(crate) fn new(model: &Model) -> Self {
190        // Names the runtime declares, then every type-level name.
191        let mut taken: BTreeSet<String> = RUNTIME_NAMES.iter().map(|s| s.to_string()).collect();
192        let mut domains = BTreeMap::new();
193        for e in &model.errors {
194            let iface = pascal(&e.type_name);
195            let names = DomainNames {
196                unknown: format!("Unknown{iface}"),
197                mapper: format!("wv{iface}"),
198                iface,
199            };
200            taken.insert(names.iface.clone());
201            taken.insert(names.unknown.clone());
202            domains.insert(e.name.clone(), names);
203        }
204        for e in &model.enums {
205            let name = pascal(&e.name);
206            for v in &e.variants {
207                taken.insert(format!("{name}{}", pascal(&v.name)));
208            }
209            taken.insert(name);
210        }
211        for r in &model.records {
212            taken.insert(pascal(&r.name));
213        }
214        for cb in &model.callbacks {
215            taken.insert(pascal(&cb.name));
216        }
217        let mut callables = BTreeMap::new();
218        for i in &model.interfaces {
219            let name = pascal(&i.name);
220            for c in &i.constructors {
221                let factory = constructor(&name, c);
222                taken.insert(factory.clone());
223                callables.insert(c.abi.symbol.clone(), factory);
224            }
225            for f in &i.statics {
226                let go = format!("{name}{}", pascal(&f.name));
227                taken.insert(go.clone());
228                callables.insert(f.abi.symbol.clone(), go);
229            }
230            for f in &i.methods {
231                callables.insert(f.abi.symbol.clone(), format!("{name}.{}", method(&f.name)));
232            }
233            taken.insert(name);
234        }
235
236        // Error codes: `{Code}Error`, or `{Domain}{Code}Error` past a clash.
237        let mut codes = BTreeMap::new();
238        for e in &model.errors {
239            let domain = &domains[&e.name].iface;
240            for c in &e.codes {
241                let plain = pascal(&type_name(&c.name, "Error"));
242                let name = if taken.contains(&plain) {
243                    pascal(&type_name(
244                        &format!("{}{}", domain.trim_end_matches("Error"), pascal(&c.name)),
245                        "Error",
246                    ))
247                } else {
248                    plain
249                };
250                taken.insert(name.clone());
251                codes.insert(c.c_const.clone(), name);
252            }
253        }
254
255        // Free functions: the bare name, with a trailing underscore while
256        // a declared name or an earlier function holds it.
257        for f in &model.functions {
258            let mut name = pascal(&f.name);
259            while taken.contains(&name) {
260                name.push('_');
261            }
262            taken.insert(name.clone());
263            callables.insert(f.abi.symbol.clone(), name);
264        }
265        Self {
266            callables,
267            domains,
268            codes,
269        }
270    }
271
272    /// The Go name of the free function, constructor, or static `f`.
273    pub(crate) fn function(&self, f: &FnBinding) -> &str {
274        &self.callables[&f.abi.symbol]
275    }
276
277    /// The doc-link spelling of the callable whose C symbol is `symbol`
278    /// (`NewOp`, `NewGadget`, `Gadget.Describe`).
279    pub(crate) fn callable(&self, symbol: &str) -> Option<&str> {
280        self.callables.get(symbol).map(String::as_str)
281    }
282
283    /// The Go names of the error domain `name`.
284    pub(crate) fn domain(&self, name: &str) -> &DomainNames {
285        &self.domains[name]
286    }
287
288    /// The Go names of the error domain `e`.
289    pub(crate) fn domain_of(&self, e: &ErrorBinding) -> &DomainNames {
290        self.domain(&e.name)
291    }
292
293    /// The Go type name of the error code `c`.
294    pub(crate) fn code(&self, c: &ErrorCodeBinding) -> &str {
295        &self.codes[&c.c_const]
296    }
297
298    /// The Go type name of the error code whose C constant is `c_const`.
299    pub(crate) fn code_by_const(&self, c_const: &str) -> Option<&str> {
300        self.codes.get(c_const).map(String::as_str)
301    }
302}
303
304/// The factory name of the constructor `c` on the interface `iface` (Go
305/// spelling): `new` on `Store` is `NewStore`, `open` is `OpenStore`.
306pub(crate) fn constructor(iface: &str, c: &FnBinding) -> String {
307    format!("{}{iface}", pascal(&c.name))
308}

380    #[test]
381    fn clashing_function_names_gain_an_underscore() {
382        let model = test_model(
383            r#"
384version: "0.13.0"
385errors:
386  - name: KvErrors
387    codes:
388      - { name: Store, code: 1, message: store }
389      - { name: user_id_missing, code: 2, message: missing }
390interfaces:
391  - name: Store
392    constructors: [{ name: open, params: [] }]
393    methods: [{ name: close, params: [] }]
394functions:
395  - { name: open_store, params: [], return: Store }
396  - { name: check, params: [] }
397  - { name: ping, params: [] }
398"#,
399        );
400        let names = GoNames::new(&model);
401        let d = names.domain("KvErrors");
402        assert_eq!(
403            (d.iface.as_str(), d.unknown.as_str(), d.mapper.as_str()),
404            ("KvError", "UnknownKvError", "wvKvError")
405        );
406        let e = &model.errors[0];
407        assert_eq!(names.code(&e.codes[0]), "StoreError");
408        assert_eq!(names.code(&e.codes[1]), "UserIDMissingError");
409        let f = &model.functions;
410        assert_eq!(names.function(&f[0]), "OpenStore_");
411        assert_eq!(names.function(&f[1]), "Check_");
412        assert_eq!(names.function(&f[2]), "Ping");
413        assert_eq!(names.callable("kv_Store_close"), Some("Store.Close_"));
414        assert_eq!(names.callable("kv_Store_open"), Some("OpenStore"));
415    }
