1//! Callback-interface rendering: the abstract class a consumer implements,
2//! the `Struct` mirroring the C vtable (the `size`/`flags`/`free` header,
3//! then one entry per method), the vtable entries, the dispatcher that
4//! delivers forwarded void methods, and the vtable itself.
5//!
6//! A method that returns a value is a `NativeCallable.isolateLocal`
7//! trampoline: it runs synchronously on the isolate's thread, receives its
8//! borrowed arguments in place, adopts object arguments, and returns its
9//! value to the producer as one C value (a scalar, an optional-scalar
10//! struct, a fresh object reference, or a `{prefix}_alloc` run). The vtable
11//! is flagged thread-affine, so the producer refuses to call such a method
12//! from any other thread (failing it with -4) instead of letting the VM
13//! abort. A thrown exception never unwinds: a method that throws a domain
14//! reports its code and payload, and anything else is reported with a
15//! runtime code. A void method is a `NativeCallable.isolateGroupBound`
16//! forwarder: it may run on any thread, copies its arguments into a message
17//! for the isolate, and returns at once; the dispatcher then runs the
18//! implementation on the event loop.
19

37/// The private global holding a callback interface's vtable.
38pub(crate) fn vtable_var(name: &str) -> String {
39    format!("_{}Vtable", lower_first(&dart_class(name)))

42/// The private dispatcher delivering a callback interface's void methods.
43pub(crate) fn dispatch_fn(name: &str) -> String {
44    format!("_{}Dispatch", lower_first(&dart_class(name)))

52/// The vtable entry function of one method.

57fn lower_first(s: &str) -> String {
58    let mut chars = s.chars();
59    chars.next().map_or_else(String::new, |c| {
60        c.to_lowercase().collect::<String>() + chars.as_str()
61    })
62}

64/// Whether a method returns a value to the producer, which makes it a
65/// synchronous, thread-affine trampoline.
66pub(crate) fn returns_value(m: &CallbackMethodBinding) -> bool {
67    m.ret.is_some()
68}

70/// Render one callback interface.
71pub(crate) fn render_callback_interface(
72    w: &mut CodeWriter,
73    model: &Model,
74    docs: &Docs,
75    cb: &CallbackBinding,
76) {
77    let class = dart_class(&cb.name);
78    render_abstract_class(w, model, docs, cb, &class);
79    render_vtable_struct(w, cb, &class);
80    for m in &cb.methods {
81        if returns_value(m) {
82            render_trampoline(w, model, cb, m, &class);
83        } else {
84            render_forwarder(w, cb, m, &class);
85        }
86    }
87    render_dispatch(w, cb, &class);
88    render_vtable(w, cb, &class);
89}

91/// The class a consumer extends or implements.
92fn render_abstract_class(
93    w: &mut CodeWriter,
94    model: &Model,
95    docs: &Docs,
96    cb: &CallbackBinding,
97    class: &str,
98) {
99    w.blank();
100    docs.write(w, &cb.doc);
101    if cb.doc.is_some() {
102        w.line("///");
103    }
104    for line in [
105        "A callback interface: implement it and pass an instance to any",
106        &format!("function taking a [{class}]. The library keeps the instance alive"),
107        "until the producer releases it.",
108        "",
109        "Methods that return a value run synchronously on this isolate's",
110        "thread, during the call from Dart that led the producer to call them.",
111        "The producer refuses to call one from any other thread: the method",
112        "doesn't run, and the producer sees it fail with",
113        "[NativeException.foreignCode]. An exception a method throws fails the",
114        "producer's call (see each method). Void methods may be called from any",
115        "thread: they're delivered asynchronously on this isolate's event loop,",
116        "in the zone that passed the instance, where a thrown exception is an",
117        "uncaught error of that zone. Object arguments are owned by the",
118        "implementation.",
119    ] {
120        w.line(if line.is_empty() {
121            "///".to_string()
122        } else {
123            format!("/// {line}")
124        });
125    }
126    docs.write_deprecated(w, &cb.deprecated);
127    w.block(format!("abstract class {class} {{"), "}", |w| {
128        for (i, m) in cb.methods.iter().enumerate() {
129            if i > 0 {
130                w.blank();
131            }
132            docs.write(w, &m.doc);
133            if returns_value(m) {
134                let note = match Throws::of(model, &m.error) {
135                    Throws::Domain(exc) => format!(
136                        "Throw a [{exc}] to fail the producer's call with its code and fields."
137                    ),
138                    Throws::Untyped => {
139                        "Throw any exception to fail the producer's call with its message.".into()
140                    }
141                    Throws::Trap => {
142                        "An exception fails the producer's call with [NativeException.foreignCode]."
143                            .into()
144                    }
145                };
146                if m.doc.is_some() {
147                    w.line("///");
148                }
149                w.line(format!("/// {note}"));
150            }
151            docs.write_deprecated(w, &m.deprecated);
152            let params: Vec<String> = m
153                .params
154                .iter()
155                .map(|p| format!("{} {}", dart_type(&p.ty), dart_ident(&p.name)))
156                .collect();
157            let ret = m.ret.as_ref().map_or("void".to_string(), dart_in_type);
158            w.line(format!(
159                "{ret} {}({});",
160                dart_ident(&m.name),
161                params.join(", ")
162            ));
163        }
164    });
165}

172/// The `Struct` whose layout is the C vtable: the `size`, `flags`, and
173/// `free` header, then one function pointer per method in declaration order.
174fn render_vtable_struct(w: &mut CodeWriter, cb: &CallbackBinding, class: &str) {
175    w.blank();
176    for m in &cb.methods {
177        let natives: Vec<String> = m.abi.params.iter().map(|p| ffi_type(&p.ty).0).collect();
178        w.line(format!(
179            "typedef {} = {} Function({});",
180            entry_typedef(cb, m),
181            ffi_type(&m.abi.ret).0,
182            natives.join(", ")
183        ));
184    }
185    w.blank();
186    w.line(format!("// The C `{}` layout.", cb.vtable_tag));
187    w.block(
188        format!("final class {} extends Struct {{", vtable_struct(class)),
189        "}",
190        |w| {
191            w.line("@Uint32()");
192            w.line("external int size;");
193            w.line("@Uint32()");
194            w.line("external int flags;");
195            w.line("external Pointer<NativeFunction<Void Function(Pointer<Void>)>> free;");
196            for m in &cb.methods {
197                w.line(format!(
198                    "external Pointer<NativeFunction<{}>> {};",
199                    entry_typedef(cb, m),
200                    dart_ident(&m.name)
201                ));
202            }
203        },
204    );
205}

222/// Whether an argument is received into a local first: objects (adopted
223/// references) and buffers (which may carry object tokens), so each
224/// reference is owned by a finalizable wrapper even if a later step throws.
225fn owns_references(p: &FieldBinding) -> bool {
226    matches!(
227        crossing(&p.ty),
228        Crossing::Object { .. } | Crossing::Buffer(_)
229    )
230}

232/// A value-returning method's `isolateLocal` trampoline: adopt object
233/// arguments first (so a later failure still leaves each reference owned by
234/// a finalizable wrapper), receive the borrowed ones in place, call the
235/// implementation, and return its value to the producer. A thrown exception
236/// is reported through `out_err` and the zero value is returned.
237fn render_trampoline(
238    w: &mut CodeWriter,
239    model: &Model,
240    cb: &CallbackBinding,
241    m: &CallbackMethodBinding,
242    class: &str,
243) {
244    let throws = &Throws::of(model, &m.error);
245    let ret = m.ret.as_ref().expect("a value-returning method");
246    let (_, dart_ret) = ffi_type(&m.abi.ret);
247    let out_err = out_err(m);
248    w.blank();
249    w.line(format!("// Vtable entry `{}.{}`.", cb.vtable_tag, m.name));
250    w.block(
251        format!(
252            "{dart_ret} {}({}) {{",
253            entry_fn(class, &m.name),
254            entry_params(m)
255        ),
256        "}",
257        |w| {
258            w.line("try {");
259            w.scope(|w| {
260                let mut args = vec![String::new(); m.params.len()];
261                for (i, p) in m
262                    .params
263                    .iter()
264                    .enumerate()
265                    .filter(|(_, p)| owns_references(p))
266                {
267                    let local = format!("_{}Arg", dart_ident(&p.name));
268                    w.line(format!(
269                        "final {local} = {};",
270                        receive(&p.ty, &dart_ident(&p.name), false)
271                    ));
272                    args[i] = local;
273                }
274                for (i, p) in m.params.iter().enumerate() {
275                    if !owns_references(p) {
276                        args[i] = receive(&p.ty, &dart_ident(&p.name), false);
277                    }
278                }
279                w.line(format!(
280                    "final _result = (_callbackTarget(ctx) as {class}).{}({});",
281                    dart_ident(&m.name),
282                    args.join(", ")
283                ));
284                w.line(format!("return {};", hand_over(ret, "_result")));
285            });
286            match throws {
287                Throws::Domain(exc) => {
288                    let payload = if domain_has_fields(model, &m.error) {
289                        format!("{}(e)", payload_fn(exc))
290                    } else {
291                        "null".into()
292                    };
293                    w.line(format!("}} on {exc} catch (e) {{"));
294                    w.scope(|w| {
295                        w.line(format!(
296                            "_failCallback({out_err}, e.code, e.message, {payload});"
297                        ));
298                    });
299                    w.line("} catch (e) {");
300                    w.scope(|w| {
301                        w.line(format!(
302                            "_reportCallbackError({out_err}, NativeException.genericCode, e);"
303                        ));
304                    });
305                }
306                Throws::Untyped => {
307                    w.line("} catch (e) {");
308                    w.scope(|w| {
309                        w.line(format!(
310                            "_reportCallbackError({out_err}, NativeException.genericCode, e);"
311                        ));
312                    });
313                }
314                Throws::Trap => {
315                    w.line("} catch (e) {");
316                    w.scope(|w| {
317                        w.line(format!(
318                            "_reportCallbackError({out_err}, NativeException.foreignCode, e);"
319                        ));
320                    });
321                }
322            }
323            w.line("}");
324            w.line(format!("return {};", zero(ret)));
325        },
326    );
327}

329/// The `_CallbackMessage` cascade step posting the argument `p` of a
330/// forwarded void method.
331fn post_arg(p: &FieldBinding) -> String {
332    let n = dart_ident(&p.name);
333    let scalar = |prim: Prim| match prim {
334        Prim::Bool => "Bool",
335        Prim::F32 | Prim::F64 => "Float64",
336        _ => "Int64",
337    };
338    match crossing(&p.ty) {
339        Crossing::Scalar(Prim::Bool) => format!("..boolean({n})"),
340        Crossing::Scalar(Prim::F32 | Prim::F64) => format!("..float64({n})"),
341        Crossing::Scalar(_) | Crossing::Enum(_) => format!("..int64({n})"),
342        Crossing::OptScalar(prim) => {
343            format!("..maybe{}({n}.has, {n}.value)", scalar(prim))
344        }
345        Crossing::OptEnum(_) => format!("..maybeInt64({n}.has, {n}.value)"),
346        Crossing::Slice(elem) => format!(
347            "..array({n}, {}, _CallbackMessage.{})",
348            elem.size().expect("a typed-array element is a scalar"),
349            typed_data_kind(elem)
350        ),
351        Crossing::Utf8 | Crossing::Bytes | Crossing::Buffer(_) => format!("..run({n})"),
352        Crossing::Object { .. } => format!("..pointer({n})"),
353    }
354}

356/// A void method's `isolateGroupBound` forwarder: it runs outside the
357/// isolate, so it only copies its slots into a message (borrowed runs are
358/// copied by the post) and touches no Dart globals.
359fn render_forwarder(
360    w: &mut CodeWriter,
361    cb: &CallbackBinding,
362    m: &CallbackMethodBinding,
363    class: &str,
364) {
365    w.blank();
366    w.line(format!(
367        "// Vtable entry `{}.{}`: forwarded to the isolate.",
368        cb.vtable_tag, m.name
369    ));
370    w.block(
371        format!("void {}({}) {{", entry_fn(class, &m.name), entry_params(m)),
372        "}",
373        |w| {
374            w.line(format!(
375                "_CallbackMessage(ctx, {}, {})",
376                m.slot,
377                m.params.len()
378            ));
379            w.scope(|w| {
380                w.scope(|w| {
381                    for p in &m.params {
382                        w.line(post_arg(p));
383                    }
384                    w.line("..send();");
385                });
386            });
387        },
388    );
389}

391/// The dispatcher running forwarded void methods on the isolate:
392/// `message` is `[key, method, arguments...]`.
393fn render_dispatch(w: &mut CodeWriter, cb: &CallbackBinding, class: &str) {
394    w.blank();
395    w.block(
396        format!(
397            "void {}(Object impl, int method, List<Object?> message) {{",
398            dispatch_fn(&cb.name)
399        ),
400        "}",
401        |w| {
402            if cb.methods.iter().all(returns_value) {
403                w.line(format!(
404                    "// Every {class} method returns a value: nothing is forwarded."
405                ));
406                return;
407            }
408            w.line(format!("final target = impl as {class};"));
409            w.block("switch (method) {", "}", |w| {
410                for m in cb.methods.iter().filter(|m| !returns_value(m)) {
411                    // Object arguments are adopted first, so a failure while
412                    // decoding the rest still leaves each transferred
413                    // reference owned by a finalizable wrapper.
414                    w.block(format!("case {}: {{", m.slot), "}", |w| {
415                        let is_object =
416                            |p: &FieldBinding| matches!(crossing(&p.ty), Crossing::Object { .. });
417                        for (i, p) in m.params.iter().enumerate().filter(|(_, p)| is_object(p)) {
418                            message_arg(w, p, i + 2);
419                        }
420                        for (i, p) in m.params.iter().enumerate().filter(|(_, p)| !is_object(p)) {
421                            message_arg(w, p, i + 2);
422                        }
423                        let args: Vec<String> =
424                            (0..m.params.len()).map(|i| format!("a{}", i + 2)).collect();
425                        w.line(format!(
426                            "target.{}({});",
427                            dart_ident(&m.name),
428                            args.join(", ")
429                        ));
430                    });
431                }
432            });
433        },
434    );
435}

437/// Decode forwarded argument `i` of `message` into the local `a{i}`.
438fn message_arg(w: &mut CodeWriter, p: &FieldBinding, i: usize) {
439    let field = format!("message[{i}]");
440    let local = format!("a{i}");
441    let value = match crossing(&p.ty) {
442        Crossing::Enum(n) => format!("{}.fromValue({field}! as int)", dart_class(n)),
443        Crossing::Scalar(_) | Crossing::Slice(_) => {
444            format!("{field}! as {}", dart_type(&p.ty))
445        }
446        Crossing::OptEnum(n) => {
447            let raw = format!("{local}Value");
448            w.line(format!("final {raw} = {field} as int?;"));
449            format!("{raw} == null ? null : {}.fromValue({raw})", dart_class(n))
450        }
451        Crossing::OptScalar(_) => format!("{field} as {}", dart_type(&p.ty)),
452        Crossing::Utf8 => format!("utf8.decode({field}! as Uint8List)"),
453        Crossing::Bytes => format!("{field}! as Uint8List"),
454        Crossing::Buffer(t) => format!("_decode({field}! as Uint8List, {})", unpack_fn(t)),
455        Crossing::Object {
456            interface,
457            nullable,
458        } => {
459            let address = format!("{local}Address");
460            w.line(format!("final {address} = {field}! as int;"));
461            let adopt = format!(
462                "{}._(Pointer<Void>.fromAddress({address}))",
463                dart_class(interface)
464            );
465            if nullable {
466                format!("{address} == 0 ? null : {adopt}")
467            } else {
468                adopt
469            }
470        }
471    };
472    w.line(format!("final {local} = {value};"));
473}

475/// The vtable every instance passed to the producer from this isolate
476/// shares; the per-instance state travels in `ctx`. It's flagged
477/// thread-affine: only this isolate's thread may call a value-returning
478/// method.
479fn render_vtable(w: &mut CodeWriter, cb: &CallbackBinding, class: &str) {
480    let strukt = vtable_struct(class);
481    w.blank();
482    w.block(
483        format!("final Pointer<{strukt}> {} = () {{", vtable_var(&cb.name)),
484        "}();",
485        |w| {
486            w.line(format!("final vtable = calloc<{strukt}>();"));
487            w.line("vtable.ref");
488            w.scope(|w| {
489                w.scope(|w| {
490                    w.line(format!("..size = sizeOf<{strukt}>()"));
491                    w.line("..flags = _vtableThreadAffine");
492                    w.line("..free = _callbackFree");
493                    for (i, m) in cb.methods.iter().enumerate() {
494                        let td = entry_typedef(cb, m);
495                        let entry = entry_fn(class, &m.name);
496                        let callable = match &m.ret {
497                            None => format!("NativeCallable<{td}>.isolateGroupBound({entry})"),
498                            Some(ret) => match exceptional_return(ret) {
499                                Some(zero) => format!(
500                                    "NativeCallable<{td}>.isolateLocal({entry}, exceptionalReturn: {zero})"
501                                ),
502                                None => format!("NativeCallable<{td}>.isolateLocal({entry})"),
503                            },
504                        };
505                        let end = if i + 1 == cb.methods.len() { ";" } else { "" };
506                        w.line(format!("..{} = _pin({callable}){end}", dart_ident(&m.name)));
507                    }
508                });
509            });
510            w.line("return vtable;");
511        },
512    );
513}

515/// Whether `error` names a domain with a code that carries fields.
516fn domain_has_fields(model: &Model, error: &ErrorStrategy) -> bool {
517    error.domain().is_some_and(|name| {
518        model
519            .error_domain(name)
520            .codes
521            .iter()
522            .any(|c| !c.fields.is_empty())
523    })
524}

526/// The error domains a value-returning callback method throws whose codes
527/// carry fields, by domain name: each needs an encoder for those fields.
528pub(crate) fn reported_domains(model: &Model) -> std::collections::BTreeSet<String> {
529    model
530        .callbacks
531        .iter()
532        .flat_map(|cb| &cb.methods)
533        .filter(|m| returns_value(m) && domain_has_fields(model, &m.error))
534        .filter_map(|m| m.error.domain().map(str::to_string))
535        .collect()
536}
