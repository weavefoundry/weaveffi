1//! Value-buffer codec emitters: the expression writing and the expression
2//! reading every wire shape, and the composite codecs, one
3//! `_pack_{stem}`/`_unpack_{stem}` pair per distinct optional, list, or map
4//! type that crosses inside a buffer (`_pack_list_Entry`,
5//! `_unpack_map_string_i64`), named by the shared
6//! [`codecs::stem`](crate::codegen::codecs::stem), so no call site inlines a
7//! loop. A record or rich enum's own pair is `_pack_{Name}`.
8//!
9//! Every dispatch goes through [`Ty::wire`], and primitives map onto the
10//! runtime's `read{Prim}`/`write{Prim}` methods by
11//! [`Prim::pascal`](weaveffi_model::ty::Prim::pascal), so this module never
12//! re-derives the wire format.
13//!
14//! An object token carries one strong reference. Writing one calls the
15//! wrapper's `_cloneRef()` so the encoding owns a fresh reference and the
16//! wrapper keeps its own; reading one adopts the pointer into a new wrapper.
17
18use crate::codegen::codecs::{composites, stem};
19use crate::codegen::CodeWriter;
20use weaveffi_model::model::Model;
21use weaveffi_model::ty::{Ty, WireType};
22
23use crate::targets::dart::types::{dart_class, dart_in_type, dart_type};
24
25/// The `_pack_{stem}` function writing a value of the buffered type `t`.
26pub(crate) fn pack_fn(t: &Ty) -> String {
27    format!("_pack_{}", stem(t))
28}
29
30/// The `_unpack_{stem}` function reading a value of the buffered type `t`.
31pub(crate) fn unpack_fn(t: &Ty) -> String {
32    format!("_unpack_{}", stem(t))
33}
34
35/// The Dart expression writing `expr` (a value of `t`) into the writer `w`.
36pub(crate) fn write_expr(t: &Ty, w: &str, expr: &str) -> String {
37    match t.wire() {
38        WireType::Prim(p) => format!("{w}.write{}({expr})", p.pascal()),
39        // A fresh reference the reader adopts.
40        WireType::Object(_) => format!("{w}.writeU64({expr}._cloneRef().address)"),
41        WireType::Enum(_) => format!("{w}.writeI32({expr}.value)"),
42        WireType::User(_) | WireType::Optional(_) | WireType::List(_) | WireType::Map(..) => {
43            format!("{}({w}, {expr})", pack_fn(t))
44        }
45    }
46}
47
48/// The Dart expression reading a value of `t` from the reader `r`; the
49/// inverse of [`write_expr`]. An object token is adopted into a new wrapper.
50pub(crate) fn read_expr(t: &Ty, r: &str) -> String {
51    match t.wire() {
52        WireType::Prim(p) => format!("{r}.read{}()", p.pascal()),
53        WireType::Object(n) => format!(
54            "{}._(Pointer<Void>.fromAddress({r}.readU64()))",
55            dart_class(n)
56        ),
57        WireType::Enum(n) => format!("{}.fromValue({r}.readI32())", dart_class(n)),
58        WireType::User(_) | WireType::Optional(_) | WireType::List(_) | WireType::Map(..) => {
59            format!("{}({r})", unpack_fn(t))
60        }
61    }
62}
63
64/// Render the composite codecs `model` uses, one `_pack`/`_unpack` pair per
65/// distinct type, built on the runtime's generic `writeOptional`,
66/// `writeList`, and `writeMap` (and their readers). A writer accepts the
67/// input flavor of its type (any `List<int>` for `bytes`), so a value of
68/// either flavor encodes.
69pub(crate) fn render_codecs(w: &mut CodeWriter, model: &Model) {
70    for t in composites(model) {
71        let (write, read) = match &t {
72            Ty::Optional(inner) => (
73                format!("w.writeOptional(v, (e) => {})", write_expr(inner, "w", "e")),
74                format!("r.readOptional(() => {})", read_expr(inner, "r")),
75            ),
76            Ty::List(inner) => (
77                format!("w.writeList(v, (e) => {})", write_expr(inner, "w", "e")),
78                format!("r.readList(() => {})", read_expr(inner, "r")),
79            ),
80            Ty::Map(k, v) => (
81                format!(
82                    "w.writeMap(v, (k) => {}, (e) => {})",
83                    write_expr(k, "w", "k"),
84                    write_expr(v, "w", "e")
85                ),
86                format!(
87                    "r.readMap(() => {}, () => {})",
88                    read_expr(k, "r"),
89                    read_expr(v, "r")
90                ),
91            ),
92            _ => unreachable!("only composites are collected"),
93        };
94        w.blank();
95        w.line(format!(
96            "void {}(_BufferWriter w, {} v) =>",
97            pack_fn(&t),
98            dart_in_type(&t)
99        ));
100        w.line(format!("    {write};"));
101        w.blank();
102        w.line(format!(
103            "{} {}(_BufferReader r) =>",
104            dart_type(&t),
105            unpack_fn(&t)
106        ));
107        w.line(format!("    {read};"));
108    }
109}
