1//! The fixed Dart runtime every generated library ships, kept as real Dart
2//! source under `runtime/` and spliced in with `{{PLACEHOLDER}}`
3//! substitution: the loader and load-time checks, the error, call-frame,
4//! and run helpers, the optional-scalar structs, the object base class, the
5//! value-buffer codec, the typed-array helpers, async and cancellation glue,
6//! the callback-interface machinery, and the iterator anchor. Each section
7//! is one part file under `lib/src/runtime/`, and sections an API doesn't
8//! use are left out. The contract rows the loader checks are rendered from
9//! the model into `contracts.dart`.

32/// Where the loader looks for the library before the system search path.
33pub(crate) enum Bundle<'a> {
34    /// Nowhere: the generated (unpackaged) bindings.
35    None,
36    /// `native/<platform>/` under the package root (then the working
37    /// directory), for the packaged library `lib` of the given platforms.
38    Packaged {
39        /// The library's base name (`kvstore` for `libkvstore.dylib`).
40        lib: &'a str,
41        /// The platforms the package bundles.
42        platforms: Vec<Platform>,
43    },
44}
45
46/// Whether the packaged library bundles a prebuilt binary for `platform`:
47/// the loader resolves `native/<platform-id>/` for the desktop matrix only
48/// (Android libraries ship through the app's own packaging, and a `.wasm`
49/// module can't be opened with `DynamicLibrary`).
50pub(crate) fn bundles_platform(platform: Platform) -> bool {
51    platform.is_desktop()
52}

54/// The `dart:ffi` `Abi` constant of a desktop platform.
55fn dart_abi(platform: Platform) -> Option<&'static str> {
56    Some(match (platform.os(), platform.arch()) {
57        (Os::MacOs, Arch::Arm64) => "Abi.macosArm64",
58        (Os::MacOs, Arch::X64) => "Abi.macosX64",
59        (Os::Linux, Arch::Arm64) => "Abi.linuxArm64",
60        (Os::Linux, Arch::X64) => "Abi.linuxX64",
61        (Os::Windows, Arch::Arm64) => "Abi.windowsArm64",
62        (Os::Windows, Arch::X64) => "Abi.windowsX64",
63        _ => return None,
64    })
65}

67/// Replace every `{{KEY}}` in `template` with its value.
68fn fill(template: &str, vars: &[(&str, &str)]) -> String {
69    let mut out = template.to_string();
70    for (key, value) in vars {
71        out = out.replace(&format!("{{{{{key}}}}}"), value);
72    }
73    debug_assert!(!out.contains("{{"), "unfilled placeholder in:\n{out}");
74    out
75}

77/// Every value type that crosses a call boundary in `model`: parameters,
78/// returns, async results, iterator elements, and callback method
79/// parameters and returns.
80fn boundary_types(model: &Model) -> impl Iterator<Item = &Ty> {
81    let callables = model.callables().flat_map(|f| {
82        f.params
83            .iter()
84            .filter_map(|p| p.ty.value())
85            .chain(f.ret.as_ref().map(|r| r.elem()))
86    });
87    let methods = model
88        .callbacks
89        .iter()
90        .flat_map(|cb| &cb.methods)
91        .flat_map(|m| m.params.iter().map(|p| &p.ty).chain(&m.ret));
92    callables.chain(methods)
93}
94
95/// Whether any value in `model` crosses as `crossing` says, in any
96/// position.
97fn crosses(model: &Model, test: impl Fn(Crossing) -> bool) -> bool {
98    boundary_types(model).any(|ty| test(crossing(ty)))
99}

101/// The runtime sections `model` needs, as `(file name, source)` pairs in
102/// library order; `package` is the Dart package name and `bundle` says
103/// where the loader looks first.
104pub(crate) fn runtime_parts(
105    model: &Model,
106    package: &str,
107    bundle: &Bundle,
108) -> Vec<(String, String)> {
109    let identity = &model.identity;
110    let (macos, linux, windows) = identity.library_files();
111    let prefix = identity.prefix.as_str();
112    let env = identity.library_env_var();
113    let library = dart_str_literal(&identity.library);
114    let name = dart_str_literal(&identity.name);
115    let abi = ABI_VERSION.to_string();
116    let common = [
117        ("PREFIX", prefix),
118        ("LIBRARY_ENV", env.as_str()),
119        ("LIBRARY", library.as_str()),
120        ("NAME", name.as_str()),
121    ];
122
123    let bundled_lookup = match bundle {
124        Bundle::None => "",
125        Bundle::Packaged { .. } => {
126            "  final bundled = _openBundled();\n  if (bundled != null) return bundled;\n"
127        }
128    };
129    let (macos, linux, windows) = (
130        dart_str_literal(&macos),
131        dart_str_literal(&linux),
132        dart_str_literal(&windows),
133    );
134    let mut vars = common.to_vec();
135    vars.extend([
136        ("MACOS_FILE", macos.as_str()),
137        ("LINUX_FILE", linux.as_str()),
138        ("WINDOWS_FILE", windows.as_str()),
139        ("BUNDLED_LOOKUP", bundled_lookup),
140        ("ABI_VERSION", abi.as_str()),
141    ]);
142    let mut parts = vec![
143        ("loader.dart".to_string(), fill(LOADER, &vars)),
144        ("contracts.dart".to_string(), render_contracts(model)),
145    ];
146    if let Bundle::Packaged { lib, platforms } = bundle {
147        let cases: String = platforms
148            .iter()
149            .filter_map(|p| {
150                let abi = dart_abi(*p)?;
151                Some(format!(
152                    "      {abi} => 'native/{}/{}',\n",
153                    p.id(),
154                    dart_str_literal(&p.lib_filename(lib))
155                ))
156            })
157            .collect();
158        let package = dart_str_literal(package);
159        let mut vars = common.to_vec();
160        vars.extend([
161            ("BUNDLED_CASES", cases.as_str()),
162            ("PACKAGE", package.as_str()),
163        ]);
164        parts.push(("bundled.dart".into(), fill(BUNDLED, &vars)));
165    }
166
167    let options = crosses(model, |c| {
168        matches!(c, Crossing::OptScalar(_) | Crossing::OptEnum(_))
169    });
170    let arrays = crosses(model, |c| matches!(c, Crossing::Slice(_)));
171    let sections = [
172        (true, "core.dart", CORE),
173        (options, "options.dart", OPTIONS),
174        (!model.interfaces.is_empty(), "object.dart", OBJECT),
175        (model.has_buffers(), "codec.dart", CODEC),
176        (arrays, "arrays.dart", ARRAYS),
177        (model.has_async(), "async.dart", ASYNC),
178        (model.has_cancellable(), "cancel.dart", CANCEL),
179        (!model.callbacks.is_empty(), "callbacks.dart", CALLBACKS),
180        (model.has_iterators(), "iterator.dart", ITERATOR),
181    ];
182    for (_, file, source) in sections.iter().filter(|(used, ..)| *used) {
183        parts.push(((*file).to_string(), fill(source, &common)));
184    }
185    parts
186}

188/// The contract rows the loader checks: each declaration these bindings
189/// were generated with, as its contract function, the hash it must return,
190/// and its path, with its canonical signature as a comment.
191fn render_contracts(model: &Model) -> String {
192    let mut w = CodeWriter::two_space();
193    w.line("// ── Contract ──");
194    w.blank();
195    w.line("/// Every declaration these bindings were generated with: its contract");
196    w.line("/// function, the hash of its canonical signature, and its path.");
197    w.line("const List<(String, int, String)> _contracts = [");
198    w.scope(|w| {
199        for row in rows(model) {
200            w.line(format!("// {}", row.signature));
201            w.line(format!(
202                "('{}', {}, '{}'),",
203                row.symbol,
204                hex(row.hash),
205                dart_str_literal(&row.path)
206            ));
207        }
208    });
209    w.line("];");
210    w.finish()
211}
