1//! How one value crosses, per [`Crossing`]: the Dart expression staging a
2//! parameter, receiving a returned value or a callback argument, and handing
3//! a callback's return to the producer.
4//!
5//! Every value is one C value in every position, so these four functions are
6//! the whole call layer's marshalling; [`calls`](super::calls) and
7//! [`callbacks`](super::callbacks) only decide where each value goes.
8
9use weaveffi_model::ty::Ty;
10
11use crate::codegen::values::{crossing, Crossing};
12use crate::targets::dart::codec::{pack_fn, unpack_fn};
13use crate::targets::dart::types::{dart_class, opt_fn, slice_fn, zero_literal};

15/// `true` when staging a value of `ty` does work that can fail (encoding,
16/// copying into the arena, borrowing an object), so a wrapper stages it into
17/// a local before registering any callback argument.
18pub(crate) fn stages(ty: &Ty) -> bool {
19    !matches!(
20        crossing(ty),
21        Crossing::Scalar(_) | Crossing::Enum(_) | Crossing::OptScalar(_) | Crossing::OptEnum(_)
22    )
23}
24
25/// The C value of the parameter `name` (of type `ty`), borrowed by the
26/// producer for one call: runs are copied into `arena`, and an object's
27/// pointer is borrowed until the arena is released.
28pub(crate) fn stage(ty: &Ty, name: &str, arena: &str) -> String {
29    match crossing(ty) {
30        Crossing::Scalar(_) => name.to_string(),
31        Crossing::Enum(_) => format!("{name}.value"),
32        Crossing::OptScalar(p) => format!("{}({name})", opt_fn(p)),
33        Crossing::OptEnum(_) => format!("_optI32({name}?.value)"),
34        Crossing::Utf8 => format!("_stageText({arena}, {name})"),
35        Crossing::Bytes => format!("_stage({arena}, {name})"),
36        Crossing::Slice(p) => format!("{}({arena}, {name})", slice_fn("stage", p)),
37        Crossing::Buffer(t) => format!("_stage({arena}, _encode({name}, {}))", pack_fn(t)),
38        Crossing::Object {
39            nullable: false, ..
40        } => format!("_borrow({arena}, {name})"),
41        Crossing::Object { nullable: true, .. } => {
42            format!("{name} == null ? nullptr : _borrow({arena}, {name})")
43        }
44    }
45}

47/// The Dart value of the arriving C value `expr` (of type `ty`), which may
48/// be evaluated more than once. An `owned` run (a return, an async result,
49/// an iterator element) is released once decoded; a borrowed one (a
50/// callback argument) is copied. An object reference is always adopted.
51pub(crate) fn receive(ty: &Ty, expr: &str, owned: bool) -> String {
52    let pick = |take: &str, copy: &str| {
53        if owned {
54            take.to_string()
55        } else {
56            copy.to_string()
57        }
58    };
59    match crossing(ty) {
60        Crossing::Scalar(_) => expr.to_string(),
61        Crossing::Enum(n) => format!("{}.fromValue({expr})", dart_class(n)),
62        Crossing::OptScalar(_) => format!("{expr}.has ? {expr}.value : null"),
63        Crossing::OptEnum(n) => {
64            format!(
65                "{expr}.has ? {}.fromValue({expr}.value) : null",
66                dart_class(n)
67            )
68        }
69        Crossing::Utf8 => format!("{}({expr})", pick("_takeText", "_readText")),
70        Crossing::Bytes => format!("{}({expr})", pick("_takeRun", "_copyRun")),
71        Crossing::Slice(p) => format!("{}({expr})", slice_fn(&pick("take", "copy"), p)),
72        Crossing::Buffer(t) if owned => format!("_takeDecoded({expr}, {})", unpack_fn(t)),
73        Crossing::Buffer(t) => format!("_decode(_viewRun({expr}), {})", unpack_fn(t)),
74        Crossing::Object {
75            interface,
76            nullable,
77        } => {
78            let class = dart_class(interface);
79            if nullable {
80                format!("{expr} == nullptr ? null : {class}._({expr})")
81            } else {
82                format!("{class}._({expr})")
83            }
84        }
85    }
86}

88/// The C value a callback method returns for the Dart value `expr`: a run
89/// the producer adopts (allocated with `{p}_alloc`), or a fresh object
90/// reference.
91pub(crate) fn hand_over(ty: &Ty, expr: &str) -> String {
92    match crossing(ty) {
93        Crossing::Scalar(_) => expr.to_string(),
94        Crossing::Enum(_) => format!("{expr}.value"),
95        Crossing::OptScalar(p) => format!("{}({expr})", opt_fn(p)),
96        Crossing::OptEnum(_) => format!("_optI32({expr}?.value)"),
97        Crossing::Utf8 => format!("_handOver(utf8.encode({expr}))"),
98        Crossing::Bytes => format!("_handOver({expr})"),
99        Crossing::Slice(p) => format!("{}({expr})", slice_fn("handOver", p)),
100        Crossing::Buffer(t) => format!("_handOver(_encode({expr}, {}))", pack_fn(t)),
101        Crossing::Object {
102            nullable: false, ..
103        } => format!("{expr}._cloneRef()"),
104        Crossing::Object { nullable: true, .. } => format!("{expr}?._cloneRef() ?? nullptr"),
105    }
106}

108/// The zero C value of `ty`: what a callback method returns when its
109/// implementation failed.
110pub(crate) fn zero(ty: &Ty) -> String {
111    match crossing(ty) {
112        Crossing::Scalar(p) => zero_literal(p).to_string(),
113        Crossing::Enum(_) => "0".into(),
114        Crossing::OptScalar(p) => format!("{}(null)", opt_fn(p)),
115        Crossing::OptEnum(_) => "_optI32(null)".into(),
116        Crossing::Utf8 | Crossing::Bytes | Crossing::Slice(_) | Crossing::Buffer(_) => {
117            "_emptyRun()".into()
118        }
119        Crossing::Object { .. } => "nullptr".into(),
120    }
121}

123/// The `exceptionalReturn` a callback trampoline returning `ty` declares:
124/// its zero for a scalar or enum, and none for a struct or pointer (which
125/// `dart:ffi` doesn't accept one for).
126pub(crate) fn exceptional_return(ty: &Ty) -> Option<String> {
127    match crossing(ty) {
128        Crossing::Scalar(p) => Some(zero_literal(p).to_string()),
129        Crossing::Enum(_) => Some("0".into()),
130        _ => None,
131    }
132}

134/// `true` when the C value of `ty` is a struct, read through a pointer with
135/// `.ref` rather than `.value`.
136pub(crate) fn is_struct(ty: &Ty) -> bool {
137    !matches!(
138        crossing(ty),
139        Crossing::Scalar(_) | Crossing::Enum(_) | Crossing::Object { .. }
140    )
141}

143#[cfg(test)]
144mod tests {
145    use super::*;
146    use weaveffi_model::ty::Prim;
147
148    #[test]
149    fn every_crossing_has_a_dart_spelling() {
150        let opt = |t| Ty::Optional(Box::new(t));
151        let store = Ty::Interface("Store".into());
152        assert_eq!(stage(&opt(Ty::Prim(Prim::U32)), "n", "a"), "_optU32(n)");
153        assert_eq!(stage(&Ty::Prim(Prim::String), "s", "a"), "_stageText(a, s)");
154        assert_eq!(
155            stage(&opt(store.clone()), "o", "a"),
156            "o == null ? nullptr : _borrow(a, o)"
157        );
158        assert_eq!(
159            receive(&opt(Ty::Enum("Kind".into())), "r", true),
160            "r.has ? Kind.fromValue(r.value) : null"
161        );
162        assert_eq!(
163            receive(&Ty::List(Box::new(Ty::Prim(Prim::F64))), "r", false),
164            "_copyF64s(r)"
165        );
166        assert_eq!(
167            receive(&Ty::Record("Entry".into()), "r", true),
168            "_takeDecoded(r, _unpack_Entry)"
169        );
170        assert_eq!(hand_over(&opt(store), "v"), "v?._cloneRef() ?? nullptr");
171        assert_eq!(zero(&Ty::Prim(Prim::Bytes)), "_emptyRun()");
172        assert_eq!(exceptional_return(&Ty::Prim(Prim::F32)), Some("0.0".into()));
173        assert!(is_struct(&opt(Ty::Prim(Prim::I64))) && !is_struct(&Ty::Prim(Prim::I64)));
174    }
175}
