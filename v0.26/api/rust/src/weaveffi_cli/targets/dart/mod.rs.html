1//! Dart (`dart:ffi`) binding generator.
2//!
3//! Emits a standalone Dart package: `pubspec.yaml`, `README.md`, and one
4//! library, `lib/{package}.dart`, whose parts live under `lib/src/`: the
5//! runtime sections under `lib/src/runtime/` and every declaration of the
6//! library's one namespace in `lib/src/api.dart`. The package is named after
7//! the identity's C prefix and loads the identity's library, honoring the
8//! `{PREFIX}_LIBRARY` override. On first use the library's ABI revision and
9//! every declaration's contract function are verified, and a mismatch
10//! throws a catchable `NativeLibraryException`.
11//!
12//! Every value crosses as one C value: scalars and C-style enums by value,
13//! optional scalars as `_Opt{Prim}` structs, strings, bytes, typed arrays,
14//! and value buffers as the `_Run` struct (all by value), and objects as
15//! pointers. Records and rich enums are value types: plain Dart classes (a
16//! sealed hierarchy for a rich enum) with value equality that cross the ABI
17//! serialized in the value buffer format, and numeric lists cross as typed
18//! arrays. Interfaces are wrapper classes owning one strong reference,
19//! released by `dispose()` or a `NativeFinalizer`, and guarded so a
20//! `dispose()` during an in-flight call defers the release until the call
21//! returns. Every call runs in its own pooled frame of native scratch, so
22//! calls nested in callbacks never share state. Async functions return a
23//! `Future` completed by a `NativeCallable.listener`; cancellable ones take
24//! a `CancelToken`. Iterators are lazy `Iterable`s. Callback interfaces are
25//! abstract classes passed as a `{ctx, vtable}` struct: value-returning
26//! methods are isolate-local trampolines on a thread-affine vtable (the
27//! producer refuses to call them off the isolate's thread), and void
28//! methods are isolate-group-bound forwarders that may run on any thread
29//! and deliver the call on the isolate's event loop.

64/// The lowest Dart SDK the generated bindings support:
65/// `NativeCallable.isolateGroupBound`, which delivers void callback methods
66/// from any thread, first shipped in Dart 3.10.
67pub const MIN_DART_SDK: &str = "3.10.0";

69/// The `ignore_for_file` line every part carries: private codec names use
70/// the shared composite stems (`_pack_list_i32`), and a runtime section an
71/// API doesn't fully use leaves private helpers unreferenced.
72const IGNORES: &str = "// ignore_for_file: camel_case_types, non_constant_identifier_names, \
73     unused_element, unused_field";

78/// Per-target configuration for [`DartGenerator`].
79#[derive(Debug, Clone, Serialize, Deserialize)]
80#[serde(default, deny_unknown_fields)]
81pub struct DartConfig {
82    /// Dart package name: the pubspec `name` and the library file
83    /// `lib/{name}.dart`. Defaults to the identity's C prefix.
84    pub name: Option<String>,
85    /// The pubspec's Dart SDK constraint (default `>=3.10.0 <4.0.0`; see
86    /// [`MIN_DART_SDK`]).
87    pub sdk: String,
88}
89
90impl Default for DartConfig {
91    fn default() -> Self {
92        Self {
93            name: None,
94            sdk: format!(">={MIN_DART_SDK} <4.0.0"),
95        }
96    }
97}

120/// Wrap a part's body: the prelude, the analyzer ignores, the `part of`
121/// directive (relative to the part), and the trailer.
122fn part_file(package: &str, rel: &Utf8PathBuf, body: &str) -> String {
123    let depth = rel.components().count() - 1;
124    let file = rel.file_name().expect("a part file name");
125    let mut out = render_prelude(CommentStyle::DoubleSlash);
126    out.push_str(IGNORES);
127    out.push('\n');
128    out.push_str(&format!(
129        "part of '{}{package}.dart';\n\n",
130        "../".repeat(depth)
131    ));
132    out.push_str(body.trim_end());
133    out.push_str("\n\n");
134    out.push_str(&render_trailer(CommentStyle::DoubleSlash, file));
135    out
136}

138/// Every declaration: the error domains, enums, records, callback
139/// interfaces, interfaces, and functions, in that order.
140fn render_api(model: &Model) -> String {
141    // Synchronous calls are leaf calls only when no callback interface
142    // exists, so no call can re-enter Dart.
143    let leaf = model.callbacks.is_empty();
144    let docs = Docs::new(model);
145    let reported = reported_domains(model);
146    let mut w = CodeWriter::two_space();
147    w.line(format!("// ── The `{}` API ──", model.identity.name));
148    if let Some(doc) = model
149        .doc
150        .as_deref()
151        .map(str::trim)
152        .filter(|d| !d.is_empty())
153    {
154        w.line("//");
155        for line in doc.lines() {
156            w.line(format!("// {line}").trim_end());
157        }
158    }
159    for table in error_tables(model, "Exception") {
160        render_error(&mut w, &docs, &table, reported.contains(&table.domain.name));
161    }
162    for e in &model.enums {
163        render_enum(&mut w, &docs, e);
164    }
165    for r in &model.records {
166        render_record(&mut w, &docs, r);
167    }
168    for cb in &model.callbacks {
169        render_callback_interface(&mut w, model, &docs, cb);
170    }
171    for i in &model.interfaces {
172        render_interface(&mut w, model, &docs, i, leaf);
173    }
174    for f in &model.functions {
175        emit_bindings(&mut w, f, leaf);
176        emit_wrapper(
177            &mut w,
178            model,
179            &docs,
180            f,
181            &DartDecl::TopLevel,
182            &dart_ident(&f.name),
183        );
184    }
185    w.finish()
186}

188/// Render the library: `lib/{package}.dart` and its parts, as paths
189/// relative to the package root.
190fn render_library(model: &Model, package: &str, bundle: &Bundle) -> Vec<(Utf8PathBuf, String)> {
191    let mut parts: Vec<(Utf8PathBuf, String)> = Vec::new();
192    for (file, body) in runtime_parts(model, package, bundle) {
193        parts.push((Utf8PathBuf::from(format!("src/runtime/{file}")), body));
194    }
195    let mut codecs = CodeWriter::two_space();
196    render_codecs(&mut codecs, model);
197    if !codecs.is_empty() {
198        let body = format!(
199            "// ── Composite codecs ──\n// One writer and reader per optional, list, or map type that crosses\n// inside a value buffer.\n{}",
200            codecs.finish()
201        );
202        parts.push(("src/runtime/composites.dart".into(), body));
203    }
204    parts.push((API_PART.into(), render_api(model)));
205
206    let mut files: Vec<(Utf8PathBuf, String)> = parts
207        .iter()
208        .map(|(rel, body)| {
209            (
210                Utf8PathBuf::from(format!("lib/{rel}")),
211                part_file(package, rel, body),
212            )
213        })
214        .collect();
215    let main = render_main(model, package, bundle, parts.iter().map(|(rel, _)| rel));
216    files.insert(0, (Utf8PathBuf::from(format!("lib/{package}.dart")), main));
217    files
218}

220/// `lib/{package}.dart`: the library's doc comment, its imports, and its
221/// parts.
222fn render_main<'a>(
223    model: &Model,
224    package: &str,
225    bundle: &Bundle,
226    parts: impl Iterator<Item = &'a Utf8PathBuf>,
227) -> String {
228    let mut w = CodeWriter::two_space();
229    w.raw(render_prelude(CommentStyle::DoubleSlash));
230    let identity = &model.identity;
231    if let Some(description) = identity.description.as_deref().map(str::trim) {
232        for line in description.lines() {
233            w.line(format!("/// {line}").trim_end());
234        }
235        w.line("///");
236    }
237    w.line(format!(
238        "/// `dart:ffi` bindings for the `{}` native library.",
239        identity.name
240    ));
241    w.line("///");
242    w.line(format!(
243        "/// The library loads on first use (set `{}` to its path to pick a",
244        identity.library_env_var()
245    ));
246    w.line("/// build); a library that can't be loaded or doesn't match these");
247    w.line("/// bindings throws a [NativeLibraryException].");
248    w.line("library;");
249    w.blank();
250    let callbacks = !model.callbacks.is_empty();
251    let bundled = matches!(bundle, Bundle::Packaged { .. });
252    if model.has_async() || callbacks {
253        w.line("import 'dart:async';");
254    }
255    w.line("import 'dart:convert';");
256    w.line("import 'dart:ffi';");
257    if bundled {
258        w.line("import 'dart:io' show Directory, File, Platform;");
259    } else {
260        w.line("import 'dart:io' show Platform;");
261    }
262    if callbacks || bundled {
263        w.line("import 'dart:isolate';");
264    }
265    w.line("import 'dart:typed_data';");
266    w.blank();
267    w.line("import 'package:ffi/ffi.dart';");
268    w.blank();
269    for rel in parts {
270        w.line(format!("part '{}';", dart_str_literal(rel.as_str())));
271    }
272    w.blank();
273    w.raw(render_trailer(
274        CommentStyle::DoubleSlash,
275        &format!("{package}.dart"),
276    ));
277    w.finish()
278}

280impl Target for DartGenerator {
281    fn name(&self) -> &'static str {
282        "dart"
283    }
284
285    fn fixed_files(&self) -> &'static [&'static str] {
286        &[
287            "README.md",
288            "pubspec.yaml",
289            "loader.dart",
290            "core.dart",
291            "options.dart",
292            "object.dart",
293            "codec.dart",
294            "arrays.dart",
295            "async.dart",
296            "cancel.dart",
297            "callbacks.dart",
298            "iterator.dart",
299        ]
300    }
301
302    fn render(&self, model: &Model) -> Vec<OutputFile> {
303        let config = &self.config;
304        let package = config.package_name(model);
305        let mut files: Vec<OutputFile> = render_library(model, &package, &Bundle::None)
306            .into_iter()
307            .map(|(path, contents)| OutputFile::new(path, contents))
308            .collect();
309        files.push(OutputFile::new(
310            "pubspec.yaml",
311            render_pubspec(&model.identity, &package, &config.sdk),
312        ));
313        files.push(OutputFile::new(
314            "README.md",
315            render_readme(&model.identity, &package, &config.sdk),
316        ));
317        files
318    }
319
320    /// A pub package directory at `dart/{package}/` with the desktop
321    /// libraries under `native/<platform>/`, which the loader tries (relative
322    /// to the package, then to the working directory) before the system
323    /// search path.
324    fn package(&self, model: &Model, ctx: &PackageContext<'_>) -> Result<Vec<Artifact>> {
325        let config = &self.config;
326        let natives = per_platform_libraries(ctx.binaries, "native", bundles_platform);
327        if natives.is_empty() {
328            return Ok(Vec::new());
329        }
330        let package = config.package_name(model);
331        let bundle = Bundle::Packaged {
332            lib: &ctx.binaries.lib_name,
333            platforms: ctx
334                .binaries
335                .platforms()
336                .filter(|p| bundles_platform(*p))
337                .collect(),
338        };
339        let mut files: Vec<PackagedFile> = render_library(model, &package, &bundle)
340            .into_iter()
341            .map(|(path, contents)| PackagedFile::text(path, contents))
342            .collect();
343        files.push(PackagedFile::text(
344            "pubspec.yaml",
345            render_pubspec(&model.identity, &package, &config.sdk),
346        ));
347        files.push(PackagedFile::text(
348            "README.md",
349            render_packaged_readme(&model.identity, ctx),
350        ));
351        files.extend(natives);
352        Ok(vec![Artifact::directory(format!("dart/{package}"), files)])
353    }
