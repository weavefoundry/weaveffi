1//! Package-surface renderers: the `.csproj` (which carries the NuGet
2//! metadata `dotnet pack` needs) and the README.
3//!
4//! Every interpolated user string routes through the shared
5//! [`xml_escape`], so markup-sensitive
6//! characters can't corrupt the XML.
7
8use crate::manifest::xml_escape;
9use crate::package::PackageContext;
10use crate::utils::{render_prelude, render_trailer, CommentStyle};
11use weaveffi_model::pkg::Identity;
12
13/// The names a generated .NET project is built from.
14pub(crate) struct Project<'a> {
15    /// The library identity.
16    pub identity: &'a Identity,
17    /// The C# namespace, which is also the assembly name and NuGet id.
18    pub namespace: &'a str,
19    /// The native library base name the bindings load.
20    pub library: &'a str,
21}
22
23/// Render the `.csproj`. `extra` is spliced in after the main property group
24/// (the native-asset item group of a packaged build).
25pub(crate) fn render_csproj(p: &Project<'_>, filename: &str, extra: &str) -> String {
26    let id = p.identity;
27    let ns = xml_escape(p.namespace);
28    let mut meta = format!(
29        "    <Version>{}</Version>\n    <Description>{}</Description>\n",
30        xml_escape(&id.version),
31        xml_escape(&id.description_or_default())
32    );
33    if !id.authors.is_empty() {
34        meta.push_str(&format!(
35            "    <Authors>{}</Authors>\n",
36            xml_escape(&id.authors.join(", "))
37        ));
38    }
39    if let Some(license) = &id.license {
40        meta.push_str(&format!(
41            "    <PackageLicenseExpression>{}</PackageLicenseExpression>\n",
42            xml_escape(license)
43        ));
44    }
45    if let Some(url) = id.homepage.as_ref().or(id.repository.as_ref()) {
46        meta.push_str(&format!(
47            "    <PackageProjectUrl>{}</PackageProjectUrl>\n",
48            xml_escape(url)
49        ));
50    }
51    if let Some(url) = &id.repository {
52        meta.push_str(&format!(
53            "    <RepositoryUrl>{}</RepositoryUrl>\n",
54            xml_escape(url)
55        ));
56    }
57    format!(
58        r#"{prelude}<Project Sdk="Microsoft.NET.Sdk">
59
60  <PropertyGroup>
61    <TargetFramework>net8.0</TargetFramework>
62    <AssemblyName>{ns}</AssemblyName>
63    <RootNamespace>{ns}</RootNamespace>
64    <PackageId>{ns}</PackageId>
65{meta}    <Nullable>enable</Nullable>
66    <AllowUnsafeBlocks>true</AllowUnsafeBlocks>
67    <IsAotCompatible>true</IsAotCompatible>
68    <GenerateDocumentationFile>true</GenerateDocumentationFile>
69    <!-- The bindings themselves reference the API's deprecated items, and
70         carry the IDL's docs, which needn't cover every member. -->
71    <NoWarn>$(NoWarn);CS0618;CS1573;CS1591</NoWarn>
72  </PropertyGroup>
73{extra}
74</Project>
75
76{trailer}"#,
77        prelude = render_prelude(CommentStyle::Xml),
78        trailer = render_trailer(CommentStyle::Xml, filename),
79    )
80}

82/// The `<ItemGroup>` shipping every bundled native library under
83/// `runtimes/<rid>/native/`, where NuGet resolves it at restore time.
84pub(crate) const NATIVE_ASSETS: &str = "  <PropertyGroup>
85    <PackageReadmeFile>README.md</PackageReadmeFile>
86  </PropertyGroup>
87  <ItemGroup>
88    <None Include=\"README.md\" Pack=\"true\" PackagePath=\"\\\" />
89    <Content Include=\"runtimes/**\" Pack=\"true\" PackagePath=\"runtimes/\">
90      <CopyToOutputDirectory>PreserveNewest</CopyToOutputDirectory>
91    </Content>
92  </ItemGroup>
93";

95/// Render the README. A packaged build lists its bundled runtime
96/// identifiers.
97pub(crate) fn render_readme(
98    p: &Project<'_>,
99    library_class: &str,
100    ctx: Option<&PackageContext>,
101) -> String {
102    let ns = p.namespace;
103    let env = p.identity.library_env_var();
104    let (mac, linux, win) = p.identity.library_files();
105    let mut out = render_prelude(CommentStyle::Xml);
106    out.push_str(&format!(
107        "# {ns} (.NET)\n\n\
108         {desc}\n\n\
109         The `{ns}` namespace binds the `{lib}` native library through\n\
110         source-generated P/Invoke (`[LibraryImport]`), so it's trim- and\n\
111         AOT-friendly.\n\n",
112        desc = p.identity.description_or_default(),
113        lib = p.library,
114    ));
115    match ctx {
116        Some(ctx) => {
117            let rids: Vec<String> = ctx
118                .binaries
119                .platforms()
120                .filter_map(|pl| pl.nuget_rid())
121                .map(|rid| format!("- `{rid}`"))
122                .collect();
123            out.push_str(&format!(
124                "## Install\n\n```bash\ndotnet add package {ns}\n```\n\n\
125                 The package bundles the native library for these runtimes:\n\n{}\n\n",
126                rids.join("\n")
127            ));
128        }
129        None => {
130            out.push_str(&format!(
131                "## Build\n\n```bash\ndotnet build\n```\n\n\
132                 Reference this project from yours (`dotnet add reference {ns}.csproj`)\n\
133                 or pack it (`dotnet pack -c Release`).\n\n"
134            ));
135        }
136    }
137    out.push_str(&format!(
138        "## Loading the native library\n\n\
139         The bindings load `{lib}` with the platform's normal rules\n\
140         (`{mac}`, `{linux}`, or `{win}` next to the app or on the\n\
141         library search path). Set `{env}` to a full path to load a\n\
142         specific file instead.\n\n\
143         The first call checks that the loaded library matches these\n\
144         bindings (its ABI revision and every declaration's contract) and\n\
145         throws `NativeLoadException` when it doesn't, as does every later\n\
146         call. Call `{library_class}.Check()` at startup to run the check\n\
147         up front.\n\n",
148        lib = p.library,
149    ));
150    out.push_str(&render_trailer(CommentStyle::Xml, "README.md"));
151    out
152}
