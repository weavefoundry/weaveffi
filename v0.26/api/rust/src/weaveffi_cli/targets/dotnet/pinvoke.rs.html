1//! The generated half of `NativeMethods`: the contract rows checked at load
2//! and one `[LibraryImport]` per C symbol the API declares.
3//!
4//! Every slot is blittable (`FfiRun`, `FfiOpt*`, `FfiCallback`, scalars,
5//! `byte` for a C `bool`, `FfiError*`, enums, function pointers) except
6//! object slots, which take the interface's `SafeHandle` subclass: the
7//! source-generated stub adds a reference to the handle for the duration of
8//! the call, so a wrapper that's disposed or collected mid-call can't free
9//! the native object under it.
10
11use weaveffi_model::abi::{AbiFn, AbiParam, CType};
12use weaveffi_model::model::{FnBinding, InterfaceBinding, Model};
13
14use crate::codegen::contract::{hex, rows};
15use crate::codegen::CodeWriter;
16use crate::targets::dotnet::types::{cs_ctype, cs_str, fn_pointer_type, safe_cs_name};
17
18/// The C# type of an interface's `SafeHandle` subclass.
19fn handle_type(ns: &str, interface: &str) -> String {
20    format!("global::{ns}.{interface}.NativeHandle")
21}
22
23/// One parameter declaration for `slot`. A borrowed object (the receiver,
24/// or an object parameter) takes its `SafeHandle`, and an async launcher's
25/// completion takes its function-pointer type.
26fn slot_decl(ns: &str, slot: &AbiParam, f: Option<&FnBinding>) -> String {
27    let name = safe_cs_name(&slot.name);
28    let ty = match &slot.ty {
29        CType::Ptr {
30            konst: true,
31            pointee,
32        } => match pointee.as_ref() {
33            CType::Object(interface) => handle_type(ns, interface),
34            _ => cs_ctype(ns, &slot.ty),
35        },
36        CType::Named(_) => match f.and_then(FnBinding::async_binding) {
37            Some(a) => fn_pointer_type(ns, &a.callback.params, &CType::Void),
38            None => cs_ctype(ns, &slot.ty),
39        },
40        ty => cs_ctype(ns, ty),
41    };
42    format!("{ty} {name}")
43}

45/// Emit one `[LibraryImport]` declaration.
46fn import(w: &mut CodeWriter, symbol: &str, ret: &str, params: &[String]) {
47    w.line(format!(
48        "[LibraryImport(LibName, EntryPoint = \"{symbol}\")]"
49    ));
50    w.line("[UnmanagedCallConv(CallConvs = new[] { typeof(CallConvCdecl) })]");
51    w.line(format!(
52        "internal static partial {ret} {symbol}({});",
53        params.join(", ")
54    ));
55    w.blank();
56}
57
58/// Emit the import for one lowered function.
59fn import_fn(w: &mut CodeWriter, ns: &str, abi: &AbiFn, f: Option<&FnBinding>) {
60    let params: Vec<String> = abi.params.iter().map(|s| slot_decl(ns, s, f)).collect();
61    import(w, &abi.symbol, &cs_ctype(ns, &abi.ret), &params);
62}
63
64/// Emit every import behind one callable: its entry point (the sync
65/// symbol, async launcher, or iterator launcher) and, for an iterator,
66/// `_next` (taking the iterator's `SafeHandle`) and `_destroy`.
67fn render_callable(w: &mut CodeWriter, ns: &str, f: &FnBinding) {
68    import_fn(w, ns, &f.abi, Some(f));
69    if let Some(it) = f.iterator() {
70        let mut params = vec!["FfiIteratorHandle iter".to_string()];
71        params.extend(it.next.params[1..].iter().map(|s| slot_decl(ns, s, None)));
72        import(w, &it.next.symbol, &cs_ctype(ns, &it.next.ret), &params);
73        import(w, &it.destroy_symbol, "void", &["IntPtr iter".to_string()]);
74    }
75}
76
77/// The imports behind one interface: `_clone`, `_destroy`, and every member.
78fn render_interface(w: &mut CodeWriter, ns: &str, i: &InterfaceBinding) {
79    import(
80        w,
81        &i.clone_symbol,
82        "IntPtr",
83        &[format!("{} self", handle_type(ns, &i.name))],
84    );
85    import(w, &i.destroy_symbol, "void", &["IntPtr self".to_string()]);
86    for f in i.members() {
87        render_callable(w, ns, f);
88    }
89}

91/// Render the generated half of `NativeMethods`: the contract check run at
92/// load (the library must export each declaration's contract function
93/// these bindings were generated with, returning the same hash) and every
94/// interface and function import in declaration order.
95pub(crate) fn render_native_methods(w: &mut CodeWriter, model: &Model, ns: &str) {
96    w.line("internal static unsafe partial class NativeMethods");
97    w.line("{");
98    w.indent();
99    w.line("static partial void VerifyContracts(IntPtr library)");
100    w.block("{", "}", |w| {
101        w.line("VerifyContract(library, new (string, ulong, string)[]");
102        w.line("{");
103        w.indent();
104        for row in rows(model) {
105            w.line(format!(
106                "(\"{}\", {}UL, \"{}\"), // {}",
107                row.symbol,
108                hex(row.hash),
109                cs_str(&row.path),
110                row.signature
111            ));
112        }
113        w.dedent();
114        w.line("});");
115    });
116    w.blank();
117    for i in &model.interfaces {
118        render_interface(w, ns, i);
119    }
120    for f in &model.functions {
121        render_callable(w, ns, f);
122    }
123    w.dedent();
124    w.line("}");
125    w.blank();
126}
