1//! Gem packaging surfaces: the `.gemspec` (source-only and per-platform)
2//! and the README.
3//!
4//! A gemspec is Ruby source, so every interpolated user string (summary,
5//! authors, license, homepage) goes through the single-quote escape in
6//! [`rb_str_literal`]; a quote or trailing backslash in package metadata
7//! can't corrupt the emitted spec.
8
9use crate::utils::{render_prelude, render_trailer, CommentStyle};
10use weaveffi_model::pkg::Identity;
11
12use crate::targets::ruby::types::rb_str_literal;
13
14/// The names one generated gem is published and loaded under.
15pub(crate) struct GemNames<'a> {
16    /// The package identity.
17    pub identity: &'a Identity,
18    /// The gem name (`{name}` unless configured).
19    pub gem: String,
20    /// The Ruby module holding the bindings (`PascalCase(name)` unless
21    /// configured).
22    pub module: String,
23}
24
25impl GemNames<'_> {
26    /// The `require` path: the identity prefix.
27    pub(crate) fn require(&self) -> &str {
28        &self.identity.prefix
29    }
30
31    /// The gemspec file name.
32    pub(crate) fn gemspec_file(&self) -> String {
33        format!("{}.gemspec", self.gem)
34    }
35}
36
37/// The `required_ruby_version` every gem declares.
38pub(crate) const REQUIRED_RUBY_VERSION: &str = ">= 3.2";
39
40/// The `ffi` gem requirement every gem declares.
41pub(crate) const FFI_REQUIREMENT: &str = "~> 1.16";
42
43/// The gem's authors. RubyGems requires a non-empty list, so an identity
44/// without authors falls back to the gem name.
45pub(crate) fn gem_authors(names: &GemNames) -> Vec<String> {
46    let id = names.identity;
47    if id.authors.is_empty() {
48        vec![names.gem.clone()]
49    } else {
50        id.authors.clone()
51    }
52}
53
54/// The metadata lines of the gemspec.
55fn gemspec_metadata(names: &GemNames) -> String {
56    let id = names.identity;
57    let authors = gem_authors(names)
58        .iter()
59        .map(|a| format!("'{}'", rb_str_literal(a)))
60        .collect::<Vec<_>>()
61        .join(", ");
62    let mut out = format!(
63        "  s.name        = '{}'\n  s.version     = '{}'\n",
64        rb_str_literal(&names.gem),
65        rb_str_literal(&id.version)
66    );
67    out.push_str(&format!(
68        "  s.summary     = '{}'\n",
69        rb_str_literal(&id.description_or_default())
70    ));
71    out.push_str(&format!("  s.authors     = [{authors}]\n"));
72    if let Some(license) = &id.license {
73        out.push_str(&format!(
74            "  s.license     = '{}'\n",
75            rb_str_literal(license)
76        ));
77    }
78    if let Some(homepage) = id.homepage.as_ref().or(id.repository.as_ref()) {
79        out.push_str(&format!(
80            "  s.homepage    = '{}'\n",
81            rb_str_literal(homepage)
82        ));
83    }
84    out
85}

87/// Render the source-only gemspec emitted by `generate`.
88pub(crate) fn render_gemspec(names: &GemNames) -> String {
89    let prelude = render_prelude(CommentStyle::Hash);
90    let trailer = render_trailer(CommentStyle::Hash, &names.gemspec_file());
91    let meta = gemspec_metadata(names);
92    format!(
93        "{prelude}Gem::Specification.new do |s|
94{meta}  s.files       = Dir['lib/**/*.rb']
95  s.require_paths = ['lib']
96  s.required_ruby_version = '{REQUIRED_RUBY_VERSION}'
97
98  s.add_dependency 'ffi', '{FFI_REQUIREMENT}'
99end
100
101{trailer}"
102    )
103}

105/// README for the source-only gem layout.
106pub(crate) fn render_readme(names: &GemNames) -> String {
107    let prelude = render_prelude(CommentStyle::Xml);
108    let trailer = render_trailer(CommentStyle::Xml, "README.md");
109    let gem = &names.gem;
110    let version = &names.identity.version;
111    let require = names.require();
112    let module = &names.module;
113    let env = names.identity.library_env_var();
114    let (macos, linux, windows) = names.identity.library_files();
115    format!(
116        r#"{prelude}# {gem} (Ruby)
117
118Ruby bindings for `{gem}` using the [ffi](https://github.com/ffi/ffi) gem.
119
120## Prerequisites
121
122- Ruby 3.2 or newer
123- The native library (`{linux}`, `{macos}`, or `{windows}`) on the library
124  search path, or its path in the `{env}` environment variable.
125
126## Install
127
128```bash
129gem build {gem}.gemspec
130gem install {gem}-{version}.gem
131```
132
133## Usage
134
135```ruby
136require '{require}'
137
138{module}::ABI_VERSION
139```
140
141{trailer}"#
142    )
143}
