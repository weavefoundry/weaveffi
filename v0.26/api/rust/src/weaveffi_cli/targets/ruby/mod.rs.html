1//! Ruby (ffi gem) binding generator.
2//!
3//! Emits a pure-Ruby gem that binds the C ABI (revision 6) with the `ffi`
4//! gem: `lib/{prefix}.rb` (the generated bindings, required as `{prefix}`),
5//! `lib/{prefix}/runtime.rb` (the fixed runtime from `runtime/runtime.rb`),
6//! a `{name}.gemspec` (Ruby 3.2 or newer), and a README. A library is one
7//! namespace, so every declaration lives directly in one Ruby module,
8//! `PascalCase(name)` unless configured.
9//!
10//! The public surface is only the idiomatic API: error classes, enum
11//! constants, records as `Data` classes, rich enums as modules of `Data`
12//! variants, interface wrapper classes, callback-interface modules, and
13//! library functions. The raw C functions live in the private `Native`
14//! module and the shared marshalling in the private `Bridge` module, both
15//! in the runtime. Every value crosses as one C value: a scalar, an
16//! `FFI::Struct` passed by value (`Run` for strings, bytes, typed arrays,
17//! and value buffers; `Opt*` for optional scalars; `Callback` for a
18//! callback-interface argument), or a pointer. A wrapper only converts its
19//! arguments and hands the C call to a `Bridge` helper as a block, value
20//! buffers are encoded by a generic codec from per-type layouts registered
21//! at the end of the file, and each callback interface's vtable is built
22//! from a row per method. Loading checks the ABI revision and every
23//! declaration's contract function and raises `{Module}::LoadError` on a
24//! mismatch.

59/// Per-target configuration for [`RubyGenerator`].
60#[derive(Debug, Clone, Default, Serialize, Deserialize)]
61#[serde(default, deny_unknown_fields)]
62pub struct RubyConfig {
63    /// Gem name written into the gemspec (default: the package name).
64    pub name: Option<String>,
65    /// The Ruby module holding the bindings (default: the package name in
66    /// PascalCase).
67    pub module_name: Option<String>,
68}
69
70impl RubyConfig {
71    /// The gem, module, and require names for `model`: the identity's,
72    /// unless overridden here.
73    fn names<'a>(&self, model: &'a Model) -> GemNames<'a> {
74        let identity = &model.identity;
75        let configured = |v: &Option<String>| {
76            v.as_deref()
77                .map(str::trim)
78                .filter(|s| !s.is_empty())
79                .map(str::to_string)
80        };
81        GemNames {
82            identity,
83            gem: configured(&self.name).unwrap_or_else(|| identity.name.clone()),
84            module: configured(&self.module_name).unwrap_or_else(|| identity.pascal_name()),
85        }
86    }
87}

98    }

101impl Target for RubyGenerator {
102    fn name(&self) -> &'static str {
103        "ruby"
104    }
105
106    /// `lib/native/`, where the runtime loads a bundled library first.
107    fn dev_bundle_dir(&self, _model: &Model) -> Option<Utf8PathBuf> {
108        Some(Utf8PathBuf::from("lib/native"))
109    }
110
111    fn fixed_files(&self) -> &'static [&'static str] {
112        &["*.gemspec", "README.md", "runtime.rb"]
113    }
114
115    fn render(&self, model: &Model) -> Vec<OutputFile> {
116        let names = self.config.names(model);
117        let lib_file = format!("{}.rb", names.require());
118        vec![
119            OutputFile::new(
120                format!("lib/{lib_file}"),
121                render_bindings(model, &names, &lib_file),
122            ),
123            OutputFile::new(
124                format!("lib/{}/runtime.rb", names.require()),
125                render_runtime(names.identity, &names.module, "runtime.rb"),
126            ),
127            OutputFile::new(names.gemspec_file(), render_gemspec(&names)),
128            OutputFile::new("README.md", render_readme(&names)),
129        ]
130    }
131
132    /// One precompiled platform gem per desktop platform, with the library
133    /// under `lib/native/` where the runtime loads it first. Platforms
134    /// RubyGems has no platform string for (Android, iOS, `wasm32`) are
135    /// skipped.
136    fn package(&self, model: &Model, ctx: &PackageContext<'_>) -> Result<Vec<Artifact>> {
137        let names = self.config.names(model);
138        let id = names.identity;
139        let lib_file = format!("{}.rb", names.require());
140        let bindings = render_bindings(model, &names, &lib_file);
141        let runtime = render_runtime(id, &names.module, "runtime.rb");
142        let readme = render_packaged_readme(&names);
143        let mut artifacts = Vec::new();
144        for nb in &ctx.binaries.binaries {
145            let Some(ruby_platform) = nb.platform.ruby_platform() else {
146                continue;
147            };
148            let files = vec![
149                PackagedFile::text(format!("lib/{lib_file}"), bindings.clone()),
150                PackagedFile::text(
151                    format!("lib/{}/runtime.rb", names.require()),
152                    runtime.clone(),
153                ),
154                PackagedFile::copy(
155                    format!("lib/native/{}", nb.platform.lib_filename(&id.library)),
156                    nb.library.clone(),
157                ),
158                PackagedFile::text("README.md", readme.clone()),
159            ];
160            let spec = GemSpec {
161                name: names.gem.clone(),
162                version: id.version.clone(),
163                platform: ruby_platform.to_string(),
164                summary: id.description_or_default(),
165                authors: gem_authors(&names),
166                license: id.license.clone(),
167                homepage: id.homepage.clone().or_else(|| id.repository.clone()),
168                required_ruby_version: REQUIRED_RUBY_VERSION.to_string(),
169                dependencies: vec![("ffi".to_string(), FFI_REQUIREMENT.to_string())],
170            };
171            artifacts.push(Artifact {
172                path: format!("ruby/{}", spec.file_name()).into(),
173                kind: ArtifactKind::Gem(spec),
174                files,
175            });
176        }
177        Ok(artifacts)
178    }
179}

181/// What every renderer of one bindings file shares.
182pub(crate) struct RbCtx<'a> {
183    /// The Ruby module name, which deprecation warnings name.
184    pub(crate) module: &'a str,
185    /// Every identifier the API declares, for rewriting doc text.
186    pub(crate) names: ApiNames,
187}

189/// Render `lib/{prefix}.rb`: the contract check, the `Native` attachments,
190/// the declarations (errors, enums, records, callback interfaces,
191/// interfaces, functions), and finally the private registrations
192/// (interface lifecycles, wire layouts, vtables).
193fn render_bindings(model: &Model, names: &GemNames, lib_file: &str) -> String {
194    let ctx = RbCtx {
195        module: &names.module,
196        names: ApiNames::new(model),
197    };
198    let tables = errors::tables(model, "Error");
199    let mut w = CodeWriter::two_space();
200    w.raw(render_prelude(CommentStyle::Hash));
201    w.line("# frozen_string_literal: true");
202    w.blank();
203    w.line(format!("require_relative '{}/runtime'", names.require()));
204    w.blank();
205    w.line(format!("# Ruby bindings for {}.", names.identity.name));
206    w.block(format!("module {}", names.module), "end", |w| {
207        render_contract(w, model);
208        render_native(w, model);
209        for table in &tables {
210            render_error(w, &ctx, table);
211        }
212        for e in &model.enums {
213            if e.is_rich() {
214                render_rich_enum(w, &ctx, e);
215            } else {
216                render_enum(w, &ctx, e);
217            }
218        }
219        for r in &model.records {
220            render_record(w, &ctx, r);
221        }
222        for cb in &model.callbacks {
223            render_callback_module(w, &ctx, cb);
224        }
225        for i in &model.interfaces {
226            render_interface(w, &ctx, i);
227        }
228        for f in &model.functions {
229            w.blank();
230            let scope = RbScope {
231                kind: ScopeKind::Free,
232                class: None,
233            };
234            render_callable(w, &ctx, f, &scope);
235        }
236        w.blank();
237        w.line("# === Private registrations ===");
238        render_layouts(w, model, &tables);
239        for cb in &model.callbacks {
240            render_vtable(w, cb);
241        }
242    });
243    w.blank();
244    w.raw(render_trailer(CommentStyle::Hash, lib_file));
245    w.finish()
246}

248/// The load-time contract check: one row per declaration, `[contract
249/// function, hash, path]` with the canonical signature as a comment, which
250/// `Bridge.check_contract!` calls and compares before anything else runs.
251fn render_contract(w: &mut CodeWriter, model: &Model) {
252    w.line("# The declarations these bindings were generated with, as [contract");
253    w.line("# function, hash, path]. Loading raises LoadError, naming the path, unless");
254    w.line("# the library exports each function and it returns the hash.");
255    w.line("Bridge.check_contract!(");
256    w.scope(|w| {
257        w.line("[");
258        w.scope(|w| {
259            for row in contract::rows(model) {
260                w.line(format!("# {}", row.signature));
261                w.line(format!(
262                    "[:{}, {}, '{}'],",
263                    row.symbol,
264                    contract::hex(row.hash),
265                    rb_str_literal(&row.path)
266                ));
267            }
268        });
269        w.line("],");
270    });
271    w.line(")");
272}

274/// The private `Native` module's attachments: every C function the
275/// wrappers call.
276fn render_native(w: &mut CodeWriter, model: &Model) {
277    w.blank();
278    w.line("# The C functions the wrappers below call (private).");
279    w.block("module Native", "end", |w| {
280        for i in &model.interfaces {
281            w.line(format!(
282                "attach_function :{}, [:pointer], :pointer",
283                i.clone_symbol
284            ));
285            w.line(format!(
286                "attach_function :{}, [:pointer], :void",
287                i.destroy_symbol
288            ));
289            for f in i.members() {
290                render_attach(w, f);
291            }
292        }
293        for f in &model.functions {
294            render_attach(w, f);
295        }
296    });
