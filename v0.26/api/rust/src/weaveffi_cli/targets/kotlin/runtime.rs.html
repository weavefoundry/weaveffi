1//! The fixed runtime sources under `runtime/`, spliced into the output with
2//! `{{PLACEHOLDER}}` substitution: the Kotlin runtime (`FfiException`,
3//! `NativeBugException`, `NativeSequence`, the JNA loader and its load-time
4//! checks, `NativeHandle`, the phantom-reference cleaner, the callback
5//! registry), the C shapes JNA maps (`Run`, `NativeError`, the callback
6//! argument, one `{p}_opt_*` structure per scalar, and the runtime surface),
7//! the value-buffer codec and content equality, and the coroutine bridge.
8
9use weaveffi_model::model::ABI_VERSION;
10use weaveffi_model::ty::Prim;

12use crate::codegen::CodeWriter;
13use crate::targets::kotlin::names::{jna_prim, kt_prim, Names};

15const RUNTIME_KT: &str = include_str!("runtime/Runtime.kt");
16const ABI_KT: &str = include_str!("runtime/Abi.kt");
17const BUFFERS_KT: &str = include_str!("runtime/Buffers.kt");
18const ASYNC_KT: &str = include_str!("runtime/Async.kt");

20/// The scalars with a `{p}_opt_*` structure, in the header's order.
21const OPT_PRIMS: [Prim; 11] = [
22    Prim::Bool,
23    Prim::I8,
24    Prim::I16,
25    Prim::I32,
26    Prim::I64,
27    Prim::U8,
28    Prim::U16,
29    Prim::U32,
30    Prim::U64,
31    Prim::F32,
32    Prim::F64,
33];
34
35/// Replace every `{{KEY}}` in `template` with its value.
36fn fill(template: &str, vars: &[(&str, &str)]) -> String {
37    let mut out = template.to_string();
38    for (key, value) in vars {
39        out = out.replace(&format!("{{{{{key}}}}}"), value);
40    }
41    debug_assert!(
42        !out.contains("{{"),
43        "unfilled placeholder in a runtime template"
44    );
45    out
46}

48/// `Runtime.kt`: always emitted. `name` is the library's display name (for
49/// load errors) and `env_var` its path override.
50pub(crate) fn runtime_kt(n: &Names, library: &str, name: &str, env_var: &str) -> String {
51    let abi = ABI_VERSION.to_string();
52    fill(
53        RUNTIME_KT,
54        &[
55            ("PACKAGE", &n.package),
56            ("LIBRARY", library),
57            ("LIBRARY_ENV", env_var),
58            ("PREFIX", &n.prefix),
59            ("NAME", name),
60            ("ABI_VERSION", &abi),
61        ],
62    )
63}

65/// `Abi.kt`: always emitted.
66pub(crate) fn abi_kt(n: &Names) -> String {
67    let optionals = optionals(&n.prefix);
68    fill(
69        ABI_KT,
70        &[
71            ("PACKAGE", &n.package),
72            ("PREFIX", &n.prefix),
73            ("OPTIONALS", &optionals),
74        ],
75    )
76}

78/// `Buffers.kt`: emitted when any value crosses as a value buffer.
79pub(crate) fn buffers_kt(n: &Names) -> String {
80    fill(BUFFERS_KT, &[("PACKAGE", &n.package)])
81}
82
83/// `Async.kt`: emitted when any callable is async.
84pub(crate) fn async_kt(n: &Names) -> String {
85    fill(ASYNC_KT, &[("PACKAGE", &n.package), ("PREFIX", &n.prefix)])
86}

88/// The JNA structure of the optional scalar `p` (`OptI32`).
89pub(crate) fn opt_class(p: Prim) -> String {
90    format!("Opt{}", p.pascal())
91}
92
93/// The function building a `{p}_opt_*` value (`optI32`).
94pub(crate) fn opt_fn(p: Prim) -> String {
95    format!("opt{}", p.pascal())
96}

98/// One structure, its `get`, and its builder per optional scalar: the
99/// `{p}_opt_{prim}` `{bool has; T value;}` shapes, with the value in its JNA
100/// carrier and converted to the public type at the edges.
101fn optionals(prefix: &str) -> String {
102    let mut w = CodeWriter::four_space();
103    for p in OPT_PRIMS {
104        let class = opt_class(p);
105        let jna = jna_prim(p);
106        let public = kt_prim(p);
107        let (lift, lower, zero) = match p {
108            Prim::Bool => ("value.asBoolean()", "value.asByte()", "0"),
109            Prim::U8 => ("value.toUByte()", "value.toByte()", "0"),
110            Prim::U16 => ("value.toUShort()", "value.toShort()", "0"),
111            Prim::U32 => ("value.toUInt()", "value.toInt()", "0"),
112            Prim::U64 => ("value.toULong()", "value.toLong()", "0"),
113            Prim::I64 => ("value", "value", "0L"),
114            Prim::F32 => ("value", "value", "0f"),
115            Prim::F64 => ("value", "value", "0.0"),
116            _ => ("value", "value", "0"),
117        };
118        w.blank();
119        w.line(format!(
120            "/** A C `{prefix}_opt_{}`: `has`, then the value (0 when absent). */",
121            p.snake()
122        ));
123        w.line("@Structure.FieldOrder(\"has\", \"value\")");
124        w.block(
125            format!("internal open class {class} : Structure() {{"),
126            "}",
127            |w| {
128                w.line("@JvmField");
129                w.line("var has: Byte = 0");
130                w.blank();
131                w.line("@JvmField");
132                w.line(format!("var value: {jna} = {zero}"));
133                w.blank();
134                w.line("/** The value, or `null` when absent. */");
135                w.line(format!(
136                    "fun get(): {public}? = if (has.asBoolean()) {lift} else null"
137                ));
138                w.blank();
139                w.line("/** The structure by value. */");
140                w.line(format!("class ByValue : {class}(), Structure.ByValue"));
141            },
142        );
143        w.blank();
144        w.line(format!("/** [value] as a `{prefix}_opt_{}`. */", p.snake()));
145        w.block(
146            format!(
147                "internal fun {}(value: {public}?): {class}.ByValue = {class}.ByValue().also {{",
148                opt_fn(p)
149            ),
150            "}",
151            |w| {
152                w.block("if (value != null) {", "}", |w| {
153                    w.line("it.has = 1");
154                    w.line(format!("it.value = {lower}"));
155                });
156                w.line("it.write()");
157            },
158        );
159    }
160    w.finish()
161}
