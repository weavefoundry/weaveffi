1//! KDoc comments and deprecation messages, with backticked API identifiers
2//! in IDL docs rewritten to their Kotlin spelling (`` `new_op` `` becomes
3//! `` `newOp` ``, `` `KvError` `` becomes `` `KvException` ``).
4
5use crate::codegen::common::DocCommentStyle;
6use crate::codegen::docs::{ApiNames, Doc, IdentKind};
7use crate::codegen::CodeWriter;
8
9use crate::targets::kotlin::names::{kt_member, kt_param, Names};
10
11/// Spells IDL identifiers the way the generated Kotlin declares them.
12pub(crate) struct Speller<'a> {
13    api: ApiNames,
14    names: &'a Names,
15}
16
17impl<'a> Speller<'a> {
18    /// A speller over every identifier `model` declares.
19    pub(crate) fn new(model: &weaveffi_model::model::Model, names: &'a Names) -> Self {
20        Self {
21            api: ApiNames::new(model),
22            names,
23        }
24    }
25
26    /// The Kotlin spelling of `ident`, or `None` to keep it as written (an
27    /// identifier naming several kinds of declaration, a variant, or an
28    /// error code).
29    pub(crate) fn spell(&self, ident: &str) -> Option<String> {
30        match self.api.kind(ident)? {
31            IdentKind::Function | IdentKind::Member | IdentKind::CallbackMethod => {
32                Some(kt_member(ident))
33            }
34            IdentKind::Param | IdentKind::Field => Some(kt_param(ident)),
35            IdentKind::Type => Some(self.names.ty(ident)),
36            IdentKind::ErrorDomain => Some(self.names.exception(ident).to_string()),
37            IdentKind::Variant | IdentKind::ErrorCode => None,
38        }
39    }
40
41    /// A declaration's doc text, rewritten.
42    pub(crate) fn text(&self, doc: &Option<String>) -> Option<String> {
43        Doc::new(doc, &None).text(|i| self.spell(i))
44    }
45
46    /// A declaration's deprecation message, rewritten.
47    pub(crate) fn deprecation(&self, deprecated: &Option<String>) -> Option<String> {
48        Doc::new(&None, deprecated).deprecation(|i| self.spell(i))
49    }
50
51    /// Emit a declaration's KDoc block (rewritten).
52    pub(crate) fn doc(&self, w: &mut CodeWriter, doc: &Option<String>) {
53        w.doc(&self.text(doc), DocCommentStyle::Javadoc);
54    }
55
56    /// Emit the KDoc block of a callable: its doc plus one `@param` tag per
57    /// documented parameter, named with its Kotlin spelling. Without
58    /// parameter docs this is the plain doc comment.
59    pub(crate) fn fn_doc<'p>(
60        &self,
61        w: &mut CodeWriter,
62        doc: Option<String>,
63        params: impl IntoIterator<Item = (&'p str, &'p Option<String>)>,
64    ) {
65        let documented: Vec<(String, String)> = params
66            .into_iter()
67            .filter_map(|(name, d)| Some((kt_param(name), self.text(d)?)))
68            .collect();
69        if documented.is_empty() {
70            w.doc(&doc, DocCommentStyle::Javadoc);
71            return;
72        }
73        let mut text = doc.unwrap_or_default();
74        if !text.is_empty() {
75            text.push_str("\n\n");
76        }
77        for (i, (name, d)) in documented.iter().enumerate() {
78            if i > 0 {
79                text.push('\n');
80            }
81            let mut lines = d.lines();
82            text.push_str(&format!("@param {name} {}", lines.next().unwrap_or("")));
83            for line in lines {
84                text.push_str(&format!("\n  {line}"));
85            }
86        }
87        w.doc(&Some(text), DocCommentStyle::Javadoc);
88    }
89}
