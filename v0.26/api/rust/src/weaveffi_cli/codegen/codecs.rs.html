1//! The value-buffer **composites** an API uses and their one canonical
2//! **stem**.
3//!
4//! Records, rich enums, and error payloads get one codec each, named after
5//! the type. Optionals, lists, and maps that cross inside a value buffer
6//! (record fields, variant fields, error payload fields, and every
7//! [`Buffer`](weaveffi_model::ty::RunKind::Buffer) run at a call boundary)
8//! are the *composites*: each distinct shape gets one codec too,
9//! and every target names it with [`stem`] so the shapes line up across
10//! languages (`list_i32`, `opt_Item`, `map_string_list_i64`).
11//!
12//! An optional scalar (OptDirect) or numeric list (Slice) at a call boundary
13//! isn't a composite: it crosses directly. The same type nested in a buffer
14//! (a record's `i64?` field) is.
15//!
16//! # Example
17//!
18//! A target that writes one reader and writer function per composite:
19//!
20//! ```ignore
21//! use crate::codegen::codecs;
22//!
23//! for ty in codecs::composites(model) {
24//!     // Dependency order: `list_i32` comes before `opt_list_i32`, so a
25//!     // language that needs definitions before use can emit in order.
26//!     let stem = codecs::stem(&ty);
27//!     w.line(format!("def _write_{stem}(w, v): ..."));
28//!     w.line(format!("def _read_{stem}(r): ..."));
29//! }
30//! ```
31
32use weaveffi_model::model::Model;
33use weaveffi_model::ty::Ty;
34
35/// Every distinct optional, list, and map type that crosses inside a value
36/// buffer anywhere in `model`, innermost first (see
37/// [`Model::composites`]).
38#[must_use]
39pub(crate) fn composites(model: &Model) -> Vec<Ty> {
40    model.composites()
41}
42
43/// The canonical identifier fragment naming `ty` in codec names (see
44/// [`Ty::stem`]).
45#[must_use]
46pub(crate) fn stem(ty: &Ty) -> String {
47    ty.stem()
48}

50#[cfg(test)]
51mod tests {
52    use super::*;
53    use crate::codegen::test_model;
54    use weaveffi_model::ty::Prim;
55
56    #[test]
57    fn stems_are_prefix_notation() {
58        let i64s = Ty::List(Box::new(Ty::Prim(Prim::I64)));
59        let map = Ty::Map(Box::new(Ty::Prim(Prim::String)), Box::new(i64s.clone()));
60        assert_eq!(stem(&map), "map_string_list_i64");
61        let opt = Ty::Optional(Box::new(Ty::Record("Item".into())));
62        assert_eq!(stem(&opt), "opt_Item");
63        assert_eq!(stem(&Ty::List(Box::new(opt))), "list_opt_Item");
64    }
65
66    #[test]
67    fn composites_are_buffered_shapes_innermost_first() {
68        let model = test_model(
69            r#"
70version: "0.13.0"
71records:
72  - name: R
73    fields:
74      - { name: a, type: "[i32]" }
75      - { name: b, type: "{string:[i32]?}" }
76functions:
77  - { name: slice, params: [{ name: xs, type: "[i32]" }], return: "[i64]" }
78  - { name: opt, params: [{ name: x, type: "i32?" }], return: "string?" }
79  - { name: it, return: "iter<[R]>" }
80"#,
81        );
82        let stems: Vec<String> = composites(&model).iter().map(stem).collect();
83        assert_eq!(
84            stems,
85            [
86                "list_i32",
87                "opt_list_i32",
88                "map_string_opt_list_i32",
89                "opt_string",
90                "list_R",
91            ]
92        );
93    }
