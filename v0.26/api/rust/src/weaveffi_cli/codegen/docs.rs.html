1//! **Doc and deprecation text** in a target's spelling.
2//!
3//! IDL docs and deprecation messages name API declarations in backticks:
4//! `` deprecated: "Use `new_op` instead" ``. A target spells `new_op` its own
5//! way (`newOp` in Swift and Kotlin, `NewOp` in Go and .NET,
6//! `kv_new_op` in C), so a message copied verbatim points at a name
7//! that doesn't exist in the generated API. [`rewrite`] replaces each
8//! backticked identifier through a closure the target supplies;
9//! [`ApiNames`] tells the closure what the identifier names, and [`Doc`]
10//! bundles a declaration's doc and deprecation message with the rewriting.
11//!
12//! Only code spans whose whole content is an identifier (or a dotted path
13//! of identifiers, such as `` `Store.get` ``) are offered to the closure;
14//! anything else in backticks (`` `x + 1` ``) is left alone, and so is an
15//! identifier the closure returns `None` for.
16//!
17//! # Example
18//!
19//! A target with camel-cased callables and parameters and suffixed error
20//! types:
21//!
22//! ```ignore
23//! use crate::codegen::docs::{ApiNames, Doc, IdentKind};
24//!
25//! let names = ApiNames::new(model);
26//! let spell = |ident: &str| match names.kind(ident)? {
27//!     IdentKind::Function | IdentKind::Member | IdentKind::Param => Some(camel(ident)),
28//!     IdentKind::ErrorDomain => Some(errors::type_name(ident, "Error")),
29//!     _ => None,
30//! };
31//! let doc = Doc::new(&f.doc, &f.deprecated);
32//! w.doc(&doc.text(&spell), DocCommentStyle::TripleSlash);
33//! if let Some(msg) = doc.deprecation(&spell) {
34//!     w.line(format!("@available(*, deprecated, message: {msg:?})"));
35//! }
36//! ```

40use weaveffi_model::model::Model;

42/// What an API identifier names.
43#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
44pub(crate) enum IdentKind {
45    /// A record, enum, interface, or callback interface (`Item`).
46    Type,
47    /// An error domain (`KitchenErrors`), which targets name through
48    /// [`weaveffi_model::errors::type_name`].
49    ErrorDomain,
50    /// A free function (`new_op`).
51    Function,
52    /// An interface constructor, method, or static (`describe`).
53    Member,
54    /// A callback interface method (`on_ready`).
55    CallbackMethod,
56    /// An enum variant (`High`).
57    Variant,
58    /// An error code (`NotFound`).
59    ErrorCode,
60    /// A record, variant, or error-code field (`label`).
61    Field,
62    /// A parameter of a function, member, or callback method (`limit`).
63    Param,
64}
65
66/// One declaration an identifier names.
67#[derive(Debug, Clone, PartialEq, Eq)]
68pub(crate) struct ApiName {
69    /// What it is.
70    pub(crate) kind: IdentKind,
71    /// Its C ABI spelling, when it has its own: a callable's symbol, a
72    /// type's tag, or an enum variant's or error code's constant. `None` for
73    /// fields, parameters, and callback methods.
74    pub(crate) c_name: Option<String>,
75}
76
77/// Every identifier the API declares, with what each one names.
78///
79/// One identifier can name several declarations (a method `describe` on two
80/// interfaces, a parameter `id` of many functions); [`get`](Self::get)
81/// returns all of them, in declaration order.
82#[derive(Debug, Clone, Default)]
83pub(crate) struct ApiNames {
84    names: HashMap<String, Vec<ApiName>>,
85}
86
87impl ApiNames {
88    /// Index every declaration in `model`.
89    #[must_use]
90    pub(crate) fn new(model: &Model) -> Self {
91        let mut index = Self::default();
92        for d in &model.errors {
93            index.add(&d.name, IdentKind::ErrorDomain, Some(&d.c_tag));
94            for c in &d.codes {
95                index.add(&c.name, IdentKind::ErrorCode, Some(&c.c_const));
96                for f in &c.fields {
97                    index.add(&f.name, IdentKind::Field, None);
98                }
99            }
100        }
101        for e in &model.enums {
102            index.add(&e.name, IdentKind::Type, Some(&e.c_tag));
103            for v in &e.variants {
104                index.add(&v.name, IdentKind::Variant, Some(&v.c_const));
105                for f in &v.fields {
106                    index.add(&f.name, IdentKind::Field, None);
107                }
108            }
109        }
110        for r in &model.records {
111            index.add(&r.name, IdentKind::Type, Some(&r.c_tag));
112            for f in &r.fields {
113                index.add(&f.name, IdentKind::Field, None);
114            }
115        }
116        for cb in &model.callbacks {
117            index.add(&cb.name, IdentKind::Type, Some(&cb.c_tag));
118            for meth in &cb.methods {
119                index.add(&meth.name, IdentKind::CallbackMethod, None);
120                for p in &meth.params {
121                    index.add(&p.name, IdentKind::Param, None);
122                }
123            }
124        }
125        for i in &model.interfaces {
126            index.add(&i.name, IdentKind::Type, Some(&i.c_tag));
127        }
128        let members = model.interfaces.iter().flat_map(|i| i.members());
129        let callables = model
130            .functions
131            .iter()
132            .map(|f| (f, IdentKind::Function))
133            .chain(members.map(|f| (f, IdentKind::Member)));
134        for (f, kind) in callables {
135            index.add(&f.name, kind, Some(&f.abi.symbol));
136            for p in &f.params {
137                index.add(&p.name, IdentKind::Param, None);
138            }
139        }
140        index
141    }
142
143    fn add(&mut self, name: &str, kind: IdentKind, c_name: Option<&String>) {
144        self.names
145            .entry(name.to_string())
146            .or_default()
147            .push(ApiName {
148                kind,
149                c_name: c_name.cloned(),
150            });
151    }
152
153    /// Every declaration `name` names, in declaration order (empty when the
154    /// API declares nothing by that name).
155    #[must_use]
156    pub(crate) fn get(&self, name: &str) -> &[ApiName] {
157        self.names.get(name).map_or(&[], Vec::as_slice)
158    }
159
160    /// What `name` names, when every declaration of it is the same kind.
161    #[must_use]
162    pub(crate) fn kind(&self, name: &str) -> Option<IdentKind> {
163        let found = self.get(name);
164        let kind = found.first()?.kind;
165        found.iter().all(|n| n.kind == kind).then_some(kind)
166    }
167
168    /// The C ABI spelling of `name`, when it names exactly one declaration
169    /// and that declaration has one.
170    #[must_use]
171    pub(crate) fn c_name(&self, name: &str) -> Option<&str> {
172        match self.get(name) {
173            [only] => only.c_name.as_deref(),
174            _ => None,
175        }
176    }
177}
178
179/// `text` with every backticked identifier the closure spells replaced by
180/// its spelling (the backticks stay).
181///
182/// `spell` receives the span's content (`new_op`, or a dotted path such as
183/// `Store.get`) and returns `None` to keep it unchanged.
184#[must_use]
185pub(crate) fn rewrite(text: &str, spell: impl Fn(&str) -> Option<String>) -> String {
186    map_code_spans(text, |ident| spell(ident).map(|s| format!("`{s}`")))
187}
188
189/// `text` with every backticked identifier the closure maps replaced by the
190/// closure's result, which replaces the whole span *including* its
191/// backticks. Use it to change the markup as well as the spelling (Javadoc
192/// `{@code newOp}`, .NET `<c>NewOp</c>`, reStructuredText ``` ``new_op`` ```).
193#[must_use]
194pub(crate) fn map_code_spans(text: &str, mut map: impl FnMut(&str) -> Option<String>) -> String {
195    let mut out = String::with_capacity(text.len());
196    let mut rest = text;
197    while let Some(open) = rest.find('`') {
198        out.push_str(&rest[..open]);
199        let after = &rest[open + 1..];
200        let Some(close) = after.find('`') else {
201            out.push_str(&rest[open..]);
202            return out;
203        };
204        let inner = &after[..close];
205        match is_ident_path(inner).then(|| map(inner)).flatten() {
206            Some(replacement) => out.push_str(&replacement),
207            None => {
208                out.push('`');
209                out.push_str(inner);
210                out.push('`');
211            }
212        }
213        rest = &after[close + 1..];
214    }
215    out.push_str(rest);
216    out
217}
218
219/// Whether `s` is an identifier or a `.`-separated path of identifiers.
220fn is_ident_path(s: &str) -> bool {
221    !s.is_empty()
222        && s.split('.').all(|part| {
223            let mut chars = part.chars();
224            chars
225                .next()
226                .is_some_and(|c| c.is_ascii_alphabetic() || c == '_')
227                && chars.all(|c| c.is_ascii_alphanumeric() || c == '_')
228        })
229}

231/// A declaration's doc and deprecation message, rendered through a target's
232/// spelling closure (see [`rewrite`]).
233#[derive(Debug, Clone, Copy)]
234pub(crate) struct Doc<'a> {
235    doc: Option<&'a str>,
236    deprecated: Option<&'a str>,
237}
238
239impl<'a> Doc<'a> {
240    /// The doc and deprecation message of one declaration, as the model
241    /// stores them.
242    #[must_use]
243    pub(crate) fn new(doc: &'a Option<String>, deprecated: &'a Option<String>) -> Self {
244        Self {
245            doc: doc.as_deref().map(str::trim).filter(|d| !d.is_empty()),
246            deprecated: deprecated
247                .as_deref()
248                .map(str::trim)
249                .filter(|d| !d.is_empty()),
250        }
251    }
252
253    /// The doc text, trimmed and rewritten, or `None` when there's none.
254    #[must_use]
255    pub(crate) fn text(&self, spell: impl Fn(&str) -> Option<String>) -> Option<String> {
256        self.doc.map(|d| rewrite(d, spell))
257    }
258
259    /// The deprecation message, trimmed and rewritten, or `None` when the
260    /// declaration isn't deprecated.
261    #[must_use]
262    pub(crate) fn deprecation(&self, spell: impl Fn(&str) -> Option<String>) -> Option<String> {
263        self.deprecated.map(|d| rewrite(d, spell))
264    }
265
266    /// The doc text followed by a `Deprecated: {message}` paragraph, for
267    /// targets (or positions) with no deprecation attribute.
268    #[must_use]
269    pub(crate) fn with_deprecation(
270        &self,
271        spell: impl Fn(&str) -> Option<String>,
272    ) -> Option<String> {
273        let note = self.deprecation(&spell).map(|m| format!("Deprecated: {m}"));
274        match (self.text(&spell), note) {
275            (Some(doc), Some(note)) => Some(format!("{doc}\n\n{note}")),
276            (doc, note) => doc.or(note),
277        }
278    }
279}

286    #[test]
287    fn rewrites_only_identifier_spans() {
288        let upper = |s: &str| Some(s.to_uppercase());
289        assert_eq!(
290            rewrite("Use `new_op` or `a.b`, not `x + 1`.", upper),
291            "Use `NEW_OP` or `A.B`, not `x + 1`."
292        );
293        assert_eq!(rewrite("unclosed `tick", upper), "unclosed `tick");
294        assert_eq!(rewrite("`keep` it", |_| None), "`keep` it");
295        assert_eq!(
296            map_code_spans("see `get`", |s| Some(format!("{{@code {s}}}"))),
297            "see {@code get}"
298        );
299    }

301    #[test]
302    fn indexes_every_declaration_kind() {
303        let model = test_model(
304            r#"
305version: "0.13.0"
306errors:
307  - name: KvError
308    codes: [{ name: NotFound, code: 1, message: missing }]
309enums:
310  - name: Mode
311    variants: [{ name: Fast }]
312records:
313  - name: Entry
314    fields: [{ name: key, type: string }]
315interfaces:
316  - name: Store
317    methods:
318      - { name: get, params: [{ name: key, type: string }], return: Entry }
319functions:
320  - { name: open, params: [{ name: path, type: string }], return: Store }
321"#,
322        );
323        let names = ApiNames::new(&model);
324        assert_eq!(names.kind("KvError"), Some(IdentKind::ErrorDomain));
325        assert_eq!(names.c_name("NotFound"), Some("kv_KvError_NotFound"));
326        assert_eq!(names.c_name("Fast"), Some("kv_Mode_Fast"));
327        assert_eq!(names.c_name("Store"), Some("kv_Store"));
328        assert_eq!(names.c_name("get"), Some("kv_Store_get"));
329        assert_eq!(names.c_name("open"), Some("kv_open"));
330        // `key` is both a field and a parameter.
331        assert_eq!(names.get("key").len(), 2);
332        assert_eq!(names.kind("key"), None);
333        assert_eq!(names.c_name("key"), None);
334        assert!(names.get("missing").is_empty());
335    }

337    #[test]
338    fn doc_appends_the_rewritten_deprecation() {
339        let doc = Some("Old `op`.".to_string());
340        let deprecated = Some(" Use `new_op` ".to_string());
341        let spell = |s: &str| Some(format!("my_{s}"));
342        let d = Doc::new(&doc, &deprecated);
343        assert_eq!(d.text(spell).as_deref(), Some("Old `my_op`."));
344        assert_eq!(d.deprecation(spell).as_deref(), Some("Use `my_new_op`"));
345        assert_eq!(
346            d.with_deprecation(spell).as_deref(),
347            Some("Old `my_op`.\n\nDeprecated: Use `my_new_op`")
348        );
349        let none = None;
350        assert_eq!(
351            Doc::new(&none, &deprecated)
352                .with_deprecation(spell)
353                .as_deref(),
354            Some("Deprecated: Use `my_new_op`")
355        );
356        assert_eq!(Doc::new(&none, &none).with_deprecation(spell), None);
357    }
