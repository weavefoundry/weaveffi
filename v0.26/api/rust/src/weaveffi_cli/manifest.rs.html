1//! Package-manifest emission helpers shared by every backend.
2//!
3//! JSON manifests (`package.json` and friends) are built as
4//! [`serde_json::Map`]s, whose keys keep their insertion order (the
5//! `preserve_order` feature) so a backend controls the conventional field
6//! order, and rendered with [`render_json`]. XML manifests (`.nuspec`,
7//! `.csproj`) are text, with values escaped by [`xml_escape`].
8
9use serde_json::{Map, Value};
10
11/// Render `object` as a pretty-printed JSON document with two-space
12/// indentation and a trailing newline, the conventional shape of
13/// `package.json`-style manifests.
14#[must_use]
15pub fn render_json(object: Map<String, Value>) -> String {
16    let mut out =
17        serde_json::to_string_pretty(&Value::Object(object)).unwrap_or_else(|_| "{}".to_string());
18    out.push('\n');
19    out
20}
21
22/// Escape `s` for placement inside XML text content or a double-quoted XML
23/// attribute value (`.nuspec`, `.csproj`, Android manifests).
24pub fn xml_escape(s: &str) -> String {
25    let mut out = String::with_capacity(s.len());
26    for c in s.chars() {
27        match c {
28            '&' => out.push_str("&amp;"),
29            '<' => out.push_str("&lt;"),
30            '>' => out.push_str("&gt;"),
31            '"' => out.push_str("&quot;"),
32            '\'' => out.push_str("&apos;"),
33            c => out.push(c),
34        }
35    }
36    out
37}

39#[cfg(test)]
40mod tests {
41    use serde_json::json;

45    #[test]
46    fn xml_escape_covers_the_five_entities() {
47        assert_eq!(
48            xml_escape(r#"a & b < c > "d" 'e'"#),
49            "a &amp; b &lt; c &gt; &quot;d&quot; &apos;e&apos;"
50        );
51    }
52
53    #[test]
54    fn json_renders_in_insertion_order_with_escapes() {
55        let Value::Object(obj) = json!({
56            "name": "kvstore",
57            "version": "1.2.0",
58            "description": "the \"best\" lib",
59            "files": [],
60            "engines": { "node": ">=18" },
61        }) else {
62            unreachable!()
63        };
64        assert_eq!(
65            render_json(obj),
66            "{\n  \"name\": \"kvstore\",\n  \"version\": \"1.2.0\",\n  \"description\": \"the \\\"best\\\" lib\",\n  \"files\": [],\n  \"engines\": {\n    \"node\": \">=18\"\n  }\n}\n"
67        );
68    }
