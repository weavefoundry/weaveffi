1//! The validation rules: library-wide name uniqueness, identifier and shape
2//! checks, type-reference resolution, ABI-representability of element
3//! shapes, callback-interface rules, interface positions, recursive value
4//! types, and error-domain consistency.
5//!
6//! Every rule pushes into a shared sink instead of returning early, so one
7//! validation pass reports every violation in the document. Each violation
8//! carries the declaration path that encloses it (the declaration, then
9//! member and parameter or field names), which the diagnostic uses to locate
10//! the offending text.
11
12use super::{Found, ValidationError};
13use crate::ir::{Api, CallbackDef, ErrorDomain, Field, Function, InterfaceDef, Throws, TypeRef};
14use crate::ty::{Prim, TypeIndex, TypeKind};
15use std::collections::{BTreeMap, BTreeSet};

17const RESERVED: &[&str] = &[
18    "if", "else", "for", "while", "loop", "match", "type", "return", "async", "await", "break",
19    "continue", "fn", "struct", "enum", "mod", "use",
20];

22/// The slots the C ABI adds to signatures, which no parameter may be named.
23const ABI_SLOTS: &[&str] = &[
24    "self",
25    "ctx",
26    "out_err",
27    "callback",
28    "context",
29    "cancel_token",
30];

32/// Primitive spellings other languages have that WeaveFFI deliberately
33/// doesn't support; a reference to one gets a dedicated diagnostic instead
34/// of "unknown type".
35const UNSUPPORTED_PRIMITIVES: &[&str] = &["usize", "isize", "u128", "i128", "char"];

37const IDENTIFIER_RULE: &str =
38    "must start with a letter or underscore and contain only alphanumeric characters or underscores";

40fn is_valid_identifier(s: &str) -> bool {
41    let mut chars = s.chars();
42    match chars.next() {
43        None => false,
44        Some(c) if !(c.is_ascii_alphabetic() || c == '_') => false,
45        _ => chars.all(|c| c.is_ascii_alphanumeric() || c == '_'),
46    }
47}

49/// `scope` extended with `names`.
50fn within(scope: &[String], names: &[&str]) -> Vec<String> {
51    let mut out = scope.to_vec();
52    out.extend(names.iter().map(|n| (*n).to_string()));
53    out
54}

56/// Run every rule over `api`, whose declarations `types` indexes.
57pub(super) fn check(api: &Api, types: &TypeIndex, found: &mut Vec<Found>) {
58    let carriers = object_carriers(api, types);
59    let mut cx = Cx {
60        types,
61        carriers: &carriers,
62        found,
63    };
64    cx.declarations(api);
65    check_names(api, cx.found);
66    check_recursion(api, types, cx.found);
67}

69/// Enforce library-wide name uniqueness: every record, enum, error domain,
70/// interface, callback interface, and free function shares one namespace,
71/// and every error-code name is unique too.
72fn check_names(api: &Api, found: &mut Vec<Found>) {
73    let mut seen: BTreeMap<&str, &'static str> = BTreeMap::new();
74    let decls = api
75        .records
76        .iter()
77        .map(|d| (d.name.as_str(), "record"))
78        .chain(api.enums.iter().map(|d| (d.name.as_str(), "enum")))
79        .chain(api.errors.iter().map(|d| (d.name.as_str(), "error domain")))
80        .chain(
81            api.interfaces
82                .iter()
83                .map(|d| (d.name.as_str(), "interface")),
84        )
85        .chain(
86            api.callbacks
87                .iter()
88                .map(|d| (d.name.as_str(), "callback interface")),
89        )
90        .chain(api.functions.iter().map(|d| (d.name.as_str(), "function")));
91    for (name, kind) in decls {
92        match seen.get(name) {
93            Some(first) => found.push((
94                ValidationError::DuplicateName {
95                    name: name.to_string(),
96                    first,
97                    second: kind,
98                },
99                vec![],
100            )),
101            None => {
102                seen.insert(name, kind);
103            }
104        }
105    }
106    let mut codes: BTreeMap<&str, &str> = BTreeMap::new();
107    for domain in &api.errors {
108        for code in &domain.codes {
109            match codes.get(code.name.as_str()) {
110                Some(first) => found.push((
111                    ValidationError::DuplicateErrorCodeName {
112                        name: code.name.clone(),
113                        first: (*first).to_string(),
114                        second: domain.name.clone(),
115                    },
116                    vec![domain.name.clone()],
117                )),
118                None => {
119                    codes.insert(&code.name, &domain.name);
120                }
121            }
122        }
123    }
124}

126/// Reject a record or rich enum that contains itself by value: directly or
127/// through optionals and other records and rich enums, but not through a
128/// list or a map, which hold their elements indirectly.
129fn check_recursion<'a>(api: &'a Api, types: &TypeIndex, found: &mut Vec<Found>) {
130    // The by-value edges: (type, field path) -> the user types the field
131    // holds by value.
132    let mut edges: BTreeMap<&str, Vec<(String, String)>> = BTreeMap::new();
133    let mut add = |name: &'a str, label: String, fields: &'a [Field]| {
134        for f in fields {
135            let mut ty = &f.ty;
136            while let TypeRef::Optional(inner) = ty {
137                ty = inner;
138            }
139            if let TypeRef::Named(target) = ty {
140                if matches!(
141                    types.kind(target),
142                    Some(TypeKind::Record | TypeKind::RichEnum)
143                ) {
144                    edges
145                        .entry(name)
146                        .or_default()
147                        .push((format!("{label}.{}", f.name), target.clone()));
148                }
149            }
150        }
151    };
152    for r in &api.records {
153        add(&r.name, r.name.clone(), &r.fields);
154    }
155    for e in &api.enums {
156        for v in &e.variants {
157            add(&e.name, format!("{}::{}", e.name, v.name), &v.fields);
158        }
159    }
160    let mut reported = BTreeSet::new();
161    for start in edges.keys() {
162        // Depth-first search for a path back to `start`.
163        let mut stack: Vec<(&str, Vec<String>)> = vec![(start, Vec::new())];
164        let mut visited = BTreeSet::new();
165        while let Some((at, path)) = stack.pop() {
166            for (field, target) in edges.get(at).into_iter().flatten() {
167                let mut next = path.clone();
168                next.push(format!("{field} -> {target}"));
169                if target == start {
170                    if reported.insert(start.to_string()) {
171                        found.push((
172                            ValidationError::RecursiveType {
173                                name: start.to_string(),
174                                path: next.join(", "),
175                            },
176                            vec![start.to_string()],
177                        ));
178                    }
179                } else if visited.insert(target.as_str()) {
180                    stack.push((target, next));
181                }
182            }
183        }
184    }
185}

187/// The rule context: the type index and the violation sink.
188struct Cx<'a> {
189    types: &'a TypeIndex,
190    /// Records and rich enums that carry an interface object somewhere in
191    /// their fields (see [`object_carriers`]).
192    carriers: &'a BTreeSet<String>,
193    found: &'a mut Vec<Found>,
194}
195
196impl Cx<'_> {
197    fn push(&mut self, scope: &[String], error: ValidationError) {
198        self.found.push((error, scope.to_vec()));
199    }

201    /// Check that `name` is a valid, non-reserved identifier.
202    fn identifier(&mut self, scope: &[String], name: &str) {
203        if !is_valid_identifier(name) {
204            self.push(
205                scope,
206                ValidationError::InvalidIdentifier {
207                    name: name.to_string(),
208                    reason: IDENTIFIER_RULE,
209                },
210            );
211        } else if RESERVED.contains(&name) {
212            self.push(
213                scope,
214                ValidationError::ReservedKeyword {
215                    name: name.to_string(),
216                },
217            );
218        }
219    }
220
221    fn kind(&self, name: &str) -> Option<TypeKind> {
222        self.types.kind(name)
223    }

225    fn declarations(&mut self, api: &Api) {
226        let top: &[String] = &[];
227        for f in &api.functions {
228            self.function(top, &f.name, f);
229            self.callable_types(std::slice::from_ref(&f.name), &f.name, f);
230        }
231
232        for r in &api.records {
233            self.identifier(top, &r.name);
234            let decl = within(top, &[&r.name]);
235            if r.fields.is_empty() {
236                self.push(
237                    &decl,
238                    ValidationError::EmptyRecord {
239                        name: r.name.clone(),
240                    },
241                );
242            }
243            let mut names = BTreeSet::new();
244            for f in &r.fields {
245                self.identifier(&decl, &f.name);
246                if !names.insert(&f.name) {
247                    self.push(
248                        &decl,
249                        ValidationError::DuplicateRecordField {
250                            record: r.name.clone(),
251                            field: f.name.clone(),
252                        },
253                    );
254                }
255                let location = || format!("field '{}' of record '{}'", f.name, r.name);
256                self.buffered_field(&decl, f, &location);
257            }
258        }
259
260        for e in &api.enums {
261            self.identifier(top, &e.name);
262            let decl = within(top, &[&e.name]);
263            if e.variants.is_empty() {
264                self.push(
265                    &decl,
266                    ValidationError::EmptyEnum {
267                        name: e.name.clone(),
268                    },
269                );
270            }
271            let mut names = BTreeSet::new();
272            let mut values = BTreeSet::new();
273            for (v, value) in e.variants.iter().zip(e.values()) {
274                self.identifier(&decl, &v.name);
275                if !names.insert(&v.name) {
276                    self.push(
277                        &decl,
278                        ValidationError::DuplicateEnumVariant {
279                            enum_name: e.name.clone(),
280                            variant: v.name.clone(),
281                        },
282                    );
283                }
284                if !values.insert(value) {
285                    self.push(
286                        &decl,
287                        ValidationError::DuplicateEnumValue {
288                            enum_name: e.name.clone(),
289                            value,
290                        },
291                    );
292                }
293                let variant = within(&decl, &[&v.name]);
294                let mut fields = BTreeSet::new();
295                for f in &v.fields {
296                    self.identifier(&variant, &f.name);
297                    if !fields.insert(&f.name) {
298                        self.push(
299                            &variant,
300                            ValidationError::DuplicateEnumVariantField {
301                                enum_name: e.name.clone(),
302                                variant: v.name.clone(),
303                                field: f.name.clone(),
304                            },
305                        );
306                    }
307                    let location =
308                        || format!("field '{}' of variant '{}::{}'", f.name, e.name, v.name);
309                    self.buffered_field(&variant, f, &location);
310                }
311            }
312        }
313
314        for i in &api.interfaces {
315            self.identifier(top, &i.name);
316            self.interface(i);
317        }
318
319        for cb in &api.callbacks {
320            self.identifier(top, &cb.name);
321            self.callback_interface(cb);
322        }
323
324        for domain in &api.errors {
325            self.error_domain(domain);
326        }
327    }

329    /// A field of a record, a rich-enum variant, or an error payload is
330    /// serialized inside a value buffer, so it obeys the buffered positional
331    /// rules: no iterators, no callback interfaces, no interface map keys,
332    /// and every reference must resolve.
333    fn buffered_field(&mut self, scope: &[String], f: &Field, location: &dyn Fn() -> String) {
334        let scope = within(scope, &[&f.name]);
335        if contains_iterator(&f.ty) {
336            self.push(
337                &scope,
338                ValidationError::IteratorInInvalidPosition {
339                    location: location(),
340                },
341            );
342        }
343        self.type_ref(&scope, &f.ty);
344        self.interface_positions(&scope, &f.ty, location);
345        self.no_callback_interface(&scope, &f.ty, location);
346    }

348    /// Validate an interface's shape: at least one member, unique member
349    /// names, constructor restrictions, and per-member signature rules.
350    fn interface(&mut self, iface: &InterfaceDef) {
351        let decl = vec![iface.name.clone()];
352        if iface.constructors.is_empty() && iface.methods.is_empty() && iface.statics.is_empty() {
353            self.push(
354                &decl,
355                ValidationError::EmptyInterface {
356                    name: iface.name.clone(),
357                },
358            );
359        }
360        let mut names = BTreeSet::new();
361        for (f, constructor) in iface.constructors.iter().map(|c| (c, true)).chain(
362            iface
363                .methods
364                .iter()
365                .chain(&iface.statics)
366                .map(|m| (m, false)),
367        ) {
368            if !names.insert(&f.name) {
369                self.push(
370                    &decl,
371                    ValidationError::DuplicateInterfaceMember {
372                        interface: iface.name.clone(),
373                        name: f.name.clone(),
374                    },
375                );
376            }
377            let display = format!("{}.{}", iface.name, f.name);
378            self.function(&decl, &display, f);
379            self.callable_types(&within(&decl, &[&f.name]), &display, f);
380            if constructor && f.returns.is_some() {
381                self.push(
382                    &decl,
383                    ValidationError::ConstructorHasReturn {
384                        interface: iface.name.clone(),
385                        constructor: f.name.clone(),
386                    },
387                );
388            }
389            if constructor && f.r#async {
390                self.push(
391                    &decl,
392                    ValidationError::AsyncConstructor {
393                        interface: iface.name.clone(),
394                        constructor: f.name.clone(),
395                    },
396                );
397            }
398        }
399    }

401    /// Validate a callback interface: at least one method, unique method
402    /// names, and per-method restrictions. A callback method is implemented
403    /// by the consumer, so it is synchronous, never takes a cancel token,
404    /// returns nothing or any value, takes no callback interface or
405    /// iterator as a parameter, and may throw only an existing domain (or
406    /// `any`).
407    fn callback_interface(&mut self, cb: &CallbackDef) {
408        let decl = vec![cb.name.clone()];
409        if cb.methods.is_empty() {
410            self.push(
411                &decl,
412                ValidationError::EmptyCallbackInterface {
413                    name: cb.name.clone(),
414                },
415            );
416        }
417        let mut names = BTreeSet::new();
418        for m in &cb.methods {
419            self.identifier(&decl, &m.name);
420            if !names.insert(&m.name) {
421                self.push(
422                    &decl,
423                    ValidationError::DuplicateCallbackMethod {
424                        interface: cb.name.clone(),
425                        name: m.name.clone(),
426                    },
427                );
428            }
429            let reject = |cx: &mut Self, reason: &'static str| {
430                cx.push(
431                    &decl,
432                    ValidationError::InvalidCallbackMethod {
433                        interface: cb.name.clone(),
434                        method: m.name.clone(),
435                        reason,
436                    },
437                );
438            };
439            if m.r#async {
440                reject(self, "cannot be async");
441            }
442            if m.cancellable {
443                reject(self, "cannot be cancellable");
444            }
445            let display = format!("{}.{}", cb.name, m.name);
446            let method = within(&decl, &[&m.name]);
447            self.throws(&method, &display, m);
448            if let Some(ret) = &m.returns {
449                let location = || format!("return type of {display}");
450                if contains_iterator(ret) {
451                    reject(self, "cannot return an iterator");
452                }
453                self.type_ref(&method, ret);
454                self.interface_positions(&method, ret, &location);
455                self.no_callback_interface(&method, ret, &location);
456            }
457            self.params(&method, &display, m);
458            for p in &m.params {
459                let param = within(&method, &[&p.name]);
460                let location = || {
461                    format!(
462                        "param '{}' of callback interface method '{display}'",
463                        p.name
464                    )
465                };
466                if contains_iterator(&p.ty) {
467                    self.push(
468                        &param,
469                        ValidationError::IteratorInInvalidPosition {
470                            location: location(),
471                        },
472                    );
473                }
474                self.type_ref(&param, &p.ty);
475                self.interface_positions(&param, &p.ty, &location);
476                self.no_callback_interface(&param, &p.ty, &location);
477            }
478        }
479    }

481    /// Valid, unique, slot-free parameter names.
482    fn params(&mut self, decl: &[String], display: &str, f: &Function) {
483        let mut names = BTreeSet::new();
484        for p in &f.params {
485            self.identifier(decl, &p.name);
486            if !names.insert(&p.name) {
487                self.push(
488                    decl,
489                    ValidationError::DuplicateParamName {
490                        function: display.to_string(),
491                        param: p.name.clone(),
492                    },
493                );
494            }
495            if ABI_SLOTS.contains(&p.name.as_str()) {
496                self.push(
497                    decl,
498                    ValidationError::SlotCollision {
499                        function: display.to_string(),
500                        slot: p.name.clone(),
501                    },
502                );
503            }
504        }
505    }

507    /// Name-level checks for one callable declared in `scope`: a valid
508    /// identifier, valid parameter names, and an existing error domain when
509    /// the callable throws one.
510    fn function(&mut self, scope: &[String], display: &str, f: &Function) {
511        self.identifier(scope, &f.name);
512        let decl = within(scope, &[&f.name]);
513        if f.cancellable && !f.r#async {
514            self.push(
515                &decl,
516                ValidationError::CancellableNotAsync {
517                    function: display.to_string(),
518                },
519            );
520        }
521        self.throws(&decl, display, f);
522        self.params(&decl, display, f);
523    }

525    /// Check that a callable's `throws` names an error domain or is `any`.
526    fn throws(&mut self, scope: &[String], display: &str, f: &Function) {
527        if let Some(Throws::Domain(domain)) = &f.throws {
528            if self.kind(domain) != Some(TypeKind::ErrorDomain) {
529                self.push(
530                    scope,
531                    ValidationError::UnknownErrorDomain {
532                        function: display.to_string(),
533                        domain: domain.clone(),
534                    },
535                );
536            }
537        }
538    }

540    /// Type-level checks for one callable's parameters and return: iterator
541    /// positions, async-iterator exclusion, reference resolution, element
542    /// shapes, interface map keys, and callback-interface positions. `scope`
543    /// ends with the callable's own name.
544    fn callable_types(&mut self, scope: &[String], display: &str, f: &Function) {
545        for p in &f.params {
546            let param = within(scope, &[&p.name]);
547            let location = || format!("param '{}' of function '{display}'", p.name);
548            if contains_iterator(&p.ty) {
549                self.push(
550                    &param,
551                    ValidationError::IteratorInInvalidPosition {
552                        location: location(),
553                    },
554                );
555            }
556            self.type_ref(&param, &p.ty);
557            self.interface_positions(&param, &p.ty, &location);
558            // A bare or optional callback interface is the one legal
559            // callback position; anything nested (`[Listener]`, a record
560            // field) is not.
561            let top = match &p.ty {
562                TypeRef::Optional(inner) => inner.as_ref(),
563                other => other,
564            };
565            match top {
566                TypeRef::Named(name) if self.kind(name) == Some(TypeKind::CallbackInterface) => {}
567                _ => self.no_callback_interface(&param, &p.ty, &location),
568            }
569        }
570        if let Some(ret) = &f.returns {
571            let location = || format!("return type of {display}");
572            if f.r#async && contains_iterator(ret) {
573                self.push(
574                    scope,
575                    ValidationError::AsyncIteratorReturn {
576                        function: display.to_string(),
577                    },
578                );
579            }
580            // An iterator is valid only as the outermost return shape.
581            let nested_iterator = match ret {
582                TypeRef::Iterator(elem) => contains_iterator(elem),
583                other => contains_iterator(other),
584            };
585            if nested_iterator {
586                self.push(
587                    scope,
588                    ValidationError::IteratorInInvalidPosition {
589                        location: format!("nested inside the {}", location()),
590                    },
591                );
592            }
593            self.type_ref(scope, ret);
594            self.interface_positions(scope, ret, &location);
595            self.no_callback_interface(scope, ret, &location);
596        }
597    }

599    /// Enforce the one place an interface reference may not appear: as a
600    /// map key. No target can hash an object by identity in a way that
601    /// survives the ABI.
602    fn interface_positions(
603        &mut self,
604        scope: &[String],
605        ty: &TypeRef,
606        location: &dyn Fn() -> String,
607    ) {
608        let mut keys = Vec::new();
609        ty.walk(&mut |t| {
610            if let TypeRef::Map(k, _) = t {
611                if let TypeRef::Named(name) = &**k {
612                    keys.push(name.clone());
613                }
614            }
615        });
616        for name in keys {
617            if self.kind(&name) == Some(TypeKind::Interface) {
618                self.push(
619                    scope,
620                    ValidationError::InterfaceInInvalidPosition {
621                        name,
622                        location: format!("map key of {}", location()),
623                    },
624                );
625            }
626        }
627    }

629    /// Reject an error-code field that carries an interface object, directly
630    /// or through a record or rich enum. Error domains are open: a consumer
631    /// generated before the code existed discards its payload undecoded, so
632    /// an object token inside it would never be released.
633    fn no_object_in_payload(&mut self, scope: &[String], f: &Field, location: &dyn Fn() -> String) {
634        let mut names = Vec::new();
635        f.ty.walk(&mut |t| {
636            if let TypeRef::Named(name) = t {
637                names.push(name.clone());
638            }
639        });
640        for name in names {
641            if self.kind(&name) == Some(TypeKind::Interface) || self.carriers.contains(&name) {
642                self.push(
643                    &within(scope, &[&f.name]),
644                    ValidationError::ObjectInErrorPayload {
645                        name,
646                        location: location(),
647                    },
648                );
649            }
650        }
651    }

653    /// Reject any callback-interface reference reachable from `ty`. Callers
654    /// that permit a bare top-level callback interface unwrap it first.
655    fn no_callback_interface(
656        &mut self,
657        scope: &[String],
658        ty: &TypeRef,
659        location: &dyn Fn() -> String,
660    ) {
661        let mut names = Vec::new();
662        ty.walk(&mut |t| {
663            if let TypeRef::Named(name) = t {
664                names.push(name.clone());
665            }
666        });
667        for name in names {
668            if self.kind(&name) == Some(TypeKind::CallbackInterface) {
669                self.push(
670                    scope,
671                    ValidationError::CallbackInterfaceInInvalidPosition {
672                        name,
673                        location: location(),
674                    },
675                );
676            }
677        }
678    }

680    /// Check that every name in `ty` resolves to a declaration, and that
681    /// every map key is a legal key type.
682    fn type_ref(&mut self, scope: &[String], ty: &TypeRef) {
683        match ty {
684            TypeRef::Named(name) => {
685                let error = if self.kind(name) == Some(TypeKind::ErrorDomain) {
686                    ValidationError::ErrorDomainAsType { name: name.clone() }
687                } else if self.kind(name).is_some() {
688                    return;
689                } else if UNSUPPORTED_PRIMITIVES.contains(&name.as_str()) {
690                    ValidationError::UnsupportedPrimitive { name: name.clone() }
691                } else {
692                    ValidationError::UnknownTypeRef { name: name.clone() }
693                };
694                self.push(scope, error);
695            }
696            TypeRef::Optional(inner) | TypeRef::List(inner) | TypeRef::Iterator(inner) => {
697                self.type_ref(scope, inner);
698            }
699            TypeRef::Map(k, v) => {
700                if !self.is_map_key(k) {
701                    self.push(
702                        scope,
703                        ValidationError::InvalidMapKey {
704                            key_type: k.to_string(),
705                        },
706                    );
707                }
708                self.type_ref(scope, k);
709                self.type_ref(scope, v);
710            }
711            TypeRef::Prim(_) => {}
712        }
713    }
714
715    /// May `ty` be a map key? Only integers, bools, strings, and C-style
716    /// enums: every target must use the key in its native dictionary with
717    /// exact equality.
718    fn is_map_key(&self, ty: &TypeRef) -> bool {
719        match ty {
720            TypeRef::Prim(p) => p.is_integer() || matches!(p, Prim::Bool | Prim::String),
721            TypeRef::Named(name) => self.kind(name) == Some(TypeKind::Enum),
722            _ => false,
723        }
724    }
725
726    fn error_domain(&mut self, domain: &ErrorDomain) {
727        let top: &[String] = &[];
728        self.identifier(top, &domain.name);
729        if domain.name == crate::ir::THROWS_ANY {
730            // `throws: any` means an untyped error, so no domain can be
731            // named `any`.
732            self.push(
733                top,
734                ValidationError::ReservedKeyword {
735                    name: domain.name.clone(),
736                },
737            );
738        }
739        let decl = within(top, &[&domain.name]);
740        let mut values = BTreeSet::new();
741        for c in &domain.codes {
742            self.identifier(&decl, &c.name);
743            // 0 means success and the whole negative range is reserved for
744            // the runtime, so domain codes must be positive.
745            if c.code <= 0 {
746                self.push(
747                    &decl,
748                    ValidationError::InvalidErrorCode {
749                        domain: domain.name.clone(),
750                        name: c.name.clone(),
751                    },
752                );
753            }
754            if !values.insert(c.code) {
755                self.push(
756                    &decl,
757                    ValidationError::DuplicateErrorCode {
758                        domain: domain.name.clone(),
759                        value: c.code,
760                    },
761                );
762            }
763            let code = within(&decl, &[&c.name]);
764            for f in &c.fields {
765                let location = || {
766                    format!(
767                        "payload field '{}' of error code '{}.{}'",
768                        f.name, domain.name, c.name
769                    )
770                };
771                self.buffered_field(&code, f, &location);
772                self.no_object_in_payload(&code, f, &location);
773            }
774        }
775    }
776}

778/// Every record and rich enum whose fields reach an interface object, directly
779/// or through other records and rich enums (a fixed point over the whole API,
780/// since types may reference each other in any order).
781fn object_carriers(api: &Api, types: &TypeIndex) -> BTreeSet<String> {
782    let mut decls: Vec<(&str, Vec<&TypeRef>)> = Vec::new();
783    for r in &api.records {
784        decls.push((&r.name, r.fields.iter().map(|f| &f.ty).collect()));
785    }
786    for e in &api.enums {
787        let fields = e.variants.iter().flat_map(|v| &v.fields);
788        decls.push((&e.name, fields.map(|f| &f.ty).collect()));
789    }
790    let mut carriers = BTreeSet::new();
791    loop {
792        let before = carriers.len();
793        for (name, fields) in &decls {
794            if carriers.contains(*name) {
795                continue;
796            }
797            let mut carries = false;
798            for ty in fields {
799                ty.walk(&mut |t| {
800                    if let TypeRef::Named(n) = t {
801                        carries |=
802                            types.kind(n) == Some(TypeKind::Interface) || carriers.contains(n);
803                    }
804                });
805            }
806            if carries {
807                carriers.insert((*name).to_string());
808            }
809        }
810        if carriers.len() == before {
811            return carriers;
812        }
813    }
814}

819    found
