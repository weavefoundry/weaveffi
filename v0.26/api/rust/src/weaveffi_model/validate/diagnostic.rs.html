1//! [`ValidationDiagnostic`]: a [`ValidationError`] plus an optional source
2//! snippet and best-effort span for fancy miette rendering.
3//!
4//! The span is located within the error's enclosing declaration: the
5//! validator records the declaration path (the declaration, then member
6//! names), and the search walks that path through the source,
7//! finding each name as the value of a `name` key, before looking for the
8//! offending text. So a duplicate parameter `x` underlines the second `x`
9//! of the right function, not the first `x` anywhere in the file.
10//!
11//! The source snippet, the span, and the miette `Diagnostic` impl are only
12//! available with the `idl` feature; without it the wrapper carries the
13//! error alone.
14

19/// Diagnostic wrapper that attaches an optional source code snippet and a
20/// best-effort byte range to a [`ValidationError`] for fancy rendering via
21/// miette. The wrapper delegates `help()` and `code()` to the inner error
22/// while exposing its own `source_code` and `labels` so the renderer can
23/// underline the offending identifier in the input.
24#[derive(Debug)]
25pub struct ValidationDiagnostic {
26    /// The underlying validation error being rendered.
27    pub error: ValidationError,
28    /// Named source snippet (filename plus contents), when an on-disk IDL was
29    /// supplied. `None` for in-memory APIs.
30    #[cfg(feature = "idl")]
31    pub src: Option<NamedSource<String>>,
32    /// Best-effort byte range of the offending identifier within `src`, used
33    /// to underline it. `None` when no span could be located.
34    #[cfg(feature = "idl")]
35    pub span: Option<SourceSpan>,
36}

50#[cfg(feature = "idl")]
51impl Diagnostic for ValidationDiagnostic {
52    fn code<'a>(&'a self) -> Option<Box<dyn std::fmt::Display + 'a>> {
53        self.error.code()
54    }
55
56    fn severity(&self) -> Option<miette::Severity> {
57        self.error.severity()
58    }
59
60    fn help<'a>(&'a self) -> Option<Box<dyn std::fmt::Display + 'a>> {
61        self.error.help()
62    }
63
64    fn url<'a>(&'a self) -> Option<Box<dyn std::fmt::Display + 'a>> {
65        self.error.url()
66    }
67
68    fn source_code(&self) -> Option<&dyn miette::SourceCode> {
69        self.src
70            .as_ref()
71            .map(|s| s as &dyn miette::SourceCode)
72            .or_else(|| self.error.source_code())
73    }
74
75    fn labels(&self) -> Option<Box<dyn Iterator<Item = miette::LabeledSpan> + '_>> {
76        if let Some(span) = self.span {
77            Some(Box::new(std::iter::once(
78                miette::LabeledSpan::new_with_span(Some("here".to_string()), span),
79            )))
80        } else {
81            self.error.labels()
82        }
83    }
84}

86impl ValidationDiagnostic {
87    /// Wrap `error`, located within the declaration path `scope`, with an
88    /// optional `(filename, contents)` source. When a source is provided the
89    /// constructor searches for the offending text within the enclosing
90    /// declaration and attaches a [`SourceSpan`] for fancy rendering. If no
91    /// span can be computed the label is omitted and miette still produces
92    /// a nicer message + help section than plain `Display`.
93    #[cfg(feature = "idl")]
94    pub(crate) fn new(
95        error: ValidationError,
96        scope: &[String],
97        source: Option<(&str, &str)>,
98    ) -> Self {
99        let (src, span) = match source {
100            Some((filename, contents)) => {
101                let span = error
102                    .needle()
103                    .and_then(|needle| locate(contents, scope, needle));
104                (Some(NamedSource::new(filename, contents.to_string())), span)
105            }
106            None => (None, None),
107        };
108        Self { error, src, span }
109    }
110
111    /// Wrap `error`. Without the `idl` feature there's no miette rendering,
112    /// so the scope and source are accepted for signature parity and
113    /// ignored.
114    #[cfg(not(feature = "idl"))]
115    pub(crate) fn new(
116        error: ValidationError,
117        _scope: &[String],
118        _source: Option<(&str, &str)>,
119    ) -> Self {
120        Self { error }
121    }
122}

124/// What to look for in the source, after walking the enclosing declaration
125/// path.
126#[cfg(feature = "idl")]
127#[derive(Clone, Copy)]
128enum Needle<'a> {
129    /// A declared name: the value of a `name` key.
130    Decl(&'a str),
131    /// The second declaration of a name (the duplicate).
132    Second(&'a str),
133    /// Any other text, such as a type reference, as a whole token.
134    Text(&'a str),
135    /// The innermost declaration of the scope itself.
136    Scope,
137}
138
139#[cfg(feature = "idl")]
140impl ValidationError {
141    /// The text a diagnostic underlines, or `None` when no one spot in the
142    /// source is to blame.
143    fn needle(&self) -> Option<Needle<'_>> {
144        use Needle::{Decl, Scope, Second, Text};
145        Some(match self {
146            Self::UnsupportedSchemaVersion { version, .. } => Text(version),
147            Self::ReservedKeyword { name }
148            | Self::InvalidIdentifier { name, .. }
149            | Self::InvalidErrorCode { name, .. }
150            | Self::DuplicateErrorCodeName { name, .. } => Decl(name),
151            Self::SlotCollision { slot: name, .. } => Decl(name),
152            Self::DuplicateName { name, .. }
153            | Self::DuplicateParamName { param: name, .. }
154            | Self::DuplicateRecordField { field: name, .. }
155            | Self::DuplicateEnumVariant { variant: name, .. }
156            | Self::DuplicateEnumVariantField { field: name, .. }
157            | Self::DuplicateInterfaceMember { name, .. }
158            | Self::DuplicateCallbackMethod { name, .. } => Second(name),
159            Self::CancellableNotAsync { .. }
160            | Self::AsyncIteratorReturn { .. }
161            | Self::EmptyRecord { .. }
162            | Self::EmptyEnum { .. }
163            | Self::EmptyInterface { .. }
164            | Self::EmptyCallbackInterface { .. }
165            | Self::RecursiveType { .. }
166            | Self::DuplicateEnumValue { .. } => Scope,
167            Self::UnknownErrorDomain { domain, .. } => Text(domain),
168            Self::ConstructorHasReturn { constructor, .. }
169            | Self::AsyncConstructor { constructor, .. } => Decl(constructor),
170            Self::InvalidCallbackMethod { method, .. } => Decl(method),
171            Self::UnknownTypeRef { name }
172            | Self::UnsupportedPrimitive { name }
173            | Self::ErrorDomainAsType { name }
174            | Self::InterfaceInInvalidPosition { name, .. }
175            | Self::CallbackInterfaceInInvalidPosition { name, .. }
176            | Self::ObjectInErrorPayload { name, .. } => Text(name),
177            Self::InvalidMapKey { key_type } => Text(key_type),
178            Self::IteratorInInvalidPosition { .. } => Text("iter"),
179            Self::SymbolCollision { .. } | Self::DuplicateErrorCode { .. } => return None,
180        })
181    }
182}

184/// Locate `needle` in `src` within the declaration path `scope`, falling
185/// back to its first occurrence anywhere when the scoped search fails.
186#[cfg(feature = "idl")]
187fn locate(src: &str, scope: &[String], needle: Needle<'_>) -> Option<SourceSpan> {
188    // A needle naming the scope itself searches for the innermost name
189    // within the scope around it.
190    let (scope, needle) = match (needle, scope.split_last()) {
191        (Needle::Scope, Some((last, outer))) => (outer, Needle::Decl(last)),
192        (Needle::Scope, None) => return None,
193        _ => (scope, needle),
194    };
195    let mut from = 0;
196    for anchor in scope {
197        if let Some(at) = find_decl(src, from, anchor) {
198            from = at + anchor.len();
199        }
200    }
201    let (text, scoped) = match needle {
202        Needle::Decl(name) => (name, find_decl(src, from, name)),
203        Needle::Second(name) => (
204            name,
205            find_decl(src, from, name)
206                .and_then(|first| find_decl(src, first + name.len(), name))
207                .or_else(|| find_decl(src, from, name)),
208        ),
209        Needle::Text(text) => (text, find_token(src, from, text)),
210        Needle::Scope => unreachable!("resolved above"),
211    };
212    let at = scoped.or_else(|| find_token(src, 0, text))?;
213    Some(span(src, at, text.len()))
214}

216#[cfg(feature = "idl")]
217fn is_ident_char(c: char) -> bool {
218    c.is_ascii_alphanumeric() || c == '_'
219}
220
221/// The byte offset of the first whole-token occurrence of `needle` at or
222/// after `from`: one not glued to identifier characters on either side.
223#[cfg(feature = "idl")]
224fn find_token(src: &str, from: usize, needle: &str) -> Option<usize> {
225    if needle.is_empty() {
226        return None;
227    }
228    src.get(from..)?
229        .match_indices(needle)
230        .map(|(i, _)| from + i)
231        .find(|&at| {
232            let before = src[..at].chars().next_back();
233            let after = src[at + needle.len()..].chars().next();
234            !before.is_some_and(is_ident_char) && !after.is_some_and(is_ident_char)
235        })
236}

238/// The byte offset of the first occurrence of `name` at or after `from` that
239/// is the value of a `name` key, in YAML (`name: x`) or JSON
240/// (`"name": "x"`) syntax.
241#[cfg(feature = "idl")]
242fn find_decl(src: &str, from: usize, name: &str) -> Option<usize> {
243    let mut at = from;
244    loop {
245        let hit = find_token(src, at, name)?;
246        let before = src[..hit].trim_end_matches(['"', '\'']).trim_end();
247        if let Some(key) = before.strip_suffix(':') {
248            let key = key.trim_end().trim_end_matches(['"', '\'']);
249            if key
250                .strip_suffix("name")
251                .is_some_and(|rest| !rest.chars().next_back().is_some_and(is_ident_char))
252            {
253                return Some(hit);
254            }
255        }
256        at = hit + name.len();
257    }
258}

260/// The span of `len` bytes at `at`, widened to its surrounding quotes when
261/// the text is a quoted scalar.
262#[cfg(feature = "idl")]
263fn span(src: &str, at: usize, len: usize) -> SourceSpan {
264    let quote = src[..at]
265        .chars()
266        .next_back()
267        .filter(|c| matches!(c, '"' | '\''));
268    match quote {
269        Some(q) if src[at + len..].starts_with(q) => SourceSpan::new((at - 1).into(), len + 2),
270        _ => SourceSpan::new(at.into(), len),
271    }
