1//! The contract: one fingerprint per declaration, which a generated binding
2//! checks against the library it loads before the first call.
3//!
4//! Every declaration has an entry: each function, interface, interface
5//! member, record, enum, callback interface, callback method, error domain,
6//! and error code. An entry has:
7//!
8//! * an **id**, [`fnv1a64`] of the declaration's dotted path (`open`,
9//!   `Store.get`, `Listener.on_put`, `KvError.NotFound`);
10//! * a **hash**, [`fnv1a64`] of its canonical signature string.
11//!
12//! A producer exports one function per entry, `uint64_t
13//! {prefix}_contract_{id:016x}(void)` ([`symbol`]), returning the hash. A
14//! binding resolves the function of every entry it was generated with and
15//! compares the hash: a missing function means the library lacks the
16//! declaration, and a different hash means it changed. Entries the library
17//! has and the binding doesn't are fine, so adding a declaration (a
18//! function, a method, a record, an error code, or a callback method at the
19//! end of its interface) never breaks a deployed binding.
20//!
21//! The canonical signature is computed from resolved types: aliases are
22//! substituted, `[u8]` is `bytes`, and each type is written in its IDL
23//! spelling by [`canonical`] (not by `Display`, so a cosmetic change to
24//! diagnostics can't change a hash). Parameter and field names, docs,
25//! deprecation text, and error messages are excluded.
26//!
27//! | Declaration | Signature |
28//! |---|---|
29//! | function or interface member | `{kind} {name}({type}, ...) -> {type or void}`, then ` throws {Domain}` or ` throws any`, ` async`, ` cancellable` when set; `kind` is `function`, `constructor`, `method`, or `static` |
30//! | interface | `interface {name}` |
31//! | record | `record {name} {{type}, ...}` |
32//! | enum | `enum {name} {{Variant} = {value}, ...}`, a variant with fields followed by ` {{type}, ...}` |
33//! | callback interface | `callback {name}` |
34//! | callback method | `callback_method #{slot} {name}({type}, ...) -> {type or void}`, then ` throws {Domain}` or ` throws any` when set |
35//! | error domain | `errors {name}` |
36//! | error code | `code {Code} = {value}`, a code with fields followed by ` {{type}, ...}` |
37//!
38//! [`Sig`] builds these strings from parts, so the producer macro, which
39//! knows each type's spelling only at compile time, assembles exactly the
40//! strings the CLI does.
41
42use crate::model::{ErrorStrategy, FieldBinding, FnBinding, Model};

45/// One declaration's contract entry.
46#[derive(Debug, Clone, PartialEq, Eq)]
47pub struct ContractEntry {
48    /// The declaration's dotted path (`Store.put`), which bindings name in
49    /// their load errors.
50    pub path: String,
51    /// [`fnv1a64`] of [`path`](Self::path).
52    pub id: u64,
53    /// [`fnv1a64`] of [`signature`](Self::signature).
54    pub hash: u64,
55    /// The canonical signature (see the [module docs](self)).
56    pub signature: String,
57    /// The producer's function returning the hash, [`symbol`].
58    pub symbol: String,
59}
60
61/// The 64-bit FNV-1a hash of `data`: tiny, dependency-free, and stable
62/// across platforms and Rust versions.
63#[must_use]
64pub const fn fnv1a64(data: &[u8]) -> u64 {
65    let mut hash: u64 = 0xcbf2_9ce4_8422_2325;
66    let mut i = 0;
67    while i < data.len() {
68        hash ^= data[i] as u64;
69        hash = hash.wrapping_mul(0x0000_0100_0000_01b3);
70        i += 1;
71    }
72    hash
73}

75/// The producer's contract function for the declaration `path`:
76/// `{prefix}_contract_{id:016x}`, the id in lower-case hex.
77#[must_use]
78pub fn symbol(prefix: &str, path: &str) -> String {
79    format!("{prefix}_contract_{:016x}", fnv1a64(path.as_bytes()))
80}

82/// The canonical IDL spelling of a value type (`i32`, `[Entry?]`,
83/// `{string:Kind}`).
84#[must_use]
85pub fn canonical(ty: &Ty) -> String {
86    match ty {
87        Ty::Prim(p) => p.snake().to_string(),
88        Ty::Record(n) | Ty::RichEnum(n) | Ty::Enum(n) | Ty::Interface(n) => n.clone(),
89        Ty::Optional(inner) => format!("{}?", canonical(inner)),
90        Ty::List(inner) => format!("[{}]", canonical(inner)),
91        Ty::Map(k, v) => format!("{{{}:{}}}", canonical(k), canonical(v)),
92    }
93}
94
95fn canonical_param(ty: &ParamTy) -> String {
96    match ty {
97        ParamTy::Value(t) => canonical(t),
98        ParamTy::Callback { name, nullable } => {
99            if *nullable {
100                format!("{name}?")
101            } else {
102                name.clone()
103            }
104        }
105    }
106}
107
108fn canonical_ret(ty: Option<&RetTy>) -> String {
109    match ty {
110        None => VOID.to_string(),
111        Some(RetTy::Value(t)) => canonical(t),
112        Some(RetTy::Iterator(t)) => format!("iter<{}>", canonical(t)),
113    }
114}
115
116/// The spelling of a missing return.
117pub const VOID: &str = "void";
118
119/// The ` throws ...` suffix of a strategy: empty, ` throws {Domain}`, or
120/// ` throws any`.
121#[must_use]
122pub fn throws_suffix(error: &ErrorStrategy) -> String {
123    match error {
124        ErrorStrategy::Trap => String::new(),
125        ErrorStrategy::Domain(d) => format!(" throws {d}"),
126        ErrorStrategy::Untyped => " throws any".to_string(),
127    }
128}

130/// One part of a canonical signature: literal text, or a slot whose text
131/// the caller supplies (a type's spelling, a ` throws` suffix).
132#[derive(Debug, Clone, PartialEq, Eq)]
133pub enum Part<T> {
134    /// Literal text.
135    Lit(String),
136    /// A caller-supplied piece.
137    Slot(T),
138}
139
140/// A canonical signature as a sequence of [`Part`]s.
141#[derive(Debug, Clone, PartialEq, Eq)]
142pub struct Sig<T>(pub Vec<Part<T>>);
143
144impl<T> Sig<T> {
145    fn new() -> Self {
146        Sig(Vec::new())
147    }
148
149    fn lit(mut self, s: impl Into<String>) -> Self {
150        let s = s.into();
151        if let Some(Part::Lit(last)) = self.0.last_mut() {
152            last.push_str(&s);
153        } else {
154            self.0.push(Part::Lit(s));
155        }
156        self
157    }
158
159    fn slot(mut self, t: T) -> Self {
160        self.0.push(Part::Slot(t));
161        self
162    }
163
164    fn list(mut self, items: Vec<T>) -> Self {
165        for (i, t) in items.into_iter().enumerate() {
166            if i > 0 {
167                self = self.lit(", ");
168            }
169            self = self.slot(t);
170        }
171        self
172    }
173
174    /// `{kind} {name}({params}) -> {ret}{throws}[ async][ cancellable]`,
175    /// where `ret` spells `void` for no return and `throws` is a
176    /// [`throws_suffix`].
177    #[must_use]
178    pub fn callable(
179        kind: &str,
180        name: &str,
181        params: Vec<T>,
182        ret: T,
183        throws: T,
184        is_async: bool,
185        cancellable: bool,
186    ) -> Self {
187        let mut sig = Self::new()
188            .lit(format!("{kind} {name}("))
189            .list(params)
190            .lit(") -> ")
191            .slot(ret)
192            .slot(throws);
193        if is_async {
194            sig = sig.lit(" async");
195        }
196        if cancellable {
197            sig = sig.lit(" cancellable");
198        }
199        sig
200    }
201
202    /// `callback_method #{slot} {name}({params}) -> {ret}{throws}`.
203    #[must_use]
204    pub fn callback_method(slot: usize, name: &str, params: Vec<T>, ret: T, throws: T) -> Self {
205        Self::new()
206            .lit(format!("callback_method #{slot} {name}("))
207            .list(params)
208            .lit(") -> ")
209            .slot(ret)
210            .slot(throws)
211    }
212
213    /// `record {name} {{fields}}`.
214    #[must_use]
215    pub fn record(name: &str, fields: Vec<T>) -> Self {
216        Self::new()
217            .lit(format!("record {name} {{"))
218            .list(fields)
219            .lit("}")
220    }
221
222    /// `enum {name} {{V = value[ {{types}}], ...}}`.
223    #[must_use]
224    pub fn enumeration(name: &str, variants: Vec<(String, i32, Vec<T>)>) -> Self {
225        let mut sig = Self::new().lit(format!("enum {name} {{"));
226        for (i, (variant, value, fields)) in variants.into_iter().enumerate() {
227            if i > 0 {
228                sig = sig.lit(", ");
229            }
230            sig = sig.case(&variant, value, fields);
231        }
232        sig.lit("}")
233    }
234
235    /// `code {name} = {value}[ {{types}}]`.
236    #[must_use]
237    pub fn code(name: &str, value: i32, fields: Vec<T>) -> Self {
238        Self::new().lit("code ").case(name, value, fields)
239    }
240
241    fn case(self, name: &str, value: i32, fields: Vec<T>) -> Self {
242        let sig = self.lit(format!("{name} = {value}"));
243        if fields.is_empty() {
244            sig
245        } else {
246            sig.lit(" {").list(fields).lit("}")
247        }
248    }
249
250    /// A signature that is all literal text (`interface Store`).
251    #[must_use]
252    pub fn text(s: impl Into<String>) -> Self {
253        Self::new().lit(s)
254    }
255}
256
257impl Sig<String> {
258    /// The signature's text.
259    #[must_use]
260    pub fn render(&self) -> String {
261        self.0
262            .iter()
263            .map(|p| match p {
264                Part::Lit(s) | Part::Slot(s) => s.as_str(),
265            })
266            .collect()
267    }
268}

270/// Every contract entry of `model`, sorted ascending by id.
271#[must_use]
272pub fn entries(model: &Model) -> Vec<ContractEntry> {
273    let prefix = model.prefix();
274    let mut out = Vec::new();
275    let mut push = |path: String, sig: Sig<String>| {
276        let signature = sig.render();
277        out.push(ContractEntry {
278            id: fnv1a64(path.as_bytes()),
279            hash: fnv1a64(signature.as_bytes()),
280            symbol: symbol(prefix, &path),
281            path,
282            signature,
283        });
284    };
285    for f in &model.functions {
286        push(f.path.clone(), callable_signature(f));
287    }
288    for i in &model.interfaces {
289        push(i.name.clone(), Sig::text(format!("interface {}", i.name)));
290        for f in i.members() {
291            push(f.path.clone(), callable_signature(f));
292        }
293    }
294    for r in &model.records {
295        push(r.name.clone(), Sig::record(&r.name, field_types(&r.fields)));
296    }
297    for e in &model.enums {
298        let variants = e
299            .variants
300            .iter()
301            .map(|v| (v.name.clone(), v.value, field_types(&v.fields)))
302            .collect();
303        push(e.name.clone(), Sig::enumeration(&e.name, variants));
304    }
305    for c in &model.callbacks {
306        push(c.name.clone(), Sig::text(format!("callback {}", c.name)));
307        for m in &c.methods {
308            push(
309                format!("{}.{}", c.name, m.name),
310                Sig::callback_method(
311                    m.slot,
312                    &m.name,
313                    field_types(&m.params),
314                    m.ret.as_ref().map_or_else(|| VOID.to_string(), canonical),
315                    throws_suffix(&m.error),
316                ),
317            );
318        }
319    }
320    for d in &model.errors {
321        push(d.name.clone(), Sig::text(format!("errors {}", d.name)));
322        for c in &d.codes {
323            push(
324                format!("{}.{}", d.name, c.name),
325                Sig::code(&c.name, c.value, field_types(&c.fields)),
326            );
327        }
328    }
329    out.sort_by_key(|e| e.id);
330    out
331}
332
333fn field_types(fields: &[FieldBinding]) -> Vec<String> {
334    fields.iter().map(|f| canonical(&f.ty)).collect()
335}
336
337fn callable_signature(f: &FnBinding) -> Sig<String> {
338    Sig::callable(
339        f.kind.keyword(),
340        &f.name,
341        f.params.iter().map(|p| canonical_param(&p.ty)).collect(),
342        canonical_ret(f.ret.as_ref()),
343        throws_suffix(&f.error),
344        f.is_async(),
345        f.cancellable(),
346    )
347}

392    fn signature(model: &Model, path: &str) -> String {
393        let entry = entries(model).into_iter().find(|e| e.path == path);
394        let entry = entry.unwrap_or_else(|| panic!("no entry {path}"));
395        assert_eq!(entry.id, fnv1a64(path.as_bytes()));
396        assert_eq!(entry.hash, fnv1a64(entry.signature.as_bytes()));
397        assert_eq!(entry.symbol, format!("kv_contract_{:016x}", entry.id));
398        entry.signature
399    }
