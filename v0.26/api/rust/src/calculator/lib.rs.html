1//! Calculator sample cdylib: the smallest useful WeaveFFI producer.
2//!
3//! Plain, safe Rust functions: `add`, `divide` (which fails with the typed
4//! `CalcError` domain on a zero divisor), `greet` (a string in and a string
5//! out), `parse` (a second domain, `ParseError`, whose message names the
6//! input), `sqrt` (an untyped `throws any` error), `mean` (a numeric list in,
7//! an optional number out), and `running_total` (a numeric list in and
8//! out). Each `#[weaveffi::...]` attribute generates the `extern "C"`
9//! functions the C ABI (and every generated binding) calls, so this file
10//! contains no `unsafe` glue.
11//!
12//! This is the example the README and the getting-started guide walk
13//! through; `kvstore` is the feature-complete sample.
14
15/// The calculator's error domain.
16#[weaveffi::error]
17#[derive(Debug)]
18pub enum CalcError {
19    /// division by zero
20    DivisionByZero = 1,
21}
22
23/// Add two integers, wrapping on overflow.
24#[weaveffi::export]
25pub fn add(a: i32, b: i32) -> i32 {
26    a.wrapping_add(b)
27}
28
29/// Divide `a` by `b`, rounding toward zero. Fails with `DivisionByZero`
30/// when `b` is zero.
31#[weaveffi::export]
32pub fn divide(a: i32, b: i32) -> Result<i32, CalcError> {
33    if b == 0 {
34        return Err(CalcError::DivisionByZero);
35    }
36    Ok(a.wrapping_div(b))
37}

39/// Greet `name`: `"Hello, {name}!"`.
40#[weaveffi::export]
41pub fn greet(name: String) -> String {
42    format!("Hello, {name}!")
43}
44
45/// The calculator's parsing errors: a second domain.
46#[weaveffi::error]
47#[derive(Debug)]
48#[repr(i32)]
49pub enum ParseError {
50    /// not a number
51    #[weaveffi(message = "not a number: {text}")]
52    NotANumber {
53        /// The text that didn't parse.
54        text: String,
55    } = 1,
56}
57
58/// Parse a decimal integer (surrounding whitespace allowed). Fails with
59/// `NotANumber` carrying the input.
60#[weaveffi::export]
61pub fn parse(text: &str) -> Result<i32, ParseError> {
62    text.trim().parse().map_err(|_| ParseError::NotANumber {
63        text: text.to_string(),
64    })
65}
66
67/// The square root of `x`. Fails with an untyped error (`throws any`) whose
68/// message is `"cannot take the square root of {x}"` for a negative `x`.
69#[weaveffi::export]
70pub fn sqrt(x: f64) -> Result<f64, String> {
71    if x < 0.0 {
72        return Err(format!("cannot take the square root of {x}"));
73    }
74    Ok(x.sqrt())
75}
76
77/// The arithmetic mean of `values`, or none for an empty list.
78#[weaveffi::export]
79pub fn mean(values: &[f64]) -> Option<f64> {
80    if values.is_empty() {
81        return None;
82    }
83    Some(values.iter().sum::<f64>() / values.len() as f64)
84}
85
86/// The running totals of `values` (wrapping on overflow): `[1, 2, 3]`
87/// gives `[1, 3, 6]`.
88#[weaveffi::export]
89pub fn running_total(values: Vec<i32>) -> Vec<i32> {
90    values
91        .iter()
92        .scan(0i32, |total, v| {
93            *total = total.wrapping_add(*v);
94            Some(*total)
95        })
96        .collect()
97}

101#[cfg(test)]
102#[allow(unsafe_code)]
103mod tests {
104    use weaveffi::__rt::{FfiError, Opt, Run};
105
106    // The library's own C functions, as a consumer calls them.
107    unsafe extern "C" {
108        fn calculator_add(a: i32, b: i32, err: *mut FfiError) -> i32;
109        fn calculator_divide(a: i32, b: i32, err: *mut FfiError) -> i32;
110        fn calculator_greet(name: Run, err: *mut FfiError) -> Run;
111        fn calculator_parse(text: Run, err: *mut FfiError) -> i32;
112        fn calculator_sqrt(x: f64, err: *mut FfiError) -> f64;
113        fn calculator_mean(values: Run, err: *mut FfiError) -> Opt<f64>;
114        fn calculator_running_total(values: Run, err: *mut FfiError) -> Run;
115        fn calculator_contract_e70bc3190530e654() -> u64;
116    }
117
118    fn message(err: &FfiError) -> &str {
119        unsafe { err.message_str() }.unwrap_or_default()
120    }
121
122    fn bytes<T: Copy>(values: &[T]) -> Run {
123        Run::borrowing(unsafe {
124            std::slice::from_raw_parts(values.as_ptr().cast(), std::mem::size_of_val(values))
125        })
126    }
127
128    #[test]
129    fn scalars_and_typed_errors() {
130        let mut err = FfiError::default();
131        assert_eq!(unsafe { calculator_add(2, 40, &mut err) }, 42);
132        assert_eq!(err.code, 0);
133        assert_eq!(unsafe { calculator_add(i32::MAX, 1, &mut err) }, i32::MIN);
134        assert_eq!(unsafe { calculator_divide(-7, 2, &mut err) }, -3);
135        assert_eq!(unsafe { calculator_divide(1, 0, &mut err) }, 0);
136        assert_eq!(err.code, 1);
137        assert_eq!(message(&err), "division by zero");
138        unsafe { weaveffi::__rt::error_clear(&mut err) };
139    }
140
141    #[test]
142    fn strings_round_trip() {
143        let mut err = FfiError::default();
144        let out = unsafe { calculator_greet(Run::borrowing("Wörld 🦀".as_bytes()), &mut err) };
145        assert_eq!(err.code, 0);
146        assert_eq!(unsafe { out.text() }.unwrap(), "Hello, Wörld 🦀!");
147        unsafe { out.free() };
148    }
149
150    #[test]
151    fn domains_carry_payloads_and_untyped_errors_messages() {
152        let mut err = FfiError::default();
153        assert_eq!(
154            unsafe { calculator_parse(Run::borrowing(b" 42 "), &mut err) },
155            42
156        );
157        assert_eq!(
158            unsafe { calculator_parse(Run::borrowing(b"4x"), &mut err) },
159            0
160        );
161        assert_eq!((err.code, message(&err)), (1, "not a number: 4x"));
162        assert_eq!(err.payload(), &[2, 0, 0, 0, b'4', b'x']);
163        unsafe { weaveffi::__rt::error_clear(&mut err) };
164        assert_eq!(unsafe { calculator_sqrt(9.0, &mut err) }, 3.0);
165        assert_eq!(unsafe { calculator_sqrt(-4.0, &mut err) }, 0.0);
166        assert_eq!(err.code, weaveffi::__rt::GENERIC_ERROR_CODE);
167        assert_eq!(message(&err), "cannot take the square root of -4");
168        unsafe { weaveffi::__rt::error_clear(&mut err) };
169    }
170
171    #[test]
172    fn lists_cross_as_typed_arrays() {
173        let mut err = FfiError::default();
174        let mean = unsafe { calculator_mean(bytes(&[1.0f64, 2.0, 6.0]), &mut err) };
175        assert_eq!((mean.has, mean.value), (1, 3.0));
176        let none = unsafe { calculator_mean(Run::EMPTY, &mut err) };
177        assert_eq!(none.has, 0);
178        let out = unsafe { calculator_running_total(bytes(&[1i32, 2, 3, i32::MAX]), &mut err) };
179        assert_eq!(
180            unsafe { out.slice::<i32>() }.unwrap(),
181            [1, 3, 6, i32::MIN + 5]
182        );
183        unsafe { out.free() };
184    }

186    #[test]
187    fn every_declaration_exports_its_contract() {
188        // `add`'s contract function, by its id: fnv1a64("add").
189        let id = weaveffi_model_free_fnv(b"add");
190        assert_eq!(format!("{id:016x}"), "e70bc3190530e654");
191        assert_eq!(
192            unsafe { calculator_contract_e70bc3190530e654() },
193            weaveffi_model_free_fnv(b"function add(i32, i32) -> i32")
194        );
195    }

197    fn weaveffi_model_free_fnv(data: &[u8]) -> u64 {
198        let mut hash: u64 = 0xcbf2_9ce4_8422_2325;
199        for b in data {
200            hash ^= u64::from(*b);
201            hash = hash.wrapping_mul(0x0000_0100_0000_01b3);
202        }
203        hash
204    }
