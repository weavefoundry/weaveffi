1//! Shared helpers for the WeaveFFI fuzz targets.
2//!
3//! Fuzz targets live in the sibling `*.rs` files in this directory and are
4//! exposed as `[[bin]]` entries in `Cargo.toml`. They require the `fuzzing`
5//! feature (and a nightly toolchain with `cargo-fuzz` installed).
6//!
7//! The value-buffer codecs are the runtime's own (`FfiType`), keyed by the
8//! tag type `export_runtime!()` defines, which this crate defines once for
9//! every target and test.
10
11#![allow(unsafe_code)]
12
13weaveffi::export_runtime!();
14
15use weaveffi::__rt::{encode, BufferDecodeError, BufferReader, FfiType};
16
17/// A type the value-buffer targets decode: a runtime value type.
18pub trait Value: FfiType<__weaveffi::Tag> {}
19
20impl<T: FfiType<__weaveffi::Tag>> Value for T {}
21
22/// Decode a whole buffer as `T` through a reader that never adopts object
23/// tokens, so no input can make the decode unsound.
24///
25/// # Errors
26///
27/// Returns the decoder's error for a malformed buffer or trailing bytes.
28pub fn decode_value<T: Value>(data: &[u8]) -> Result<T, BufferDecodeError> {
29    let mut r = BufferReader::token_free(data);
30    // SAFETY: a token-free reader fails on an object token instead of
31    // adopting it, which is the only unsafe part of a decode.
32    let value = unsafe { T::read(&mut r) }?;
33    r.expect_end()?;
34    Ok(value)
35}
36
37/// Encode `value` as a whole buffer.
38#[must_use]
39pub fn encode_value<T: Value>(value: &T) -> Vec<u8> {
40    encode(|w| value.write(w))
41}
