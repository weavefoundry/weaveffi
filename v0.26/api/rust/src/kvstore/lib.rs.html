1//! Kvstore sample cdylib: an in-memory key-value store that uses every
2//! feature WeaveFFI supports, written as plain, safe Rust.
3//!
4//! This is the feature-complete producer the conformance harness runs in
5//! every language:
6//!
7//! * a reference-counted `Store` interface with fallible and infallible
8//!   constructors, methods, statics, and a deprecated method;
9//! * the `KvError` domain, whose codes carry payload fields, with its
10//!   `Display` generated from message templates;
11//! * records (`Entry`, `StoreInfo`), a C-style enum (`EntryKind`), and a rich
12//!   enum (`Change`), with optionals, lists, and maps (including maps keyed by
13//!   the C-style enum);
14//! * `Store` objects in every position: parameters, returns, optionals,
15//!   lists, map values, record fields, iterator elements, an async result,
16//!   and both a parameter and the return of a callback method;
17//! * four callback interfaces the consumer implements: `Listener` (retained,
18//!   and notified from a producer thread during compaction), `Policy`
19//!   (rich returns, an optional-scalar parameter and return, and methods
20//!   that throw `KvError`, whose typed errors, payload included, reach the
21//!   original caller typed), `Loader` (string, bytes, and optional-object
22//!   returns, passed as an optional parameter), and `Scorer` (a typed array
23//!   in and out);
24//! * optional scalars and numeric lists crossing directly (an optional TTL
25//!   parameter, an optional expiry return, a `[u64]` return), `usize`
26//!   counts, and a `throws any` method (`import_lines`, failing with a
27//!   `String`);
28//! * lazy iterators of strings, records, objects, and optional scalars;
29//! * async methods and functions, including a cancellable one that stops its
30//!   background work cooperatively, with a test hook (`Store::active_jobs`)
31//!   that shows it did, and ones completing with an optional scalar and a
32//!   typed array;
33//! * declarations in other Rust modules (`stats`, `report`) and files, which
34//!   join the library's one namespace.
35//!
36//! Time is a logical clock per store (`now`, advanced by `tick`), so TTLs are
37//! deterministic: an entry put with `ttl_seconds: Some(t)` expires once the
38//! clock reaches `now + t`.
39//!
40//! No lock is held while a consumer callback runs: each operation snapshots
41//! the callbacks it needs, releases its locks, and then calls out, so a
42//! callback may call back into the store.
43

53/// The store's error domain. Each code's fields travel as the error's
54/// payload, so every binding raises a typed error carrying them. The
55/// message of each is its template, filled from the fields.
56#[weaveffi::error]
57#[derive(Debug, Clone, PartialEq, Eq)]
58#[repr(i32)]
59pub enum KvError {
60    /// key not found
61    #[weaveffi(message = "key not found: {key}")]
62    KeyNotFound {
63        /// The key that was looked up.
64        key: String,
65    } = 1001,
66    /// entry expired
67    #[weaveffi(message = "entry {key} expired at {expired_at}")]
68    Expired {
69        /// The expired entry's key.
70        key: String,
71        /// The logical time at which it expired.
72        expired_at: i64,
73    } = 1002,
74    /// store is full
75    #[weaveffi(message = "store is full ({capacity} entries)")]
76    StoreFull {
77        /// The store's capacity.
78        capacity: u32,
79    } = 1003,
80    /// invalid path
81    InvalidPath = 1004,
82    /// write rejected by policy
83    #[weaveffi(message = "write to {key} rejected: {reason}")]
84    Rejected {
85        /// The rejected key.
86        key: String,
87        /// Why the policy rejected it.
88        reason: String,
89    } = 1005,
90    /// a consumer callback failed
91    #[weaveffi(message = "{message}")]
92    CallbackFailed {
93        /// The consumer's message.
94        message: String,
95    } = 1006,
96}
97
98/// A consumer callback that fails with anything but a `KvError` code
99/// (or is called off its thread) fails the store operation with
100/// `CallbackFailed`, carrying its message.
101impl From<ForeignError> for KvError {
102    fn from(e: ForeignError) -> Self {
103        Self::CallbackFailed { message: e.message }
104    }
105}

188/// An observer of a store's changes, retained from `subscribe` until
189/// `unsubscribe`, a failure, or the store's release. Changes made by
190/// `compact` arrive on a producer thread.
191#[weaveffi::callback_interface]
192pub trait Listener: Send + Sync {
193    /// Whether the listener wants changes to `key` (`Cleared` changes
194    /// are always delivered).
195    fn accepts(&self, key: &str) -> Result<bool, ForeignError>;
196
197    /// A change the listener accepted. A failure here (or in `accepts`)
198    /// unsubscribes the listener; the store operation still succeeds.
199    fn on_change(&self, change: &Change) -> Result<(), ForeignError>;
200}

202/// A store's write policy, consulted by every `put` while installed
203/// with `set_policy`.
204#[weaveffi::callback_interface]
205pub trait Policy: Send + Sync {
206    /// The TTL a write of `key` gets, given the TTL the caller
207    /// `requested` (absent for none): return it unchanged to keep it,
208    /// another value to override it, or none to store without one.
209    /// Consulted before `admit`.
210    fn ttl_for(&self, key: &str, requested: Option<i64>) -> Result<Option<i64>, KvError>;
211
212    /// Admit an entry about to be stored, returning it as it should be
213    /// stored. The policy may change its value, kind, TTL, tags, and
214    /// metadata (its key and version are kept). Fail with
215    /// `Rejected` to veto the write: `put` then fails with that same
216    /// error.
217    fn admit(&self, entry: &Entry) -> Result<Entry, KvError>;
218
219    /// The store `key` belongs in: return `home` (the store `put` was
220    /// called on) to keep it there, or another store to redirect it.
221    fn route(&self, key: &str, home: Arc<Store>) -> Result<Arc<Store>, ForeignError>;
222}

232/// A read-through source that `get_or_load` consults on a miss.
233#[weaveffi::callback_interface]
234pub trait Loader: Send + Sync {
235    /// The loader's name, recorded in a loaded entry's metadata under
236    /// `source`.
237    fn name(&self) -> Result<String, ForeignError>;
238
239    /// A store that may already hold `key`, consulted first, if any.
240    fn fallback(&self, key: &str) -> Result<Option<Arc<Store>>, ForeignError>;
241
242    /// The value for `key`. Fail with `KeyNotFound` naming
243    /// `key` when there's none.
244    fn load(&self, key: &str) -> Result<Vec<u8>, KvError>;
245}

257/// An embedded key-value store. Each object owns its entries, logical
258/// clock, listeners, and policy; the last release drops them (and frees
259/// the consumer's callbacks).
260#[weaveffi::interface]
261pub struct Store {
262    path: String,
263    entries: Mutex<BTreeMap<String, Entry>>,
264    clock: AtomicI64,
265    capacity: AtomicU32,
266    listeners: Mutex<Vec<(u32, Arc<dyn Listener>)>>,
267    next_listener: AtomicU32,
268    policy: Mutex<Option<Arc<dyn Policy>>>,
269}
270
271impl Store {
272    fn with_path(path: String) -> Store {
273        Store {
274            path,
275            entries: Mutex::new(BTreeMap::new()),
276            clock: AtomicI64::new(0),
277            capacity: AtomicU32::new(DEFAULT_CAPACITY),
278            listeners: Mutex::new(Vec::new()),
279            next_listener: AtomicU32::new(1),
280            policy: Mutex::new(None),
281        }
282    }
283
284    /// Tell every listener that accepts `change`'s key about it, with no
285    /// lock held, unsubscribing a listener that fails.
286    fn notify(&self, change: &Change) {
287        let key = match change {
288            Change::Put { entry, .. } => Some(entry.key.as_str()),
289            Change::Removed { key, .. } => Some(key.as_str()),
290            Change::Cleared { .. } => None,
291        };
292        let listeners = lock(&self.listeners).clone();
293        for (id, listener) in listeners {
294            let delivered = match key {
295                Some(k) => listener.accepts(k).and_then(|wanted| {
296                    if wanted {
297                        listener.on_change(change)
298                    } else {
299                        Ok(())
300                    }
301                }),
302                None => listener.on_change(change),
303            };
304            if delivered.is_err() {
305                lock(&self.listeners).retain(|(i, _)| *i != id);
306            }
307        }
308    }
309
310    /// Store `entry` (whose key and version this sets), enforcing the
311    /// capacity, and notify listeners.
312    fn insert(&self, mut entry: Entry, key: String) -> Result<Entry, KvError> {
313        let replaced = {
314            let mut entries = lock(&self.entries);
315            let previous = entries.get(&key).map(|e| e.version);
316            let capacity = self.capacity.load(Ordering::Relaxed);
317            if previous.is_none() && entries.len() >= capacity as usize {
318                return Err(KvError::StoreFull { capacity });
319            }
320            entry.version = previous.map_or(1, |v| v.wrapping_add(1));
321            entry.key = key.clone();
322            entries.insert(key, entry.clone());
323            previous.is_some()
324        };
325        self.notify(&Change::Put {
326            entry: entry.clone(),
327            replaced,
328        });
329        Ok(entry)
330    }
331
332    /// The live entries whose keys start with `prefix`, in key order.
333    fn live(&self, prefix: Option<&str>) -> Vec<Entry> {
334        let now = self.now();
335        lock(&self.entries)
336            .values()
337            .filter(|e| !e.is_expired(now))
338            .filter(|e| prefix.is_none_or(|p| e.key.starts_with(p)))
339            .cloned()
340            .collect()
341    }
342
343    fn copy_of(&self, entries: Vec<Entry>, path: String) -> Store {
344        let store = Store::with_path(path);
345        store.clock.store(self.now(), Ordering::Relaxed);
346        *lock(&store.entries) = entries.into_iter().map(|e| (e.key.clone(), e)).collect();
347        store
348    }
349
350    /// Remove every expired entry and notify listeners; returns how many
351    /// were removed.
352    fn sweep(&self) -> u32 {
353        let now = self.now();
354        let removed: Vec<String> = {
355            let mut entries = lock(&self.entries);
356            let keys: Vec<String> = entries
357                .values()
358                .filter(|e| e.is_expired(now))
359                .map(|e| e.key.clone())
360                .collect();
361            for k in &keys {
362                entries.remove(k);
363            }
364            keys
365        };
366        for key in &removed {
367            self.notify(&Change::Removed {
368                key: key.clone(),
369                expired: true,
370            });
371        }
372        removed.len() as u32
373    }
374}

461    /// Store `value` under `key`, returning the stored entry. The
462    /// installed policy (if any) picks the TTL (`ttl_for`), admits the
463    /// entry, and may route it to another store, where it's stored
464    /// instead. Fails with `StoreFull`, or with the policy's failure
465    /// (its own `KvError`, such as `Rejected`, unchanged, and any other
466    /// as `CallbackFailed`).
467    pub fn put(
468        self: Arc<Self>,
469        key: String,
470        value: Vec<u8>,
471        kind: EntryKind,
472        ttl_seconds: Option<i64>,
473    ) -> Result<Entry, KvError> {
474        let policy = lock(&self.policy).clone();
475        let ttl_seconds = match &policy {
476            Some(policy) => policy.ttl_for(&key, ttl_seconds)?,
477            None => ttl_seconds,
478        };
479        let mut entry = Entry {
480            key: key.clone(),
481            value,
482            kind,
483            version: 0,
484            expires_at: ttl_seconds.map(|t| self.now() + t),
485            tags: Vec::new(),
486            metadata: BTreeMap::new(),
487        };
488        let target = match policy {
489            Some(policy) => {
490                entry = policy.admit(&entry)?;
491                policy.route(&key, Arc::clone(&self))?
492            }
493            None => self,
494        };
495        target.insert(entry, key)
496    }

498    /// The live entry for `key`. Fails with `KeyNotFound`, or with
499    /// `Expired` for an entry whose TTL elapsed, which is
500    /// then removed (and listeners told).
501    pub fn get(&self, key: String) -> Result<Entry, KvError> {
502        let now = self.now();
503        let expired_at = {
504            let mut entries = lock(&self.entries);
505            match entries.get(&key) {
506                None => return Err(KvError::KeyNotFound { key }),
507                Some(e) if !e.is_expired(now) => return Ok(e.clone()),
508                Some(e) => {
509                    let at = e.expires_at.unwrap_or(now);
510                    entries.remove(&key);
511                    at
512                }
513            }
514        };
515        self.notify(&Change::Removed {
516            key: key.clone(),
517            expired: true,
518        });
519        Err(KvError::Expired { key, expired_at })
520    }

522    /// The live entry for `key`, if any (an expired entry stays until
523    /// `get` or `compact` removes it).
524    pub fn find(&self, key: String) -> Option<Entry> {
525        let now = self.now();
526        lock(&self.entries)
527            .get(&key)
528            .filter(|e| !e.is_expired(now))
529            .cloned()
530    }

532    /// The live entry for `key`, else one from `loader`: its `fallback`
533    /// store's entry if that has one, otherwise a new `Volatile` entry
534    /// holding what `load` returns, with metadata `source` set to the
535    /// loader's name (either way it's stored here). Returns no entry
536    /// with no loader, or when `load` fails with `KeyNotFound` for
537    /// this same key; any other loader failure fails the call (a
538    /// `KvError` unchanged, anything else as `CallbackFailed`).
539    pub fn get_or_load(
540        &self,
541        key: String,
542        loader: Option<Arc<dyn Loader>>,
543    ) -> Result<Option<Entry>, KvError> {
544        if let Some(entry) = self.find(key.clone()) {
545            return Ok(Some(entry));
546        }
547        let Some(loader) = loader else {
548            return Ok(None);
549        };
550        if let Some(entry) = loader.fallback(&key)?.and_then(|s| s.find(key.clone())) {
551            return Ok(Some(self.insert(entry, key)?));
552        }
553        let value = match loader.load(&key) {
554            Ok(value) => value,
555            Err(KvError::KeyNotFound { key: missing }) if missing == key => return Ok(None),
556            Err(e) => return Err(e),
557        };
558        let entry = Entry {
559            key: key.clone(),
560            value,
561            kind: EntryKind::Volatile,
562            version: 0,
563            expires_at: None,
564            tags: Vec::new(),
565            metadata: BTreeMap::from([("source".to_string(), loader.name()?)]),
566        };
567        Ok(Some(self.insert(entry, key)?))
568    }

570    /// Remove `key`, returning whether it existed (listeners are told).
571    pub fn delete(&self, key: String) -> bool {
572        if lock(&self.entries).remove(&key).is_none() {
573            return false;
574        }
575        self.notify(&Change::Removed {
576            key,
577            expired: false,
578        });
579        true
580    }

583    /// told once).

585        let count = {
586            let mut entries = lock(&self.entries);
587            let n = entries.len() as u32;
588            entries.clear();
589            n
590        };
591        self.notify(&Change::Cleared { count });
592        count

620    /// The live keys ordered by `scorer`'s scores, highest first (ties
621    /// keep key order). Fails with `CallbackFailed` when the scorer
622    /// fails or returns a score count that doesn't match the entries.
623    pub fn rank(&self, scorer: Arc<dyn Scorer>) -> Result<Vec<String>, KvError> {
624        let entries = self.live(None);
625        let sizes: Vec<u64> = entries.iter().map(|e| e.value.len() as u64).collect();
626        let scores = scorer.scores(&sizes)?;
627        if scores.len() != entries.len() {
628            return Err(KvError::CallbackFailed {
629                message: format!("expected {} scores, got {}", entries.len(), scores.len()),
630            });
631        }
632        let mut ranked: Vec<(f64, String)> = scores
633            .into_iter()
634            .zip(entries.into_iter().map(|e| e.key))
635            .collect();
636        ranked.sort_by(|a, b| b.0.total_cmp(&a.0));
637        Ok(ranked.into_iter().map(|(_, k)| k).collect())
638    }

640    /// Store one `Volatile` entry per line of `text`, each written
641    /// `key=value` (blank lines are skipped), returning how many were
642    /// stored. Fails, as an untyped error, with `line {n}: expected
643    /// key=value` (`n` counts from 1) at the first malformed line, after
644    /// storing the lines before it.
645    pub fn import_lines(self: Arc<Self>, text: &str) -> Result<usize, String> {
646        let mut stored = 0;
647        for (i, line) in text.lines().enumerate() {
648            if line.trim().is_empty() {
649                continue;
650            }
651            let Some((key, value)) = line.split_once('=') else {
652                return Err(format!("line {}: expected key=value", i + 1));
653            };
654            Arc::clone(&self)
655                .put(
656                    key.to_string(),
657                    value.as_bytes().to_vec(),
658                    EntryKind::Volatile,
659                    None,
660                )
661                .map_err(|e| e.to_string())?;
662            stored += 1;
663        }
664        Ok(stored)
665    }
666
667    /// The live keys in order, optionally only those starting with
668    /// `prefix`, pulled lazily from a snapshot. Fails with
669    /// `KeyNotFound` naming the prefix when a prefix matches
670    /// nothing.
671    pub fn keys(&self, prefix: Option<String>) -> Result<weaveffi::Iter<String>, KvError> {
672        let keys: Vec<String> = self
673            .live(prefix.as_deref())
674            .into_iter()
675            .map(|e| e.key)
676            .collect();
677        match prefix {
678            Some(p) if keys.is_empty() => Err(KvError::KeyNotFound { key: p }),
679            _ => Ok(weaveffi::Iter::new(keys)),
680        }
681    }
682
683    /// The live entries in key order, optionally only those whose keys
684    /// start with `prefix`.
685    pub fn entries(&self, prefix: Option<String>) -> weaveffi::Iter<Entry> {
686        weaveffi::Iter::new(self.live(prefix.as_deref()))
687    }
688
689    /// Each live entry's expiry time (none for an entry without a TTL),
690    /// in key order, pulled lazily from a snapshot.
691    pub fn expirations(&self) -> weaveffi::Iter<Option<i64>> {
692        weaveffi::Iter::new(self.live(None).into_iter().map(|e| e.expires_at))
693    }

695    /// One new store per prefix, created lazily as the iterator is
696    /// pulled, holding copies of the live entries under that prefix
697    /// (with the prefix as its path).
698    pub fn partition(self: Arc<Self>, prefixes: Vec<String>) -> weaveffi::Iter<Arc<Store>> {
699        weaveffi::Iter::new(prefixes.into_iter().map(move |p| {
700            let entries = self.live(Some(&p));
701            Arc::new(self.copy_of(entries, p))
702        }))
703    }

705    /// Subscribe `listener`, returning its subscription id.
706    pub fn subscribe(&self, listener: Arc<dyn Listener>) -> u32 {
707        let id = self.next_listener.fetch_add(1, Ordering::Relaxed);
708        lock(&self.listeners).push((id, listener));
709        id
710    }
711
712    /// Unsubscribe a listener by id, returning whether it was
713    /// subscribed (its consumer `free` runs once no call holds it).
714    pub fn unsubscribe(&self, id: u32) -> bool {
715        let removed = {
716            let mut listeners = lock(&self.listeners);
717            let at = listeners.iter().position(|(i, _)| *i == id);
718            at.map(|at| listeners.remove(at))
719        };
720        // Released here, with no lock held.
721        removed.is_some()
722    }
723
724    /// The number of subscribed listeners.
725    pub fn listener_count(&self) -> usize {
726        lock(&self.listeners).len()
727    }
728
729    /// Install a write policy, replacing (and releasing) any previous
730    /// one; passing none removes it.
731    pub fn set_policy(&self, policy: Option<Arc<dyn Policy>>) {
732        let previous = std::mem::replace(&mut *lock(&self.policy), policy);
733        drop(previous);
734    }
735
736    /// Whether a policy is installed.
737    pub fn has_policy(&self) -> bool {
738        lock(&self.policy).is_some()
739    }
740
741    /// Another reference to this same store.
742    pub fn share(self: Arc<Self>) -> Arc<Store> {
743        self
744    }
745
746    /// A new, independent store holding a copy of every live entry (with
747    /// the same path and clock, and no listeners or policy).
748    pub fn fork(&self) -> Arc<Store> {
749        Arc::new(self.copy_of(self.live(None), self.path.clone()))
750    }
751
752    /// Whichever of this store and `other` holds more live entries (this
753    /// one on a tie), or no store when `other` is absent and this store is
754    /// empty.
755    pub fn larger(self: Arc<Self>, other: Option<Arc<Store>>) -> Option<Arc<Store>> {
756        match other {
757            Some(o) if o.count() > self.count() => Some(o),
758            Some(_) => Some(self),
759            None if self.count() > 0 => Some(self),
760            None => None,
761        }
762    }
763
764    /// Snapshot this store into a record that carries the store itself.
765    pub fn describe(self: Arc<Self>, label: String, mirror: Option<Arc<Store>>) -> StoreInfo {
766        let count = self.count() as u32;
767        StoreInfo {
768            label,
769            store: self,
770            mirror,
771            count,
772        }
773    }

775    /// Remove every expired entry, telling listeners from a producer
776    /// thread, and complete with how many were removed. First pauses for
777    /// `pause_ms` on a background thread that polls the cancel token
778    /// (counted by `active_jobs` while it runs). Cancelling
779    /// completes the call with the cancelled code at once; the
780    /// background pause then notices and stops.
781    pub async fn compact(&self, pause_ms: u32, cancel: CancelToken) -> u32 {
782        Pause::start(pause_ms, &cancel).await;
783        self.sweep()
784    }
785
786    /// The live entry for each key, in order (absent where there's
787    /// none), resolved on a producer thread.
788    pub async fn get_many(&self, keys: Vec<String>) -> Vec<Option<Entry>> {
789        keys.into_iter().map(|k| self.find(k)).collect()
790    }
791
792    /// The version of `key`'s live entry, if any, resolved on a producer
793    /// thread.
794    pub async fn version_of(&self, key: String) -> Option<u32> {
795        self.find(key).map(|e| e.version)
796    }
797
798    /// The version of each key's live entry, in order (`0` where
799    /// there's none), resolved on a producer thread.
800    pub async fn versions(&self, keys: Vec<String>) -> Vec<u32> {
801        keys.into_iter()
802            .map(|k| self.find(k).map_or(0, |e| e.version))
803            .collect()
804    }

820/// A pause that runs on a background thread polling a cancel token, so
821/// a cancelled call stops its work instead of sleeping on.
822struct Pause {
823    state: Option<Arc<PauseState>>,
824}
825
826#[derive(Default)]
827struct PauseState {
828    done: std::sync::atomic::AtomicBool,
829    waker: Mutex<Option<std::task::Waker>>,
830}
831
832impl Pause {
833    #[cfg(not(target_arch = "wasm32"))]
834    fn start(ms: u32, cancel: &CancelToken) -> Pause {
835        if ms == 0 {
836            return Pause { state: None };
837        }
838        let state = Arc::new(PauseState::default());
839        let shared = Arc::clone(&state);
840        let cancel = cancel.clone();
841        let deadline = std::time::Instant::now() + std::time::Duration::from_millis(ms.into());
842        ACTIVE_JOBS.fetch_add(1, Ordering::SeqCst);
843        std::thread::spawn(move || {
844            while !cancel.is_cancelled() && std::time::Instant::now() < deadline {
845                std::thread::sleep(std::time::Duration::from_millis(1));
846            }
847            drop(cancel);
848            shared.done.store(true, Ordering::SeqCst);
849            if let Some(w) = lock(&shared.waker).take() {
850                w.wake();
851            }
852            ACTIVE_JOBS.fetch_sub(1, Ordering::SeqCst);
853        });
854        Pause { state: Some(state) }
855    }
856
857    // `wasm32-unknown-unknown` has neither threads nor a clock, so the
858    // pause elapses at once there.
859    #[cfg(target_arch = "wasm32")]
860    fn start(ms: u32, cancel: &CancelToken) -> Pause {
861        let _ = (ms, cancel);
862        Pause { state: None }
863    }
864}
865
866impl Future for Pause {
867    type Output = ();
868
869    fn poll(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<()> {
870        let Some(state) = &self.state else {
871            return Poll::Ready(());
872        };
873        *lock(&state.waker) = Some(cx.waker().clone());
874        if state.done.load(Ordering::SeqCst) {
875            Poll::Ready(())
876        } else {
877            Poll::Pending
878        }
879    }
880}

882/// Aggregate statistics: functions in their own Rust module that take the
883/// `Store` and report its `KvError` domain (a library is one namespace, so
884/// the module only organizes the Rust source).
885pub mod stats {
886    use std::collections::BTreeMap;
887    use std::sync::Arc;
888
889    use crate::{EntryKind, KvError, Store};
890
891    /// Aggregate statistics over a set of entries.
892    #[weaveffi::record]
893    #[derive(Clone, Debug, PartialEq, Default)]
894    pub struct Stats {
895        /// The number of live entries.
896        pub entries: u32,
897        /// The total size of their values in bytes.
898        pub bytes: u64,
899        /// Live entries per kind (kinds with none are absent).
900        pub by_kind: BTreeMap<EntryKind, u32>,
901    }
902
903    fn add(stats: &mut Stats, store: &Store, prefix: Option<&str>) {
904        for e in store.live(prefix) {
905            stats.entries += 1;
906            stats.bytes += e.value.len() as u64;
907            *stats.by_kind.entry(e.kind).or_insert(0) += 1;
908        }
909    }
910
911    /// Statistics over `store`'s live entries, optionally only those
912    /// whose keys start with `prefix`. Fails with
913    /// `KeyNotFound` naming the prefix when it matches
914    /// nothing.
915    #[weaveffi::export]
916    pub fn summarize(store: &Store, prefix: Option<String>) -> Result<Stats, KvError> {
917        let mut stats = Stats::default();
918        add(&mut stats, store, prefix.as_deref());
919        match prefix {
920            Some(p) if stats.entries == 0 => Err(KvError::KeyNotFound { key: p }),
921            _ => Ok(stats),
922        }
923    }
924
925    /// Statistics over every store's live entries, computed on a
926    /// producer thread.
927    #[weaveffi::export]
928    pub async fn summarize_all(stores: Vec<Arc<Store>>) -> Stats {
929        let mut stats = Stats::default();
930        for store in &stores {
931            add(&mut stats, store, None);
932        }
933        stats
934    }
935}

937/// Plain-text reports over entries, with their own error domain, in a
938/// second Rust module.
939pub mod report {
940    use crate::Entry;
941
942    /// The report module's error domain.
943    #[weaveffi::error]
944    #[derive(Debug)]
945    pub enum ReportError {
946        /// nothing to report
947        NothingToReport = 2001,
948    }
949
950    /// One line per entry, sorted by key: `"{key}: {n} bytes, {kind}"`, with
951    /// `, v{version}` appended when the version is above 1. Fails with
952    /// `NothingToReport` for an empty list.
953    #[weaveffi::export]
954    pub fn render_report(mut entries: Vec<Entry>) -> Result<Vec<String>, ReportError> {
955        if entries.is_empty() {
956            return Err(ReportError::NothingToReport);
957        }
958        entries.sort_by(|a, b| a.key.cmp(&b.key));
959        Ok(entries
960            .iter()
961            .map(|e| {
962                let mut line = format!("{}: {} bytes, {:?}", e.key, e.value.len(), e.kind);
963                if e.version > 1 {
964                    line.push_str(&format!(", v{}", e.version));
965                }
966                line
967            })
968            .collect())
969    }
970}
