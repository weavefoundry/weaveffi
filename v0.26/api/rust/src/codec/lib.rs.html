1//! Codec sample cdylib: the wire oracle for the WeaveFFI value-buffer
2//! protocol, built on shared test vectors.
3//!
4//! Every generated binding ships its own encoder and decoder for records,
5//! rich enums, optionals, lists, maps, and object tokens. This producer holds
6//! one fixed table of test values, the *vectors*, covering every wire shape:
7//! each primitive at its extremes (NaN, the infinities, `-0.0`, and
8//! subnormals for floats), empty and non-empty strings and bytes with
9//! non-ASCII text and interior NUL bytes, optionals present and absent,
10//! lists, maps keyed by strings, integers, and C-style enums, nested records,
11//! C-style and rich enums, and objects inside buffers. Each vector is one
12//! variant of the [`Vector`] rich enum.
13//!
14//! # The vector protocol
15//!
16//! A consumer's codec check is one loop over the table:
17//!
18//! 1. `vector_count()` is the number of vectors; indexes run from `0`.
19//! 2. `vector(i)` returns vector `i` (the producer encodes, the consumer
20//!    decodes). An index past the end fails with
21//!    [`CodecError::OutOfRange`], whose payload carries the index and
22//!    the count.
23//! 3. `check_vector(i, v)` takes the decoded value straight back (the
24//!    consumer re-encodes, the producer decodes) and returns `true` only when
25//!    it equals vector `i` exactly: integers bit for bit, floats with their
26//!    sign (any NaN matches NaN), strings and bytes byte for byte, optionals,
27//!    list order, map contents, enum variants, and the values of the objects
28//!    inside. A lossy decode or encode in either direction makes it `false`.
29//! 4. `vector_name(i)` labels vector `i` and `describe_vector(v)` renders
30//!    any vector as text, so a failing consumer can print what it sent and
31//!    what the producer saw.
32//!
33//! A round trip alone can't catch a symmetric bug (a decoder and an encoder
34//! that swap the same two fields), so a consumer also builds a few vectors
35//! from literals and checks those, and spot-checks some decoded fields
36//! against literals.
37//!
38//! Buffers aren't the only way values cross. The `echo_*` functions return
39//! their argument unchanged through the direct ABI families (scalars by
40//! value, strings and bytes as `(ptr, len)` runs, the C-style enum as an
41//! `int32_t`, optional scalars as a presence flag plus a value, numeric
42//! lists as typed arrays, `usize` as a `u64`, a `char` as a one-scalar
43//! string, and the custom `Hex` type as its string repr), so a consumer
44//! feeds every primitive vector's value through the matching echo too.
45//! `chunks` streams typed arrays through an iterator. The `Token` interface
46//! with `primary_of`, `same_primary`, and `sum_holder` checks object
47//! identity and reference counting through buffers.
48//!
49//! The producer itself is pure safe Rust; the `#[weaveffi::...]` expansions
50//! supply the codecs the consumers are checked against.

55/// Parse a `Hex`: lowercase or uppercase hex digits, no prefix.
56fn parse_hex(text: String) -> Result<Hex, std::num::ParseIntError> {
57    u32::from_str_radix(&text, 16).map(Hex)
58}
59
60/// Format a `Hex`: lowercase hex digits, no prefix.
61fn format_hex(value: &Hex) -> String {
62    format!("{:x}", value.0)
63}

65/// The oracle's error domain.
66#[weaveffi::error]
67#[derive(Debug, PartialEq)]
68#[repr(i32)]
69pub enum CodecError {
70    /// vector index out of range
71    #[weaveffi(message = "vector {index} is out of range (count {count})")]
72    OutOfRange {
73        /// The requested index.
74        index: u32,
75        /// The number of vectors.
76        count: u32,
77    } = 1,
78}

539/// Wrap a value in the `Vector` variant of its shape.
540macro_rules! vector_from {
541    ($($t:ty => $variant:ident),* $(,)?) => {
542        $(
543            impl From<$t> for Vector {
544                fn from(value: $t) -> Self {
545                    Vector::$variant { value }
546                }
547            }
548        )*
549    };
550}

730pub fn vector(index: u32) -> Result<Vector, CodecError> {
731    let mut all = table();
732    let count = all.len() as u32;
733    if index >= count {
734        return Err(CodecError::OutOfRange { index, count });
735    }
736    Ok(all.swap_remove(index as usize).1)
737}

739/// A short label for vector `index`. Fails with
740/// `OutOfRange` for an index past the end.
741#[weaveffi::export]
742pub fn vector_name(index: u32) -> Result<String, CodecError> {
743    let all = table();
744    let count = all.len() as u32;
745    all.get(index as usize)
746        .map(|(name, _)| (*name).to_string())
747        .ok_or(CodecError::OutOfRange { index, count })
748}

750/// Whether `value` equals vector `index` exactly (`false` for an index
751/// past the end).
752#[weaveffi::export]
753pub fn check_vector(index: u32, value: Vector) -> bool {
754    table()
755        .get(index as usize)
756        .is_some_and(|(_, expected)| same(expected, &value))
757}

916/// `values` split into consecutive arrays of `size` elements (the last
917/// may be shorter), pulled lazily; nothing for a `size` of `0`.
918#[weaveffi::export]
919pub fn chunks(values: &[i32], size: u32) -> weaveffi::Iter<Vec<i32>> {
920    let chunks: Vec<Vec<i32>> = if size == 0 {
921        Vec::new()
922    } else {
923        values.chunks(size as usize).map(<[i32]>::to_vec).collect()
924    };
925    weaveffi::Iter::new(chunks)
926}

928/// The sum of every token value inside `holder` (wrapping on overflow).
929#[weaveffi::export]
930pub fn sum_holder(holder: &Holder) -> i64 {
931    std::iter::once(&holder.primary)
932        .chain(holder.spare.iter())
933        .chain(holder.many.iter())
934        .chain(holder.by_name.values())
935        .fold(0i64, |acc, t| acc.wrapping_add(t.value()))
936}

961    #[test]
962    fn every_vector_round_trips_through_its_codec() {
963        let n = vector_count();
964        assert!(n > 60, "the table covers every shape");
965        for i in 0..n {
966            let v = vector(i).unwrap();
967            let mut w = weaveffi::__rt::BufferWriter::new();
968            <Vector as Codec<Tag>>::encode(&v, &mut w);
969            let bytes = w.finish();
970            let mut r = BufferReader::new(&bytes);
971            let back = unsafe { <Vector as Codec<Tag>>::decode(&mut r) }.unwrap();
972            r.expect_end().unwrap();
973            assert!(check_vector(i, back), "vector {i}: {v:?}");
974        }
975        assert_eq!(
976            vector(n).unwrap_err(),
977            CodecError::OutOfRange { index: n, count: n }
978        );
979    }
