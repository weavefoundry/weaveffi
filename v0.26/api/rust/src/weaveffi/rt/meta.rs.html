1//! Compile-time metadata: type spellings, contract signatures, and the
2//! metadata frames a producer embeds.
3//!
4//! The producer macros only see the tokens of the item they annotate, so a
5//! parameter's IDL spelling (`[Entry?]`, `{string:u64}`) comes from its
6//! type's [`FfiType::NAME`](crate::rt::FfiType::NAME), a [`Name`] each type
7//! builds at compile time from its parts. A declaration's metadata frame and
8//! contract signature are then sequences of [`Part`]s (literal text from the
9//! macro, names from the types) that [`frame`] and [`hash`] assemble in
10//! `const` context. The CLI computes the same signature strings from the IR
11//! (`weaveffi_model::contract`), which is what makes a binding's contract
12//! check agree with the library it loads.
13
14/// The capacity of a [`Name`] in bytes.
15pub const NAME_CAP: usize = 256;
16
17/// A type's IDL spelling, built at compile time (`i32`, `[Entry?]`).
18#[derive(Clone, Copy)]
19pub struct Name {
20    bytes: [u8; NAME_CAP],
21    len: usize,
22}
23
24impl Name {
25    /// The empty name (a callable that throws nothing has an empty
26    /// [`FfiReturn::DOMAIN`](crate::rt::FfiReturn::DOMAIN)).
27    pub const EMPTY: Name = Name {
28        bytes: [0; NAME_CAP],
29        len: 0,
30    };
31
32    /// The name `s`.
33    #[must_use]
34    pub const fn new(s: &str) -> Name {
35        Name::EMPTY.push(s.as_bytes())
36    }
37
38    /// This name followed by `more`.
39    ///
40    /// # Panics
41    ///
42    /// Panics (at compile time) when the result exceeds [`NAME_CAP`] bytes.
43    #[must_use]
44    pub const fn push(mut self, more: &[u8]) -> Name {
45        assert!(
46            self.len + more.len() <= NAME_CAP,
47            "a WeaveFFI type spelling is longer than 256 bytes"
48        );
49        let mut i = 0;
50        while i < more.len() {
51            self.bytes[self.len + i] = more[i];
52            i += 1;
53        }
54        self.len += more.len();
55        self
56    }
57
58    /// `inner?`.
59    #[must_use]
60    pub const fn optional(inner: &Name) -> Name {
61        Name::EMPTY.push(inner.as_bytes()).push(b"?")
62    }
63
64    /// `[inner]`.
65    #[must_use]
66    pub const fn list(inner: &Name) -> Name {
67        Name::new("[").push(inner.as_bytes()).push(b"]")
68    }
69
70    /// `{key:value}`.
71    #[must_use]
72    pub const fn map(key: &Name, value: &Name) -> Name {
73        Name::new("{")
74            .push(key.as_bytes())
75            .push(b":")
76            .push(value.as_bytes())
77            .push(b"}")
78    }
79
80    /// `iter<inner>`.
81    #[must_use]
82    pub const fn iter(inner: &Name) -> Name {
83        Name::new("iter<").push(inner.as_bytes()).push(b">")
84    }
85
86    /// A contract signature's ` throws` suffix for the domain `domain`
87    /// (`KvError`, or `any` for an untyped error): ` throws {domain}`, or
88    /// nothing for the empty domain of an infallible callable.
89    #[must_use]
90    pub const fn suffix(domain: &Name) -> Name {
91        if domain.len == 0 {
92            Name::EMPTY
93        } else {
94            Name::new(" throws ").push(domain.as_bytes())
95        }
96    }
97
98    /// The spelling's bytes.
99    #[must_use]
100    pub const fn as_bytes(&self) -> &[u8] {
101        self.bytes.split_at(self.len).0
102    }
103
104    /// Whether the spelling is `s`.
105    #[must_use]
106    pub const fn is(&self, s: &str) -> bool {
107        let s = s.as_bytes();
108        if s.len() != self.len {
109            return false;
110        }
111        let mut i = 0;
112        while i < s.len() {
113            if s[i] != self.bytes[i] {
114                return false;
115            }
116            i += 1;
117        }
118        true
119    }
120}
121
122impl std::fmt::Debug for Name {
123    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
124        f.write_str(&String::from_utf8_lossy(self.as_bytes()))
125    }
126}

128/// One part of a frame or a signature.
129#[derive(Clone, Copy)]
130pub enum Part<'a> {
131    /// Literal bytes.
132    Lit(&'a str),
133    /// A type's spelling (or a ` throws` suffix).
134    Name(&'a Name),
135    /// A source path, with `\` written as `/` (it lands inside a JSON
136    /// string, and paths never contain `"`).
137    Path(&'a str),
138    /// A decimal number.
139    Num(u32),
140}
141
142const fn digits(mut n: u32) -> usize {
143    let mut d = 1;
144    while n >= 10 {
145        n /= 10;
146        d += 1;
147    }
148    d
149}
150
151const fn part_len(part: &Part<'_>) -> usize {
152    match part {
153        Part::Lit(s) | Part::Path(s) => s.len(),
154        Part::Name(n) => n.len,
155        Part::Num(n) => digits(*n),
156    }
157}
158
159/// The total length of `parts`.
160#[must_use]
161pub const fn len(parts: &[Part<'_>]) -> usize {
162    let mut total = 0;
163    let mut i = 0;
164    while i < parts.len() {
165        total += part_len(&parts[i]);
166        i += 1;
167    }
168    total
169}
170
171/// Call `f` on each byte of `parts`, in order (as a const-friendly loop the
172/// callers inline).
173const fn byte_at(parts: &[Part<'_>], mut index: usize) -> u8 {
174    let mut i = 0;
175    while i < parts.len() {
176        let n = part_len(&parts[i]);
177        if index < n {
178            return match &parts[i] {
179                Part::Lit(s) => s.as_bytes()[index],
180                Part::Path(s) => match s.as_bytes()[index] {
181                    b'\\' => b'/',
182                    b => b,
183                },
184                Part::Name(name) => name.bytes[index],
185                Part::Num(num) => {
186                    let mut v = *num;
187                    let mut k = n - 1 - index;
188                    while k > 0 {
189                        v /= 10;
190                        k -= 1;
191                    }
192                    b'0' + (v % 10) as u8
193                }
194            };
195        }
196        index -= n;
197        i += 1;
198    }
199    0
200}

202/// The 64-bit FNV-1a hash of the concatenation of `parts`: a contract
203/// signature's hash.
204#[must_use]
205pub const fn hash(parts: &[Part<'_>]) -> u64 {
206    let mut hash: u64 = 0xcbf2_9ce4_8422_2325;
207    let total = len(parts);
208    let mut i = 0;
209    while i < total {
210        hash ^= byte_at(parts, i) as u64;
211        hash = hash.wrapping_mul(0x0000_0100_0000_01b3);
212        i += 1;
213    }
214    hash
215}
216
217/// The byte length of the frame [`frame`] builds from `parts`: a `u32`
218/// length prefix and the parts.
219#[must_use]
220pub const fn frame_len(parts: &[Part<'_>]) -> usize {
221    4 + len(parts)
222}
223
224/// A metadata frame: a little-endian `u32` length, then the concatenation
225/// of `parts` (compact JSON). `N` must be [`frame_len`] of `parts`.
226///
227/// # Panics
228///
229/// Panics (at compile time) when `N` is wrong.
230#[must_use]
231pub const fn frame<const N: usize>(parts: &[Part<'_>]) -> [u8; N] {
232    let body = len(parts);
233    assert!(N == body + 4, "frame length mismatch");
234    let mut out = [0u8; N];
235    let prefix = (body as u32).to_le_bytes();
236    out[0] = prefix[0];
237    out[1] = prefix[1];
238    out[2] = prefix[2];
239    out[3] = prefix[3];
240    let mut i = 0;
241    while i < body {
242        out[4 + i] = byte_at(parts, i);
243        i += 1;
244    }
245    out
246}

248#[cfg(test)]
249mod tests {
250    use super::*;
251
252    const ENTRY: Name = Name::new("Entry");
253    const SPELLING: Name = Name::map(&Name::new("string"), &Name::list(&Name::optional(&ENTRY)));
254
255    #[test]
256    fn names_compose() {
257        assert_eq!(SPELLING.as_bytes(), b"{string:[Entry?]}");
258        assert!(Name::iter(&ENTRY).is("iter<Entry>"));
259        assert!(Name::suffix(&Name::new("any")).is(" throws any"));
260        assert!(Name::suffix(&Name::EMPTY).is(""));
261        assert!(Name::EMPTY.is(""));
262    }
263
264    #[test]
265    fn frames_and_hashes_concatenate_parts() {
266        const PARTS: &[Part<'static>] = &[
267            Part::Lit("method get("),
268            Part::Name(&Name::new("string")),
269            Part::Lit(") -> "),
270            Part::Name(&ENTRY),
271            Part::Name(&Name::suffix(&Name::new("KvError"))),
272        ];
273        let text = b"method get(string) -> Entry throws KvError";
274        let expected = {
275            let mut h: u64 = 0xcbf2_9ce4_8422_2325;
276            for b in text {
277                h ^= u64::from(*b);
278                h = h.wrapping_mul(0x0000_0100_0000_01b3);
279            }
280            h
281        };
282        assert_eq!(hash(PARTS), expected);
283        const LEN: usize = frame_len(PARTS);
284        const FRAME: [u8; LEN] = frame::<LEN>(PARTS);
285        assert_eq!(&FRAME[..4], &(text.len() as u32).to_le_bytes());
286        assert_eq!(&FRAME[4..], text);
287
288        const LOC: &[Part<'static>] = &[
289            Part::Path("src\\lib.rs"),
290            Part::Lit(":"),
291            Part::Num(1207),
292            Part::Lit(":"),
293            Part::Num(5),
294            Part::Num(0),
295        ];
296        const L: usize = len(LOC);
297        const BYTES: [u8; L + 4] = frame::<{ L + 4 }>(LOC);
298        assert_eq!(&BYTES[4..], b"src/lib.rs:1207:50");
299    }
