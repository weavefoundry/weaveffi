1//! The value layer: how every Rust type a producer exports crosses the C
2//! ABI, chosen by the type itself.
3//!
4//! In ABI revision 6 every value is one C value in every position, so a
5//! type's crossing is a property of the type: [`FfiType::Abi`] is its C
6//! value (a scalar, an [`Opt`], a [`Run`], or an object pointer), and its
7//! methods move it across in each direction. The producer macros never
8//! inspect a type's syntax to marshal it; they emit
9//! `<T as FfiType<Tag>>::lift(abi)` and let rustc resolve `T` wherever it's
10//! declared, through aliases, in other modules, or in other crates.
11//!
12//! `UT` is the producer crate's **tag**, the `crate::__weaveffi::Tag` type
13//! `weaveffi::export_runtime!()` defines: a crate implements these traits
14//! for its own types, and for a foreign type it exports as a custom type,
15//! under its own tag, which the orphan rules allow.
16//!
17//! * [`FfiType`]: a value type, legal everywhere (parameters, returns, async
18//!   results, iterator items, callback arguments and returns, and inside
19//!   value buffers). Its `Option` and `Vec` crossings are hooks on the
20//!   element type, so `Option<i32>` is an [`Opt`], `Option<Arc<Store>>` a
21//!   nullable pointer, `Vec<f64>` a typed array, `Vec<u8>` raw bytes, and
22//!   everything else a value buffer.
23//! * [`FfiRef`]: a borrowed parameter (`&str`, `&[u8]`, `&Store`, `&Entry`)
24//!   and a borrowed callback argument.
25//! * [`FfiReturn`]: what a callable returns: a value, `()`, an interface by
26//!   value (`Self`), or a `Result` whose error is an
27//!   [`ErrorType`](crate::rt::ErrorType).
28

38/// Keeps the encodings of a callback call's borrowed arguments alive until
39/// the call returns.
40#[derive(Debug, Default)]
41pub struct Keep(Vec<Vec<u8>>);
42
43impl Keep {
44    /// An empty keeper.
45    #[must_use]
46    pub fn new() -> Self {
47        Self::default()
48    }
49
50    /// A run borrowing `bytes`, which the keeper holds.
51    pub fn hold(&mut self, bytes: Vec<u8>) -> Run {
52        let run = Run::borrowing(&bytes);
53        // Moving the vector doesn't move its heap buffer.
54        self.0.push(bytes);
55        run
56    }
57}

65/// Name the parameter an argument came from in its lift failure: a
66/// marshalling failure reports `"{param}: {message}"`, and any other
67/// failure passes through unchanged.
68///
69/// # Errors
70///
71/// Returns `lifted`'s error, renamed as above.
72pub fn in_param<V>(lifted: Result<V, FfiError>, param: &str) -> Result<V, FfiError> {
73    lifted.map_err(|e| {
74        if e.code != MARSHAL_ERROR_CODE {
75            return e;
76        }
77        // SAFETY: the runtime filled `e`, so its message run is live.
78        let message = unsafe { e.message_str() }.unwrap_or_default();
79        marshal(&format!("{param}: {message}"))
80    })
81}

87/// Decode a value buffer with `read`, requiring every byte to be consumed.
88///
89/// # Errors
90///
91/// Returns a marshalling error for a malformed run or buffer.
92///
93/// # Safety
94///
95/// The run must be readable for the call, and every object token in it
96/// must carry an unadopted reference (see
97/// [`FfiType::read`]).
98pub unsafe fn decode_run<T>(
99    run: Run,
100    read: impl FnOnce(&mut BufferReader<'_>) -> Result<T, BufferDecodeError>,
101) -> Result<T, FfiError> {
102    // SAFETY: forwarded from the caller.
103    let bytes = unsafe { run.bytes() }?;
104    let mut r = BufferReader::new(bytes);
105    let value = read(&mut r).map_err(|e| decode_error(&e))?;
106    r.expect_end().map_err(|e| decode_error(&e))?;
107    Ok(value)
108}

110/// Encode with `write` into a fresh buffer.
111#[must_use]
112pub fn encode(write: impl FnOnce(&mut BufferWriter)) -> Vec<u8> {
113    let mut w = BufferWriter::new();
114    write(&mut w);
115    w.finish()
116}

118/// A value type: the trait every type a producer exports implements (see
119/// the [module docs](self)).
120#[diagnostic::on_unimplemented(
121    message = "`{Self}` can't cross the WeaveFFI C ABI",
122    label = "not a WeaveFFI value type",
123    note = "a value is a scalar (`i32`, `f64`, `bool`, `usize`, ...), `char`, `String`, `Vec<T>`, \
124            `Option<T>`, a `HashMap` or `BTreeMap`, a `#[weaveffi::record]`, a \
125            `#[weaveffi::enumeration]`, a `#[weaveffi::custom]` type, or a \
126            `#[weaveffi::interface]` as `Arc<T>`",
127    note = "a parameter may also be borrowed (`&str`, `&[T]`, `&T`), and only a whole return may \
128            be a `weaveffi::Iter<T>`"
129)]
130pub trait FfiType<UT>: Sized {
131    /// The IDL spelling (`i32`, `string`, `[Entry?]`).
132    const NAME: Name;
133
134    /// The one C value the type crosses as.
135    type Abi: Copy;
136
137    /// The zero value a failed call returns.
138    const ZERO: Self::Abi;
139
140    /// Move the value to the receiver: a return, an async result, or an
141    /// iterator item. A run is allocated; an object reference moves.
142    fn lower(self) -> Self::Abi;
143
144    /// Lift a parameter the consumer lends for the call: runs are copied
145    /// or decoded (adopting any object tokens), and an object gains a
146    /// reference.
147    ///
148    /// # Errors
149    ///
150    /// Returns a marshalling error when the value is malformed.
151    ///
152    /// # Safety
153    ///
154    /// `abi` must be a valid C value of the type, as the ABI defines it.
155    unsafe fn lift(abi: Self::Abi) -> Result<Self, FfiError>;
156
157    /// Release a C value the receiver owns without lifting it: free a run,
158    /// or drop an object reference.
159    ///
160    /// # Safety
161    ///
162    /// `abi` must own what it releases, released once.
163    unsafe fn release(abi: Self::Abi);
164
165    /// Lift a value the consumer moved to the producer (a callback
166    /// method's return): lift it, then release the original.
167    ///
168    /// # Errors
169    ///
170    /// Returns a marshalling error when the value is malformed.
171    ///
172    /// # Safety
173    ///
174    /// As [`lift`](Self::lift) and [`release`](Self::release).
175    unsafe fn adopt(abi: Self::Abi) -> Result<Self, FfiError> {
176        // SAFETY: forwarded from the caller.
177        let value = unsafe { <Self as FfiType<UT>>::lift(abi) };
178        // SAFETY: forwarded from the caller.
179        unsafe { <Self as FfiType<UT>>::release(abi) };
180        value
181    }
182
183    /// Lend the value as a callback argument: runs borrow `self` or an
184    /// encoding `keep` holds; an object moves a fresh reference.
185    fn lower_arg(&self, keep: &mut Keep) -> Self::Abi;
186
187    /// Append the value's value-buffer encoding.
188    fn write(&self, w: &mut BufferWriter);
189
190    /// Decode one value from a value buffer.
191    ///
192    /// # Errors
193    ///
194    /// Returns an error when the buffer is exhausted or malformed.
195    ///
196    /// # Safety
197    ///
198    /// Unless `r` was built with [`BufferReader::token_free`], every object
199    /// token the value contains must carry one unadopted strong reference to
200    /// a live object of its type; reading adopts it.
201    unsafe fn read(r: &mut BufferReader<'_>) -> Result<Self, BufferDecodeError>;
202
203    /// The C value of `Option<Self>`.
204    type OptAbi: Copy;
205
206    /// The zero `Option<Self>`.
207    const OPT_ZERO: Self::OptAbi;
208
209    /// [`lower`](Self::lower) for `Option<Self>`.
210    fn lower_opt(value: Option<Self>) -> Self::OptAbi;
211
212    /// [`lift`](Self::lift) for `Option<Self>`.
213    ///
214    /// # Errors
215    ///
216    /// Returns a marshalling error when the value is malformed.
217    ///
218    /// # Safety
219    ///
220    /// As [`lift`](Self::lift).
221    unsafe fn lift_opt(abi: Self::OptAbi) -> Result<Option<Self>, FfiError>;
222
223    /// [`release`](Self::release) for `Option<Self>`.
224    ///
225    /// # Safety
226    ///
227    /// As [`release`](Self::release).
228    unsafe fn release_opt(abi: Self::OptAbi);
229
230    /// [`lower_arg`](Self::lower_arg) for `Option<Self>`.
231    fn lower_opt_arg(value: Option<&Self>, keep: &mut Keep) -> Self::OptAbi;
232
233    /// The IDL spelling of `Vec<Self>` (`bytes` for `u8`).
234    const LIST_NAME: Name = Name::list(&Self::NAME);
235
236    /// [`lower`](Self::lower) for `Vec<Self>`: a value buffer, or a typed
237    /// array for the numeric types.
238    fn lower_list(values: Vec<Self>) -> Run {
239        Run::copy_of(&encode(|w| <Self as FfiType<UT>>::write_list(&values, w)))
240    }
241
242    /// [`lift`](Self::lift) for `Vec<Self>`.
243    ///
244    /// # Errors
245    ///
246    /// Returns a marshalling error when the run is malformed.
247    ///
248    /// # Safety
249    ///
250    /// As [`lift`](Self::lift).
251    unsafe fn lift_list(run: Run) -> Result<Vec<Self>, FfiError> {
252        // SAFETY: forwarded from the caller.
253        unsafe { decode_run(run, |r| <Self as FfiType<UT>>::read_list(r)) }
254    }
255
256    /// Lift `&[Self]`: borrowed in place for the numeric types, decoded
257    /// otherwise.
258    ///
259    /// # Errors
260    ///
261    /// Returns a marshalling error when the run is malformed.
262    ///
263    /// # Safety
264    ///
265    /// As [`lift`](Self::lift), and the run outlives `'a`.
266    unsafe fn lift_slice<'a>(run: Run) -> Result<Slice<'a, Self>, FfiError> {
267        // SAFETY: forwarded from the caller.
268        unsafe { <Self as FfiType<UT>>::lift_list(run) }.map(Slice::Owned)
269    }
270
271    /// [`lower_arg`](Self::lower_arg) for `[Self]`.
272    fn lower_list_arg(values: &[Self], keep: &mut Keep) -> Run {
273        keep.hold(encode(|w| <Self as FfiType<UT>>::write_list(values, w)))
274    }
275
276    /// Append a list's value-buffer encoding: a `u32` count, then each
277    /// element.
278    fn write_list(values: &[Self], w: &mut BufferWriter) {
279        w.write_len(values.len());
280        for v in values {
281            v.write(w);
282        }
283    }
284
285    /// Decode a list from a value buffer.
286    ///
287    /// # Errors
288    ///
289    /// Returns an error when the buffer is exhausted or malformed.
290    ///
291    /// # Safety
292    ///
293    /// As [`read`](Self::read).
294    unsafe fn read_list(r: &mut BufferReader<'_>) -> Result<Vec<Self>, BufferDecodeError> {
295        let count = r.read_count()?;
296        let mut out = Vec::with_capacity(count.min(r.remaining()));
297        for _ in 0..count {
298            // SAFETY: forwarded from the caller.
299            out.push(unsafe { <Self as FfiType<UT>>::read(r) }?);
300        }
301        Ok(out)
302    }
303}

464/// Implement [`FfiRef`] (held by value), [`FfiLend`], and [`FfiReturn`] for a type that
465/// implements [`FfiType`] under the tag `$ut` (or every tag, when `$ut` is
466/// a generic parameter the caller binds). The producer macros use it for
467/// every record, enum, and custom type.
468#[doc(hidden)]
469#[macro_export]
470macro_rules! __weaveffi_value_traits {
471    ([$($gen:tt)*] $ut:ty, $t:ty $(where $($bound:tt)*)?) => {
472        impl<$($gen)*> $crate::__rt::FfiRef<$ut> for $t $(where $($bound)*)? {
473            const NAME: $crate::__rt::Name = <$t as $crate::__rt::FfiType<$ut>>::NAME;
474            type Abi = <$t as $crate::__rt::FfiType<$ut>>::Abi;
475            type Holder<'a> = $crate::__rt::Owned<$t> where Self: 'a;
476            type OptAbi = <$t as $crate::__rt::FfiType<$ut>>::OptAbi;
477            type OptHolder<'a> = $crate::__rt::Owned<$t> where Self: 'a;
478            type Owned = $t;
479            unsafe fn lift_owned(abi: Self::Abi) -> ::core::result::Result<$t, $crate::__rt::FfiError> {
480                // SAFETY: forwarded from the caller.
481                unsafe { <$t as $crate::__rt::FfiType<$ut>>::lift(abi) }
482            }
483            unsafe fn lift_ref<'a>(
484                abi: Self::Abi,
485            ) -> ::core::result::Result<Self::Holder<'a>, $crate::__rt::FfiError>
486            where
487                Self: 'a,
488            {
489                // SAFETY: forwarded from the caller.
490                unsafe { <$t as $crate::__rt::FfiType<$ut>>::lift(abi) }.map($crate::__rt::Owned)
491            }
492            unsafe fn lift_opt_ref<'a>(
493                abi: Self::OptAbi,
494            ) -> ::core::result::Result<
495                ::core::option::Option<Self::OptHolder<'a>>,
496                $crate::__rt::FfiError,
497            >
498            where
499                Self: 'a,
500            {
501                // SAFETY: forwarded from the caller.
502                unsafe { <$t as $crate::__rt::FfiType<$ut>>::lift_opt(abi) }
503                    .map(|v| v.map($crate::__rt::Owned))
504            }
505        }
506
507        impl<$($gen)*> $crate::__rt::FfiLend<$ut> for $t $(where $($bound)*)? {
508            const NAME: $crate::__rt::Name = <$t as $crate::__rt::FfiType<$ut>>::NAME;
509            type Abi = <$t as $crate::__rt::FfiType<$ut>>::Abi;
510            type OptAbi = <$t as $crate::__rt::FfiType<$ut>>::OptAbi;
511            fn lower_ref(&self, keep: &mut $crate::__rt::Keep) -> Self::Abi {
512                <$t as $crate::__rt::FfiType<$ut>>::lower_arg(self, keep)
513            }
514            fn lower_opt_ref(
515                value: ::core::option::Option<&Self>,
516                keep: &mut $crate::__rt::Keep,
517            ) -> Self::OptAbi {
518                <$t as $crate::__rt::FfiType<$ut>>::lower_opt_arg(value, keep)
519            }
520        }
521
522        impl<$($gen)*> $crate::__rt::FfiReturn<$ut> for $t $(where $($bound)*)? {
523            const NAME: $crate::__rt::Name = <$t as $crate::__rt::FfiType<$ut>>::NAME;
524            type Abi = <$t as $crate::__rt::FfiType<$ut>>::Abi;
525            const ZERO: Self::Abi = <$t as $crate::__rt::FfiType<$ut>>::ZERO;
526            fn lower_return(self) -> ::core::result::Result<Self::Abi, $crate::__rt::FfiError> {
527                ::core::result::Result::Ok(<$t as $crate::__rt::FfiType<$ut>>::lower(self))
528            }
529        }
530    };
531}

555/// Implement every value trait for a type that crosses as a value buffer
556/// (a record or a rich enum) from its [`Codec`].
557#[doc(hidden)]
558#[macro_export]
559macro_rules! __weaveffi_buffered {
560    ($ut:ty, $t:ty) => {
561        impl $crate::__rt::FfiType<$ut> for $t {
562            const NAME: $crate::__rt::Name = <$t as $crate::__rt::Codec<$ut>>::NAME;
563            type Abi = $crate::__rt::Run;
564            const ZERO: $crate::__rt::Run = $crate::__rt::Run::EMPTY;
565            fn lower(self) -> $crate::__rt::Run {
566                $crate::__rt::Run::copy_of(&$crate::__rt::encode(|w| {
567                    <$t as $crate::__rt::Codec<$ut>>::encode(&self, w)
568                }))
569            }
570            unsafe fn lift(abi: $crate::__rt::Run) -> ::core::result::Result<Self, $crate::__rt::FfiError> {
571                // SAFETY: forwarded from the caller.
572                unsafe {
573                    $crate::__rt::decode_run(abi, |r| <$t as $crate::__rt::Codec<$ut>>::decode(r))
574                }
575            }
576            unsafe fn release(abi: $crate::__rt::Run) {
577                // SAFETY: forwarded from the caller.
578                unsafe { abi.free() }
579            }
580            fn lower_arg(&self, keep: &mut $crate::__rt::Keep) -> $crate::__rt::Run {
581                keep.hold($crate::__rt::encode(|w| {
582                    <$t as $crate::__rt::Codec<$ut>>::encode(self, w)
583                }))
584            }
585            fn write(&self, w: &mut $crate::__rt::BufferWriter) {
586                <$t as $crate::__rt::Codec<$ut>>::encode(self, w)
587            }
588            unsafe fn read(
589                r: &mut $crate::__rt::BufferReader<'_>,
590            ) -> ::core::result::Result<Self, $crate::__rt::BufferDecodeError> {
591                // SAFETY: forwarded from the caller.
592                unsafe { <$t as $crate::__rt::Codec<$ut>>::decode(r) }

859/// list encoding inside value buffers on little-endian targets.
860macro_rules! numeric {
861    ($($t:ty => $name:literal, $write:ident, $read:ident;)*) => {
862        $(
863            impl<UT> FfiType<UT> for $t {
864                scalar_core!($t, $name, $write, $read);
865                fn lower_list(values: Vec<Self>) -> Run {
866                    Run::copy_of(pod_bytes(&values))
867                }
868                unsafe fn lift_list(run: Run) -> Result<Vec<Self>, FfiError> {
869                    // SAFETY: forwarded from the caller.
870                    unsafe { run.slice::<$t>() }.map(<[$t]>::to_vec)
871                }
872                unsafe fn lift_slice<'a>(run: Run) -> Result<Slice<'a, Self>, FfiError> {
873                    // SAFETY: forwarded from the caller.

880                    w.write_len(values.len());

882                        w.write_raw(pod_bytes(values));
883                    } else {
884                        for v in values {
885                            w.$write(*v);

887                    }
888                }

894                        )?;

897                            .chunks_exact(std::mem::size_of::<$t>())
898                            .map(|c| <$t>::from_le_bytes(c.try_into().expect("exact chunk")))
899                            .collect())

902                    }

907    };

1300macro_rules! map_type {
1301    ($map:ident, [$($bound:tt)*], $with_capacity:expr) => {
1302        impl<UT, K: FfiType<UT> + $($bound)*, V: FfiType<UT>> FfiType<UT> for $map<K, V> {
1303            const NAME: Name = Name::map(&K::NAME, &V::NAME);
1304            type Abi = Run;
1305            const ZERO: Run = Run::EMPTY;
1306            fn lower(self) -> Run {
1307                Run::copy_of(&encode(|w| self.write(w)))
1308            }
1309            unsafe fn lift(abi: Run) -> Result<Self, FfiError> {
1310                // SAFETY: forwarded from the caller.
1311                unsafe { decode_run(abi, |r| <Self as FfiType<UT>>::read(r)) }
1312            }
1313            unsafe fn release(abi: Run) {
1314                // SAFETY: forwarded from the caller.
1315                unsafe { abi.free() }
1316            }
1317            fn lower_arg(&self, keep: &mut Keep) -> Run {
1318                keep.hold(encode(|w| self.write(w)))
1319            }
1320            fn write(&self, w: &mut BufferWriter) {
1321                w.write_len(self.len());
1322                for (k, v) in self {
1323                    k.write(w);
1324                    v.write(w);
1325                }
1326            }
1327            unsafe fn read(r: &mut BufferReader<'_>) -> Result<Self, BufferDecodeError> {
1328                let len = r.read_count()?;
1329                let with_capacity: fn(usize) -> Self = $with_capacity;
1330                let mut out = with_capacity(len.min(r.remaining()));
1331                for _ in 0..len {
1332                    // SAFETY: forwarded from the caller.
1333                    let (k, v) = unsafe { (K::read(r)?, V::read(r)?) };
1334                    if out.insert(k, v).is_some() {
1335                        return Err(DUPLICATE_KEY);
1336                    }
1337                }
1338                Ok(out)
1339            }
1340            crate::__weaveffi_buffered_opt!(UT);
1341        }
1342        crate::__weaveffi_value_traits!(
1343            [UT, K: FfiType<UT> + $($bound)*, V: FfiType<UT>] UT, $map<K, V>
1344        );
1345    };
1346}
