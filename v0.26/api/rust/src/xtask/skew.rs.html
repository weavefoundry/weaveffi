1//! `cargo xtask skew`: bindings against a library of a different version.
2//!
3//! The contract (see the C ABI reference) lets a library grow: bindings
4//! generated from an older API load a newer library, a declaration the
5//! library changed is reported by its path, and one the library lacks
6//! fails to link (or, for a binding that loads the library at run time, to
7//! resolve). Each scenario here edits the calculator sample's extracted
8//! API, generates the C target from the edited copy, and builds
9//! `xtask/skew/check.c` against the real calculator library.
10
11use std::path::Path;
12use std::process::Command;
13
14use serde_json::Value;
15
16use crate::{run, Result, Workspace};
17
18/// One version-skew scenario: how the bindings' API differs from the
19/// library's, and what loading must do.
20struct Scenario {
21    name: &'static str,
22    edit: fn(&mut Value),
23    expect: Expect,
24}
25
26enum Expect {
27    /// The contract check passes and calls work.
28    Loads,
29    /// The contract check reports this declaration path.
30    Mismatch(&'static str),
31    /// Linking fails, naming the contract function of this declaration
32    /// path, which the library doesn't export.
33    NoLink(&'static str),
34}
35
36fn functions(api: &mut Value) -> &mut Vec<Value> {
37    api["functions"].as_array_mut().expect("functions")
38}
39
40fn function<'a>(api: &'a mut Value, name: &str) -> &'a mut Value {
41    functions(api)
42        .iter_mut()
43        .find(|f| f["name"] == name)
44        .expect("function")
45}
46
47const SCENARIOS: [Scenario; 5] = [
48    Scenario {
49        name: "the same API",
50        edit: |_| {},
51        expect: Expect::Loads,
52    },
53    Scenario {
54        // An older binding: the library has since added `mean`,
55        // `running_total`, `parse`, and the `ParseError` domain.
56        name: "bindings older than the library",
57        edit: |api| {
58            functions(api).retain(|f| {
59                !matches!(f["name"].as_str(), Some("mean" | "running_total" | "parse"))
60            });
61            api["errors"]
62                .as_array_mut()
63                .expect("errors")
64                .retain(|e| e["name"] != "ParseError");
65        },
66        expect: Expect::Loads,
67    },
68    Scenario {
69        name: "a parameter type changed",
70        edit: |api| function(api, "add")["params"][1]["type"] = "i64".into(),
71        expect: Expect::Mismatch("add"),
72    },
73    Scenario {
74        name: "an error code's value changed",
75        edit: |api| {
76            let domain = api["errors"]
77                .as_array_mut()
78                .expect("errors")
79                .iter_mut()
80                .find(|e| e["name"] == "CalcError")
81                .expect("CalcError");
82            domain["codes"][0]["code"] = 7.into();
83        },
84        expect: Expect::Mismatch("CalcError.DivisionByZero"),
85    },
86    Scenario {
87        // A newer binding: it declares `subtract`, which the library lacks.
88        name: "bindings newer than the library",
89        edit: |api| {
90            let mut sub = function(api, "add").clone();
91            sub["name"] = "subtract".into();
92            functions(api).push(sub);
93        },
94        expect: Expect::NoLink("subtract"),
95    },
96];
97
98/// Run every scenario.
99pub(crate) fn run_all(ws: &Workspace) -> Result<bool> {
100    if cfg!(windows) {
101        println!("skew: skipped on Windows (the check links with cc)");
102        return Ok(true);
103    }
104    ws.build_cli()?;
105    run(Command::new(crate::cargo())
106        .args(["build", "-q", "-p", "calculator"])
107        .current_dir(&ws.root))?;
108    let library = ws.sample_lib("calculator");
109    let lib_dir = library.parent().expect("library directory").to_path_buf();
110    let output = Command::new(ws.weaveffi())
111        .args(["-q", "extract", "-f", "json", "--library"])
112        .arg(&library)
113        .arg(ws.root.join("samples/calculator"))
114        .output()
115        .map_err(|e| format!("failed to run weaveffi extract: {e}"))?;
116    if !output.status.success() {
117        return Err(format!(
118            "weaveffi extract failed: {}",
119            String::from_utf8_lossy(&output.stderr)
120        ));
121    }
122    let api: Value = serde_json::from_slice(&output.stdout)
123        .map_err(|e| format!("extract printed invalid JSON: {e}"))?;
124
125    let scratch = ws.target_dir.join("skew");
126    let _ = std::fs::remove_dir_all(&scratch);
127    let mut ok = true;
128    for (i, scenario) in SCENARIOS.iter().enumerate() {
129        let dir = scratch.join(i.to_string());
130        let mut edited = api.clone();
131        (scenario.edit)(&mut edited);
132        let outcome = check(ws, &dir, &edited, &lib_dir)?;
133        let pass = match (&scenario.expect, &outcome) {
134            (Expect::Loads, Outcome::Loaded) => true,
135            (Expect::Mismatch(path), Outcome::Mismatch(got)) => got == path,
136            (Expect::NoLink(path), Outcome::NoLink(log)) => {
137                log.contains(&format!("calculator_contract_{:016x}", fnv1a64(path)))
138            }
139            _ => false,
140        };
141        println!(
142            "[{}] skew: {} ({outcome})",
143            if pass { "OK" } else { "FAIL" },
144            scenario.name
145        );
146        ok &= pass;
147    }
148    Ok(ok)
149}

151/// A declaration path's contract id (see the C ABI reference).
152fn fnv1a64(text: &str) -> u64 {
153    text.bytes().fold(0xcbf2_9ce4_8422_2325, |hash, b| {
154        (hash ^ u64::from(b)).wrapping_mul(0x0000_0100_0000_01b3)
155    })
156}
157
158/// What building and running the check against the library did.
159enum Outcome {
160    Loaded,
161    Mismatch(String),
162    NoLink(String),
163    Failed(String),
164}
165
166impl std::fmt::Display for Outcome {
167    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
168        match self {
169            Outcome::Loaded => f.write_str("loaded"),
170            Outcome::Mismatch(path) => write!(f, "contract mismatch at {path}"),
171            Outcome::NoLink(_) => f.write_str("didn't link"),
172            Outcome::Failed(why) => write!(f, "failed: {why}"),
173        }
174    }
175}
176
177/// Generate the C target from `api` into `dir`, then build and run the
178/// check program against the library in `lib_dir`.
179fn check(ws: &Workspace, dir: &Path, api: &Value, lib_dir: &Path) -> Result<Outcome> {
180    std::fs::create_dir_all(dir).map_err(|e| format!("{}: {e}", dir.display()))?;
181    let idl = dir.join("calculator.json");
182    std::fs::write(&idl, api.to_string()).map_err(|e| format!("{}: {e}", idl.display()))?;
183    std::fs::write(
184        dir.join("weaveffi.toml"),
185        "[package]\nname = \"calculator\"\nversion = \"1.0.0\"\n",
186    )
187    .map_err(|e| format!("weaveffi.toml: {e}"))?;
188    run(Command::new(ws.weaveffi())
189        .args(["-q", "generate"])
190        .arg(&idl)
191        .args(["--target", "c", "-o"])
192        .arg(dir.join("out")))?;
193
194    let exe = dir.join("check");
195    let cc = std::env::var("CC").unwrap_or_else(|_| "cc".to_string());
196    let build = Command::new(cc)
197        .args(["-std=c11", "-Wall", "-Wextra", "-Werror", "-I"])
198        .arg(dir.join("out/c"))
199        .arg(ws.root.join("xtask/skew/check.c"))
200        .arg("-L")
201        .arg(lib_dir)
202        .args(["-lcalculator", "-o"])
203        .arg(&exe)
204        .output()
205        .map_err(|e| format!("failed to run cc: {e}"))?;
206    if !build.status.success() {
207        return Ok(Outcome::NoLink(
208            String::from_utf8_lossy(&build.stderr).into_owned(),
209        ));
210    }
211    let loader_var = if cfg!(target_os = "macos") {
212        "DYLD_LIBRARY_PATH"
213    } else {
214        "LD_LIBRARY_PATH"
215    };
216    let ran = Command::new(&exe)
217        .env(loader_var, lib_dir)
218        .output()
219        .map_err(|e| format!("failed to run the check: {e}"))?;
220    let stdout = String::from_utf8_lossy(&ran.stdout).trim().to_string();
221    Ok(match ran.status.code() {
222        Some(0) => Outcome::Loaded,
223        Some(3) => Outcome::Mismatch(stdout),
224        _ => Outcome::Failed(format!(
225            "{}{}",
226            stdout,
227            String::from_utf8_lossy(&ran.stderr)
228        )),
229    })
230}
